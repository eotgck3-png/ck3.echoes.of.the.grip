# ── eotg_elven_monarchy_succession_events.txt ─────────────────────────────────
# Eternal Court confirmation event chain for the Elven Monarchy government.
#
# Confirmation (fires on new ruler 7 days after previous ruler's death, from
# eotg_on_death_elven_monarch):
#   court.0001 — The Court convenes. All three outcomes resolve INSIDE .0001:
#     Strong (>=3)   → eotg_se_elven_court_confirmed_bonus, vassals notified
#                      via send_interface_message
#     Moderate (>=1) → eotg_se_elven_court_confirmed_penalty, vassals notified
#     None (0)       → penalty + eotg_elven_court_contested, and .0004 fires at
#                      a hostile powerful vassal
#   court.0004 — Succession crisis. Rival claimant declared.
#
# Aftermath (fires from eotg_on_yearly_elven_flavor, gated on the modifier the
# confirmation left behind — these are the long tail of how the Court ruled):
#   court.0005 — The Court calls in the debt of a unanimous confirmation.
#                Requires eotg_elven_court_confirmed.
#   court.0006 — Ruling without the Court's blessing. Requires
#                eotg_elven_court_contested.
#
# NOTE: .0002 and .0003 were retired 2026-08-28. They were notification events
# from an earlier design in which .0001 branched outward; .0001 now resolves
# those outcomes inline and notifies vassals with send_interface_message, so the
# pair was unreachable and would have double-notified if wired. .0005/.0006
# replace them with reachable consequence events. IDs .0002/.0003 stay retired —
# do not reuse them.
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_elven_court

# The Eternal Court convenes to ratify the new ruler's succession.
eotg_elven_court.0001 = {
	type = character_event
	title = eotg_elven_court_0001_title
	theme = diplomacy

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { eotg_st_elven_court_strong_support = yes }
				desc = eotg_elven_court_0001_desc_strong
			}
			triggered_desc = {
				trigger = { eotg_st_elven_court_moderate_support = yes }
				desc = eotg_elven_court_0001_desc_moderate
			}
			desc = eotg_elven_court_0001_desc_weak
		}
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Confirmed with full Court support — legitimacy and prestige bonus
	option = {
		name = eotg_elven_court_0001_opt_accept_strong
		trigger = { eotg_st_elven_court_strong_support = yes }
		eotg_se_elven_court_confirmed_bonus = yes
		show_as_tooltip = {
			add_prestige = eotg_sv_elven_confirmation_prestige
			add_legitimacy = eotg_sv_elven_confirmation_bonus_legitimacy
		}
		every_vassal = {
			limit = { is_powerful_vassal = yes }
			send_interface_message = {
				title = eotg_elven_court_confirmed_strong_title
				desc = eotg_elven_court_confirmed_strong_desc
				left_icon = root
			}
		}
		ai_chance = { base = 100 }
	}

	# Confirmed with moderate support — small prestige, legitimacy penalty
	option = {
		name = eotg_elven_court_0001_opt_accept_moderate
		trigger = {
			eotg_st_elven_court_moderate_support = yes
			NOT = { eotg_st_elven_court_strong_support = yes }
		}
		eotg_se_elven_court_confirmed_penalty = yes
		show_as_tooltip = {
			add_prestige = 50
			add_legitimacy = -15
		}
		every_vassal = {
			limit = { is_powerful_vassal = yes }
			send_interface_message = {
				title = eotg_elven_court_confirmed_weak_title
				desc = eotg_elven_court_confirmed_weak_desc
				left_icon = root
			}
		}
		ai_chance = { base = 100 }
	}

	# No support — succession crisis, rival claimant backed by the Court
	option = {
		name = eotg_elven_court_0001_opt_crisis
		trigger = {
			NOT = { eotg_st_elven_court_moderate_support = yes }
		}
		add_internal_flag = dangerous
		eotg_se_elven_court_confirmed_penalty = yes
		add_character_modifier = {
			modifier = eotg_elven_court_contested
			years = 10
		}
		# Find a powerful vassal hostile to the ruler to back as rival claimant
		random_vassal = {
			limit = {
				is_powerful_vassal = yes
				opinion = { target = root  value < 0 }
			}
			trigger_event = { id = eotg_elven_court.0004  days = 3 }
		}
		ai_chance = { base = 100 }
	}
}

# Succession crisis — powerful vassal backs a rival claimant
eotg_elven_court.0004 = {
	type = character_event
	title = eotg_elven_court_0004_title
	theme = diplomacy

	desc = eotg_elven_court_0004_desc

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = liege
		animation = personality_rational
	}

	immediate = {
		liege = { save_scope_as = eotg_new_sovereign }
	}

	# Back the rival — add prestige, increase rivalry
	option = {
		name = eotg_elven_court_0004_opt_back_rival
		add_internal_flag = dangerous
		add_prestige = 100
		scope:eotg_new_sovereign = {
			add_legitimacy = -20
			add_character_modifier = {
				modifier = eotg_elven_succession_crisis
				years = 5
			}
		}
		stress_impact = {
			content = medium_stress_impact_gain
		}
		ai_chance = {
			base = 60
			modifier = { add = 20  has_trait = ambitious }
			modifier = { add = -20  has_trait = content }
		}
	}

	# Defer — accept the new ruler despite personal misgivings
	option = {
		name = eotg_elven_court_0004_opt_defer
		add_prestige = -50
		add_opinion = {
			target = scope:eotg_new_sovereign
			modifier = eotg_elven_grudging_acceptance
		}
		ai_chance = {
			base = 40
			modifier = { add = 20  has_trait = content }
			modifier = { add = -20  has_trait = ambitious }
		}
	}
}

# ── Confirmation aftermath ───────────────────────────────────────────────────

# The Eternal Court calls in the debt implied by a unanimous confirmation.
# Fired from eotg_on_yearly_elven_flavor while eotg_elven_court_confirmed holds.
# Elves remember. A mandate given freely is still a mandate given, and the Court
# has been waiting for a convenient moment to say so.
eotg_elven_court.0005 = {
	type = character_event
	title = eotg_elven_court_0005_title
	theme = diplomacy

	desc = eotg_elven_court_0005_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:eotg_court_elder
		animation = personality_bold
		trigger = { exists = scope:eotg_court_elder }
	}

	immediate = {
		# An elder of the Court delivers the reminder in person where possible.
		random_vassal = {
			limit = {
				is_alive = yes
				is_powerful_vassal = yes
			}
			save_scope_as = eotg_court_elder
		}
	}

	# Honor the debt — the Court is repaid in influence, the mandate deepens.
	option = {
		name = eotg_elven_court_0005_opt_honor
		add_legitimacy = 20
		add_prestige = -50
		if = {
			limit = { exists = scope:eotg_court_elder }
			reverse_add_opinion = {
				target = scope:eotg_court_elder
				modifier = eotg_elven_court_debt_honored
			}
		}
		ai_chance = {
			base = 45
			modifier = { add = 20  has_trait = humble }
			modifier = { add = -20  has_trait = arrogant }
		}
	}

	# Remind the Court that a mandate is not a leash.
	option = {
		name = eotg_elven_court_0005_opt_refuse
		add_prestige = 100
		add_legitimacy = -15
		remove_character_modifier = eotg_elven_court_confirmed
		stress_impact = {
			humble = minor_stress_impact_gain
		}
		ai_chance = {
			base = 30
			modifier = { add = 25  has_trait = arrogant }
			modifier = { add = 20  has_trait = ambitious }
			modifier = { add = -20  has_trait = humble }
		}
	}

	# Defer the matter. Elves are patient; so, it turns out, are you.
	option = {
		name = eotg_elven_court_0005_opt_defer
		add_stress = 10
		ai_chance = {
			base = 25
			modifier = { add = 20  has_trait = patient }
		}
	}
}

# Ruling without the Court's blessing — the slow cost of a contested seat.
# Fired from eotg_on_yearly_elven_flavor while eotg_elven_court_contested holds.
eotg_elven_court.0006 = {
	type = character_event
	title = eotg_elven_court_0006_title
	theme = diplomacy

	desc = eotg_elven_court_0006_desc

	left_portrait = {
		character = root
		animation = worry
	}

	# Court the Court — spend coin and time buying the ratification you lacked.
	option = {
		name = eotg_elven_court_0006_opt_court
		remove_short_term_gold = 75
		add_legitimacy = 15
		if = {
			limit = { has_character_modifier = eotg_elven_court_contested }
			remove_character_modifier = eotg_elven_court_contested
		}
		ai_chance = {
			base = 45
			modifier = { add = 25  gold > 200 }
			modifier = { add = -30  gold < 100 }
		}
	}

	# Rule without them. Let the Court learn to follow a sovereign it did not choose.
	option = {
		name = eotg_elven_court_0006_opt_defy
		add_dread = 15
		add_prestige = 50
		stress_impact = {
			content = minor_stress_impact_gain
		}
		ai_chance = {
			base = 30
			modifier = { add = 25  has_trait = stubborn }
			modifier = { add = 20  dread > 30 }
		}
	}

	# Outlast it. The Court measures time in centuries — but so do you.
	option = {
		name = eotg_elven_court_0006_opt_endure
		add_stress = 15
		add_legitimacy = 5
		ai_chance = {
			base = 25
			modifier = { add = 25  has_trait = patient }
		}
	}
}
