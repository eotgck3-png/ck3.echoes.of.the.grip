# -- eotg_augmentation_fracture.txt -------------------------------------------
# Neurofracture transformation event and Neurofractured ongoing events.
#
# eotg_fracture.0001 -- Neural Cascade          (transformation reveal)
# eotg_fracture.002  -- Containment Failure     (random ongoing)
# eotg_fracture.003  -- Lucid Moment            (rare ongoing)
# eotg_fracture.004  -- The Court Massacre      (high-stress trigger)
# eotg_fracture.005  -- What the Heir Saw       (primary heir witness)
# eotg_fracture.006  -- The Warrant             (rebel declaration)
# eotg_fracture.007  -- Dead Reckoning          (rare clarity)
# Phase 3a (docs/specs/cybernetics_v2_phase3.md §3):
# eotg_fracture.008-.015 -- Flicker band pool
# eotg_fracture.016-.024 -- Fracture band pool (.020 is .019's chain stage)
# eotg_fracture.025-.026 -- Storm band pool (with .002, .004, .006, .007, .018)
# eotg_fracture.027      -- The Cascade (terminal, risk >= 95)
# eotg_fracture.028/.029 -- chain stages of .017 / .027
# Phase 3b: .007.f, .026.d and .027's outcomes fire the endgames
# (events/eotg_augmentation_endgame.txt, namespace eotg_aug_end).
#
# Fires from: eotg_trigger_neurofracture scripted effect (0001), called from
#             eotg_on_yearly_aug_overclocked_check and eotg_aug_tier3.002 (c);
#             eotg_on_yearly_aug_neurofractured_check (band lists and the
#             terminal step), via yearly_playable_pulse; .005 from the Heir's
#             Arc story (eotg_story_aug_heir_arc, stage 1).
# Cooldown flags (eotg_flag_nf_event_cooldown, eotg_flag_aug_scramble_cooldown,
# eotg_flag_aug_last_lucid, eotg_flag_aug_terminal_cooldown) are set and
# checked in the on_action only. Event triggers mirror world state only.
# Resource: post-cascade, eotg_fracture_risk is "episode pressure": it drifts
# up yearly, its band (Flicker / Fracture / Storm) selects the pool, and
# options here move it.
#
# cybernetics_v2 Phase 1 (docs/specs/cybernetics_v2_phase1.md §5): victims from
# the weighted picker, named in the desc, flagged eotg_flag_aug_breach_victim
# for 30 days (audit §1.5); stress helpers on every violent option (audit
# §3.3); trait-gated options; dominated-option fixes; scaled gold.
# -----------------------------------------------------------------------------

# -- eotg_fracture.0001 -------------------------------------------------------
# Neural Cascade -- Transformation Event

eotg_fracture.0001 = {
    type = character_event
    title = eotg_fracture.0001.t
    desc = {
        desc = eotg_fracture.0001.desc
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:eotg_victim }
                desc = eotg_fracture.0001.desc_victim
            }
            desc = eotg_fracture.0001.desc_none
        }
    }
    theme = battle

    left_portrait = {
        character = root
        animation = stress
    }
    # Guarded: 60% of cascades hurt no one (vanilla
    # events/activities/activity_system_events.txt:89-92).
    right_portrait = {
        trigger = { exists = scope:eotg_victim }
        character = scope:eotg_victim
        animation = pain
    }

    # 40% that someone is hurt, as before; now picked and named (family 0.05).
    immediate = {
        hidden_effect = {
            add_dread = 40
            random = {
                chance = 40
                eotg_aug_pick_victim_effect = { NAME = eotg_victim  FAMILY_FACTOR = 0.05 }
            }
            if = {
                limit = { exists = scope:eotg_victim }
                scope:eotg_victim = {
                    increase_wounds_effect = { REASON = attacked }
                    add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
                }
            }
        }
    }

    option = {
        name = eotg_fracture.0001.a
        stress_impact = { base = massive_stress_impact_gain }
    }
}


# -- eotg_fracture.002 --------------------------------------------------------
# Containment Failure (renamed from Containment Breach, lore review)
# The Neurofractured character loses control -- court casualties, picked and
# named: wounded 50% (scope:eotg_victim), killed 20% (scope:eotg_victim_dead,
# never the wounded one). Family factor 0.05, or 0.25 in the Storm band: the
# event sits in both the Fracture and the Storm pools, and the picker's
# Storm-band rule (eotg_aug_pick_victim_effect header) follows the band it
# fired in. The parameter is textual, so each pick is an if/else pair.

eotg_fracture.002 = {
    type = character_event
    title = eotg_fracture.002.t
    desc = {
        desc = eotg_fracture.002.desc
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:eotg_victim }
                desc = eotg_fracture.002.desc_wounded
            }
            triggered_desc = {
                trigger = { NOT = { exists = scope:eotg_victim_dead } }
                desc = eotg_fracture.002.desc_none
            }
        }
        triggered_desc = {
            trigger = { exists = scope:eotg_victim_dead }
            desc = eotg_fracture.002.desc_killed
        }
    }
    theme = battle

    left_portrait = {
        character = root
        animation = stress
    }
    # Guarded (vanilla events/activities/activity_system_events.txt:89-92).
    # Shows the wounded victim; falls back to the dead one when only that one
    # exists. scope:eotg_victim_portrait is saved at the end of immediate.
    # Dead-portrait shape: vanilla events/activities/chariot_race_activity/
    # chariot_ongoing_events_jp.txt:2544 (animation = dead, animate_if_dead).
    right_portrait = {
        trigger = { exists = scope:eotg_victim_portrait }
        character = scope:eotg_victim_portrait
        triggered_animation = {
            trigger = { is_alive = no }
            animation = dead
        }
        animation = pain
        animate_if_dead = yes
    }

    immediate = {
        hidden_effect = {
            random = {
                chance = 50
                if = {
                    limit = { eotg_aug_pressure_storm = yes }
                    eotg_aug_pick_victim_effect = { NAME = eotg_victim  FAMILY_FACTOR = 0.25 }
                }
                else = {
                    eotg_aug_pick_victim_effect = { NAME = eotg_victim  FAMILY_FACTOR = 0.05 }
                }
            }
            if = {
                limit = { exists = scope:eotg_victim }
                scope:eotg_victim = {
                    increase_wounds_effect = { REASON = attacked }
                    add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
                }
            }
            random = {
                chance = 20
                if = {
                    limit = { eotg_aug_pressure_storm = yes }
                    eotg_aug_pick_second_victim_effect = {
                        NAME = eotg_victim_dead
                        FAMILY_FACTOR = 0.25
                        EXCLUDE = eotg_victim
                    }
                }
                else = {
                    eotg_aug_pick_second_victim_effect = {
                        NAME = eotg_victim_dead
                        FAMILY_FACTOR = 0.05
                        EXCLUDE = eotg_victim
                    }
                }
            }
            if = {
                limit = { exists = scope:eotg_victim_dead }
                scope:eotg_victim_dead = {
                    death = { death_reason = death_murder  killer = root }
                }
            }
            # .d [sadistic]: pre-pick the extra victim so the option text can
            # name them. After the death above, so the picker's is_alive test
            # excludes the dead victim; EXCLUDE keeps the wounded one out.
            if = {
                limit = {
                    has_trait = sadistic
                    eotg_aug_pressure_storm = yes
                }
                eotg_aug_pick_second_victim_effect = {
                    NAME = eotg_victim_extra
                    FAMILY_FACTOR = 0.25
                    EXCLUDE = eotg_victim
                }
            }
            else_if = {
                limit = { has_trait = sadistic }
                eotg_aug_pick_second_victim_effect = {
                    NAME = eotg_victim_extra
                    FAMILY_FACTOR = 0.05
                    EXCLUDE = eotg_victim
                }
            }
            # Portrait scope: the wounded victim, else the dead one
            if = {
                limit = { exists = scope:eotg_victim }
                scope:eotg_victim = { save_scope_as = eotg_victim_portrait }
            }
            else_if = {
                limit = { exists = scope:eotg_victim_dead }
                scope:eotg_victim_dead = { save_scope_as = eotg_victim_portrait }
            }
        }
    }

    # "No chains"
    option = {
        name = eotg_fracture.002.a
        add_dread = 30
        add_tyranny = 5
        eotg_add_fracture_risk = { AMOUNT = 10 }
        eotg_aug_stress_cruelty_effect = yes
        ai_chance = {
            base = 50
            modifier = { add = 20  has_trait = callous }
            modifier = { add = -20  has_trait = compassionate }
        }
    }

    # "Someone help me"
    option = {
        name = eotg_fracture.002.b
        stress_impact = { base = medium_stress_impact_gain }
        add_character_modifier = { modifier = eotg_mod_withdrawn_from_court  years = 2 }
        eotg_add_fracture_risk = { AMOUNT = -10 }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = compassionate }
            modifier = { add = 10  has_trait = humble }
        }
    }

    # "Execute the survivors" -- only the people hurt in an episode in the last
    # 30 days (eotg_flag_aug_breach_victim), never every wounded courtier
    # (audit §1.5). Hidden when there is no survivor to execute (an option with
    # no target is a dead button; vanilla gates target options on existence).
    # QA round 2: an execution, so vanilla death_execution
    # (game/common/deathreasons/00_event_deaths.txt), and a deliberate killing,
    # so family victims make the ruler a kinslayer through vanilla
    # add_kinslayer_trait_or_nothing_effect (00_secret_effects.txt:433), called
    # before the death as vanilla execute_prisoner_effect does
    # (00_prison_effects.txt). The involuntary episode deaths (immediate above,
    # and every other Containment Failure / Massacre victim) carry no
    # kinslayer: the machine did it, not a decision (QA round 2 M4).
    option = {
        name = eotg_fracture.002.c
        trigger = {
            any_courtier = {
                is_alive = yes
                has_character_flag = eotg_flag_aug_breach_victim
            }
        }
        every_courtier = {
            limit = {
                is_alive = yes
                has_character_flag = eotg_flag_aug_breach_victim
            }
            save_scope_as = eotg_executed
            root = { add_kinslayer_trait_or_nothing_effect = { VICTIM = scope:eotg_executed } }
            death = { death_reason = death_execution  killer = root }
        }
        add_dread = 50
        add_tyranny = 20
        eotg_aug_stress_murder_effect = yes
        ai_chance = {
            base = 20
            modifier = { add = 30  has_trait = callous }
            modifier = { add = 20  has_trait = wrathful }
        }
    }

    # [sadistic] "Again." -- one more courtier hurt (never the first), picked
    # in immediate as scope:eotg_victim_extra so the option text names them
    # (Phase 1 QA). Hidden when no one is left to hurt, as .c is (an option
    # with no target is a dead button).
    # Stress: the wound helper gives sadistic minor relief; the extra minor
    # line brings it to the spec's medium (-15 + -15 = -30, vanilla
    # common/script_values/00_stress_values.txt).
    option = {
        name = eotg_fracture.002.d
        trigger = {
            has_trait = sadistic
            exists = scope:eotg_victim_extra
        }
        trait = sadistic
        if = {
            limit = { exists = scope:eotg_victim_extra }
            scope:eotg_victim_extra = {
                increase_wounds_effect = { REASON = attacked }
                add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
            }
        }
        add_dread = 30
        eotg_add_fracture_risk = { AMOUNT = 15 }
        eotg_aug_stress_wound_effect = yes
        stress_impact = {
            sadistic = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = sadistic }
            modifier = { add = 10  has_trait = wrathful }
        }
    }

    # [compassionate] "Tend them with my own hands."
    # eotg_opinion_aug_grateful_patient (Phase 2), as the spec directs.
    option = {
        name = eotg_fracture.002.e
        trigger = { has_trait = compassionate }
        trait = compassionate
        if = {
            limit = { exists = scope:eotg_victim }
            scope:eotg_victim = {
                add_opinion = { modifier = eotg_opinion_aug_grateful_patient  target = root  years = 10 }
            }
        }
        add_prestige = -50
        eotg_add_fracture_risk = { AMOUNT = -5 }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = compassionate }
            modifier = { add = 10  has_trait = humble }
        }
    }
}

# -- eotg_fracture.005 --------------------------------------------------------
# What the Heir Saw
# The primary heir was present during a fracture episode.
# Caller (Phase 3a): the Heir's Arc, eotg_story_aug_heir_arc stage 1; no
# longer in the random pool. scope:watching_heir is the story's eotg_heir.
# Each option moves the story's eotg_heir_dread (a +1, b -1, c +2, d -1, e 0).

eotg_fracture.005 = {
    type = character_event
    title = eotg_fracture.005.t
    desc = eotg_fracture.005.desc
    theme = family

    trigger = {
        any_owned_story = {
            story_type = eotg_story_aug_heir_arc
            exists = var:eotg_heir
            var:eotg_heir = { is_alive = yes }
        }
    }

    left_portrait = {
        character = root
        animation = stress
    }
    right_portrait = {
        trigger = { exists = scope:watching_heir }
        character = scope:watching_heir
        animation = worry
    }

    immediate = {
        hidden_effect = {
            random_owned_story = {
                limit = { story_type = eotg_story_aug_heir_arc }
                save_scope_as = eotg_heir_story
                var:eotg_heir = { save_scope_as = watching_heir }
            }
        }
    }

    # "They will understand eventually." — heir fears you
    option = {
        name = eotg_fracture.005.a
        scope:eotg_heir_story = { change_variable = { name = eotg_heir_dread  add = 1 } }
        scope:watching_heir = {
            add_opinion = { modifier = eotg_opinion_aug_fear  target = root  years = 10 }
        }
        add_dread = 15
        eotg_add_fracture_risk = { AMOUNT = 5 }
        eotg_aug_stress_cruelty_effect = yes
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = callous }
            modifier = { add = -30  has_trait = compassionate }
        }
    }

    # "Call them in. Explain as much as you can." — heir uneasy but closer
    option = {
        name = eotg_fracture.005.b
        scope:eotg_heir_story = { change_variable = { name = eotg_heir_dread  add = -1 } }
        scope:watching_heir = {
            add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
        }
        stress_impact = { base = medium_stress_impact_gain }
        eotg_add_fracture_risk = { AMOUNT = -5 }
        ai_chance = {
            base = 40
            modifier = { add = 30  has_trait = compassionate }
            modifier = { add = 20  has_trait = humble }
        }
    }

    # "It was necessary. Prepare them." — heir disgusted, may gain callous
    option = {
        name = eotg_fracture.005.c

# -- eotg_fracture.029 --------------------------------------------------------
# It Passes (notification). The Cascade passes: risk set to 60, two people
# near you are hurt (picker, Storm family factor 0.25), +50 dread.
# Chain stage of eotg_fracture.027 (the "it passes" outcome).

eotg_fracture.029 = {
    type = character_event
    title = eotg_fracture.029.t
    desc = {
        desc = eotg_fracture.029.desc
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:eotg_victim_2 }
                desc = eotg_fracture.029.desc_two
            }
            triggered_desc = {
                trigger = { exists = scope:eotg_victim }
                desc = eotg_fracture.029.desc_one
            }
            desc = eotg_fracture.029.desc_none
        }
    }
    theme = mental_health

    left_portrait = {
        character = root
        animation = stress
    }
    right_portrait = {
        trigger = { exists = scope:eotg_victim }
        character = scope:eotg_victim
        animation = pain
    }
    lower_right_portrait = {
        trigger = { exists = scope:eotg_victim_2 }
        character = scope:eotg_victim_2
    }

    immediate = {
        set_variable = { name = eotg_fracture_risk  value = 60 }
        add_dread = 50
        hidden_effect = {
            eotg_aug_pick_victim_effect = { NAME = eotg_victim  FAMILY_FACTOR = 0.25 }
            if = {
                limit = { exists = scope:eotg_victim }
                scope:eotg_victim = {
                    increase_wounds_effect = { REASON = attacked }
                    add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
                }
                eotg_aug_pick_second_victim_effect = {
                    NAME = eotg_victim_2
                    FAMILY_FACTOR = 0.25
                    EXCLUDE = eotg_victim
                }
            }
            if = {
                limit = { exists = scope:eotg_victim_2 }
                scope:eotg_victim_2 = {
                    increase_wounds_effect = { REASON = attacked }
                    add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
                }
            }
        }
    }

    # "Again it passes."
    option = {
        name = eotg_fracture.029.a
        ai_chance = {
            base = 50
            modifier = { add = 20  has_trait = arrogant }
            modifier = { add = 20  has_trait = callous }
        }
    }

    # [humble] "Next time it won't."
    option = {
        name = eotg_fracture.029.b
        trigger = { has_trait = humble }
        trait = humble
        eotg_add_fracture_risk = { AMOUNT = -5 }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = humble }
            modifier = { add = 10  has_trait = patient }
        }
    }
}
