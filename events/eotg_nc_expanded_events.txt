# eotg_nc_flavor.0001 — Senate Debate Turns Hostile
# Senators openly attack presidential leadership during debate.
eotg_nc_flavor.0001 = {
	type = character_event
	title = eotg_nc_flavor_0001_title
	desc = eotg_nc_flavor_0001_desc
	theme = diplomacy

	left_portrait = { character = root  animation = stress }

	trigger = {
		eotg_st_is_president = yes
		eotg_st_nc_senate_hostile = yes
		NOT = { has_character_flag = eotg_nc_0001_recent }
	}

	option = {
		name = eotg_nc_flavor_0001_opt_negotiate
		add_diplomacy_skill = 1
		every_vassal = {
			limit = { eotg_st_is_nc_senator = yes }
			add_opinion = { target = root  modifier = eotg_pmc_loyalty_restored }
		}
		remove_character_modifier = eotg_nc_senate_gridlock
		add_character_flag = { flag = eotg_nc_0001_recent  years = 3 }
		ai_chance = { base = 45  modifier = { add = 15  has_trait = gregarious } }
	}
	option = {
		name = eotg_nc_flavor_0001_opt_shame
		add_dread = 10
