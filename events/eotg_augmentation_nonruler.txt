# ── eotg_augmentation_nonruler.txt ───────────────────────────────────────────
# The non-ruler lifecycle: docs/specs/cybernetics_v2_phase6.md.
# Augmented knights and courtiers progress, break, and reach their liege.
#
# eotg_aug_nr.001 — The Champion Volunteers
# eotg_aug_nr.002 — Your Champion's New Edge
# eotg_aug_nr.003 — Something Is Wrong With Them
# eotg_aug_nr.004 — The Champion's Mistake
# eotg_aug_nr.005 — The Familiar Change
# eotg_aug_nr.006 — The Broken Champion       (the non-ruler cascade)
#
# Fires from: eotg_on_yearly_aug_nonruler_check (via random_yearly_everyone_pulse)
#   for .001-.005; eotg_aug_nr_cascade_effect for .006.
# Root is the LIEGE; scope:eotg_champion is the non-ruler. The liege cooldown
#   eotg_flag_aug_nr_cooldown is set and checked in the on_action only
#   (lesson 5): these triggers are the world-state guard "the champion still
#   exists, lives, and is still root's courtier or knight" (QA round 2 M9:
#   the 1-30 day delay lets them leave court, and no option may jail or pay
#   a stranger).
# Resource: every event moves the champion's own eotg_fracture_risk or
#   changes their tier (nr.001 install, nr.006.b removal).
# Victims come from the picker run in the liege's scope (root = liege, so its
#   root exclusions are right), with the champion excluded through the
#   second-victim picker's EXCLUDE.
# Hardware-state options (docs/specs/cybernetics_v2_tier_options.md): nr.003
#   f (Enhanced+) and g (Seamless). Residue pacing (lore ruling, 2026-10-05):
#   eotg_aug_residue only runs 0-3 and nr.003 recurs, so the Seamless row g,
#   which moves the champion's risk, does not also lower the ruler's residue
#   (Flicker, end.042, and the desc_residue_high lines stay reachable).
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_aug_nr


# ── eotg_aug_nr.001 ───────────────────────────────────────────────────────────
# The Champion Volunteers (NR-01). An unaugmented knight asks for the
# implant. Caller: eotg_on_yearly_aug_nonruler_check (step 1).

eotg_aug_nr.001 = {
    type = character_event
    title = eotg_aug_nr.001.t
    desc = eotg_aug_nr.001.desc
    theme = martial_chivalry_focus

    trigger = {
        exists = scope:eotg_champion
        scope:eotg_champion = {
            is_alive = yes
            OR = {    # QA round 2 M9: still at root's court
                is_courtier_of = root
                is_knight_of = root
            }
            eotg_is_augmented_any = no
        }
    }

    left_portrait = {
        character = root
        animation = thinking
    }
    right_portrait = {
        trigger = { exists = scope:eotg_champion }
        character = scope:eotg_champion
        animation = personality_bold
    }

    # "Granted."
    option = {
        name = eotg_aug_nr.001.a
        trigger = { gold >= minor_gold_value }
        remove_short_term_gold = minor_gold_value
        scope:eotg_champion = {
            eotg_aug_initiate_effect = yes
            eotg_add_fracture_risk = { AMOUNT = 0 }
        }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = ambitious }
            modifier = { add = 10  has_trait = brave }
        }
    }

    # "Denied."
    option = {
        name = eotg_aug_nr.001.b
        scope:eotg_champion = {
            add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = zealous }
            modifier = { add = 10  has_trait = content }
        }
    }

    # "Granted, if they pay for it."
    option = {
        name = eotg_aug_nr.001.c
        scope:eotg_champion = {
            eotg_aug_initiate_effect = yes
            eotg_add_fracture_risk = { AMOUNT = 0 }
            add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = greedy }
            modifier = { add = 10  has_trait = arbitrary }
        }
    }

    # [generous] "At my expense, and properly."
    option = {
        name = eotg_aug_nr.001.d
        trigger = {
            has_trait = generous
            gold >= medium_gold_value
        }
        trait = generous
        remove_short_term_gold = medium_gold_value
        scope:eotg_champion = {
            eotg_aug_initiate_effect = yes
            eotg_add_fracture_risk = { AMOUNT = 0 }
            add_character_modifier = { modifier = eotg_mod_implant_calibrated  years = 3 }
            add_opinion = { modifier = eotg_opinion_aug_grateful_patient  target = root  years = 10 }
        }
        stress_impact = {
            generous = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = generous }
            modifier = { add = 10  has_trait = compassionate }
        }
    }

    # [paranoid] "Why do you want to be stronger than me?"
    option = {
        name = eotg_aug_nr.001.e
        trigger = { has_trait = paranoid }
        trait = paranoid
        scope:eotg_champion = {
            add_opinion = { modifier = eotg_opinion_aug_fear  target = root  years = 10 }
        }
        stress_impact = {
            paranoid = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = paranoid }
            modifier = { add = 10  has_trait = vengeful }
        }
    }
}

# ── eotg_aug_nr.004 ───────────────────────────────────────────────────────────
# The Champion's Mistake (NR-04). Victim: the second-victim picker in the
# liege's scope, family factor 0.05, EXCLUDE = eotg_champion; wounded
# (attacked) and flagged. No victim (desc_none, "a door, a bulkhead,
# themselves"): the champion gets wounded_1 if unwounded.
# a-d carry the wound helper for the liege (their champion, their
# responsibility). e: the victim is killed by the champion.
# Imprisonment: vanilla imprison_character_effect (00_prison_effects.txt:1644;
# QA round 2 M5), which picks house arrest or the dungeon and charges
# tyranny and family opinion like every other arrest in the system.
# Caller: eotg_on_yearly_aug_nonruler_check (step 5).

eotg_aug_nr.004 = {
    type = character_event
    title = eotg_aug_nr.004.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:eotg_victim }
                desc = eotg_aug_nr.004.desc
            }
            desc = eotg_aug_nr.004.desc_none
        }
    }
    theme = dread

    trigger = {
        exists = scope:eotg_champion
        scope:eotg_champion = {
            is_alive = yes
            is_imprisoned = no
            OR = {    # QA round 2 M9: still at root's court
                is_courtier_of = root
                is_knight_of = root
            }
        }
    }

    left_portrait = {
        character = root
        animation = shock
    }
    right_portrait = {
        trigger = { exists = scope:eotg_champion }
        character = scope:eotg_champion
        animation = rage
    }
    lower_right_portrait = {
        trigger = { exists = scope:eotg_victim }
        character = scope:eotg_victim
        animate_if_dead = yes
    }

    immediate = {
        hidden_effect = {
            eotg_aug_pick_second_victim_effect = {
                NAME = eotg_victim
                FAMILY_FACTOR = 0.05
                EXCLUDE = eotg_champion
            }
        }
        if = {
            limit = { exists = scope:eotg_victim }
            scope:eotg_victim = {
                increase_wounds_effect = { REASON = attacked }
                add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
            }
        }
        else = {
            scope:eotg_champion = {
                if = {
                    limit = { NOT = { has_trait = wounded } }
                    add_trait = wounded_1
                }
            }
        }
    }

    # "Imprison them."
    option = {
        name = eotg_aug_nr.004.a
        # Ungated: the event trigger guarantees the champion is free.
        imprison_character_effect = {
            TARGET = scope:eotg_champion
            IMPRISONER = root
        }
        scope:eotg_champion = {
            eotg_add_fracture_risk = { AMOUNT = -5 }
        }
        eotg_aug_stress_tyranny_effect = yes
        eotg_aug_stress_wound_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = just }
            modifier = { add = 10  has_trait = paranoid }
        }
    }

    # "Pay off the victim."
    option = {
        name = eotg_aug_nr.004.b
        # The wound in immediate can kill; no paying off the dead.
        trigger = {
            exists = scope:eotg_victim
            scope:eotg_victim = { is_alive = yes }
            gold >= minor_gold_value
        }
        remove_short_term_gold = minor_gold_value
        scope:eotg_champion = {
            eotg_add_fracture_risk = { AMOUNT = 5 }
        }
        eotg_aug_stress_wound_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = greedy }
            modifier = { add = 10  has_trait = deceitful }
        }
    }

    # "Restrain them, and have them treated."
    option = {
        name = eotg_aug_nr.004.c
        trigger = { gold >= medium_gold_value }
        remove_short_term_gold = medium_gold_value
        scope:eotg_champion = {
            eotg_add_fracture_risk = { AMOUNT = -10 }
        }
        eotg_aug_stress_wound_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = compassionate }
            modifier = { add = 10  has_trait = diligent }
        }
    }

    # [just] "A trial." Then as a.
    option = {
        name = eotg_aug_nr.004.d
        trigger = {
            has_trait = just
            scope:eotg_champion = { is_imprisoned = no }
        }
        trait = just
        add_prestige = 50
        imprison_character_effect = {
            TARGET = scope:eotg_champion
            IMPRISONER = root
        }
        scope:eotg_champion = {
            eotg_add_fracture_risk = { AMOUNT = -5 }
        }
        eotg_aug_stress_wound_effect = yes
        stress_impact = {
            just = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = just }
            modifier = { add = 10  has_trait = honest }
        }
    }

    # [sadistic] "Let them finish."
    option = {
        name = eotg_aug_nr.004.e
        # Killing: family must be impossible here, not merely unlikely
        trigger = {
            has_trait = sadistic
            exists = scope:eotg_victim
            scope:eotg_victim = {
                is_alive = yes
                NOT = { is_close_family_of = root }
                NOT = { is_spouse_of = root }
            }
        }
        trait = sadistic
        scope:eotg_victim = {
            death = {
                death_reason = death_murder
                killer = scope:eotg_champion
            }
        }
        add_dread = 20
        scope:eotg_champion = {
            eotg_add_fracture_risk = { AMOUNT = 5 }
        }
        eotg_aug_stress_murder_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = sadistic }
            modifier = { add = 10  has_trait = callous }
        }
    }

    # [forgiving] "They did not choose this. Let it go." - no prison, no
    # treatment: the champion's risk climbs (+5, as b); a living victim
    # resents it. Omits eotg_aug_stress_wound_effect, whose forgiving gain
    # would cancel this option's relief (conformance ruling M10). Inline
    # stress after vanilla court.0111.b (events/court_events/
    # court_events_general.txt:1246); vengeful dropped (forgiving's
    # opposite). The event trigger guarantees the champion is free.
    # Spec cybernetics_v2_trait_depth §5.1 #8.
    option = {
        name = eotg_aug_nr.004.f
        trigger = { has_trait = forgiving }
        trait = forgiving
        scope:eotg_champion = {
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
            eotg_add_fracture_risk = { AMOUNT = 5 }
        }
        if = {
            limit = {
                exists = scope:eotg_victim
                scope:eotg_victim = { is_alive = yes }
            }
            scope:eotg_victim = {
                add_opinion = { modifier = eotg_opinion_aug_disgust  target = root  years = 5 }
            }
        }
        stress_impact = {
            forgiving = minor_stress_impact_loss
            just = minor_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = forgiving }
            modifier = { add = 10  has_trait = compassionate }
        }
    }
}
