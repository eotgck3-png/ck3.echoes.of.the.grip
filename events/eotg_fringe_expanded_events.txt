# The Grip is slipping. Captains grow bold.
eotg_fringe.0014 = {
	type = character_event
	title = eotg_fringe_0014_title
	theme = intrigue

	desc = eotg_fringe_0014_desc

	left_portrait = {
		character = root
		animation = stress
	}

	trigger = {
		eotg_st_is_fringe_overlord = yes
		eotg_st_fringe_grip_low = yes
		realm_size >= 8
	}

	# Tighten discipline — dread gain, grip gain, resentment.
	option = {
		name = eotg_fringe_0014_opt_discipline
		add_dread = 10
		eotg_se_fringe_gain_grip = { AMOUNT = 10 }
		add_character_modifier = {
			modifier = eotg_fringe_cracking_authority
			years = 2
		}
		every_vassal = {
			limit = { eotg_st_is_fringe_captain = yes }
			add_opinion = {
				target = root
				modifier = eotg_fringe_fear_rule_resentment
			}
		}
		stress_impact = {
			base = minor_stress_impact_gain
			compassionate = medium_stress_impact_gain
		}
		ai_chance = {
			base = 40
			modifier = { add = 20  dread > 20 }
		}
	}

	# Let them squabble — Grip falls, captains direct anger at each other.
	option = {
		name = eotg_fringe_0014_opt_squabble
		eotg_se_fringe_lose_grip = { AMOUNT = 8 }
		add_prestige = -50
		random = {
			chance = 40
			trigger_event = { id = eotg_fringe.0007  days = 14 }
		}
		ai_chance = {
			base = 30
			modifier = { add = -20  eotg_st_fringe_grip_critical = yes }
		}
	}

	# Buy their loyalty — gold for opinion and grip.
	option = {
		name = eotg_fringe_0014_opt_buy
		remove_short_term_gold = 150
		eotg_se_fringe_gain_grip = { AMOUNT = 5 }
		add_character_modifier = {
			modifier = eotg_fringe_bought_loyalty
			years = 2
		}
		every_vassal = {
			limit = { eotg_st_is_fringe_captain = yes }
			add_opinion = {
				target = root
				modifier = eotg_fringe_spoils_distributed
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 30  gold > 300 }
		}
	}
}
