# ── eotg_augmentation_procedures.txt ─────────────────────────────────────────
# Procedures, providers and repairs, and the former marker's aftermath
# (docs/specs/cybernetics_v2_procedures.md §4.4, §4.10).
#
# eotg_aug_proc.001 — Who Takes It Out?    (removal provider)
# eotg_aug_proc.002 — After the Procedure  (outcome report)
# eotg_aug_proc.003 — Spare Parts          (repair offer)
# eotg_aug_proc.004 — A Part Replaced      (Seamless repair; lore N1)
# eotg_aug_proc.020 — Phantom Static       (after leaving the system, G3)
#
# Fires from:
#   eotg_decision_partial_removal, eotg_decision_overclock_regression,
#     eotg_decision_remove_implants -> .001;
#   eotg_aug_procedure_effect (common/scripted_effects/
#     eotg_augmentation_effects.txt) -> .002;
#   eotg_on_trait_gained_aug_repair (on_trait_gained) -> .003 / .004;
#   eotg_on_yearly_aug_former_check (yearly_playable_pulse) -> .020.
# Cooldowns live in those on_actions and decisions only
# (eotg_flag_aug_repair_cooldown, eotg_flag_aug_phantom_done); no event
# trigger here reads a cooldown flag (index §1 rule 5).
# Resource: eotg_fracture_risk and tier (track XP, moved only by
# eotg_aug_set_integration_effect). Every paid option of .001 changes tier;
# every paid option of .003 moves risk inside the shared roll
# (eotg_aug_procedure_roll_effect). .002 and .004 are chain stages (.004
# reads the Seamless residue, eotg_aug_residue). .020 reads the former marker
# (eotg_aug_former) and c/d lead into an install. All risk moves are hidden;
# no text quantifies them.
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_aug_proc


# ── eotg_aug_proc.001 ─────────────────────────────────────────────────────────
# Who Takes It Out? The removal the decision booked
# (var:eotg_aug_removal_kind, 30 days) is performed by the provider chosen
# here. Paid options: pay (eotg_aug_pay_procedure_effect at
# eotg_aug_removal_base_value x the provider), perform
# (eotg_aug_removal_perform_effect: the decision's old body, with its own
# stress, so no surgery helper here), then the shared roll, hidden.
# Callers: eotg_decision_partial_removal, eotg_decision_overclock_regression,
# eotg_decision_remove_implants (trigger_event in the decision effect, as Seek
# fires init.018).
# Procedures QA: every paid option re-checks the booking (has_variable
# eotg_aug_removal_kind), so a second queued copy cannot perform an unbooked
# removal; b needs a court physician other than root
# (eotg_aug_has_court_physician), so a physician ruler without one takes e.

eotg_aug_proc.001 = {
    type = character_event
    title = eotg_aug_proc.001.t
    desc = {
        desc = eotg_aug_proc.001.desc
        first_valid = {
            triggered_desc = {
                trigger = { var:eotg_aug_removal_kind ?= flag:partial }
                desc = eotg_aug_proc.001.desc_partial
            }
            triggered_desc = {
                trigger = { var:eotg_aug_removal_kind ?= flag:downgrade }
                desc = eotg_aug_proc.001.desc_downgrade
            }
            triggered_desc = {
                trigger = { var:eotg_aug_removal_kind ?= flag:full }
                desc = eotg_aug_proc.001.desc_full
            }
        }
        triggered_desc = {
            trigger = { eotg_aug_has_surgeon_for = { PATIENT = root } }
            desc = eotg_aug_proc.001.desc_physician
        }
    }
    theme = medicine

    # World-state guard: the booking, and the tier it was booked at
    trigger = {
        has_variable = eotg_aug_removal_kind
        OR = {
            AND = {
                var:eotg_aug_removal_kind = flag:partial
                eotg_is_aug_tier2 = yes
            }
            AND = {
                var:eotg_aug_removal_kind = flag:downgrade
                eotg_is_aug_tier3 = yes
            }
            AND = {
                var:eotg_aug_removal_kind = flag:full
                eotg_is_aug_tier1 = yes
            }
        }
    }

    left_portrait = {
        character = root
        animation = worry
    }
    # Guarded (vanilla events/activities/activity_system_events.txt:89-92)
    right_portrait = {
        trigger = {
            exists = scope:eotg_proc_surgeon
            NOT = { scope:eotg_proc_surgeon = root }
        }
        character = scope:eotg_proc_surgeon
        animation = personality_rational
    }

    immediate = {
        hidden_effect = {
            eotg_aug_save_surgeon_effect = yes
        }
    }

    # "The clinic."
    option = {
        name = eotg_aug_proc.001.a
        show_as_unavailable = { eotg_aug_clinic_open = no }
        trigger = {
            # Realm spec §3.2: sanctioned clinics close under a Ban
            custom_description = {
                text = eotg_aug_clinic_closed_tt
                eotg_aug_clinic_open = yes
            }
            has_variable = eotg_aug_removal_kind
            eotg_aug_can_afford_procedure = { BASE = eotg_aug_removal_base_value  MULT = 1 }
        }
        eotg_aug_pay_procedure_effect = { BASE = eotg_aug_removal_base_value  MULT = 1 }
        eotg_aug_removal_perform_effect = yes
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = clinic  PROCEDURE = removal  SURGEON = root }
        }
        custom_tooltip = eotg_aug_proc.001.tt
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = diligent }
            modifier = { add = 10  has_trait = craven }
            modifier = { add = -20  has_trait = greedy }
        }
    }

    # (a surgeon other than root: a court physician or a technician) "My
    # own physician." / "My implant technician."
    option = {
        name = eotg_aug_proc.001.b
        # Realm spec §3.2: the technician (own or borrowed) is offered here
        name = {
            trigger = { eotg_aug_surgeon_is_technician = yes }
            text = eotg_aug_opt_technician
        }
        trigger = {
            has_variable = eotg_aug_removal_kind
            eotg_aug_has_surgeon_for = { PATIENT = root }
            eotg_aug_can_afford_procedure = { BASE = eotg_aug_removal_base_value  MULT = eotg_aug_price_mult_physician }
        }
        eotg_aug_pay_procedure_effect = { BASE = eotg_aug_removal_base_value  MULT = eotg_aug_price_mult_physician }
        eotg_aug_removal_perform_effect = yes
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = physician  PROCEDURE = removal  SURGEON = scope:eotg_proc_surgeon }
        }
        custom_tooltip = eotg_aug_proc.001.tt
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = trusting }
            modifier = { add = 10  has_trait = diligent }
        }
    }

    # "Someone cheaper."
    option = {
        name = eotg_aug_proc.001.c
        trigger = {
            has_variable = eotg_aug_removal_kind
            eotg_aug_can_afford_procedure = { BASE = eotg_aug_removal_base_value  MULT = eotg_aug_price_mult_backstreet_removal }
        }
        eotg_aug_pay_procedure_effect = { BASE = eotg_aug_removal_base_value  MULT = eotg_aug_price_mult_backstreet_removal }
        eotg_aug_removal_perform_effect = yes
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = backstreet  PROCEDURE = removal  SURGEON = root }
        }
        custom_tooltip = eotg_aug_proc.001.tt
        ai_chance = {
            base = 20
            modifier = { add = 20  has_trait = greedy }
            modifier = { add = 10  has_trait = deceitful }
            modifier = { add = -20  has_trait = craven }
        }
    }

    # "Not yet." Ungated. The booking lapses (after).
    option = {
        name = eotg_aug_proc.001.d
        stress_impact = {
            ambitious = minor_stress_impact_loss
        }
        ai_chance = {
            base = 10
            modifier = { add = 20  has_trait = ambitious }
            modifier = { add = 10  has_trait = stubborn }
        }
    }

    # [lifestyle_physician] "I'll guide their hands myself." Self-surgery:
    # the patient is the surgeon (eotg_aug_proc_bad_factor x1.5).
    option = {
        name = eotg_aug_proc.001.e
        trigger = {
            has_variable = eotg_aug_removal_kind
            has_trait = lifestyle_physician
            NOT = { has_trait = blind }
            eotg_aug_can_afford_procedure = { BASE = eotg_aug_removal_base_value  MULT = eotg_aug_price_mult_self }
        }
        trait = lifestyle_physician
        eotg_aug_pay_procedure_effect = { BASE = eotg_aug_removal_base_value  MULT = eotg_aug_price_mult_self }
        eotg_aug_removal_perform_effect = yes
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = self  PROCEDURE = removal  SURGEON = root }
        }
        custom_tooltip = eotg_aug_proc.001.tt
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = lifestyle_physician }
            modifier = { add = 10  has_trait = arrogant }
        }
    }

    # [zealous] (full removal) "Every piece. I want none of it left." The
    # clinic at full price; faith-neutral (index §5 deferral note).
    option = {
        name = eotg_aug_proc.001.f
        show_as_unavailable = { eotg_aug_clinic_open = no }
        trigger = {
            # Realm spec §3.2: sanctioned clinics close under a Ban
            custom_description = {
                text = eotg_aug_clinic_closed_tt
                eotg_aug_clinic_open = yes
            }
            has_variable = eotg_aug_removal_kind
            has_trait = zealous
            var:eotg_aug_removal_kind ?= flag:full
            eotg_aug_can_afford_procedure = { BASE = eotg_aug_removal_base_value  MULT = 1 }
        }
        trait = zealous
        eotg_aug_pay_procedure_effect = { BASE = eotg_aug_removal_base_value  MULT = 1 }
        eotg_aug_removal_perform_effect = yes
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = clinic  PROCEDURE = removal  SURGEON = root }
        }
        custom_tooltip = eotg_aug_proc.001.tt
        add_piety = 50
        stress_impact = {
            zealous = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = zealous }
            modifier = { add = 10  has_trait = humble }
        }
    }

    # The booking is spent, whichever option is taken
    after = {
        remove_variable = eotg_aug_removal_kind
    }
}


# ── eotg_aug_proc.002 ─────────────────────────────────────────────────────────

# applied here, in immediate, so nothing lands before this explains it.
# A hidden flaw reads as clean (spec §6.3, thread T6).
# Chain stage: no world-state guard beyond being alive (the patient may have
# left the system).
# Caller: eotg_aug_procedure_effect, 14-45 days (also reached from the
# interactions: Demand Implant Removal, Salvage Implants, the fault repairs).

                desc = eotg_aug_proc.002.desc_salvaged

                trigger = { scope:eotg_proc_outcome ?= flag:blind }

            desc = eotg_aug_proc.outcome_clean

    theme = recovery

    trigger = { is_alive = yes }

    left_portrait = {
        character = root
        triggered_animation = {

    # "It is done." / "It will have to do." Two names for one option
    # (vanilla events/health_events.txt:1141-1148, `name = { trigger text }`).

        name = eotg_aug_proc.002.a
        name = {
            trigger = {
                OR = {
                    scope:eotg_proc_outcome ?= flag:infection
                    scope:eotg_proc_outcome ?= flag:rejection
                    scope:eotg_proc_outcome ?= flag:discovered
                    scope:eotg_proc_outcome ?= flag:maimed
                    scope:eotg_proc_outcome ?= flag:one_eyed
                    scope:eotg_proc_outcome ?= flag:blind

            text = eotg_aug_proc.002.a_grim

            base = 30
            modifier = { add = 10  has_trait = content }
            modifier = { add = 10  has_trait = lazy }

    # (infection or fragments) "Fetch a physician." As init.011.b: free with

        name = eotg_aug_proc.002.b
        trigger = {
            OR = {
                scope:eotg_proc_outcome ?= flag:infection

            limit = { eotg_has_physician_access = no }
            remove_short_term_gold = minor_gold_value
        }
        if = {
            limit = { has_character_modifier = eotg_mod_aug_infection }
            remove_character_modifier = eotg_mod_aug_infection
        }
        if = {
            limit = { has_character_modifier = eotg_mod_aug_fragments }
            remove_character_modifier = eotg_mod_aug_fragments
        }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = diligent }
            modifier = { add = 20  has_trait = craven }
            modifier = { add = -20  has_trait = greedy }
        }
    }

    # (infection, still augmented) "Sweat it out." As init.011.c; the risk
    # move is hidden, and gated on the system so a former never regains risk.
    option = {
        name = eotg_aug_proc.002.c
        trigger = {
            scope:eotg_proc_outcome ?= flag:infection
            eotg_is_augmented_any = yes
        }
        hidden_effect = {
            eotg_add_fracture_risk = { AMOUNT = 5 }
        }
        random = {
            chance = 25
            add_trait = ill
        }
        eotg_aug_stress_neglect_effect = yes
        ai_chance = {
            base = 20
            modifier = { add = 20  has_trait = stubborn }
            modifier = { add = 20  has_trait = greedy }
        }
    }
}
