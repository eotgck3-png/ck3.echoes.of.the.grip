# ── eotg_augmentation_activities.txt ─────────────────────────────────────────
# Augmented people at vanilla activities (docs/specs/cybernetics_v2_realm.md
# §4.7-4.8, G6).
#
# eotg_aug_act.001 — The Rules of the Field  (tournament host; AI hosts too)
# eotg_aug_act.002 — Past Spec               (augmented player contestant)
# eotg_aug_act.003 — Foul Play?              (player tournament host)
# eotg_aug_act.004 — A Hum at the Table      (feast; player)
# eotg_aug_act.005 — Quarry in the Overlay   (hunt; player)
# eotg_aug_act.006 — At the Holy Site        (pilgrimage; player)
#
# Fires from (common/on_action/eotg_augmentation_on_actions.txt; each custom
# on_action is a child of vanilla activity hooks, extended additively):
#   eotg_on_tournament_opening_aug (tournament_opening_on_action) -> .001;
#   eotg_on_tournament_contest_aug (the seven contest random pulses) -> .002;
#   eotg_on_tournament_active_aug (tournament_active_state_pulse) -> .003;
#   eotg_on_feast_aug (feast_default_event_selection) -> .004;
#   eotg_on_hunt_aug (hunt_random_pulse) -> .005;
#   eotg_on_pilgrimage_aug (pilgrimage_destination_events) -> .006.
# Cooldown authority lives in those on_actions (character flags, or a
# variable on the activity); no event trigger here reads them.
# Tournaments need Tours & Tournaments (the vanilla hooks never fire without
# it); the feast, hunt and pilgrimage events are the fallback (lesson 12).
# Resource: eotg_fracture_risk and tier. Every event moves risk in an option
# or reads the band / tier in a desc or branch trigger (spec §2). All risk
# moves are hidden; no text quantifies them.
# Register (lore R5-R7): the field, the bout, the arena, the marshal; the
# quarry and the trail; the holy site, faith-neutral; physical evidence only.
# Hardware-state options (docs/specs/cybernetics_v2_tier_options.md): the
# Seamless rows act.001.f and act.003.g. Residue pacing (lore ruling,
# 2026-10-05): eotg_aug_residue only runs 0-3 and these hosts recur, so a
# Seamless row that moves another character's risk (act.001.f, the
# entrants) does not also lower the ruler's residue, and a residue-only row
# (act.003.g) is hidden once none is left; otherwise a few clicks would
# drain it and Flicker (end.042) and the end.040/.041 desc_residue_high
# lines could never be reached.
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_aug_act


# ── eotg_aug_act.001 ──────────────────────────────────────────────────────────
# The Rules of the Field. The marshals ask the host about entrants with
# hardware. A chosen target (index rule 7 note): the most visible entrant,
# named in loc. "All entrants" = every attending eotg_aug_tournament_entrant.
# Caller: eotg_on_tournament_opening_aug, 1-3 days (scope:activity saved).

eotg_aug_act.001 = {
    type = activity_event
    window = widget_activity_locale_fullscreen_event
    title = eotg_aug_act.001.t
    desc = {
        desc = eotg_aug_act.001.desc
        triggered_desc = {
            trigger = {
                involved_activity = {
                    any_attending_character = {
                        count >= 2
                        eotg_aug_tournament_entrant = yes
                    }
                }
            }
            desc = eotg_aug_act.001.desc_many
        }
        first_valid = {
            triggered_desc = {
                trigger = { scope:eotg_aug_entrant = { has_trait = eotg_neurofractured } }
                desc = eotg_aug_act.001.desc_entrant_nf
            }
            triggered_desc = {
                trigger = { scope:eotg_aug_entrant = { eotg_is_aug_tier3 = yes } }
                desc = eotg_aug_act.001.desc_entrant_oc
            }
        }
        # Lore R1: only an independent ruler reads "your law"
        triggered_desc = {
            trigger = {
                eotg_aug_under_policy = { LAW = ban }
                is_independent_ruler = yes
            }
            desc = eotg_aug_act.001.desc_ban
        }
        triggered_desc = {
            trigger = {
                eotg_aug_under_policy = { LAW = ban }
                is_independent_ruler = no
            }
            desc = eotg_aug_act.001.desc_ban_v
        }
    }
    theme = tournament_grounds

    trigger = {
        exists = scope:activity
        involved_activity ?= {
            any_attending_character = { eotg_aug_tournament_entrant = yes }
        }
    }

    left_portrait = {
        character = root
        animation = thinking
    }
    right_portrait = {
        trigger = { exists = scope:eotg_aug_entrant }
        character = scope:eotg_aug_entrant
        animation = personality_bold
    }

    immediate = {
        involved_activity = {
            random_attending_character = {
                limit = { eotg_aug_tournament_entrant = yes }
                weight = {
                    base = 1
                    modifier = {
                        add = 2
                        eotg_is_aug_tier3 = yes
                    }
                }
                save_scope_as = eotg_aug_entrant
            }
        }
    }

    # "Let them compete." They will run hot.
    option = {
        name = eotg_aug_act.001.a
        involved_activity = {
            every_attending_character = {
                limit = { eotg_aug_tournament_entrant = yes }
                hidden_effect = { eotg_add_fracture_risk = { AMOUNT = 3 } }
            }
            every_attending_character = {
                limit = {
                    has_trait = zealous
                    NOT = { this = root }
                }
                add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
            }
        }
        scope:eotg_aug_entrant ?= {
            add_opinion = { modifier = eotg_opinion_aug_admiration  target = root }
        }
        if = {
            limit = { eotg_aug_under_policy = { LAW = ban } }
            eotg_aug_stress_lie_effect = yes
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = cynical }
            modifier = { add = 10  has_trait = just }
            modifier = { add = -20  has_trait = zealous }
        }
    }

    # "Bar them from the field."
    option = {
        name = eotg_aug_act.001.b
        involved_activity = {
            every_attending_character = {
                limit = { eotg_aug_tournament_entrant = yes }
                eotg_aug_act_withdraw_effect = { HOST = root }
            }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = zealous }
            modifier = { add = 10  has_trait = just }
            modifier = { add = -10  has_trait = cynical }
        }
    }

    # "No powered output on the field." A handicap (no-op for anyone not
    # competing).
    option = {
        name = eotg_aug_act.001.c
        involved_activity = {
            every_attending_character = {
                limit = { eotg_aug_tournament_entrant = yes }
                activity_tournament_change_contestant_score_effect = { SCORE = decrease_minor }
            }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = diligent }
            modifier = { add = 10  has_trait = content }
        }
    }

    # [cynical] "Make them the main event."
    option = {
        name = eotg_aug_act.001.d
        trigger = { has_trait = cynical }
        trait = cynical
        involved_activity = {
            every_attending_character = {
                limit = { eotg_aug_tournament_entrant = yes }
                hidden_effect = { eotg_add_fracture_risk = { AMOUNT = 5 } }
            }
            every_attending_character = {
                limit = {
                    has_trait = zealous
                    NOT = { this = root }
                }
                add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
            }
        }
        add_prestige = minor_prestige_value
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = cynical }
            modifier = { add = 10  has_trait = arrogant }
        }
    }

    # [zealous] "No one fights with hardware on my field."
    option = {
        name = eotg_aug_act.001.e
        trigger = { has_trait = zealous }
        trait = zealous
        involved_activity = {
            every_attending_character = {
                limit = { eotg_aug_tournament_entrant = yes }
                eotg_aug_act_withdraw_effect = { HOST = root }
            }
        }
        add_piety = minor_piety_value
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = zealous }
            modifier = { add = 10  has_trait = stubborn }
        }
    }

    # [Seamless] "Units above tolerance are barred. The rest may ride."
    # Tier options spec row 1: a ruling by rating, not by faith (no zealous
    # unease loop). Overclocked and Neurofractured entrants withdraw; the rest
    # compete and run hot. Its signature move is the entrants' risk; it does
    # not also lower the ruler's residue (residue pacing ruling, 2026-10-05:
    # the host recurs and residue only runs 0-3).
    option = {
        name = eotg_aug_act.001.f
        add_internal_flag = special    # W3: gated by a rare state (Seamless)
        trigger = { has_trait = eotg_total_integration }
        trait = eotg_total_integration
        involved_activity = {
            every_attending_character = {
                limit = {
                    eotg_aug_tournament_entrant = yes
                    OR = {
                        eotg_is_aug_tier3 = yes
                        has_trait = eotg_neurofractured
                    }
                }
                eotg_aug_act_withdraw_effect = { HOST = root }
            }
            every_attending_character = {
                limit = {
                    eotg_aug_tournament_entrant = yes
                    OR = {
                        eotg_is_aug_tier1 = yes
                        eotg_is_aug_tier2 = yes
                    }
                }
                hidden_effect = { eotg_add_fracture_risk = { AMOUNT = 3 } }
            }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = just }
            modifier = { add = 10  has_trait = diligent }
        }
    }
}


# ── eotg_aug_act.002 ──────────────────────────────────────────────────────────
# Past Spec. The hardware could give more than the field's rules expect. The
# "rigged contest" beat; a carries "someone saw it" for the player.
# Caller: eotg_on_tournament_contest_aug (player contestant), 1 day.

eotg_aug_act.002 = {
    type = activity_event
    window = widget_activity_locale_fullscreen_event
    title = eotg_aug_act.002.t
    desc = {
        desc = eotg_aug_act.002.desc
        first_valid = {
            triggered_desc = {
                trigger = { involved_activity ?= { has_current_phase = tournament_phase_board_game } }
                desc = eotg_aug_act.002.desc_board
            }
            desc = eotg_aug_act.002.desc_physical
        }
        triggered_desc = {
            trigger = {
                OR = {
                    eotg_is_aug_tier3 = yes
                    has_trait = eotg_neurofractured
                    eotg_aug_pressure_flicker = no
                }
            }
            desc = eotg_aug_act.002.desc_hot
        }
        triggered_desc = {
            trigger = {
                involved_activity.activity_host ?= { eotg_aug_under_policy = { LAW = ban } }
            }
            desc = eotg_aug_act.002.desc_ban
        }
    }
    theme = tournament_contest

    trigger = {
        activity_is_competing_trigger = yes
        eotg_is_augmented_any = yes
        NOT = { has_trait = eotg_total_integration }
    }

    left_portrait = {
        character = root
        animation = personality_bold
    }

    # "Run it past spec."
    option = {
        name = eotg_aug_act.002.a
        activity_tournament_change_contestant_score_effect = { SCORE = increase_major }
        hidden_effect = {
            if = {
                limit = { has_trait = eotg_neurofractured }
                eotg_add_fracture_risk = { AMOUNT = 8 }
            }
            else_if = {
                limit = { eotg_is_aug_tier3 = yes }
                eotg_add_fracture_risk = { AMOUNT = 6 }
            }
            else = {
                eotg_add_fracture_risk = { AMOUNT = 4 }
            }
        }
        random = {
            chance = 25
            involved_activity.activity_host ?= {
                if = {
                    limit = { NOT = { this = root } }
                    add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
                }
            }
            add_prestige = {
                value = minor_prestige_value
                multiply = -1
            }
            send_interface_toast = {
                type = event_toast_effect_bad
                title = eotg_aug_act.002.a.seen
                left_icon = root
            }
        }
        eotg_aug_stress_embrace_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = ambitious }
            modifier = { add = 20  has_trait = arrogant }
            modifier = { add = -20  has_trait = craven }
        }
    }

    # "Within the rules."
    option = {
        name = eotg_aug_act.002.b
        activity_tournament_change_contestant_score_effect = { SCORE = increase_minor }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = just }
            modifier = { add = 10  has_trait = honest }
        }
    }

    # "Throttle it down."
    option = {
        name = eotg_aug_act.002.c
        activity_tournament_change_contestant_score_effect = { SCORE = decrease_minor }
        add_prestige = minor_prestige_value
        if = {
            limit = { eotg_is_aug_tier3 = yes }
            hidden_effect = { eotg_add_fracture_risk = { AMOUNT = -3 } }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = humble }
            modifier = { add = 10  has_trait = content }
        }
    }

    # [deceitful] "Let them think it's skill." No discovery roll.
    option = {
        name = eotg_aug_act.002.d
        trigger = { has_trait = deceitful }
        trait = deceitful
        activity_tournament_change_contestant_score_effect = { SCORE = increase_medium }
        hidden_effect = { eotg_add_fracture_risk = { AMOUNT = 3 } }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = deceitful }
            modifier = { add = 10  has_trait = ambitious }
        }
    }

    # [honest] "Declare the hardware to the marshal."
    option = {
        name = eotg_aug_act.002.e
        trigger = { has_trait = honest }
        trait = honest
        involved_activity.activity_host ?= {
            if = {
                limit = { NOT = { this = root } }
                add_opinion = { modifier = eotg_opinion_aug_admiration  target = root }
            }
        }
        add_prestige = minor_prestige_value
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = honest }
            modifier = { add = 10  has_trait = just }
        }
    }

    # [wrathful, Overclocked] "Let the reflexes answer." Tier options spec
    # row 2: the reflex layer runs unthrottled (Overheated); no "seen" roll.
    option = {
        name = eotg_aug_act.002.f
        trigger = {
            eotg_is_aug_tier3 = yes
            has_trait = wrathful
        }
        trait = wrathful
        trait = eotg_cybernetics
        activity_tournament_change_contestant_score_effect = { SCORE = increase_major }
        add_character_modifier = { modifier = eotg_mod_aug_overheated  years = 1 }
        hidden_effect = { eotg_add_fracture_risk = { AMOUNT = 8 } }
        stress_impact = {
            wrathful = minor_stress_impact_loss
        }
        eotg_aug_stress_embrace_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = wrathful }
            modifier = { add = 10  has_trait = brave }
        }
    }

    # [Neurofractured, voice 2+] "Run the schedule it has already drafted."
    # Tier options spec row 3: the model times every exchange inside the
    # rules' tolerance, so there is no "seen" roll.
    option = {
        name = eotg_aug_act.002.g
        add_internal_flag = special    # W3: gated by a rare state (Neurofractured)
        trigger = {
            has_trait = eotg_neurofractured
            has_variable = eotg_aug_voice
            var:eotg_aug_voice >= 2
        }
        trait = eotg_neurofractured
        activity_tournament_change_contestant_score_effect = { SCORE = increase_major }
        hidden_effect = { eotg_add_fracture_risk = { AMOUNT = 10 } }
        eotg_aug_stress_embrace_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = ambitious }
            modifier = { add = 10  has_trait = arrogant }
        }
    }

# ── eotg_aug_act.003 ──────────────────────────────────────────────────────────
# Foul Play? A contestant accuses an augmented one of winning with hardware.
# Both are chosen targets, named in loc. Evidence is physical only (lore R6);
# .c.hot / .c.clean never say who examined (e reaches them too).
# Caller: eotg_on_tournament_active_aug (player host), 1-5 days.

eotg_aug_act.003 = {
    type = activity_event
    window = widget_activity_locale_fullscreen_event
    title = eotg_aug_act.003.t
    desc = {
        desc = eotg_aug_act.003.desc
        triggered_desc = {
            trigger = {
                scope:eotg_aug_accused ?= {
                    OR = {
                        eotg_is_aug_tier3 = yes
                        has_trait = eotg_neurofractured
                        eotg_aug_pressure_flicker = no
                    }
                }
            }
            desc = eotg_aug_act.003.desc_hot
        }
        # Lore R1 variants
        first_valid = {
            triggered_desc = {
                trigger = {
                    eotg_aug_under_policy = { LAW = ban }
                    is_independent_ruler = yes
                }
                desc = eotg_aug_act.003.desc_ban
            }
            triggered_desc = {
                trigger = { eotg_aug_under_policy = { LAW = ban } }
                desc = eotg_aug_act.003.desc_ban_v
            }
            triggered_desc = {
                trigger = {
                    eotg_aug_under_policy = { LAW = license }
                    is_independent_ruler = yes
                }
                desc = eotg_aug_act.003.desc_license
            }
            triggered_desc = {
                trigger = { eotg_aug_under_policy = { LAW = license } }
                desc = eotg_aug_act.003.desc_license_v
            }
        }
    }
    theme = tournament_grounds

    # World-state guard, mirroring the on_action
    trigger = {
        exists = scope:activity
        scope:activity.activity_host ?= root
        scope:activity = {
            any_attending_character = {
                activity_is_competing_trigger = yes
                eotg_aug_tamper_target = yes
                NOT = { this = root }
            }
            any_attending_character = {
                count >= 2
                activity_is_competing_trigger = yes
                NOT = { this = root }
            }
        }
    }

    left_portrait = {
        character = root
        animation = thinking
    }
    right_portrait = {
        trigger = { exists = scope:eotg_aug_accused }
        character = scope:eotg_aug_accused
        animation = worry
    }
    lower_right_portrait = {
        trigger = { exists = scope:eotg_aug_accuser }
        character = scope:eotg_aug_accuser
    }

    immediate = {
        scope:activity = {
            random_attending_character = {
                limit = {
                    activity_is_competing_trigger = yes
                    eotg_aug_tamper_target = yes
                    NOT = { this = root }
                }
                weight = {
                    base = 1
                    modifier = {
                        add = 2
                        eotg_is_aug_tier3 = yes
                    }
                }
                save_scope_as = eotg_aug_accused
            }
            random_attending_character = {
                limit = {
                    activity_is_competing_trigger = yes
                    NOT = { this = root }
                    NOT = { this = scope:eotg_aug_accused }
                }
                weight = {
                    base = 1
                    modifier = {
                        add = 5
                        has_relation_rival = scope:eotg_aug_accused
                    }
                }
                save_scope_as = eotg_aug_accuser
            }
        }
    }

    # "Disqualify them."
    option = {
        name = eotg_aug_act.003.a
        scope:eotg_aug_accused = { eotg_aug_act_withdraw_effect = { HOST = root } }
        scope:eotg_aug_accuser ?= {
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = zealous }
            modifier = { add = 10  has_trait = just }
        }
    }

    # "The accusation fails."
    option = {
        name = eotg_aug_act.003.b
        scope:eotg_aug_accuser ?= {
            add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
        }
        scope:eotg_aug_accused = {
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = cynical }
            modifier = { add = 10  has_trait = trusting }
        }
    }

    # "Have them examined." The branch reads the accused's band.
    option = {
        name = eotg_aug_act.003.c
        trigger = { eotg_aug_has_surgeon_access = yes }
        eotg_aug_act003_examine_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = diligent }
            modifier = { add = 10  has_trait = paranoid }
        }
    }

    # "Let them settle it on the field."
    option = {
        name = eotg_aug_act.003.d
        scope:eotg_aug_accused = {
            hidden_effect = { eotg_add_fracture_risk = { AMOUNT = 4 } }
            activity_tournament_change_contestant_score_effect = { SCORE = increase_minor }
        }
        scope:eotg_aug_accuser ?= {
            activity_tournament_change_contestant_score_effect = { SCORE = decrease_minor }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = arrogant }
            modifier = { add = 10  has_trait = brave }
        }
    }

    # [just] "Hear them both, then rule." As c without the surgeon (the
    # host's own judgment).
    option = {
        name = eotg_aug_act.003.e
        trigger = { has_trait = just }
        trait = just
        eotg_aug_act003_examine_effect = yes
        add_prestige = minor_prestige_value
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = just }
            modifier = { add = 10  has_trait = diligent }
        }
    }

    # [diligent, Enhanced+] "Run the heat readings through the overlay."
    # Tier options spec row 4: the same verdict as c/e, told by toast; no
    # surgeon needed. Not for a just host, who has e (never shown together).
    option = {
        name = eotg_aug_act.003.f
        trigger = {
            OR = {
                eotg_is_aug_tier2 = yes
                eotg_is_aug_tier3 = yes
            }
            has_trait = diligent
            NOT = { has_trait = just }
        }
        trait = diligent
        trait = eotg_cybernetics
        eotg_aug_act003_examine_effect = yes
        hidden_effect = { eotg_add_fracture_risk = { AMOUNT = 2 } }
        stress_impact = {
            diligent = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = diligent }
            modifier = { add = 10  has_trait = paranoid }
        }
    }

    # [Seamless] "Casing temperatures noted. A ruling follows." Tier options
    # spec row 5: as f, by tolerance; residue, not risk. Not for a just host.
    # Its only move is residue, so it is hidden once none is left (residue
    # pacing ruling, 2026-10-05; a world-state guard, not a cooldown).
