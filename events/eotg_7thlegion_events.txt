# ── eotg_7thlegion.0001 ───────────────────────────────────────────────────────
# Fires from: on_game_started (day 14)

    title = eotg_7thlegion.0001.t
    desc = eotg_7thlegion.0001.desc
    theme = martial

    option = {
        name = eotg_7thlegion.0001.a

        add_gold = 100
        add_character_modifier = { modifier = eotg_professional_contractors years = 10 }
        trigger_event = { id = eotg_7thlegion.0002 days = 270 }
    }

    option = {
        name = eotg_7thlegion.0001.b
        # "Contracts are leverage." (Lyssa influence)

        trigger_event = { id = eotg_7thlegion.0002 days = 270 }
    }

    option = {
        name = eotg_7thlegion.0001.c
        # "We decide where the Legion marches." (Corven influence)
        add_character_modifier = { modifier = eotg_autonomous_command years = 10 }
        add_dread = 10
        add_character_flag = eotg_legion_autonomy_path
        trigger_event = { id = eotg_7thlegion.0002 days = 270 }
    }
