# ── eotg_cartel_loyalty_tax_events.txt ────────────────────────────────────────
# Loyalty Tax collection event chain for the Cartel government.
#
# Chain:
#   Decision → tax.0001 to each eligible Boss (pay or refuse)
#     Refuse → tax.0002 to Grand Underlord (accept or punish)
#       Punish → tax.0003 to refusing Boss (consequences)
#
# Collective defiance (fires from eotg_on_yearly_cartel_flavor, not from the
# chain above):
#   tax.0005 — two or more Bosses are carrying eotg_cartel_defiant_boss at the
#              same time, so the refusals have started talking to each other.
#
# NOTE: .0004 was retired 2026-08-28. It covered this same premise but was
# referenced from nowhere, and its trigger condition ("enough Bosses refuse")
# required a refusal accumulator that does not exist anywhere in the codebase.
# .0005 replaces it by counting the defiance modifiers live at evaluation time.
# ID .0004 stays retired — do not reuse it.
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_cartel_tax

# Boss receives Loyalty Tax demand
eotg_cartel_tax.0001 = {
	type = character_event
	title = eotg_cartel_tax_0001_title
	theme = intrigue

	desc = eotg_cartel_tax_0001_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = liege
		animation = personality_bold
	}

	immediate = {
		liege = { save_scope_as = eotg_grand_underlord }
	}

	# Pay the tax
	option = {
		name = eotg_cartel_tax_0001_opt_pay
		eotg_se_cartel_boss_pays_tax = yes
		show_as_tooltip = {
			add_gold = {
				value = eotg_sv_cartel_loyalty_tax_demand
				multiply = -1
			}
			scope:eotg_grand_underlord = { add_prestige = 25 }
		}
		stress_impact = {
			greedy = minor_stress_impact_gain
			arbitrary = minor_stress_impact_loss
		}
		ai_chance = {
			base = 60
			modifier = { add = 20  has_trait = content }
			modifier = { add = -20  has_trait = ambitious }
			modifier = { add = -30  gold < 50 }
		}
	}

	# Refuse to pay
	option = {
		name = eotg_cartel_tax_0001_opt_refuse
		add_internal_flag = dangerous
		eotg_se_cartel_boss_refuses_tax = yes
		show_as_tooltip = {
			add_character_modifier = {
				modifier = eotg_cartel_defiant_boss
				years = 2
			}
			scope:eotg_grand_underlord = { add_prestige = -30 }
		}
		stress_impact = {
			content = medium_stress_impact_gain
			greedy = minor_stress_impact_loss
		}
		ai_chance = {
			base = 40
			modifier = { add = 30  has_trait = ambitious }
			modifier = { add = -30  has_trait = content }
			modifier = { add = 20  gold < 30 }
		}
	}
}

# Grand Underlord is notified a Boss refused — accept or escalate
eotg_cartel_tax.0002 = {
	type = character_event
	title = eotg_cartel_tax_0002_title
	theme = intrigue

	desc = eotg_cartel_tax_0002_desc

	left_portrait = {
		character = root
		animation = anger
	}

	immediate = {
		# Find the most recently-defiant Boss to personalize the event portrait
		random_vassal = {
			limit = { has_character_modifier = eotg_cartel_defiant_boss }
			save_scope_as = eotg_defiant_boss
		}
	}

	right_portrait = {
		character = scope:eotg_defiant_boss
		animation = personality_bold
	}

	# Absorb the defiance — lose face, avoid escalation
	option = {
		name = eotg_cartel_tax_0002_opt_accept
		add_prestige = -50
		stress_impact = {
			arbitrary = medium_stress_impact_gain
		}
		ai_chance = {
			base = 30
			modifier = { add = 20  has_trait = content }
			modifier = { add = -20  dread < 20 }
		}
	}

	# Issue punishment — frighten the defiant Boss
	option = {
		name = eotg_cartel_tax_0002_opt_punish
		add_internal_flag = dangerous
		scope:eotg_defiant_boss = {
			trigger_event = { id = eotg_cartel_tax.0003  days = 2 }
		}
		add_dread = 5
		ai_chance = {
			base = 70
			modifier = { add = 20  has_trait = arbitrary }
			modifier = { add = 20  dread > 40 }
		}
	}
}

# Refusing Boss receives punishment from the Grand Underlord
eotg_cartel_tax.0003 = {
	type = character_event
	title = eotg_cartel_tax_0003_title
	theme = intrigue

	desc = eotg_cartel_tax_0003_desc

	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = liege
		animation = anger
	}

	option = {
		name = eotg_cartel_tax_0003_opt_endure
		add_prestige = -100
		remove_short_term_gold = 25
		stress_impact = {
			base = medium_stress_impact_gain
		}
		ai_chance = { base = 100 }
	}
}

# The Bosses compare notes — collective defiance hardens into a standoff.
#
# Fired from eotg_on_yearly_cartel_flavor when two or more Bosses are currently
# carrying eotg_cartel_defiant_boss. This is the beat .0004 was written for
# ("too many Bosses are refusing"), but counted live off the defiance modifiers
# with any_vassal = { count >= 2 } rather than off an accumulator variable that
# was never built.
eotg_cartel_tax.0005 = {
	type = character_event
	title = eotg_cartel_tax_0005_title
	theme = intrigue

	desc = eotg_cartel_tax_0005_desc

	left_portrait = {
		character = root
		animation = personality_bold
	}

	right_portrait = {
		character = scope:eotg_ringleader
		animation = personality_bold
		trigger = { exists = scope:eotg_ringleader }
	}

	immediate = {
		# The most powerful of the refusing Bosses reads as the ringleader.
		ordered_vassal = {
			limit = { has_character_modifier = eotg_cartel_defiant_boss }
			order_by = max_military_strength
			max = 1
			save_scope_as = eotg_ringleader
		}
	}

	# Break the ringleader publicly — dread now, a standoff to follow.
	option = {
		name = eotg_cartel_tax_0005_opt_make_example
		add_internal_flag = dangerous
		add_dread = 15
		add_character_modifier = {
			modifier = eotg_cartel_shakedown_standoff
			years = 3
		}
		custom_tooltip = eotg_cartel_shakedown_standoff_tooltip
		if = {
			limit = { exists = scope:eotg_ringleader }
			scope:eotg_ringleader = {
				add_prestige = -100
				add_opinion = {
					target = root
					modifier = eotg_cartel_made_an_example
				}
			}
		}
		ai_chance = {
			base = 50
			modifier = { add = 25  has_trait = arbitrary }
			modifier = { add = 20  dread > 40 }
			modifier = { add = -20  has_trait = forgiving }
		}
	}

	# Buy the quiet ones back before the refusal becomes a bloc.
	option = {
		name = eotg_cartel_tax_0005_opt_buy_back
		remove_short_term_gold = 100
		every_vassal = {
			limit = { has_character_modifier = eotg_cartel_defiant_boss }
			remove_character_modifier = eotg_cartel_defiant_boss
			add_opinion = {
				target = root
				modifier = eotg_cartel_bought_back
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 30  gold > 300 }
			modifier = { add = -40  gold < 150 }
		}
	}

	# Concede the cycle. Cheaper than a war, and everyone will remember it.
	option = {
		name = eotg_cartel_tax_0005_opt_concede
		add_prestige = -75
		add_dread = -10
		stress_impact = {
			arbitrary = medium_stress_impact_gain
		}
		ai_chance = {
			base = 20
			modifier = { add = 20  gold < 100 }
			modifier = { add = -20  dread > 50 }
		}
	}
}
