# ── eotg_aug_tier1.001 ────────────────────────────────────────────────────────
# Phantom Sensation
# The implants surface in consciousness. A choice between acceptance,
# maintenance, or dangerous curiosity.

eotg_aug_tier1.001 = {
    type = character_event
    title = eotg_aug_tier1.001.t
    desc = eotg_aug_tier1.001.desc
    theme = physical_health

    left_portrait = {
        character = root
        animation = stress
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
        }
    }

    # "Ignore it"
    option = {
        name = eotg_aug_tier1.001.a
        stress_impact = { base = minor_stress_impact_gain }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = callous }
            modifier = { add = -20  has_trait = content }
        }
    }

    # "Recalibrate" — pay to stabilize
    option = {
        name = eotg_aug_tier1.001.b
        trigger = { gold >= 25 }
        remove_short_term_gold = 25
        add_character_modifier = { modifier = eotg_mod_implant_calibrated  years = 2 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = content }
            modifier = { add = 10  has_trait = patient }
        }
    }

    # "Fascinating" — lean into it, small chance to accelerate risk
    option = {
        name = eotg_aug_tier1.001.c
        add_learning_skill = 1
        hidden_effect = {
            if = {
                limit = { eotg_can_progress_to_enhanced = yes }
                eotg_add_fracture_risk = { AMOUNT = 2 }
            }
        }
        ai_chance = {
            base = 20
            modifier = { add = 20  has_trait = curious }
            modifier = { add = 10  has_trait = ambitious }
        }
    }
}


# ── eotg_aug_tier1.002 ────────────────────────────────────────────────────────
# The Upgrade — Progression Gate
# Augmented → Enhanced transition event.

eotg_aug_tier1.002 = {
    type = character_event
    title = eotg_aug_tier1.002.t
    desc = eotg_aug_tier1.002.desc
    theme = physical_health

    left_portrait = {
        character = root
        animation = personality_bold
    }

    # "Proceed" — pay and upgrade
    option = {
        name = eotg_aug_tier1.002.a
        trigger = { gold >= 100 }
        remove_short_term_gold = 100
        remove_trait = eotg_augmented
        add_trait = eotg_enhanced
        eotg_apply_edu_synergy_enhanced = yes
        stress_impact = {
            base = 0
            compassionate = minor_stress_impact_gain
            humble = minor_stress_impact_gain
        }
        ai_chance = {
            base = 40
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  has_trait = brave }
            modifier = { add = -50  has_trait = zealous }
            modifier = { add = -20  has_trait = content }
        }
    }

    # "Not yet"
    option = {
        name = eotg_aug_tier1.002.b
        stress_impact = {
            base = 0
            ambitious = minor_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = patient }
            modifier = { add = 20  gold < 100 }
        }
    }

    # "This concerns me" — suppress progression for 5 years
    option = {
        name = eotg_aug_tier1.002.c
        add_character_flag = { flag = eotg_flag_suppress_progression  years = 5 }
        stress_impact = {
            base = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = content }
            modifier = { add = 20  has_trait = humble }
        }
    }
}


# ── eotg_aug_tier1.003 ────────────────────────────────────────────────────────
# An Uncomfortable Question
# A vassal privately asks what their lord has become. One-time event.

eotg_aug_tier1.003 = {
    type = character_event
    title = eotg_aug_tier1.003.t
    desc = eotg_aug_tier1.003.desc
    theme = realm

    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        character = scope:concerned_vassal
        animation = worry
    }

    trigger = {
        any_vassal = { is_alive = yes }
        NOT = { has_character_flag = eotg_flag_tier1_vassal_suspicion_fired }
    }

    immediate = {
        hidden_effect = {
            add_character_flag = eotg_flag_tier1_vassal_suspicion_fired
            random_vassal = {
                limit = { is_alive = yes }
                save_scope_as = concerned_vassal
            }
        }
    }

    # "Reassure them." — opinion gain
    option = {
        name = eotg_aug_tier1.003.a
        scope:concerned_vassal = {
            add_opinion = { modifier = eotg_opinion_aug_admiration  target = root }
        }
        stress_impact = { base = minor_stress_impact_loss }
        ai_chance = {
            base = 50
            modifier = { add = 20  has_trait = gregarious }
            modifier = { add = 10  has_trait = compassionate }
        }
    }

    # "It is not their concern." — dread, opinion penalty
    option = {
        name = eotg_aug_tier1.003.b
        add_dread = 10
        scope:concerned_vassal = {
            add_opinion = { modifier = eotg_opinion_aug_fear  target = root }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = callous }
            modifier = { add = 10  has_trait = arrogant }
        }
    }

    # "Show them what I can do." — dread and intimidated admiration
    option = {
        name = eotg_aug_tier1.003.c
        add_dread = 20
        scope:concerned_vassal = {
            add_opinion = { modifier = eotg_opinion_aug_admiration  target = root }
        }
        stress_impact = { base = minor_stress_impact_gain }
        ai_chance = {
            base = 20
            modifier = { add = 20  has_trait = brave }
            modifier = { add = 20  has_trait = arrogant }
        }
    }
}

        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = generous }
            modifier = { add = 10  has_trait = brave }

    # "Deny it." — knight grows uneasy
    option = {
        name = eotg_aug_tier1.004.b

        }

            base = 40

        }
    }

    # "This ambition concerns me." — knight grows resentful
    option = {
        name = eotg_aug_tier1.004.c
        scope:petitioning_knight = {

            base = 20

            modifier = { add = 10  has_trait = suspicious }
