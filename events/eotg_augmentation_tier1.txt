# ── eotg_aug_tier1.002 ────────────────────────────────────────────────────────
# The Upgrade — Progression Gate
# Augmented → Enhanced transition event. Phase 4a (A-12): stage 1 of 3 of The
# First Upgrade. Paying no longer sets integration: a, d and e pay and fire
# eotg_aug_tier1.021 (What Will You Change?), which fires .022 (Installation),
# where the tier changes. Price: eotg_aug_pay_upgrade_effect (medium gold,
# x0.85 after tier1.020.c).
# Callers: eotg_on_yearly_aug_tier1_check (The Upgrade branch, unsolicited),
# eotg_decision_aug_pursue_next_stage (sought; balance spec §5.1).
# Sought (eotg_flag_aug_sought_upgrade, 60 days, set by the decision): the
# sought desc replaces the base one, and the AI follows through (b, c x0.1).
# tier1.022 clears the flag on delivery; b and c clear it here.

eotg_aug_tier1.002 = {
    type = character_event
    title = eotg_aug_tier1.002.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { has_character_flag = eotg_flag_aug_sought_upgrade }
                desc = eotg_aug_tier1.002.desc_sought
            }
            desc = eotg_aug_tier1.002.desc
        }
    }
    theme = physical_health

    # World-state tier guard (mirrors the on_action branch): a regression
    # between roll and firing must not let the upgrade skip a tier.
    trigger = { eotg_is_aug_tier1 = yes }

    left_portrait = {
        character = root
        animation = personality_bold
    }

    # "Proceed" — pay; the surgeons ask what to change first (.021)
    # Balance spec §5.6: how well you vet the surgeons is a stewardship check
    # against medium_skill_rating (10); success sets eotg_flag_aug_chain_edge
    # (2 years), read by the Installation (tier1.022: clean +15) and cleared
    # there. Shape: vanilla duel against a fixed value,
    # events/activities/chariot_race_activity/chariot_race_events.txt:1947.
    option = {
        name = eotg_aug_tier1.002.a
        trigger = { eotg_aug_can_afford_upgrade = yes }
        eotg_aug_pay_upgrade_effect = yes
        # Balance spec §5.6 (B2 QA): clear any other chain's edge/strain first
        eotg_aug_clear_chain_flags_effect = yes
        duel = {
            skill = stewardship
            value = medium_skill_rating
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = 3.5
                    min = -49
                }
                desc = eotg_aug_tier1.002.a.success
                hidden_effect = {
                    add_character_flag = { flag = eotg_flag_aug_chain_edge  years = 2 }
                }
            }
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = -3.5
                    min = -49
                }
                desc = eotg_aug_tier1.002.a.failure
            }
        }
        trigger_event = { id = eotg_aug_tier1.021  days = 1 }
        stress_impact = {
            base = 0
            compassionate = minor_stress_impact_gain
            humble = minor_stress_impact_gain
        }
        ai_chance = {
            base = 40
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  has_trait = brave }
            modifier = { add = -50  has_trait = zealous }
            modifier = { add = -20  has_trait = content }
        }
    }

    # "Not yet"
    option = {
        name = eotg_aug_tier1.002.b
        remove_character_flag = eotg_flag_aug_sought_upgrade
        stress_impact = {
            base = 0
            ambitious = minor_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = patient }
            modifier = { add = 20  gold < medium_gold_value }
            modifier = { add = 10  has_trait = content }
            modifier = {    # balance spec §5.1: the ruler asked for this
                factor = 0.1
                has_character_flag = eotg_flag_aug_sought_upgrade
            }
        }
    }

    # "This concerns me" — suppress progression for 5 years
    option = {
        name = eotg_aug_tier1.002.c
        remove_character_flag = eotg_flag_aug_sought_upgrade
        add_character_flag = { flag = eotg_flag_suppress_progression  years = 5 }
        stress_impact = {
            base = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = content }
            modifier = { add = 20  has_trait = humble }
            modifier = {    # balance spec §5.1: the ruler asked for this
                factor = 0.1
                has_character_flag = eotg_flag_aug_sought_upgrade
            }
        }
    }

    # [ambitious] "Because I can." - as a, leaning in (risk +3)
    option = {
        name = eotg_aug_tier1.002.d
        trigger = {
            has_trait = ambitious
            eotg_aug_can_afford_upgrade = yes
        }
        trait = ambitious
        eotg_aug_pay_upgrade_effect = yes
        eotg_add_fracture_risk = { AMOUNT = 3 }
        trigger_event = { id = eotg_aug_tier1.021  days = 1 }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = -30  has_trait = zealous }
        }
    }

    # [craven] "Is it safe? Tell me it's safe." - as a; .021 opens with the
    # surgeon's reassurance (flag read by its desc, cleared by .022)
    option = {
        name = eotg_aug_tier1.002.e
        trigger = {
            has_trait = craven
            eotg_aug_can_afford_upgrade = yes
        }
        trait = craven
        eotg_aug_pay_upgrade_effect = yes
        add_character_flag = { flag = eotg_flag_aug_upgrade_reassured  days = 120 }
        trigger_event = { id = eotg_aug_tier1.021  days = 1 }
        stress_impact = {
            craven = minor_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = craven }
            modifier = { add = 10  has_trait = ambitious }
        }
    }
}
