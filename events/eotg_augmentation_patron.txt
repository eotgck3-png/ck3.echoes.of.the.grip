# ── eotg_aug_patron.001 ───────────────────────────────────────────────────────
# The Patron's Offer. "We would like to pay for it."
# Callers: eotg_on_yearly_aug_initiation_check (Patron entry, weight 15),
# eotg_aug_init.018 (e), 1 day later.
# a/c/d accept through eotg_aug_patron_accept_effect (install at risk 0, the
# envoy from eotg_aug_patron_envoy_template with root's culture and faith by
# scope, the story with standard / read / greedy terms).

eotg_aug_patron.001 = {
    type = character_event
    title = eotg_aug_patron.001.t
    desc = eotg_aug_patron.001.desc
    theme = stewardship_wealth_focus

    trigger = {
        eotg_is_augmented_any = no
        eotg_aug_has_patron = no
    }

    left_portrait = {
        character = root
        animation = thinking
    }

    # "Accept."
    option = {
        name = eotg_aug_patron.001.a
        eotg_aug_patron_accept_effect = { TERMS = standard }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = ambitious }
            modifier = { add = 10  has_trait = brave }
            modifier = { add = -20  has_trait = zealous }
        }
    }

    # "Decline."
    option = {
        name = eotg_aug_patron.001.b
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = content }
            modifier = { add = 20  has_trait = zealous }
        }
    }

    # "Accept, after reading every clause." Slower demands, smaller bills.
    option = {
        name = eotg_aug_patron.001.c
        eotg_aug_patron_accept_effect = { TERMS = read }
        stress_impact = { base = minor_stress_impact_gain }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = diligent }
            modifier = { add = 10  has_trait = patient }
        }
    }

    # [greedy] "Accept, and ask for a stipend." Bills x1.5.
    option = {
        name = eotg_aug_patron.001.d
        trigger = { has_trait = greedy }
        trait = greedy
        eotg_aug_patron_accept_effect = { TERMS = greedy }
        add_gold = minor_gold_value
        stress_impact = {
            greedy = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = greedy }
            modifier = { add = 10  has_trait = ambitious }
        }
    }

    # [paranoid] "Who are you, really?" Decline.
    option = {
        name = eotg_aug_patron.001.e
        trigger = { has_trait = paranoid }
        trait = paranoid
        add_character_modifier = { modifier = eotg_mod_aug_lesson_intrigue  years = 5 }
        stress_impact = {
            paranoid = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = paranoid }
            modifier = { add = 10  has_trait = craven }
        }
    }
}


# ── eotg_aug_patron.002 ───────────────────────────────────────────────────────
# Repayment. "A small matter of the invoice." Bills scale with the terms
# (eotg_aug_patron_pay_effect: greedy x1.5, read x0.75).
# c: diplomacy duel against the envoy (shape of tier2.005.e, vanilla
# false_conversion.2000.a).
# Caller: eotg_story_aug_patron (demand 0).

eotg_aug_patron.002 = {
    type = character_event
    title = eotg_aug_patron.002.t
    desc = eotg_aug_patron.002.desc
    theme = stewardship_wealth_focus

    trigger = { eotg_aug_has_patron = yes }

    left_portrait = {
        character = root
        animation = personality_greedy
    }
    right_portrait = {
        trigger = { exists = scope:eotg_patron_envoy }
        character = scope:eotg_patron_envoy
        animation = steward
    }

    immediate = {
        hidden_effect = {
            random_owned_story = {
                limit = { story_type = eotg_story_aug_patron }
                save_scope_as = eotg_patron_story
                var:eotg_envoy ?= { save_scope_as = eotg_patron_envoy }
            }
        }
    }

    # "Pay."
    option = {
        name = eotg_aug_patron.002.a
        trigger = { eotg_aug_patron_can_pay = { BASE = medium_gold_value  MULT = 1 } }
        eotg_aug_patron_pay_effect = { BASE = medium_gold_value  MULT = 1 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = honest }
            modifier = { add = 10  has_trait = content }
            modifier = { add = -20  has_trait = greedy }
        }
    }

    # "Refuse." The firmware hesitates.
    option = {
        name = eotg_aug_patron.002.b
        eotg_aug_patron_grievance_effect = { AMOUNT = 1 }
        scope:eotg_patron_envoy ?= {
            add_opinion = { modifier = eotg_opinion_aug_disgust  target = root  years = 5 }
        }
        # Owner-side risk only while the owner still has hardware (final QA: the
        # debt outlives Remove Implants; nothing may recreate the variable).
        # Balance spec §5.8: +8 at Overclocked (the firmware lever scales).
        if = {
            limit = { eotg_is_aug_tier3 = yes }
            eotg_add_fracture_risk = { AMOUNT = 8 }
        }
        else_if = {
            limit = { eotg_is_augmented_any = yes }
            eotg_add_fracture_risk = { AMOUNT = 5 }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = greedy }
            modifier = { add = 10  has_trait = stubborn }
        }
    }

    # "Renegotiate."
    option = {
        name = eotg_aug_patron.002.c
        trigger = {
            exists = scope:eotg_patron_envoy
            eotg_aug_patron_can_pay = { BASE = major_gold_value  MULT = 1 }
        }
        duel = {
            skill = diplomacy
            target = scope:eotg_patron_envoy
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = 2.5
                }
                desc = eotg_aug_patron.002.c.success
                eotg_aug_patron_pay_effect = { BASE = minor_gold_value  MULT = 1 }
            }
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = -2.5
                }
                desc = eotg_aug_patron.002.c.failure
                eotg_aug_patron_pay_effect = { BASE = major_gold_value  MULT = 1 }
            }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = gregarious }
            modifier = { add = 10  has_trait = ambitious }
        }
    }

    # [deceitful] "Pay in promises." The envoy holds a debt hook.
    option = {
        name = eotg_aug_patron.002.d
        trigger = {
            has_trait = deceitful
            exists = scope:eotg_patron_envoy
        }
        trait = deceitful
        scope:eotg_patron_envoy = {
            add_hook = {
                type = indebted_hook
                target = root
            }
        }
        eotg_aug_stress_lie_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = deceitful }
            modifier = { add = 10  has_trait = greedy }
        }
    }

    # [honest] "Ask for honest terms." 50%: half the bill; 50%: offence taken.
    option = {
        name = eotg_aug_patron.002.e
        trigger = {
            has_trait = honest
            eotg_aug_patron_can_pay = { BASE = medium_gold_value  MULT = 0.5 }
        }
        trait = honest
        random_list = {
            50 = {
                desc = eotg_aug_patron.002.e.success
                eotg_aug_patron_pay_effect = { BASE = medium_gold_value  MULT = 0.5 }
            }
            50 = {
                desc = eotg_aug_patron.002.e.failure
                eotg_aug_patron_grievance_effect = { AMOUNT = 1 }
            }
        }
        stress_impact = {
            honest = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = honest }
            modifier = { add = 10  has_trait = just }
        }
    }

    # (Overclocked) "Let their technicians service the debt." Balance spec
    # §5.8: universal at Overclocked. No gold; the demand counts as paid (the
    # `after` block increments eotg_demand, as for every option here); their
    # firmware work costs risk +10, hidden. No trait icon: the gate is the
    # tier, not a trait.
    option = {
        name = eotg_aug_patron.002.f
        trigger = { eotg_is_aug_tier3 = yes }
        hidden_effect = {
            eotg_add_fracture_risk = { AMOUNT = 10 }
        }
        eotg_aug_stress_embrace_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = greedy }
            modifier = { add = 10  has_trait = ambitious }
            modifier = { add = -20  has_trait = paranoid }
        }
    }

    after = {
        hidden_effect = { eotg_aug_patron_demand_effect = yes }
    }
}
