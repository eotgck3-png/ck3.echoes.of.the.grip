# ── eotg_pmc_contract_events.txt ─────────────────────────────────────────────
# Contract War events for the PMC government.
#
# eotg_pmc_contract.0001 — Contractor-General receives a contract offer
# eotg_pmc_contract.0002 — Marshal's recommendation fires (hidden, same tick)
# eotg_pmc_contract.0003 — Client notified of controversy spreading to their lands
# eotg_pmc_contract.0004 — Yearly controversy backlash warning to Contractor-General
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_pmc_contract

# Contractor-General receives a contract offer from a ruler who entered a war
eotg_pmc_contract.0001 = {
	type = character_event
	title = eotg_pmc_contract_0001_title
	theme = stewardship

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = eotg_pmc_contract_client }
				desc = eotg_pmc_contract_0001_desc_named
			}
			desc = eotg_pmc_contract_0001_desc_generic
		}
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = root.var:eotg_pmc_contract_client
		animation = personality_rational
		trigger = { has_variable = eotg_pmc_contract_client }
	}

	immediate = {
		if = {
			limit = { has_variable = eotg_pmc_contract_client }
			var:eotg_pmc_contract_client = { save_scope_as = eotg_client }
		}
		# Fire Marshal recommendation as a hidden follow-up
		trigger_event = { id = eotg_pmc_contract.0002  days = 1 }
	}

	# Accept: Minor Contract
	option = {
		name = eotg_pmc_contract_0001_opt_minor
		eotg_se_pmc_apply_minor_contract = yes
		ai_chance = {
			base = 30
			modifier = { add = -10  gold > 500 }
			modifier = { add = 20  gold < 100 }
		}
	}

	# Accept: Standard Contract
	option = {
		name = eotg_pmc_contract_0001_opt_standard
		eotg_se_pmc_apply_standard_contract = yes
		ai_chance = {
			base = 40
			modifier = { add = 10  gold > 200 }
		}
	}

	# Accept: Exclusive Contract
	option = {
		name = eotg_pmc_contract_0001_opt_exclusive
		trigger = { NOT = { has_character_flag = eotg_pmc_active_contract } }
		eotg_se_pmc_apply_exclusive_contract = yes
		ai_chance = {
			base = 20
			modifier = { add = 20  gold < 50 }
			modifier = { add = -10  has_trait = ambitious }
		}
	}

	# Decline
	option = {
		name = eotg_pmc_contract_0001_opt_decline
		add_prestige = 25
		stress_impact = {
			greedy = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
			modifier = { add = 20  has_trait = content }
		}
	}
}

# Hidden: Marshal fires a recommendation shortly after the contract offer
eotg_pmc_contract.0002 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { has_character_flag = eotg_pmc_active_contract }
			# Contract already accepted — no recommendation needed
			break = yes
		}
		# Marshal (if exists) recommends Standard tier
		if = {
			limit = {
				any_councillor = {
					councillor_task = task_train_troops
				}
			}
			trigger_event = { id = eotg_pmc_contract.0002  days = 2 }
		}
	}
}

# Client notified that PMC controversy is rippling into their territory
eotg_pmc_contract.0003 = {
	type = character_event
	title = eotg_pmc_contract_0003_title
	theme = stewardship

	desc = eotg_pmc_contract_0003_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Tolerate it — the contract is worth the friction
	option = {
		name = eotg_pmc_contract_0003_opt_tolerate
		add_prestige = -25
		stress_impact = {
			base = minor_stress_impact_gain
		}
		ai_chance = { base = 70 }
	}

	# Demand the PMC curtail their activities — prestige cost to both
	option = {
		name = eotg_pmc_contract_0003_opt_demand
		add_prestige = -50
		ai_chance = { base = 30 }
	}
}

# Yearly controversy backlash — fires to Contractor-General when active controversy is building
eotg_pmc_contract.0004 = {
	type = character_event
	title = eotg_pmc_contract_0004_title
	theme = stewardship

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_modifier = eotg_pmc_controversy_exclusive }
				desc = eotg_pmc_contract_0004_desc_exclusive
			}
			triggered_desc = {
				trigger = { has_character_modifier = eotg_pmc_controversy_standard }
				desc = eotg_pmc_contract_0004_desc_standard
			}
			desc = eotg_pmc_contract_0004_desc_minor
		}
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Manage it — prestige cost, no lasting effect
	option = {
		name = eotg_pmc_contract_0004_opt_manage
		add_prestige = -50
		stress_impact = {
			base = minor_stress_impact_gain
		}
		ai_chance = { base = 70 }
	}

	# Ignore it — risk the pressure building
	option = {
		name = eotg_pmc_contract_0004_opt_ignore
		add_prestige = 25
		stress_impact = {
			content = minor_stress_impact_gain
		}
		ai_chance = { base = 30 }
	}
}
