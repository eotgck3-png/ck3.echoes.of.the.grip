# -- eotg_augmentation_endgame.txt --------------------------------------------
# The Neurofractured endgames (docs/specs/cybernetics_v2_phase3.md §7).
#
# eotg_aug_end.001 -- Excision: The Surgeons      (leave the system, or die)
# eotg_aug_end.002 -- Silence                     (Excision survivors)
# eotg_aug_end.009 -- Hand Over the Controls      (choose Total Integration)
# eotg_aug_end.010 -- There Is No Static          (Total Integration notice)
# eotg_aug_end.011 -- The Empty Hall              (the court leaves)
# eotg_aug_end.020 -- Appoint a Warden            (Containment Regency by choice)
# eotg_aug_end.030 -- Into Restraints             (abdication)
# eotg_aug_end.031 -- The Locked Wing             (to the successor)
# eotg_aug_end.040 -- The Ledger Balances         (Seamless, G2)
# eotg_aug_end.041 -- A Resignation               (Seamless, G2)
# eotg_aug_end.042 -- A Flicker                   (Seamless, G2)
#
# Fires from: eotg_decision_aug_excision, eotg_fracture.026 (d),
#   eotg_fracture.027 (excision outcome), eotg_fracture.007 (f) and
#   eotg_aug_end.031 (c) -> .001; eotg_aug_excision_surgery_effect -> .002;
#   eotg_decision_aug_embrace_cascade -> .009;
#   eotg_aug_total_integration_effect -> .010; .010 immediate -> .011;
#   eotg_decision_aug_appoint_warden -> .020; eotg_fracture.027 (abdication
#   outcome) and .020 (d) -> .030; eotg_aug_abdicate_effect -> .031;
#   eotg_on_yearly_aug_seamless_check (yearly_playable_pulse) -> .040-.042.
# Cooldowns live in the decisions (cooldown = {} or flag) and, for .040-.042,
# in eotg_on_yearly_aug_seamless_check (eotg_flag_aug_seamless_cooldown);
# chain stages carry none (index §1 rule 5).
# Resource: every event here changes tier (exits the system, becomes
# Seamless, or dies) or moves eotg_fracture_risk; chain stages (.002, .010,
# .011) couple through the tier change that fired them. The Seamless events
# (.040-.042) read and move the Seamless resource, eotg_aug_residue (hidden,
# 0-3, only falls; procedures spec §2), since a Seamless ruler has no risk.
# Loc register: "self" / "personhood", never "humanity" (index §5.5).
# -----------------------------------------------------------------------------

namespace = eotg_aug_end


# -- eotg_aug_end.001 ---------------------------------------------------------
# Excision — The Surgeons. The odds are given in words (desc variants by tier,
# physician, promise and a free table), never numbers; the roll is hidden in
# eotg_aug_excision_surgery_effect. Surgical odds, not fracture risk.

eotg_aug_end.001 = {
    type = character_event
    title = eotg_aug_end.001.t
    desc = {
        desc = eotg_aug_end.001.desc
        first_valid = {
            triggered_desc = {
                trigger = { has_trait = eotg_neurofractured }
                desc = eotg_aug_end.001.desc_odds_grim
            }
            triggered_desc = {
                trigger = { eotg_is_aug_tier3 = yes }
                desc = eotg_aug_end.001.desc_odds_poor
            }
            desc = eotg_aug_end.001.desc_odds_fair
        }
        triggered_desc = {
            trigger = { eotg_has_physician_access = yes }
            desc = eotg_aug_end.001.desc_physician
        }
        # Free outranks promised: a free table makes the half-price promise
        # moot, so only one of the two lines shows.
        first_valid = {
            triggered_desc = {
                trigger = { has_character_flag = eotg_flag_aug_excision_free }
                desc = eotg_aug_end.001.desc_free
            }
            triggered_desc = {
                trigger = { has_character_flag = eotg_flag_aug_promised_treatment }
                desc = eotg_aug_end.001.desc_promised
            }
        }
    }
    theme = medicine
    override_background = { reference = eotg_bg_clinic }

    trigger = {
        OR = {
            has_trait = eotg_cybernetics
            has_trait = eotg_neurofractured
        }
    }

    left_portrait = {
        character = root
        animation = worry
    }
    right_portrait = {
        trigger = { exists = scope:eotg_surgeon }
        character = scope:eotg_surgeon
        animation = personality_rational
    }

    immediate = {
        hidden_effect = {
            if = {
                limit = { employs_court_position = court_physician_court_position }
                court_position:court_physician_court_position = { save_scope_as = eotg_surgeon }
            }
        }
    }

    # "Begin."
    option = {
        name = eotg_aug_end.001.a
        add_internal_flag = dangerous    # W3: the excision can kill root
        trigger = { eotg_aug_can_afford_excision = yes }
        eotg_aug_excision_pay_effect = yes
        custom_tooltip = eotg_aug_end.001.tt
        hidden_effect = {
            eotg_aug_excision_surgery_effect = { DEATH_BASE = 15  BRAVE = no  MAIMED_BASE = 30  PHYSICIAN = yes }
        }
        eotg_aug_stress_reject_effect = yes
        ai_chance = {
            base = 50
            modifier = { add = 20  has_trait = content }
            modifier = { add = 20  has_trait = zealous }
            modifier = { add = -20  has_trait = ambitious }
        }
    }

    # "Not today." The free table does not wait.
    option = {
        name = eotg_aug_end.001.b
        # Tooltip sweep: the hidden consequence (a free excision is lost)
        # is named only when there is one.
        if = {
            limit = { has_character_flag = eotg_flag_aug_excision_free }
            custom_tooltip = eotg_aug_end.001.b.tt
        }
        hidden_effect = {
            remove_character_flag = eotg_flag_aug_excision_free
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = craven }
            modifier = { add = 20  has_trait = ambitious }
        }
    }

    # [craven] "Put me under. Don't wake me if it goes wrong." As a, death +10.
    option = {
        name = eotg_aug_end.001.c
        add_internal_flag = dangerous    # W3: the excision can kill root
        trigger = {
            has_trait = craven
            eotg_aug_can_afford_excision = yes
        }
        trait = craven
        eotg_aug_excision_pay_effect = yes
        custom_tooltip = eotg_aug_end.001.tt
        hidden_effect = {
            eotg_aug_excision_surgery_effect = { DEATH_BASE = 25  BRAVE = no  MAIMED_BASE = 30  PHYSICIAN = yes }
        }
        stress_impact = {
            craven = medium_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = craven }
            modifier = { add = 10  has_trait = content }
        }
    }

    # [brave] "Awake. I want to watch them take it out." As a; the maimed
    # outcome becomes clean, +100 prestige on survival.
    # (Procedures spec §4.7: a, c and d keep the physician at the table,
    # PHYSICIAN = yes, MAIMED_BASE = 30.)
    option = {
        name = eotg_aug_end.001.d
        add_internal_flag = dangerous    # W3: the excision can kill root
        trigger = {
            has_trait = brave
            eotg_aug_can_afford_excision = yes
        }
        trait = brave
        eotg_aug_excision_pay_effect = yes
        custom_tooltip = eotg_aug_end.001.d.tt
        hidden_effect = {
            eotg_aug_excision_surgery_effect = { DEATH_BASE = 15  BRAVE = yes  MAIMED_BASE = 30  PHYSICIAN = yes }
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = brave }
            modifier = { add = 10  has_trait = stubborn }
        }
    }

    # "Not their table. Somewhere cheaper." (procedures spec §4.7) The back
    # streets: x0.4 of the full price (the Heir's Arc promise was to the real
    # surgeons, so it does not apply), death 25 and maimed 40 (the gaps
    # doc's back-street table, +10 each), and no physician at the table
    # (PHYSICIAN = no; lore ruling (d), upheld 2026-10-04). Not offered when
    # the surgeons are already in the room (the free table).
    option = {
        name = eotg_aug_end.001.e
        add_internal_flag = dangerous    # W3: the excision can kill root
        trigger = {
            NOT = { has_character_flag = eotg_flag_aug_excision_free }
            eotg_aug_can_afford_procedure = { BASE = major_gold_value  MULT = eotg_aug_price_mult_backstreet_removal }
        }
        eotg_aug_pay_procedure_effect = { BASE = major_gold_value  MULT = eotg_aug_price_mult_backstreet_removal }
        custom_tooltip = eotg_aug_end.001.tt
        hidden_effect = {
            eotg_aug_excision_surgery_effect = { DEATH_BASE = 25  BRAVE = no  MAIMED_BASE = 40  PHYSICIAN = no }
        }
        eotg_aug_stress_reject_effect = yes
        ai_chance = {
            base = 15
            modifier = { add = 20  has_trait = greedy }
            modifier = { add = 10  has_trait = brave }
            modifier = { add = -20  has_trait = craven }
        }
    }
}

# -- eotg_aug_end.011 ---------------------------------------------------------
# The Empty Hall. Every courtier who is not close family or a spouse and
# thinks ill of you has already gone (move_to_pool in immediate); the count
# is root variable eotg_hall_gone for the desc
# ([ROOT.Char.MakeScope.Var('eotg_hall_gone').GetValue], vanilla loc form),
# cleared in `after`. Option c only sets the tone: the departures happened.
# Caller: eotg_aug_end.010 (180 days). Coupling: reads the Seamless trait.

eotg_aug_end.011 = {
    type = character_event
    title = eotg_aug_end.011.t
    desc = {
        first_valid = {
            # The most notable leaver, saved in immediate before move_to_pool.
            triggered_desc = {
                trigger = { exists = scope:eotg_departed_courtier }
                desc = eotg_aug_end.011.desc_named
            }
            triggered_desc = {
                trigger = { var:eotg_hall_gone > 0 }
                desc = eotg_aug_end.011.desc
            }
            desc = eotg_aug_end.011.desc_none
        }
    }
    theme = court
    override_background = { reference = eotg_bg_council }

    trigger = {
        has_trait = eotg_total_integration
    }

    left_portrait = {
        character = root
        animation = idle
    }

    immediate = {
        set_variable = {
            name = eotg_hall_gone
            value = {
                value = 0
                every_courtier = {
                    limit = {
                        is_imprisoned = no
                        NOT = { is_close_family_of = root }
                        NOT = { is_spouse_of = root }
                        opinion = { target = root  value < 0 }
                    }
                    add = 1
                }
            }
        }
        # The most notable leaver, named in the desc: a court-position holder
        # first, then the highest prestige. Saved before move_to_pool; the
        # character still exists in the pool, so the name resolves.
        ordered_courtier = {
            limit = {
                is_imprisoned = no
                NOT = { is_close_family_of = root }
                NOT = { is_spouse_of = root }
                opinion = { target = root  value < 0 }
            }
            order_by = {
                value = prestige
                if = {
                    limit = { has_any_court_position = yes }
                    add = 1000000
                }
            }
            save_scope_as = eotg_departed_courtier
        }
        every_courtier = {
            limit = {
                is_imprisoned = no
                NOT = { is_close_family_of = root }
                NOT = { is_spouse_of = root }
                opinion = { target = root  value < 0 }
            }
            move_to_pool = yes
        }
    }

    # "Hire replacements." QA round 2 M11: gated on the gold, and it hires.
    # One or two lowborn courtiers from vanilla pool templates
    # (common/scripted_character_templates/
    # 00_pool_repopulation_character_templates.txt: pool_repopulate_stewardship
    # for the household, pool_repopulate_prowess for the guard; both take
    # root's culture and faith by scope, so map-agnostic), created and added
    # the way vanilla hires a lowborn (create_character + add_courtier,
    # events/activities/hold_court_activity/hold_court_events_general.txt:2173-2181).
    option = {
        name = eotg_aug_end.011.a
        trigger = { gold >= major_gold_value }
        remove_short_term_gold = major_gold_value
        create_character = {
            template = pool_repopulate_stewardship
            location = root.location
            dynasty = none
            save_scope_as = eotg_hire_first
        }
        add_courtier = scope:eotg_hire_first
        random = {
            chance = 50
            create_character = {
                template = pool_repopulate_prowess
                location = root.location
                dynasty = none
                save_scope_as = eotg_hire_second
            }
            add_courtier = scope:eotg_hire_second
            send_interface_toast = {    # W3 (event_quality_v1)
                type = event_toast_effect_good
                title = eotg_aug_end.011.toast_a_second
                left_icon = root
                right_icon = scope:eotg_hire_second
            }
        }
        ai_chance = {
            base = 40
            modifier = { add = 20  gold >= major_gold_value }
            modifier = { add = 10  is_at_war = yes }
        }
    }

    # "Let them go."
    option = {
        name = eotg_aug_end.011.b
        ai_chance = {
            base = 50
            modifier = { add = 20  gold < major_gold_value }
            modifier = { add = 10  is_at_war = no }
        }
    }

    # "Make them stay." Too late: it only sets the tone.
    option = {
        name = eotg_aug_end.011.c
        add_dread = 50
        add_tyranny = 20
        ai_chance = {
            base = 20
            modifier = { add = 20  dread >= 50 }
            modifier = { add = 10  gold < minor_gold_value }
        }
    }

    after = {
        remove_variable = eotg_hall_gone
    }
}
