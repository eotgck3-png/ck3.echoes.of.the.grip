# eotg_gobcorp_flavor.0001 — A Combine Gains Dominance
# One Trade Combine begins economically dominating the realm.
eotg_gobcorp_flavor.0001 = {
	type = character_event
	title = eotg_gobcorp_flavor_0001_title
	desc = eotg_gobcorp_flavor_0001_desc
	theme = stewardship

	left_portrait = { character = root  animation = personality_bold }

	trigger = {
		eotg_st_is_director_general = yes
		is_powerful_vassal = no
		any_vassal = { is_powerful_vassal = yes }
		NOT = { has_character_flag = eotg_gobcorp_0001_recent }
	}

	option = {
		name = eotg_gobcorp_flavor_0001_opt_support
		add_gold = 30
		add_character_modifier = { modifier = eotg_gobcorp_faction_favoritism  years = 3 }
		every_vassal = {
			limit = { is_powerful_vassal = yes  NOT = { this = prev } }
			add_opinion = { target = root  modifier = eotg_gobcorp_faction_rivalry }
		}
		add_character_flag = { flag = eotg_gobcorp_0001_recent  years = 4 }
		ai_chance = { base = 40  modifier = { add = 20  has_trait = greedy } }
	}
	option = {
		name = eotg_gobcorp_flavor_0001_opt_rivals
		add_intrigue_skill = 1
		add_character_modifier = { modifier = eotg_gobcorp_market_monopoly  years = 2 }
		add_character_flag = { flag = eotg_gobcorp_0001_recent  years = 4 }
		ai_chance = { base = 30  modifier = { add = 15  has_trait = deceitful } }
	}
	option = {
		name = eotg_gobcorp_flavor_0001_opt_regulate
		add_prestige = 20
		every_vassal = {
			limit = { is_powerful_vassal = yes }
			add_opinion = { target = root  modifier = eotg_fringe_fear_rule_resentment }
		}
		add_character_flag = { flag = eotg_gobcorp_0001_recent  years = 4 }
		ai_chance = { base = 30  modifier = { add = 15  has_trait = just } }
	}
}
