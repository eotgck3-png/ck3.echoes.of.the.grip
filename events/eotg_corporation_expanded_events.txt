# eotg_corp_flavor.0001 — Financial Irregularities Detected
# Evidence emerges that a rival executive's division suffers unexplained losses.
eotg_corp_flavor.0001 = {
	type = character_event
	title = eotg_corp_flavor_0001_title
	desc = eotg_corp_flavor_0001_desc
	theme = intrigue

	left_portrait = { character = root  animation = personality_rational }

	trigger = {
		eotg_st_is_ceo = yes
		intrigue >= 10
		vassal_count >= 2
		NOT = { has_character_flag = eotg_corp_0001_recent }
	}

	option = {
		name = eotg_corp_flavor_0001_opt_leak
		every_vassal = {
			limit = { is_powerful_vassal = yes  intrigue < 8 }
			add_prestige = -25
		}
		add_character_flag = { flag = eotg_corp_0001_recent  years = 3 }
		ai_chance = { base = 35  modifier = { add = 15  has_trait = brave } }
	}
	option = {
		name = eotg_corp_flavor_0001_opt_blackmail
		every_vassal = {
			limit = { is_powerful_vassal = yes  intrigue < 8 }
			add_hook = {
				type = strong_favor_hook
				target = prev
			}
		}
		add_character_modifier = { modifier = eotg_corporation_financial_sabotage  years = 2 }
		add_character_flag = { flag = eotg_corp_0001_recent  years = 3 }
		ai_chance = { base = 40  modifier = { add = 20  has_trait = deceitful } }
	}
	option = {
		name = eotg_corp_flavor_0001_opt_profit
		add_gold = 40
		add_character_flag = { flag = eotg_corp_0001_recent  years = 3 }
		ai_chance = { base = 25  modifier = { add = 15  has_trait = greedy } }
	}
}
