#   (eotg_flag_aug_contraband_cd) is set there and never read here.
# Resource: tier. b removes the subject's implants through the shared roll
#   (a tier change); the desc reads the subject's tier.
# Lore Q1 boundary: the arrest is made by the subject's own liege, inside the
#   realm, under the realm's own law; discovery is hearsay ("word has
#   reached you"); no apparatus. Lore R1: "your law" only for an independent
#   ruler, "the law of your realm" otherwise.
#
# event_quality_v1 batch B8 (W5 re-gating, W6 portraits; §13): e re-axed
# from [just] to a duchy tier gate (AI base 10, counter-trait negatives),
# just keeping its weight and a stress row there as a response; f (greedy,
# 6 gates against just 16) is the kept personality gate, paired with its
# stress row (§13.1 step 5).

namespace = eotg_aug_realm


# ── eotg_aug_realm.001 ────────────────────────────────────────────────────────
# Contraband Hardware. scope:eotg_contraband_subject, saved by the effect, is
# a chosen target named in loc. The opinion the effect gave root is the lawful
# reason to imprison; c and d forgive it.
# Caller: eotg_aug_contraband_effect, 3-10 days.

eotg_aug_realm.001 = {
    type = character_event
    title = eotg_aug_realm.001.t
    desc = {
        desc = eotg_aug_realm.001.desc
        first_valid = {
            triggered_desc = {
                trigger = {
                    eotg_aug_under_policy = { LAW = ban }
                    is_independent_ruler = yes
                }
                desc = eotg_aug_realm.001.desc_ban
            }
            triggered_desc = {
                trigger = { eotg_aug_under_policy = { LAW = ban } }
                desc = eotg_aug_realm.001.desc_ban_v
            }
            triggered_desc = {
                trigger = {
                    eotg_aug_under_policy = { LAW = license }
                    is_independent_ruler = yes
                }
                desc = eotg_aug_realm.001.desc_license
            }
            triggered_desc = {
                trigger = { eotg_aug_under_policy = { LAW = license } }
                desc = eotg_aug_realm.001.desc_license_v
            }
        }
        first_valid = {
            triggered_desc = {
                trigger = { scope:eotg_contraband_subject = { has_trait = eotg_neurofractured } }
                desc = eotg_aug_realm.001.desc_nf
            }
            triggered_desc = {
                trigger = { scope:eotg_contraband_subject = { eotg_is_aug_tier3 = yes } }
                desc = eotg_aug_realm.001.desc_oc
            }
            triggered_desc = {
                trigger = { scope:eotg_contraband_subject = { eotg_is_aug_tier2 = yes } }
                desc = eotg_aug_realm.001.desc_enh
            }
            triggered_desc = {
                trigger = { scope:eotg_contraband_subject = { eotg_is_aug_tier1 = yes } }
                desc = eotg_aug_realm.001.desc_aug
            }
        }
        triggered_desc = {
            trigger = { scope:eotg_contraband_subject = { is_courtier_of = root } }
            desc = eotg_aug_realm.001.desc_courtier
        }
    }
    theme = intrigue
    override_background = { reference = eotg_bg_underlevel_day }

    # World-state guard
    trigger = {
        # Seamless exposure (event_quality_v1 §12.5.1 S1, rule (a), event
        # trigger): the desc and options need a self deliberating; a state
        # guard re-checked at fire time, not a cooldown (lesson 5 n/a)
        NOT = { has_trait = eotg_total_integration }
        scope:eotg_contraband_subject ?= {
            is_alive = yes
            is_imprisoned = no
        }
    }

    left_portrait = {
        character = root
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { OR = { has_trait = wrathful  has_trait = zealous } }
            animation = anger
        }
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = greedy }
            animation = personality_greedy
        }
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = compassionate }
            animation = worry
        }
        animation = thinking
    }
    right_portrait = {
        character = scope:eotg_contraband_subject
        animation = worry
    }

    # "Arrest them." The opinion is the lawful reason.
    option = {
        name = eotg_aug_realm.001.a
        imprison_character_effect = {
            TARGET = scope:eotg_contraband_subject
            IMPRISONER = root
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = just }
            modifier = { add = 20  has_trait = zealous }
            modifier = { add = -20  has_trait = compassionate }
        }
    }

    # "Have it taken out." A household member, on the liege's order: the
    # interactions spec's removal effect and its roll, run with
    # scope:actor = root and scope:recipient = the subject. The "forced" case,
    # so scope:hook is set for the effect: Forced Procedure on the subject and
    # the tyranny stress on root (spec §4.4). Not over a booking the subject
    # made themself (procedures QA booking guard).
    option = {
        name = eotg_aug_realm.001.b
        trigger = {
            scope:eotg_contraband_subject = {
                is_courtier_of = root
                is_landed = no
                # Tiers 1-3 only, matching Demand Implant Removal's is_shown:
                # a Neurofractured subject's exit is Excision, and Seamless
                # hardware cannot be taken out
                has_trait = eotg_cybernetics
                NOT = { has_variable = eotg_aug_removal_kind }
            }
            trigger_if = {
                limit = { eotg_aug_has_surgeon_for = { PATIENT = scope:eotg_contraband_subject } }
                eotg_aug_can_afford_procedure = { BASE = medium_gold_value  MULT = eotg_aug_price_mult_physician }
            }
            trigger_else = {
                eotg_aug_can_afford_procedure = { BASE = medium_gold_value  MULT = eotg_aug_price_mult_backstreet_removal }
            }
        }
        save_scope_as = actor
        scope:eotg_contraband_subject = { save_scope_as = recipient }
        save_scope_value_as = { name = hook  value = yes }
        if = {
            limit = { eotg_aug_has_surgeon_for = { PATIENT = scope:eotg_contraband_subject } }
            eotg_aug_demand_removal_effect = { PROVIDER = physician }
        }
        else = {
            eotg_aug_demand_removal_effect = { PROVIDER = backstreet }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = zealous }
            modifier = { add = 10  has_trait = arbitrary }
            modifier = { add = -20  has_trait = compassionate }
        }
    }

    # "A fine will do."
    option = {
        name = eotg_aug_realm.001.c
        trigger = {
            scope:eotg_contraband_subject = { gold >= minor_gold_value }
        }
        scope:eotg_contraband_subject = {
            pay_short_term_gold = {
                target = root
                gold = minor_gold_value
            }
        }
        eotg_aug_realm001_forgive_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = greedy }
            modifier = { add = 10  has_trait = just }
        }
    }

    # "Look the other way." Root breaks its own law.
    option = {
        name = eotg_aug_realm.001.d
        eotg_aug_realm001_forgive_effect = yes
        eotg_aug_stress_lie_effect = yes
        ai_chance = {
            base = 20
            modifier = { add = 20  has_trait = cynical }
            modifier = { add = 10  has_trait = lazy }
            modifier = { add = -20  has_trait = zealous }
        }
    }

    # [duchy+] "The realm's law holds, for everyone." As a, plus prestige.
    # W5 (event_quality_v1 §13.1 3b): was [just]; it moves prestige, which a
    # does not, so it cannot fold. Making the arrest a public example of the
    # law is an act of the seat, so a duchy tier gate, no icon (root is the
    # subject's liege; any landed liege below duke keeps a). just had no
    # stress row; it gets one here as a response (§13.1 step 4) and keeps
    # its AI weight.
    option = {
        name = eotg_aug_realm.001.e
        trigger = { highest_held_title_tier >= tier_duchy }
        imprison_character_effect = {
            TARGET = scope:eotg_contraband_subject
            IMPRISONER = root
        }
        add_prestige = minor_prestige_value
        stress_impact = {
            just = minor_stress_impact_loss
        }
        ai_chance = {
            base = 10    # B1 lesson 4: tier gates show to every AI duke
            modifier = { add = 30  has_trait = just }
            modifier = { add = 10  has_trait = diligent }
            modifier = { add = -20  has_trait = arbitrary }
            modifier = { add = -10  has_trait = compassionate }
        }
    }

    # [greedy] "A fine. A large one."
    option = {
        name = eotg_aug_realm.001.f
        trigger = {
            has_trait = greedy
            scope:eotg_contraband_subject = { gold >= medium_gold_value }
        }
        trait = greedy
        scope:eotg_contraband_subject = {
            pay_short_term_gold = {
                target = root
                gold = medium_gold_value
            }
            add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
        }
        eotg_aug_realm001_forgive_effect = yes
        # W5 step 5 (event_quality_v1 §13.1): the kept personality gate,
        # paired the vanilla way (icon, trigger, stress row)
        stress_impact = {
            greedy = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = greedy }
            modifier = { add = 10  has_trait = arbitrary }
        }
    }
}
