# -- eotg_augmentation_inherit.txt --------------------------------------------
# A Fracturing Inheritance (docs/specs/cybernetics_v2_fracturing_inheritance.md).
# The ruler is sound; the primary heir is Neurofractured. The ruler decides
# what the dynasty does about it, over 3-5 events and 17 outcomes.
#
# eotg_aug_inherit.001 -- The Report                    (entry)
# eotg_aug_inherit.002 -- The Conversation
# eotg_aug_inherit.003 -- The Physicians
# eotg_aug_inherit.004 -- The Watch
# eotg_aug_inherit.005 -- The Council
# eotg_aug_inherit.006 -- The Terms
# eotg_aug_inherit.007 -- The Answer
# eotg_aug_inherit.008 -- The Ward
# eotg_aug_inherit.009 -- The Pattern                   (danger hub, once)
# eotg_aug_inherit.010 -- The Table
# eotg_aug_inherit.011 -- Passage Out                   (L6)
# eotg_aug_inherit.012 -- The Shuttle That Never Docked (L8)
# eotg_aug_inherit.013 -- Struck from the Line          (L7, or a reaction)
# eotg_aug_inherit.014 -- The Hand-Off                  (L9, L10)
# eotg_aug_inherit.015 -- The Offer
# eotg_aug_inherit.016 -- The List                      (L13)
# eotg_aug_inherit.017 -- The Long Watch                (L14)
# eotg_aug_inherit.018 -- The Last Audience             (L15, or .024)
# eotg_aug_inherit.019 -- The Quiet Room                (L12)
# eotg_aug_inherit.020 -- Two Machines                  (L16)
# eotg_aug_inherit.021 -- After the Table               (L1, L2, L3)
# eotg_aug_inherit.022 -- Kept Terms                    (L4)
# eotg_aug_inherit.023 -- The Locked Wing               (L5)
# eotg_aug_inherit.024 -- Still Breathing               (after a failed attempt)
# eotg_aug_inherit.025 -- The Seat                      (to the heir, after L15)
# eotg_aug_inherit.026 -- The Warden's Hand             (to the heir, after L9 / L10)
# eotg_aug_inherit.027 -- The Line Shortens             (L11)
# eotg_aug_inherit.028 -- (hidden) Resolution           (AI rulers, owner Q5)
#
# Fires from: eotg_aug_inherit_start_effect -> .001 (player ruler) or .028
#   (AI ruler); the start effect is called by eotg_on_yearly_aug_inherit_check
#   (yearly_playable_pulse) and eotg_aug_nr_cascade_effect (the non-ruler
#   cascade, for a primary heir). Every other event is a chain stage fired by
#   trigger_event from an earlier stage or a leaf / route effect in
#   common/scripted_effects/eotg_augmentation_effects.txt (each header names it).
# Cooldown: eotg_flag_aug_inh_cooldown lives in the start effect and the gate
#   eotg_aug_inherit_can_start only. No event trigger here reads it (lesson 5);
#   triggers are the world-state guard eotg_aug_inherit_chain_valid.
# Cleanup: every ruler-root event has on_trigger_fail = the close effect, and
#   every leaf calls it.
# Resource: the heir's eotg_fracture_risk (hidden). Every non-terminal event
#   moves it in at least one option; the heir's band picks descs (.003, .022)
#   and weights the rolls; leaves change a tier (death, excision, the ruler's
#   cascade). No number, "risk" or odds in any tooltip (index §1 rule 3).
# Loc: dot-form keys, localization/english/eotg_aug_inherit_l_english.yml.
#   Profile names (Running Hot, The Ledger, Flatline, Certainty) are
#   internal only (spec §7.1).
# AI: the tree runs only for player rulers (eotg_aug_inherit_tree_runs); the
#   ai_chance blocks matter only if the ruler passes to the AI mid-chain.
# -----------------------------------------------------------------------------

namespace = eotg_aug_inherit
