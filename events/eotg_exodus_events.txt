namespace = eotg_exodus

# A battered refugee fleet limps into the ruler's home systems.
eotg_exodus.0001 = {
	type = character_event
	title = eotg_exodus_0001_title
	theme = realm

	desc = eotg_exodus_0001_desc

	left_portrait = {
		character = root
		animation = worry
	}

	immediate = {
		capital_county = { save_scope_as = eotg_exodus_county }
	}

	# Shelter them — pay for their settlement, strain now, growth later.
	option = {
		name = eotg_exodus_0001_opt_shelter
		remove_short_term_gold = 50
		scope:eotg_exodus_county = {
			add_county_modifier = {
				modifier = eotg_exodus_refugee_strain
				years = 3
			}
		}
		if = {
			limit = { has_trait = compassionate }
			add_stress = -15
		}
		hidden_effect = {
			trigger_event = { id = eotg_exodus.0002  days = { 360 720 } }
		}
		ai_chance = {
			base = 40
			modifier = { add = 20  gold > 300 }
			modifier = { add = 20  has_trait = compassionate }
			modifier = { add = -20  has_trait = greedy }
		}
	}

	# Turn them away — no cost, but the desperate do desperate things.
	option = {
		name = eotg_exodus_0001_opt_refuse
		add_prestige = -25
		if = {
			limit = { has_trait = compassionate }
			add_stress = 20
		}
		hidden_effect = {
			random = {
				chance = 35
				trigger_event = { id = eotg_exodus.0003  days = { 90 180 } }
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20  has_trait = callous }
			modifier = { add = -20  has_trait = compassionate }
		}
	}

	# Press them into service — manpower now, resentment later.
	option = {
		name = eotg_exodus_0001_opt_press
		add_character_modifier = {
			modifier = eotg_exodus_pressed_levies
			years = 5
		}
		add_dread = 10
		if = {
			limit = {
				OR = {
					has_trait = compassionate
					has_trait = just
				}
			}
			add_stress = 20
		}
		ai_chance = {
			base = 30
			modifier = { add = 20  has_trait = sadistic }
			modifier = { add = 10  has_trait = ambitious }
			modifier = { add = -20  has_trait = just }
		}
	}
}
