# -- 0001 The Scar Reopens (contact) -------------------------------------------
eotg_void.0001 = {
    type = character_event
    title = eotg_void.0001.t
    desc = eotg_void.0001.desc
    theme = disaster
    override_background = { reference = eotg_bg_barren_world }
    left_portrait = {
        character = root
        animation = disbelief
    }

    # [education: learning] Survey it properly.
    option = {
        name = eotg_void.0001.a
        trigger = { has_trait = education_learning }
        skill = learning
        add_learning_skill = 1
        eotg_void_add_ward_effect = { YEARS = 5 }
        eotg_void_seed_effect = yes
        eotg_void_change_exposure_effect = { AMOUNT = 5 }
        ai_chance = {
            base = 60
            modifier = { add = 20  has_trait = curious }
        }
    }

    # [tier: duchy] Wall it off. Re-gated from martial/stewardship education:
    # an order to guards is authority (event_quality_v1.md §13.1 3b).
    option = {
        name = eotg_void.0001.b
        trigger = { highest_held_title_tier >= tier_duchy }
        remove_short_term_gold = 50
        add_prestige = 40
        eotg_void_seed_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = paranoid }
            modifier = { add = -30  gold < 50 }
            modifier = { add = -15  has_trait = brave }
        }
    }

    # [brave] Go in myself. The cult also opens it (inert, ruling R3).
    option = {
        name = eotg_void.0001.c
        trigger = {
            OR = {
                has_trait = brave
                eotg_void_is_cult_faith = yes
            }
        }
        trait = brave
        eotg_void_seed_effect = yes
        eotg_void_change_exposure_effect = { AMOUNT = 20 }
        add_prestige = 75
        stress_impact = {
            base = minor_stress_impact_gain
            brave = minor_stress_impact_loss
            craven = medium_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 40  eotg_void_is_cult_faith = yes }
            modifier = { add = 25  has_trait = brave }
        }
    }

    # Rebuild the barrier and say nothing.
    option = {
        name = eotg_void.0001.d
        eotg_void_seed_effect = yes
        eotg_void_change_exposure_effect = { AMOUNT = 10 }
        stress_impact = { base = minor_stress_impact_gain }
        ai_chance = {
            base = 20
            modifier = { add = 30  has_trait = craven }
            modifier = { add = 20  has_trait = lazy }
        }
    }
}

# -- 0004 The Voidwalker at the Gate -----------------------------------------------
# Fixed (ruling R5): the Voidwalker is a real character, created here and
# seated by option a, flagged so eotg_void_has_voidwalker matches. A seated
# Voidwalker ends the re-rolls (the on_action's gate). Reframed (ruling R4):
# a warrior of the Malvrick Orders who contains Void relics, not a hunted
# heretic (REVIEW_866.md:64; the SETTING LORE body's hunts are outranked).
eotg_void.0004 = {
    type = character_event
    title = eotg_void.0004.t
    desc = eotg_void.0004.desc
    theme = secret
    override_background = { reference = eotg_bg_corridor_night }
    left_portrait = {
        character = root
        animation = scheme
    }
    right_portrait = {
        character = scope:eotg_void_voidwalker
        animation = personality_rational
    }

    immediate = {
        hidden_effect = {
            eotg_void_create_voidwalker_effect = yes
        }
    }

    # Give them a place at court.
    option = {
        name = eotg_void.0004.a
        eotg_void_shelter_voidwalker_effect = yes
        eotg_void_change_exposure_effect = { AMOUNT = -10 }
        add_piety = -100
        every_courtier = {
            limit = { has_trait = zealous }
            add_opinion = { modifier = eotg_opinion_void_shelters_voidwalker  target = root }
        }
        ai_chance = {
            base = 30
            modifier = { add = 40  eotg_void_is_cult_faith = yes }
            modifier = { add = 25  has_trait = cynical }
            modifier = { add = 20  has_trait = education_learning }
            modifier = { add = -60  has_trait = zealous }
        }
    }

    # [education: learning or diplomacy] Hear them out, take the ward, let
    # them go on.
    option = {
        name = eotg_void.0004.b
        trigger = {
            OR = {
                has_trait = education_learning
                has_trait = education_diplomacy
            }
        }
        add_learning_skill = 1
        eotg_void_change_exposure_effect = { AMOUNT = -5 }
        eotg_void_add_ward_effect = { YEARS = 6 }
        ai_chance = {
            base = 50
            modifier = { add = 20  has_trait = patient }
        }
    }

    # Turn them away in front of the clergy.
    option = {
        name = eotg_void.0004.c
        add_piety = 150
        add_prestige = 50
        eotg_void_change_exposure_effect = { AMOUNT = 5 }
        ai_chance = {
            base = 40
            modifier = { add = 50  has_trait = zealous }
            modifier = { add = 20  has_trait = just }
            modifier = { add = -50  eotg_void_is_cult_faith = yes }
        }
    }

    # Have them killed at the gate. Made real: v1 killed no one.
    # death shape: events/eotg_augmentation_kingpin.txt:1135-1138.
    option = {
        name = eotg_void.0004.d
        scope:eotg_void_voidwalker = {
            death = {
                death_reason = death_murder
                killer = root
            }
        }
        add_dread = 20
        add_piety = 75
        eotg_void_change_exposure_effect = { AMOUNT = 8 }
        stress_impact = {
            compassionate = medium_stress_impact_gain
            just = minor_stress_impact_gain
        }
        ai_chance = {
            base = 15
            modifier = { add = 35  has_trait = callous }
            modifier = { add = 30  has_trait = sadistic }
            modifier = { add = -50  has_trait = compassionate }
        }
    }
}
