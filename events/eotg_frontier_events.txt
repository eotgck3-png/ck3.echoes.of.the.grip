# -- eotg_frontier_events.txt -------------------------------------------------
# Frontier Systems v1 events (docs/specs/frontier_v1.md §9): 6 core events
# (owner ruling Q2). Every event reads or moves Establishment
# (var:eotg_frontier_progress) or its pressure (var:eotg_frontier_strain):
# CLAUDE.md invariant 5.
#
# Every event is fired by something (invariant 4):
#   .001  eotg_decision_frontier_establish
#   .002  eotg_frontier_event_roll_effect (yearly tick; player holders)
#   .003  eotg_frontier_event_roll_effect, or eotg_frontier.010
#   .004  eotg_frontier_tick_effect (completion)
#   .005  eotg_frontier_tick_effect (strain 4)
#   .010  eotg_decision_frontier_sponsor
#   Phase 2 (docs/specs/frontier_v2.md §9), at the end of the file:
#   .020  eotg_decision_frontier_survey
#   .021  eotg_decision_frontier_build
#   .030-.036  eotg_frontier_flavor_pick_effect (the same yearly roll)
# Event triggers are world-state guards only; the flavor cooldown
# (eotg_frontier_event_cd) lives in the tick and is never read here.
# Root is always a character (the Region's holder, or the decision taker);
# the Region arrives as scope:eotg_frontier_county.
#
# theme = realm: confirmed (verification §3). Every option that changes a
# Region re-checks its state (verification Q-B1), and each event that can be
# left open while the state moves on has a "the moment has passed" option, so
# it never opens with nothing to click. Gains are halved with the doubled
# pacing and strain bands follow the 0-8 scale (verification §4).
# -----------------------------------------------------------------------------

namespace = eotg_frontier

# -- eotg_frontier.001 --------------------------------------------------------
# An Opportunity Here: the start. Pick the project type; the cost is paid here
# (the decision has no `cost`). The founder candidate was picked by the
# decision (the best steward at court, or the taker).
eotg_frontier.001 = {
    type = character_event
    title = eotg_frontier.001.t
    desc = {
        desc = eotg_frontier.001.desc
        # Lore I1: never tell the player about themselves in the third person
        first_valid = {
            triggered_desc = {
                trigger = {
                    exists = scope:eotg_frontier_candidate
                    NOT = { scope:eotg_frontier_candidate = root }
                }
                desc = eotg_frontier.001.desc_other
            }
            desc = eotg_frontier.001.desc_self
        }
        triggered_desc = {
            trigger = {
                scope:eotg_frontier_county = { eotg_frontier_is_abandoned = yes }
            }
            desc = eotg_frontier.001.desc_resettle
        }
        # Expansion B2 (frontier_v2.md §B2)
        triggered_desc = {
            trigger = {
                scope:eotg_frontier_county = { has_variable = eotg_frontier_salvage }
            }
            desc = eotg_frontier.001.desc_salvage
        }
        # Phase 2 (frontier_v2.md §9.3): what is already known shapes the choice
        triggered_desc = {
            trigger = {
                scope:eotg_frontier_county = { eotg_frontier_trait_count_value >= 1 }
            }
            desc = eotg_frontier.001.desc_known
        }
    }
    theme = realm

    trigger = {
        exists = scope:eotg_frontier_county
        scope:eotg_frontier_county = { eotg_frontier_can_establish = yes }
    }

    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        trigger = {
            exists = scope:eotg_frontier_candidate
            NOT = { scope:eotg_frontier_candidate = root }
        }
        character = scope:eotg_frontier_candidate
        animation = personality_bold
    }

    # "Settlers first."
    option = {
        name = eotg_frontier.001.a
        trigger = {
            gold >= eotg_frontier_establish_cost_value
            scope:eotg_frontier_county = { eotg_frontier_can_establish = yes }
        }
        remove_short_term_gold = eotg_frontier_establish_cost_value
        scope:eotg_frontier_county = {
            eotg_frontier_start_effect = { TYPE = settlement  FOUNDER = scope:eotg_frontier_candidate }
        }
        eotg_frontier_start_tooltip_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  scope:eotg_frontier_county = { has_county_modifier = eotg_frontier_trait_habitable } }
        }
    }

    # "A trading post."
    option = {
        name = eotg_frontier.001.b
        trigger = {
            gold >= eotg_frontier_establish_cost_value
            scope:eotg_frontier_county = { eotg_frontier_can_establish = yes }
        }
        remove_short_term_gold = eotg_frontier_establish_cost_value
        scope:eotg_frontier_county = {
            eotg_frontier_start_effect = { TYPE = trade  FOUNDER = scope:eotg_frontier_candidate }
        }
        eotg_frontier_start_tooltip_effect = yes
        ai_chance = {
            base = 15
            modifier = { add = 10  has_trait = greedy }
            modifier = { add = 30  scope:eotg_frontier_county = { has_county_modifier = eotg_frontier_trait_trade_route } }
        }
    }

    # "Mine it."
    option = {
        name = eotg_frontier.001.c
        trigger = {
            gold >= eotg_frontier_establish_cost_value
            scope:eotg_frontier_county = { eotg_frontier_can_establish = yes }
        }
        remove_short_term_gold = eotg_frontier_establish_cost_value
        scope:eotg_frontier_county = {
            eotg_frontier_start_effect = { TYPE = mining  FOUNDER = scope:eotg_frontier_candidate }
        }
        eotg_frontier_start_tooltip_effect = yes
        ai_chance = {
            base = 15
            modifier = { add = 30  scope:eotg_frontier_county = { has_county_modifier = eotg_frontier_trait_rich_resources } }
        }
    }

    # "Hold it."
    option = {
        name = eotg_frontier.001.d
        # Phase 2 (frontier_v2.md §6.1): a garrison costs standing as well
        trigger = {
            gold >= eotg_frontier_establish_cost_value
            prestige >= minor_prestige_value
            scope:eotg_frontier_county = { eotg_frontier_can_establish = yes }
        }
        remove_short_term_gold = eotg_frontier_establish_cost_value
        add_prestige = {
            value = minor_prestige_value
            multiply = -1
        }
        scope:eotg_frontier_county = {
            eotg_frontier_start_effect = { TYPE = military  FOUNDER = scope:eotg_frontier_candidate }
        }
        eotg_frontier_start_tooltip_effect = yes
        ai_chance = {
            base = 10
            modifier = { add = 15  martial >= 12 }
            modifier = { add = 10  is_at_war = yes }
            modifier = {
                add = 30
                scope:eotg_frontier_county = {
                    OR = {
                        has_county_modifier = eotg_frontier_trait_strategic
                        has_county_modifier = eotg_frontier_trait_dangerous
                    }
                }
            }
        }
    }

    # "A place of faith."
    option = {
        name = eotg_frontier.001.e
        # Phase 2 (§6.1): a mission costs devotion as well
        trigger = {
            gold >= eotg_frontier_establish_cost_value
            piety >= minor_piety_value
            scope:eotg_frontier_county = { eotg_frontier_can_establish = yes }
        }
        remove_short_term_gold = eotg_frontier_establish_cost_value
        add_piety = {
            value = minor_piety_value
            multiply = -1
        }
        scope:eotg_frontier_county = {
            eotg_frontier_start_effect = { TYPE = religious  FOUNDER = scope:eotg_frontier_candidate }
        }
        eotg_frontier_start_tooltip_effect = yes
        ai_chance = {
            base = 5
            modifier = { add = 20  has_trait = zealous }
        }
    }

    # "Study what is out there."
    option = {
        name = eotg_frontier.001.f
        # Phase 2 (§6.1): someone has to be able to read the surveys
        trigger = {
            gold >= eotg_frontier_establish_cost_value
            OR = {
                learning >= decent_skill_rating
                scope:eotg_frontier_candidate ?= { learning >= decent_skill_rating }
            }
            scope:eotg_frontier_county = { eotg_frontier_can_establish = yes }
        }
        remove_short_term_gold = eotg_frontier_establish_cost_value
        scope:eotg_frontier_county = {
            eotg_frontier_start_effect = { TYPE = research  FOUNDER = scope:eotg_frontier_candidate }
        }
        eotg_frontier_start_tooltip_effect = yes
        ai_chance = {
            base = 5
            modifier = { add = 15  learning >= 12 }
            modifier = {
                add = 30
                scope:eotg_frontier_county = {
                    OR = {
                        has_county_modifier = eotg_frontier_trait_anomalous
                        has_county_modifier = eotg_frontier_trait_ancient_ruins
                    }
                }
            }
        }
    }

    # "Bring it into the realm's order."
    option = {
        name = eotg_frontier.001.g
        trigger = {
            gold >= eotg_frontier_establish_cost_value
            scope:eotg_frontier_county = { eotg_frontier_admin_eligible = yes }
        }
        remove_short_term_gold = eotg_frontier_establish_cost_value
        scope:eotg_frontier_county = {
            eotg_frontier_start_effect = { TYPE = administrative  FOUNDER = scope:eotg_frontier_candidate }
        }
        eotg_frontier_start_tooltip_effect = yes
        ai_chance = {
            base = 5
            modifier = { add = 10  stewardship >= 14 }
        }
    }

    # "Not now."
    option = {
        name = eotg_frontier.001.h
        ai_chance = {
            base = 0
        }
    }
}

# -- eotg_frontier.002 --------------------------------------------------------
# A Hard Season (complication) / The Frontier Without a Founder (variant,
# owner ruling Q2: no founder-death event; an invalid founder is the strain
# cause No Founder, and this is where the player replaces them).
# The desc reads strain by band; no number is ever shown.
eotg_frontier.002 = {
    type = character_event
    title = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:eotg_frontier_county = { eotg_frontier_has_valid_founder = no }
                }
                desc = eotg_frontier.002.t_no_founder
            }
            desc = eotg_frontier.002.t
        }
    }
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:eotg_frontier_county = { eotg_frontier_has_valid_founder = no }
                }
                desc = eotg_frontier.002.desc_no_founder
            }
            triggered_desc = {
                trigger = {
                    scope:eotg_frontier_county = { var:eotg_frontier_strain >= 6 }
                }
                desc = eotg_frontier.002.desc_high
            }
            triggered_desc = {
                trigger = {
                    scope:eotg_frontier_county = { var:eotg_frontier_strain >= 4 }
                }
                desc = eotg_frontier.002.desc_mid
            }
            desc = eotg_frontier.002.desc_low
        }
        # Expansion B1 (frontier_v2.md §B1): the place itself is part of the trouble
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:eotg_frontier_county = {
                        eotg_frontier_has_valid_founder = yes
                        has_county_modifier = eotg_frontier_trait_hostile
                        NOT = { has_county_modifier = eotg_frontier_infra_orbital_station }
                    }
                }
                desc = eotg_frontier.002.desc_hostile
            }
            triggered_desc = {
                trigger = {
                    scope:eotg_frontier_county = {
                        eotg_frontier_has_valid_founder = yes
                        has_county_modifier = eotg_frontier_trait_remote
                        NOT = { has_county_modifier = eotg_frontier_infra_navigation_beacon }
                    }
                }
                desc = eotg_frontier.002.desc_remote
            }
            triggered_desc = {
                trigger = {
                    scope:eotg_frontier_county = {
                        eotg_frontier_has_valid_founder = yes
                        has_county_modifier = eotg_frontier_trait_dangerous
                        NOT = { has_county_modifier = eotg_frontier_infra_garrison_post }
                    }
                }
                desc = eotg_frontier.002.desc_dangerous
            }
        }
    }
    theme = realm

    trigger = {
        exists = scope:eotg_frontier_county
        scope:eotg_frontier_county = { eotg_frontier_is_frontier = yes }
    }

    left_portrait = {
        character = root
        animation = worry
    }
    right_portrait = {
        trigger = {
            exists = scope:eotg_frontier_candidate
            NOT = { scope:eotg_frontier_candidate = root }
        }
        character = scope:eotg_frontier_candidate
        animation = personality_bold
    }

    immediate = {
        hidden_effect = {
            # A replacement founder for the variant: the best steward at
            # court who is not the old founder
            if = {
                limit = {
                    scope:eotg_frontier_county = { eotg_frontier_has_valid_founder = no }
                }
                ordered_courtier = {
                    limit = {
                        is_adult = yes
                        is_imprisoned = no
                        NOT = { scope:eotg_frontier_county = { var:eotg_frontier_founder ?= prev } }
                    }
                    order_by = stewardship
                    save_scope_as = eotg_frontier_candidate
                }
            }
        }
    }

    # "Ship in supplies."
    option = {
        name = eotg_frontier.002.a
        trigger = {
            scope:eotg_frontier_county = {
                eotg_frontier_is_frontier = yes
                eotg_frontier_has_valid_founder = yes
            }
            gold >= minor_gold_value
        }
        remove_short_term_gold = minor_gold_value
        scope:eotg_frontier_county = { eotg_frontier_ease_strain_effect = yes }
        custom_tooltip = eotg_frontier.002.ease_tt
        ai_chance = {
            base = 40
            modifier = { add = -20  has_trait = greedy }
        }
    }

    # "Push through." The ungated fallback (error.log qa_noopt: an event must
    # never open with no selectable option), so it is offered in the Without
    # a Founder variant too. Its effects are guarded instead: a Region that
    # left the Frontier state while the event was open gets nothing.
    option = {
        name = eotg_frontier.002.b
        scope:eotg_frontier_county = {
            if = {
                limit = { eotg_frontier_is_frontier = yes }
                add_county_modifier = { modifier = eotg_frontier_mod_hard_year  years = 2 }
                eotg_frontier_add_progress_effect = { AMOUNT = 4 }
                eotg_frontier_add_strain_effect = { CAUSE = hardship }
                custom_tooltip = eotg_frontier.002.push_tt
            }
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = stubborn }
        }
    }

    # (a sponsor who can pay) "Call on the backer."
    option = {
        name = eotg_frontier.002.c
        trigger = {
            scope:eotg_frontier_county = {
                eotg_frontier_is_frontier = yes
                eotg_frontier_has_valid_founder = yes
                eotg_frontier_has_valid_sponsor = yes
                var:eotg_frontier_sponsor = { gold >= eotg_frontier_sponsor_payment_value }
            }
        }
        scope:eotg_frontier_county = {
            var:eotg_frontier_sponsor = {
                remove_short_term_gold = eotg_frontier_sponsor_payment_value
            }
            eotg_frontier_ease_strain_effect = yes
        }
        custom_tooltip = eotg_frontier.002.ease_tt
        ai_chance = {
            base = 40
        }
    }

    # (no founder; a candidate at court) "Appoint them."
    option = {
        name = eotg_frontier.002.d
        trigger = {
            scope:eotg_frontier_county = {
                eotg_frontier_is_frontier = yes
                eotg_frontier_has_valid_founder = no
            }
            exists = scope:eotg_frontier_candidate
        }
        scope:eotg_frontier_county = {
            eotg_frontier_set_founder_effect = { FOUNDER = scope:eotg_frontier_candidate }
        }
        custom_tooltip = eotg_frontier.002.founder_tt
        ai_chance = {
            base = 50
        }
    }

    # (no founder) "Lead it yourself."
    option = {
        name = eotg_frontier.002.e
        trigger = {
            scope:eotg_frontier_county = {
                eotg_frontier_is_frontier = yes
                eotg_frontier_has_valid_founder = no
            }
        }
        scope:eotg_frontier_county = {
            eotg_frontier_set_founder_effect = { FOUNDER = root }
        }
        custom_tooltip = eotg_frontier.002.founder_tt
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = diligent }
        }
    }

    # (no founder) "They will manage." Strain keeps building.
    option = {
        name = eotg_frontier.002.f
        trigger = {
            scope:eotg_frontier_county = {
                eotg_frontier_is_frontier = yes
                eotg_frontier_has_valid_founder = no
            }
        }
        custom_tooltip = eotg_frontier.002.f_tt
        ai_chance = {
            base = 5
            modifier = { add = 20  has_trait = lazy }
        }
    }
    # The Region is no longer in this state (it moved on while the event was
    # open): a way out with no effect (verification Q-B1)
    option = {
        name = eotg_frontier.option_passed
        trigger = {
            scope:eotg_frontier_county = { eotg_frontier_is_frontier = no }
        }
        ai_chance = {
            base = 100
        }
    }
}

# -- eotg_frontier.003 --------------------------------------------------------
# An Offer of Backing. scope:eotg_frontier_sponsor offers; the holder (root)
# decides. Sponsorship gives the backer no claim and no ownership (spec §R).
eotg_frontier.003 = {
    type = character_event
    title = eotg_frontier.003.t
    desc = {
        desc = eotg_frontier.003.desc
        triggered_desc = {
            trigger = {
                scope:eotg_frontier_county = { exists = var:eotg_frontier_sponsor }
            }
            desc = eotg_frontier.003.desc_replace
        }
    }
    theme = realm

    trigger = {
        exists = scope:eotg_frontier_county
        exists = scope:eotg_frontier_sponsor
        scope:eotg_frontier_county = { eotg_frontier_is_frontier = yes }
        scope:eotg_frontier_sponsor = { is_alive = yes }
        NOT = { scope:eotg_frontier_sponsor = root }
    }

    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        character = scope:eotg_frontier_sponsor
        animation = personality_honorable
    }

    # "Accept the backing."
    option = {
        name = eotg_frontier.003.a
        trigger = {
            scope:eotg_frontier_county = { eotg_frontier_is_frontier = yes }
            scope:eotg_frontier_sponsor = { eotg_frontier_can_sponsor = yes }
        }
        scope:eotg_frontier_county = {
            eotg_frontier_set_sponsor_effect = { SPONSOR = scope:eotg_frontier_sponsor }
            eotg_frontier_add_progress_effect = { AMOUNT = 3 }
        }
        custom_tooltip = eotg_frontier.003.accept_tt
        hidden_effect = {
            if = {
                limit = { exists = scope:eotg_frontier_old_sponsor }
                scope:eotg_frontier_old_sponsor = {
                    send_interface_toast = {
                        title = eotg_frontier_toast_sponsor_replaced
                        left_icon = root
                    }
                }
            }
        }
        ai_chance = {
            base = 50
            modifier = {
                add = 30
                opinion = { target = scope:eotg_frontier_sponsor  value >= 20 }
            }
            modifier = {
                factor = 0
                opinion = { target = scope:eotg_frontier_sponsor  value < -20 }
            }
        }
    }

    # "We will manage without them."
    option = {
        name = eotg_frontier.003.b
        custom_tooltip = eotg_frontier.003.refuse_tt
        ai_chance = {
            base = 20
            modifier = { add = 20  has_trait = arrogant }
        }
    }
}

# -- eotg_frontier.042 --------------------------------------------------------
# The Expedition Returns (spec §4): one step of exploration, run in immediate.
# At Known, the charts are filed as a survey (Phase 2's
# eotg_frontier_file_survey_effect): +2 progress on a running Frontier, a head
# start for the next project otherwise. That is how the event couples to progress.
eotg_frontier.042 = {
    type = character_event
    title = eotg_frontier.042.t
    desc = {
        desc = eotg_frontier.042.desc
        first_valid = {
            triggered_desc = {
                trigger = { scope:eotg_frontier_exploration ?= flag:known }
                desc = eotg_frontier.042.desc_known
            }
            triggered_desc = {
                trigger = { scope:eotg_frontier_exploration ?= flag:partial }
                desc = eotg_frontier.042.desc_partial
            }
            desc = eotg_frontier.042.desc_nothing
        }
    }
    theme = realm

    trigger = {
        exists = scope:eotg_frontier_county
    }

    left_portrait = {
        character = root
        animation = personality_rational
    }

    immediate = {
        hidden_effect = {
            clear_saved_scope = eotg_frontier_exploration
            scope:eotg_frontier_county = { eotg_frontier_explore_step_effect = yes }
        }
    }

    # "Send them straight back out." From Partial, with gold: on to Known now.
    option = {
        name = eotg_frontier.042.b
        trigger = {
            gold >= eotg_frontier_explore_cost_value
            scope:eotg_frontier_county = { eotg_frontier_is_partial = yes }
        }
        remove_short_term_gold = eotg_frontier_explore_cost_value
        custom_tooltip = eotg_frontier.042.b_tt
        scope:eotg_frontier_county = {
            eotg_frontier_explore_step_effect = yes
            # At Known, the charts are filed as a survey: +2 progress on a
            # Frontier, survey data otherwise (Phase 2, never data on a Frontier)
            if = {
                limit = { scope:eotg_frontier_exploration ?= flag:known }
                eotg_frontier_file_survey_effect = yes
            }
        }
        ai_chance = {
            base = 20
            modifier = { add = 10  learning >= 12 }
        }
    }

    # "File the charts." Always available; at Known the charts are a head start.
    option = {
        name = eotg_frontier.042.a
        scope:eotg_frontier_county = {
            if = {
                limit = { scope:eotg_frontier_exploration ?= flag:known }
                eotg_frontier_file_survey_effect = yes
            }
            else_if = {
                limit = { scope:eotg_frontier_exploration ?= flag:partial }
                custom_tooltip = eotg_frontier.042.a_partial_tt
            }
        }
        ai_chance = {
            base = 50
        }
    }
}

# -- eotg_frontier.040 --------------------------------------------------------
# Guns for Hire (spec §5): a mercenary company offers to guard the convoys of a
# Dangerous Region, or, for a Military project with no backer, to back the venture. The captain
# gains no claim and no say (LAW AT 866). Comes through the yearly roll's pool
# for a Military project with no backer, or a Dangerous Region of any other
# project with no escort.
eotg_frontier.040 = {
    type = character_event
    title = eotg_frontier.040.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:eotg_frontier_company }
                desc = eotg_frontier.040.desc
            }
            desc = eotg_frontier.040.desc_none
        }
    }
    theme = realm

    trigger = {
        exists = scope:eotg_frontier_county
        scope:eotg_frontier_county = {
            eotg_frontier_is_frontier = yes
            OR = {
                var:eotg_frontier_type ?= flag:military
                has_county_modifier = eotg_frontier_trait_dangerous
            }
        }
    }

    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        trigger = { exists = scope:eotg_frontier_company }
        character = scope:eotg_frontier_company
        animation = personality_bold
    }

    immediate = {
        hidden_effect = {
            if = {
                limit = { NOT = { exists = scope:eotg_frontier_company } }
                scope:eotg_frontier_county = { eotg_frontier_find_company_effect = yes }
            }
        }
    }

    # (a Dangerous Region, a project that is not Military) "Hire them to guard
    # the convoys." The fee goes to the captain: the amount is fixed once, in
    # the payer's scope, so both sides see the same number (a scope value as a
    # gold amount: confirmed, spec frontier_v3.md §15 item 8).
    option = {
        name = eotg_frontier.040.a
        trigger = {
            exists = scope:eotg_frontier_company
            gold >= eotg_frontier_escort_cost_value
            scope:eotg_frontier_county = {
                eotg_frontier_is_frontier = yes
                has_county_modifier = eotg_frontier_trait_dangerous
                NOT = { var:eotg_frontier_type ?= flag:military }
                NOT = { has_county_modifier = eotg_frontier_mod_merc_escort }
            }
        }
        save_temporary_scope_value_as = { name = eotg_frontier_fee  value = eotg_frontier_escort_cost_value }
        pay_short_term_gold = { target = scope:eotg_frontier_company  gold = scope:eotg_frontier_fee }
        scope:eotg_frontier_county = {
            add_county_modifier = { modifier = eotg_frontier_mod_merc_escort  years = 5 }
            eotg_frontier_ease_strain_effect = yes
        }
        custom_tooltip = eotg_frontier.040.a_tt
        ai_chance = {
            base = 40
        }
    }

    # (a Military project with no backer) "Let the company back the venture."
    option = {
        name = eotg_frontier.040.b
        trigger = {
            exists = scope:eotg_frontier_company
            scope:eotg_frontier_company = { eotg_frontier_can_back = yes }
            scope:eotg_frontier_county = {
                eotg_frontier_is_frontier = yes
                var:eotg_frontier_type ?= flag:military
                NOT = { exists = var:eotg_frontier_sponsor }
            }
        }
        scope:eotg_frontier_county = {
            eotg_frontier_set_sponsor_effect = { SPONSOR = scope:eotg_frontier_company }
            eotg_frontier_add_progress_effect = { AMOUNT = 3 }
        }
        custom_tooltip = eotg_frontier.040.b_tt
        ai_chance = {
            base = 30
        }
    }

    # "We will hold it ourselves." Always available; changes nothing.
    option = {
        name = eotg_frontier.040.c
        ai_chance = {
            base = 20
            modifier = { add = 20  has_trait = arrogant }
        }
    }
}
