# ── eotg_augmentation_initiation.txt ─────────────────────────────────────────
# Entry events for the Cybernetic Augmentation system.
#
# eotg_aug_init.001 — The Cost of Survival     (post-battle wound offer)
# eotg_aug_init.002 — A Corporate Offer        (wealthy landed ruler)
# eotg_aug_init.003 — Aging Hands              (aged ruler with low prowess)
#
# eotg_aug_init.004 — Desperation Protocol     (at war + wounded)
# eotg_aug_init.005 — A Familiar Change        (augmented peer at court)
# eotg_aug_init.019 — The Arms Race            (Phase 5: two augmented knights)
#
# Fires from: eotg_on_yearly_aug_initiation_check
#   (common/on_action/eotg_augmentation_on_actions.txt, via yearly_playable_pulse).
# Cooldown eotg_flag_aug_event_cooldown is set and checked there only.
# Resource: every accept option adds eotg_cybernetics at XP 0 (eotg_aug_initiate_effect)
# and sets eotg_fracture_risk.
#
# cybernetics_v2 Phase 1 (docs/specs/cybernetics_v2_phase1.md §1): trait-gated
# options (trigger has_trait + trait icon, vanilla form), stress helpers on the
# install/refuse options, scaled gold, lesson modifiers instead of permanent skill.
# Where an option had a bespoke stress line for a trait the helper also covers,
# the helper's line replaces it (spec "How to read the tables").
#
# Procedures spec §4.5 (docs/specs/cybernetics_v2_procedures.md): every ruler
# self-install runs the shared roll on the click, after its install lines,
# with a provider by fiction (eotg_aug_procedure_effect, hidden, plus the
# eotg_aug_proc.install_tt tooltip). Clinic: init.001 a/d/e, .002 a, .003
# a/d, .005 a/d, .006 a/e/f, .009 a/d, .016 a/e, .018 a, .019 a/d.
# Physician: init.006 c, .017 a/d, .018 c. Back streets: init.002 d, .004
# a/e (their existing illegal modifier and hidden flaw switch the matching
# outcomes off), and init.010 a/c/d and .018 b, which roll only and keep
# their own reveal (init.011). Not rolled: init.007/.008 (the Bridge's own
# settle stage), .013 (a parent's hardware: own flaw), .014/.015 (the child),
# .017 c and .002 c (a knight), .020 (a retainer, not root).
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_aug_init


# ── eotg_aug_init.001 ─────────────────────────────────────────────────────────
# The Cost of Survival
# A battle wound opens the door to cybernetic augmentation.

eotg_aug_init.001 = {
    type = character_event
    title = eotg_aug_init.001.t
    desc = eotg_aug_init.001.desc
    theme = battle
    override_background = { reference = eotg_bg_battlefield }

    left_portrait = {
        character = root
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = craven }
            animation = fear
        }
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = zealous }
            animation = disapproval
        }
        animation = stress
    }

    # Install guard, as the Phase 2 install events (closes a double install
    # via Seek Augmentation in the days before this fires).
    trigger = { eotg_is_augmented_any = no }

    # CB-42 (docs/specs/cybernetics_v2_seller_names.md §5.1): the seller, rolled
    # once into a character variable before the desc and options render (no
    # random_valid: the name never changes while the window is open)
    immediate = {
        hidden_effect = {
            eotg_aug_roll_company_effect = { VAR = eotg_aug_seller_company }
        }
    }
    # Accept — receive the augmentation, remove the wound
    option = {
        name = eotg_aug_init.001.a
        eotg_aug_initiate_effect = yes
        eotg_aug_heal_wounds_effect = yes
        hidden_effect = {
            set_variable = { name = eotg_fracture_risk  value = 0 }
        }
        custom_tooltip = eotg_aug_proc.install_tt
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = clinic  PROCEDURE = install  SURGEON = root }
            eotg_aug_record_clinic_effect = yes    # CB-42: clinic of record
        }
        stress_impact = {
            base = minor_stress_impact_loss
            content = minor_stress_impact_gain
        }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 50
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  has_trait = brave }
            modifier = { add = -50  has_trait = zealous }
            modifier = { add = -20  has_trait = content }
        }
    }

    # Find another way — conventional care: 50% chance the wound eases one rank
    option = {
        name = eotg_aug_init.001.b
        random = {
            chance = 50
            eotg_aug_ease_wound_effect = yes
            send_interface_toast = {    # W3 (event_quality_v1)
                type = event_toast_effect_good
                title = eotg_aug_init.001.toast_b_eased
                left_icon = root
            }
        }
        stress_impact = {
            zealous = minor_stress_impact_loss
            ambitious = minor_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = zealous }
            modifier = { add = 10  has_trait = content }
        }
    }

    # Refuse entirely — zealous or principled characters only.
    # W5 (event_quality_v1 §13.1 step 2b): the event's one kept personality
    # gate (content is protected, 3 gates). Step 5 pairing: trait icons and
    # stress rows for content and humble added (zealous had both).
    option = {
        name = eotg_aug_init.001.c
        trigger = {
            OR = { has_trait = zealous  has_trait = content  has_trait = humble }
        }
        trait = zealous
        trait = content
        trait = humble
        add_piety = 100
        stress_impact = {
            zealous = minor_stress_impact_loss
            content = minor_stress_impact_loss
            humble = minor_stress_impact_loss
            ambitious = medium_stress_impact_gain
        }
        ai_chance = {
            base = 20
            modifier = { add = 30  has_trait = zealous }
            modifier = { add = 20  has_trait = content }
        }
    }

    # [craven] "Anything, so I never feel that again." — as a, but the fear
    # buys more hardware than the wound needs: risk starts at 5.
    # Stress is written out rather than calling the surgery helper: the helper
    # gives craven a medium GAIN, and this option is the craven's relief. The
    # other surgery lines are kept (craven and brave are opposites).
    # W5 (event_quality_v1 §13.1 3a): was [craven]; it moves risk (5), so it
    # cannot fold. Re-axed as competence: a soldier's read of what the next
    # wound will cost, armouring past it, is martial. craven keeps its stress
    # row here as a response (§13.1 step 4); c is the kept personality gate.
    option = {
        name = eotg_aug_init.001.d
        trigger = { has_trait = education_martial }
        skill = martial
        eotg_aug_initiate_effect = yes
        eotg_aug_heal_wounds_effect = yes
        hidden_effect = {
            set_variable = { name = eotg_fracture_risk  value = 5 }
        }
        custom_tooltip = eotg_aug_proc.install_tt
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = clinic  PROCEDURE = install  SURGEON = root }
            eotg_aug_record_clinic_effect = yes    # CB-42: clinic of record
        }
        stress_impact = {
            craven = minor_stress_impact_loss
            zealous = medium_stress_impact_gain
            humble = minor_stress_impact_gain
            cynical = minor_stress_impact_loss
            ambitious = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = education_martial_3 }
            modifier = { add = 30  has_trait = education_martial_4 }
            modifier = { add = 30  has_trait = education_martial_5 }
            modifier = { add = 10  has_trait = craven }
            modifier = { add = -30  has_trait = zealous }
        }
    }

    # [education_diplomacy] "Flesh was only ever a first draft." — as a, plus
    # prestige for the bravado and a piety cost for saying it aloud.
    # W5 (event_quality_v1 §13.1 3a): was [cynical]; it moves prestige and
    # piety, so it cannot fold. Saying it aloud so the camp repeats it is
    # rhetoric, so diplomacy. cynical keeps its relief through the surgery
    # helper (§13.1 step 4); c is the kept personality gate.
    option = {
        name = eotg_aug_init.001.e
        trigger = { has_trait = education_diplomacy }
        skill = diplomacy
        eotg_aug_initiate_effect = yes
        eotg_aug_heal_wounds_effect = yes
        hidden_effect = {
            set_variable = { name = eotg_fracture_risk  value = 0 }
        }
        custom_tooltip = eotg_aug_proc.install_tt
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = clinic  PROCEDURE = install  SURGEON = root }
            eotg_aug_record_clinic_effect = yes    # CB-42: clinic of record
        }
        add_prestige = 50
        add_piety = -50
        stress_impact = {
            base = minor_stress_impact_loss
            content = minor_stress_impact_gain
        }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = education_diplomacy_3 }
            modifier = { add = 30  has_trait = education_diplomacy_4 }
            modifier = { add = 30  has_trait = education_diplomacy_5 }
            modifier = { add = 10  has_trait = cynical }
        }
    }
}


# ── eotg_aug_init.002 ─────────────────────────────────────────────────────────
# A Corporate Offer
# A medical syndicate approaches a wealthy landed ruler.

eotg_aug_init.002 = {
    type = character_event
    title = eotg_aug_init.002.t
    desc = eotg_aug_init.002.desc
    theme = stewardship_wealth_focus
    override_background = { reference = eotg_bg_council }

    left_portrait = {
        character = root
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = greedy }
            animation = interested
        }
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = zealous }
            animation = disapproval
        }
        animation = personality_rational
    }

    # Install guard, as the Phase 2 install events (closes a double install
    # via Seek Augmentation in the days before this fires).
    trigger = { eotg_is_augmented_any = no }

    # CB-42 (docs/specs/cybernetics_v2_seller_names.md §5.1): the seller, rolled
    # once into a character variable before the desc and options render (no
    # random_valid: the name never changes while the window is open)
    immediate = {
        hidden_effect = {
            eotg_aug_roll_company_effect = { VAR = eotg_aug_seller_company }
        }
    }
    # Accept the offer — a purchase, not a gift (Phase 1: the syndicate sells)
    option = {
        name = eotg_aug_init.002.a
        trigger = { gold >= medium_gold_value }
        remove_short_term_gold = medium_gold_value
        eotg_aug_initiate_effect = yes
        hidden_effect = {
            set_variable = { name = eotg_fracture_risk  value = 0 }
        }
        custom_tooltip = eotg_aug_proc.install_tt
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = clinic  PROCEDURE = install  SURGEON = root }
            eotg_aug_record_clinic_effect = yes    # CB-42: clinic of record
        }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 40
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  eotg_ai_wants_augmentation = yes }
            modifier = { add = -100  has_trait = zealous }
        }
    }

    # Decline — you learned how they sell (B6: lesson, not permanent skill)
    option = {
        name = eotg_aug_init.002.b
        add_character_modifier = { modifier = eotg_mod_aug_lesson_intrigue  years = 5 }
        stress_impact = {
            base = 0
            ambitious = minor_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = zealous }
            modifier = { add = 20  has_trait = content }
        }
    }

    # Offer it to a knight instead
    option = {
        name = eotg_aug_init.002.c
        trigger = { any_knight = { is_alive = yes  eotg_is_augmented_any = no } }
        random_knight = {
            limit = { is_alive = yes  eotg_is_augmented_any = no }
            eotg_aug_initiate_effect = yes
            # Start the knight's risk at 0, as tier1.004.a/.d do (Phase 1 QA)
            eotg_add_fracture_risk = { AMOUNT = 0 }
        }
        add_dread = 5
        # QA round 2: putting someone else under the knife first is the
        # cruelty helper, matching init.017.c "On a volunteer first."
        eotg_aug_stress_cruelty_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = callous }
            modifier = { add = -10  has_trait = compassionate }
        }
    }

    # [greedy] "Haggle them down." — cheaper hardware, and cheap hardware
    # comes due: thread T6 hidden flaw (+4 Overclocked accrual), risk starts at 5.
    option = {
        name = eotg_aug_init.002.d
        trigger = {
            has_trait = greedy
            gold >= minor_gold_value
        }
        trait = greedy
        remove_short_term_gold = minor_gold_value
        eotg_aug_initiate_effect = yes
        hidden_effect = {
            set_variable = { name = eotg_fracture_risk  value = 5 }
            add_character_flag = eotg_flag_aug_hidden_flaw
        }
        custom_tooltip = eotg_aug_proc.install_tt
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = root  PROVIDER = backstreet  PROCEDURE = install  SURGEON = root }
        }
        eotg_aug_stress_surgery_effect = yes
        # W5 §13.1 step 5: the kept personality gate gets its stress row
        stress_impact = {
            greedy = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = greedy }
            modifier = { add = -50  has_trait = zealous }
        }
    }
}
