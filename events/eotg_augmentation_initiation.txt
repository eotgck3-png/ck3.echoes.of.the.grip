# ── eotg_augmentation_initiation.txt ─────────────────────────────────────────
# Entry events for the Cybernetic Augmentation system.
#
# eotg_aug_init.001 — The Cost of Survival     (post-battle wound offer)
# eotg_aug_init.002 — A Corporate Offer        (wealthy landed ruler)
# eotg_aug_init.003 — Aging Hands              (aged ruler with low prowess)
#
# Fires from: eotg_augmentation_on_actions.txt yearly checks
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_aug_init


# ── eotg_aug_init.001 ─────────────────────────────────────────────────────────
# The Cost of Survival
# A battle wound opens the door to cybernetic augmentation.

eotg_aug_init.001 = {
    type = character_event
    title = eotg_aug_init.001.t
    desc = eotg_aug_init.001.desc
    theme = battle

    left_portrait = {
        character = root
        animation = stress
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
        }
    }

    # Accept — receive the augmentation, remove the wound
    option = {
        name = eotg_aug_init.001.a
        add_trait = eotg_augmented
        remove_trait = wounded
        eotg_apply_edu_synergy_augmented = yes
        hidden_effect = {
            set_variable = { name = eotg_fracture_risk  value = 0 }
        }
        stress_impact = {
            base = minor_stress_impact_loss
            humble = minor_stress_impact_gain
            content = minor_stress_impact_gain
            zealous = medium_stress_impact_gain
            ambitious = minor_stress_impact_loss
        }
        ai_chance = {
            base = 50
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  has_trait = brave }
            modifier = { add = -50  has_trait = zealous }
            modifier = { add = -20  has_trait = content }
        }
    }

    # Find another way — wound heals naturally
    option = {
        name = eotg_aug_init.001.b
        remove_trait = wounded
        stress_impact = {
            zealous = minor_stress_impact_loss
            ambitious = minor_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = zealous }
            modifier = { add = 10  has_trait = content }
        }
    }

    # Refuse entirely — zealous or principled characters only
    option = {
        name = eotg_aug_init.001.c
        trigger = {
            OR = { has_trait = zealous  has_trait = content  has_trait = humble }
        }
        add_piety = 100
        stress_impact = {
            zealous = minor_stress_impact_loss
            ambitious = medium_stress_impact_gain
        }
        ai_chance = {
            base = 20
            modifier = { add = 30  has_trait = zealous }
            modifier = { add = 20  has_trait = content }
        }
    }
}

    theme = stewardship_wealth_focus

    left_portrait = {

    }

    # Accept the offer
    option = {
        name = eotg_aug_init.002.a
        add_trait = eotg_augmented
        eotg_apply_edu_synergy_augmented = yes
        hidden_effect = {
            set_variable = { name = eotg_fracture_risk  value = 0 }
        }
        stress_impact = {
            base = 0
            zealous = medium_stress_impact_gain
            humble = minor_stress_impact_gain
        }
        ai_chance = {
            base = 40
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  eotg_ai_wants_augmentation = yes }
            modifier = { add = -100  has_trait = zealous }
        }
    }

    # Decline
    option = {
        name = eotg_aug_init.002.b
        add_intrigue_skill = 1
        stress_impact = {
            base = 0
            ambitious = minor_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = zealous }
            modifier = { add = 20  has_trait = content }
        }
    }

    # Offer it to a knight instead
    option = {
        name = eotg_aug_init.002.c
        trigger = { any_knight = { is_alive = yes } }
        random_knight = {
            limit = { is_alive = yes }
            add_trait = eotg_augmented
        }
        add_dread = 5
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = callous }
        }
    }
}

# ── eotg_aug_init.005 ─────────────────────────────────────────────────────────
# A Familiar Change
# Observing a peer who has already been augmented.

eotg_aug_init.005 = {
    type = character_event
    title = eotg_aug_init.005.t
    desc = eotg_aug_init.005.desc
    theme = stewardship_wealth_focus

    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        character = scope:aug_peer
        animation = personality_bold
    }

    trigger = {
        OR = {
            any_vassal = { has_trait = eotg_augmented }
            any_courtier = { is_alive = yes  has_trait = eotg_augmented }
        }
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 3 }
            if = {
                limit = { any_vassal = { has_trait = eotg_augmented } }
                random_vassal = {
                    limit = { has_trait = eotg_augmented }
                    save_scope_as = aug_peer
                }
            }
            else = {
                random_courtier = {
                    limit = { is_alive = yes  has_trait = eotg_augmented }
                    save_scope_as = aug_peer
                }
            }
        }
    }

    # "If it works for them, it should work for me." — become augmented
    option = {
        name = eotg_aug_init.005.a
        add_trait = eotg_augmented
        eotg_apply_edu_synergy_augmented = yes
        hidden_effect = {
            set_variable = { name = eotg_fracture_risk  value = 0 }
        }
        scope:aug_peer = {
            add_opinion = { modifier = eotg_opinion_aug_admiration  target = root }
        }
        stress_impact = {
            base = 0
            zealous = medium_stress_impact_gain
        }
        ai_chance = {
            base = 40
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = -100  has_trait = zealous }
        }
    }

    # "It unsettles me." — resist, gain piety, suppress initiation
    option = {
        name = eotg_aug_init.005.b
        add_piety = 50
        add_character_flag = { flag = eotg_flag_suppress_progression  years = 5 }
        scope:aug_peer = {
            add_opinion = { modifier = eotg_opinion_aug_unease  target = root }
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = zealous }
            modifier = { add = 20  has_trait = content }
        }
    }

    # "Study it carefully before deciding." — delay, gain learning
    option = {
        name = eotg_aug_init.005.c
        add_learning_skill = 1
        add_character_flag = { flag = eotg_flag_suppress_progression  years = 2 }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = patient }
            modifier = { add = 20  has_trait = curious }
        }
    }
}
