# Captains expect their cut. Fire after transitioning from war to peace.
eotg_fringe.0001 = {
	type = character_event
	title = eotg_fringe_0001_title
	theme = martial

	desc = eotg_fringe_0001_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {
		random_vassal = {
			limit = {
				eotg_st_is_fringe_captain = yes
			}
			save_scope_as = eotg_demanding_captain
		}
	}

	right_portrait = {
		character = scope:eotg_demanding_captain
		animation = personality_bold
		trigger = { exists = scope:eotg_demanding_captain }
	}

	# Share generously — buy loyalty, lose gold.
	option = {
		name = eotg_fringe_0001_opt_generous
		remove_short_term_gold = eotg_sv_fringe_spoils_cost
		eotg_se_fringe_gain_grip = { AMOUNT = 8 }
		every_vassal = {
			limit = { eotg_st_is_fringe_captain = yes }
			add_opinion = {
				target = root
				modifier = eotg_fringe_spoils_distributed
			}
		}
		add_character_flag = { flag = eotg_fringe_captain_paid  years = 2 }
		ai_chance = {
			base = 50
			modifier = { add = 30  gold > 400 }
		}
	}

	# Keep most — gain gold, lose Grip and loyalty.
	option = {
		name = eotg_fringe_0001_opt_keep
		eotg_se_fringe_lose_grip = { AMOUNT = 8 }
		every_vassal = {
			limit = { eotg_st_is_fringe_captain = yes }
			add_opinion = {
				target = root
				modifier = eotg_fringe_spoils_kept
			}
		}
		ai_chance = {
			base = 50
			modifier = { add = -30  eotg_st_fringe_grip_low = yes }
		}
	}

	# Favor one captain — boost one, alienate others.
	option = {
		name = eotg_fringe_0001_opt_favor
		trigger = { exists = scope:eotg_demanding_captain }
		scope:eotg_demanding_captain = {
			add_opinion = {
				target = root
				modifier = eotg_fringe_spoils_distributed
			}
		}
		every_vassal = {
			limit = {
				eotg_st_is_fringe_captain = yes
				NOT = { this = scope:eotg_demanding_captain }
			}
			add_opinion = {
				target = root
				modifier = eotg_fringe_spoils_kept
			}
		}
		eotg_se_fringe_lose_grip = { AMOUNT = 3 }
		remove_short_term_gold = 50
		ai_chance = { base = 20 }
	}
}

# A captain with low opinion refuses a direct order from the Overlord.
eotg_fringe.0002 = {
	type = character_event

	theme = intrigue

	desc = eotg_fringe_0002_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {
		random_vassal = {
			limit = {
				eotg_st_is_fringe_captain = yes
				opinion = { target = root  value < 0 }
			}
			save_scope_as = eotg_defiant_captain
		}
	}

	right_portrait = {
		character = scope:eotg_defiant_captain
		animation = personality_bold
		trigger = { exists = scope:eotg_defiant_captain }
	}

	# Tolerate — show weakness but avoid escalation.
	option = {
		name = eotg_fringe_0002_opt_tolerate
		eotg_se_fringe_lose_grip = { AMOUNT = 5 }
		ai_chance = { base = 30 }
	}

	# Public punishment — gain dread, risk rebellion.
	option = {
		name = eotg_fringe_0002_opt_punish
		trigger = { exists = scope:eotg_defiant_captain }
		add_dread = 10
		scope:eotg_defiant_captain = {
			add_opinion = {
				target = root
				modifier = eotg_fringe_public_punishment
			}
		}
		random = {
			chance = 30
			scope:eotg_defiant_captain = {
				add_character_modifier = {
					modifier = eotg_fringe_captain_defiant
					years = 3
				}
			}
		}
		stress_impact = {
			base = minor_stress_impact_gain
			compassionate = medium_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = { add = 20  dread > 20 }
		}
	}

	# Attempt assassination — high risk, high reward.
	option = {
		name = eotg_fringe_0002_opt_eliminate
		trigger = { exists = scope:eotg_defiant_captain }
		add_prestige = -50
		scope:eotg_defiant_captain = {
			add_character_modifier = {
				modifier = eotg_fringe_captain_hunted
				years = 2
			}
		}
		ai_chance = {
			base = 20
			modifier = { add = 20  intrigue > 14 }
		}
	}
}

# Blood Feud — two captains have formed a deadly rivalry.

		ai_chance = {
			base = 50

		}

	# Let them fight — the stronger captain asserts dominance.
	option = {
		name = eotg_fringe_0007_opt_let_fight
		add_prestige = 25
		eotg_se_fringe_lose_grip = { AMOUNT = 5 }
		ai_chance = { base = 30 }

			50 = {

			base = 20
			modifier = { add = 20  intrigue > 12 }
		}
	}
