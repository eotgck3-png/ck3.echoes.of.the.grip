# eotg_elven_flavor.0001 — The Eternal Court Convenes
# The Eternal Court gathers to deliberate on matters of the realm.
eotg_elven_flavor.0001 = {
	type = character_event
	title = eotg_elven_flavor_0001_title
	desc = eotg_elven_flavor_0001_desc
	theme = diplomacy

	left_portrait = { character = root  animation = personality_rational }

	trigger = {
		eotg_st_is_starborn_sovereign = yes
		vassal_count >= 3
		NOT = { has_character_flag = eotg_elven_0001_recent }
	}

	option = {
		name = eotg_elven_flavor_0001_opt_respect
		add_prestige = 20
		add_character_modifier = { modifier = eotg_elven_court_session  years = 2 }
		every_vassal = {
			limit = { eotg_st_is_celestial_lord = yes }
			add_opinion = { target = root  modifier = eotg_pmc_loyalty_restored }
		}
		add_character_flag = { flag = eotg_elven_0001_recent  years = 4 }
		ai_chance = { base = 45  modifier = { add = 10  has_trait = patient } }
	}
	option = {
		name = eotg_elven_flavor_0001_opt_manipulate
		add_intrigue_skill = 2
		add_character_flag = { flag = eotg_elven_0001_recent  years = 4 }
		ai_chance = { base = 30  modifier = { add = 15  has_trait = deceitful } }
	}
	option = {
		name = eotg_elven_flavor_0001_opt_pressure
		add_dread = 10
		every_vassal = {
			limit = { eotg_st_is_celestial_lord = yes }
			add_opinion = { target = root  modifier = eotg_fringe_fear_rule_resentment }
