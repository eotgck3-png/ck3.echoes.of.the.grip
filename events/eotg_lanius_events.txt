# ── eotg_lanius_events.txt ────────────────────────────────────────────────────
# Event chain for Caelorin Solvaerath, "Prince of Myr" — Lanius Principality.
# Fires from: eotg_char_arc_actions.txt (on_game_started)
# Scope: character (Caelorin)
#
# Arc summary:
#   I   — The Title Without a Throne (0001)
#   II  — The Price of Recognition   (0002)
#   III — The Hollow Throne          (0003)
#   IV  — The Whispers of Myr        (0004)
#   V   — The Prince Acknowledged    (0005)
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_lanius

# ── eotg_lanius.0001 ──────────────────────────────────────────────────────────
# Fires from: on_game_started (day 60)
# Scope: Caelorin Solvaerath
eotg_lanius.0001 = {
    type = character_event
    title = eotg_lanius.0001.t
    desc = eotg_lanius.0001.desc
    theme = intrigue

    option = {
        name = eotg_lanius.0001.a
        # "If history is weak, it deserves replacement."
        add_prestige = 150
        add_character_modifier = {
            modifier = eotg_manufactured_legacy
            years = 10
        }
        trigger_event = { id = eotg_lanius.0002 days = 180 }
    }

    option = {
        name = eotg_lanius.0001.b
        # "I will prove worthy of the title."
        add_piety = 100
        add_character_modifier = {
            modifier = eotg_seeking_recognition
            years = 5
        }
        trigger_event = { id = eotg_lanius.0002 days = 270 }
    }

    option = {
        name = eotg_lanius.0001.c
        # "Let them whisper. Fear works too."
        add_dread = 15
        add_character_modifier = {
            modifier = eotg_rule_through_presence
            years = 5
        }
        trigger_event = { id = eotg_lanius.0002 days = 180 }
    }
}

# ── eotg_lanius.0002 ──────────────────────────────────────────────────────────
# Fires from: eotg_lanius.0001
# The Synod demands a sacrifice before recognition.
eotg_lanius.0002 = {
    type = character_event
    title = eotg_lanius.0002.t
    desc = eotg_lanius.0002.desc
    theme = faith

    option = {
        name = eotg_lanius.0002.a
        # "Fund the restoration of the Ashen Cathedrals." — lose gold
        remove_short_term_gold = 200
        add_character_modifier = {
            modifier = eotg_patron_of_restoration
            years = 10
        }
        trigger_event = { id = eotg_lanius.0003 days = 90 }
    }

    option = {
        name = eotg_lanius.0002.b
        # "Send your finest soldiers to defend pilgrimage routes." — lose levies (via opinion hit)
        add_character_modifier = {
            modifier = eotg_guardians_pilgrim_roads
            years = 5
        }
        add_prestige = -50
        trigger_event = { id = eotg_lanius.0003 days = 90 }
    }

    option = {
        name = eotg_lanius.0002.c
        # "Grant the Synod judicial authority."
        add_character_modifier = {
            modifier = eotg_ecclesiastical_concessions
            years = 10
        }
        add_piety = 75
        trigger_event = { id = eotg_lanius.0003 days = 90 }
    }
}

# ── eotg_lanius.0003 ──────────────────────────────────────────────────────────
# Fires from: eotg_lanius.0002
# The Wardens of the Hollow Throne summon Caelorin for an ideological test.
eotg_lanius.0003 = {
    type = character_event
    title = eotg_lanius.0003.t
    desc = eotg_lanius.0003.desc
    theme = faith

    option = {
        name = eotg_lanius.0003.a
        # "Because truth alone never saved kingdoms."
        add_intrigue_skill = 1
        add_piety = -50
        trigger_event = { id = eotg_lanius.0004 days = 730 }
    }

    option = {
        name = eotg_lanius.0003.b
        # "Because people need something to believe."
        add_diplomacy_skill = 1
        add_prestige = 100
        add_piety = 50
        trigger_event = { id = eotg_lanius.0004 days = 730 }
    }

    option = {
        name = eotg_lanius.0003.c
        # "Because I will succeed where the real rulers failed."
        add_dread = 10
        add_martial_skill = 1
        trigger_event = { id = eotg_lanius.0004 days = 730 }
    }
}

# ── eotg_lanius.0004 ──────────────────────────────────────────────────────────
# Fires from: eotg_lanius.0003 (~2 years later)
# The propaganda spreads organically — the Synod begins to fracture.
eotg_lanius.0004 = {
    type = character_event
    title = eotg_lanius.0004.t
    desc = eotg_lanius.0004.desc
    theme = intrigue

    option = {
        name = eotg_lanius.0004.a
        # "Encourage the stories quietly."
        add_legitimacy = 20
        add_piety = 25
        add_prestige = 100
        trigger_event = { id = eotg_lanius.0005 days = 1460 }
    }

    option = {
        name = eotg_lanius.0004.b
        # "Turn the myth into doctrine."
        add_character_modifier = { modifier = eotg_sanctioned_mythmaking years = 10 }
        trigger_event = { id = eotg_lanius.0005 days = 1095 }
    }

    option = {
        name = eotg_lanius.0004.c
        # "Remain silent."
        add_prestige = 50
        trigger_event = { id = eotg_lanius.0005 days = 1825 }
    }
}

# ── eotg_lanius.0005 ──────────────────────────────────────────────────────────
# Fires from: eotg_lanius.0004 (5-8 years into campaign)
# The Synod and Wardens issue their carefully-worded declaration.
eotg_lanius.0005 = {
    type = character_event
    title = eotg_lanius.0005.t
    desc = eotg_lanius.0005.desc
    theme = faith

    trigger = {
        prestige >= 500
        NOT = { has_character_modifier = eotg_manufactured_legacy }
    }

    option = {
        name = eotg_lanius.0005.a
        # "Then let Myr remember my name."
        add_character_modifier = { modifier = eotg_recognized_prince_of_myr years = 20 }
        add_legitimacy = 40
    }

    option = {
        name = eotg_lanius.0005.b
        # "This is only the beginning."
        add_prestige = 300
        add_martial_skill = 1
    }

    option = {
        name = eotg_lanius.0005.c
        # "I never needed their blessing."
        add_dread = 20
        add_piety = -100
        add_legitimacy = 15
    }
}
