# ── eotg_aug_tier3.001 ────────────────────────────────────────────────────────
# Violent Impulse
# The overclocked system registers a threat and the body responds before
# the mind can intervene.

eotg_aug_tier3.001 = {
    type = character_event
    title = eotg_aug_tier3.001.t
    desc = {
        desc = eotg_aug_tier3.001.desc
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:eotg_victim }
                desc = eotg_aug_tier3.001.desc_victim
            }
            desc = eotg_aug_tier3.001.desc_none
        }
    }
    theme = intrigue
    override_background = { reference = eotg_bg_council }

    left_portrait = {
        character = root
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { OR = { has_trait = wrathful  has_trait = callous } }
            animation = anger
        }
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = paranoid }
            animation = paranoia
        }
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = compassionate }
            animation = worry
        }
        animation = stress
    }
    # Guarded: the picker may find no one (vanilla
    # events/activities/activity_system_events.txt:89-92).
    right_portrait = {
        trigger = { exists = scope:eotg_victim }
        character = scope:eotg_victim
        animation = fear
    }

    # The courtier who moved too quickly (phase 0 §2.2 picker; Overclocked
    # family factor 0.05). May find no one: b and c are then unavailable.
    immediate = {
        hidden_effect = {
            eotg_aug_pick_victim_effect = { NAME = eotg_victim  FAMILY_FACTOR = 0.05 }
        }
    }

    # "Control yourself" — resist, gain stress, reduce risk
    option = {
        name = eotg_aug_tier3.001.a
        stress_impact = { base = medium_stress_impact_gain }
        eotg_add_fracture_risk = { AMOUNT = -3 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = patient }
            modifier = { add = 10  has_trait = content }
        }
    }

    # "Strike first" — wound a courtier, add dread and risk
    option = {
        name = eotg_aug_tier3.001.b
        trigger = { exists = scope:eotg_victim }
        scope:eotg_victim = {
            increase_wounds_effect = { REASON = fight }
            add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
        }
        add_dread = 15
        eotg_add_fracture_risk = { AMOUNT = 5 }
        eotg_aug_stress_wound_effect = yes
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = wrathful }
            modifier = { add = 10  has_trait = callous }
        }
    }

    # "Kill them" — execute a courtier, heavy dread and risk.
    # QA round 2 M4: a deliberate killing, so a family victim (picker 0.05)
    # makes a kinslayer: vanilla add_kinslayer_trait_or_nothing_effect
    # (00_secret_effects.txt:433), before the death. Option b (a wound) and
    # the involuntary episodes carry none.
    option = {
        name = eotg_aug_tier3.001.c
        trigger = { exists = scope:eotg_victim }
        add_kinslayer_trait_or_nothing_effect = { VICTIM = scope:eotg_victim }
        scope:eotg_victim = {
            death = { death_reason = death_murder  killer = root }
        }
        add_dread = 30
        add_tyranny = 10
        eotg_add_fracture_risk = { AMOUNT = 15 }
        eotg_aug_stress_murder_effect = yes
        ai_chance = {
            base = 20
            modifier = { add = 30  has_trait = callous }
            modifier = { add = -30  has_trait = compassionate }
        }
    }

    # [calm] "Breathe. Count. Walk out." - you leave mid-audience
    option = {
        name = eotg_aug_tier3.001.d
        trigger = { has_trait = calm }
        trait = calm
        add_prestige = -50
        eotg_add_fracture_risk = { AMOUNT = -6 }
        stress_impact = {
            calm = medium_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = calm }
            modifier = { add = 10  has_trait = patient }
        }
    }

    # [education_martial] "Take it out on the practice posts." W5
    # (event_quality_v1 §13.1 3a): was [wrathful]; it moves risk (+5) and
    # dread, so it cannot fold. Burning the reflex off in drill is a
    # soldier's discipline, so martial. wrathful keeps its stress loss here as
    # a response (§13.1 step 4); calm (d) is the kept personality gate (calm
    # 4 gates, the thinnest of the pair).
    option = {
        name = eotg_aug_tier3.001.e
        trigger = { has_trait = education_martial }
        skill = martial
        add_dread = 10
        eotg_add_fracture_risk = { AMOUNT = 5 }
        stress_impact = {
            wrathful = medium_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = education_martial_3 }
            modifier = { add = 30  has_trait = education_martial_4 }
            modifier = { add = 30  has_trait = education_martial_5 }
            modifier = { add = 10  has_trait = wrathful }
        }
    }
}
