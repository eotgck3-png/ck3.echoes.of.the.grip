# ── eotg_augmentation_tier3.txt ──────────────────────────────────────────────
# Events for Overclocked-tier characters.
#
# eotg_aug_tier3.001 — Violent Impulse         (random, adds fracture risk)
# eotg_aug_tier3.002 — The Mirror              (random, identity confrontation)
# eotg_aug_tier3.003 — Sleepless               (random, rest deprivation)
# eotg_aug_tier3.004 — Signal Noise            (paranoid ideation)
# eotg_aug_tier3.005 — Clarity                 (at-war tactical vision)
# eotg_aug_tier3.006 — The Delegation          (vassal ultimatum)
# eotg_aug_tier3.007 — Two Machines            (overclocked peer encounter)
#
# Fires from: eotg_on_yearly_aug_overclocked_check
#   (common/on_action/eotg_augmentation_on_actions.txt, via yearly_playable_pulse).
# Cooldown eotg_flag_oc_event_cooldown is set and checked there only.
# Resource: every event moves eotg_fracture_risk in at least one option.
#
# cybernetics_v2 Phase 1 (docs/specs/cybernetics_v2_phase1.md §4): trait-gated
# options (trigger has_trait + trait icon), stress helpers on every violent

# (audit §1.5), scaled gold, lesson modifiers instead of repeatable skill.

namespace = eotg_aug_tier3


# ── eotg_aug_tier3.001 ────────────────────────────────────────────────────────
# Violent Impulse
# The overclocked system registers a threat and the body responds before
# the mind can intervene.

eotg_aug_tier3.001 = {
    type = character_event
    title = eotg_aug_tier3.001.t
    desc = {
        desc = eotg_aug_tier3.001.desc
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:eotg_victim }
                desc = eotg_aug_tier3.001.desc_victim
            }
            desc = eotg_aug_tier3.001.desc_none
        }
    }
    theme = intrigue

    left_portrait = {
        character = root
        animation = stress
    }
    # Guarded: the picker may find no one (vanilla
    # events/activities/activity_system_events.txt:89-92).
    right_portrait = {
        trigger = { exists = scope:eotg_victim }
        character = scope:eotg_victim
        animation = fear
    }

    # The courtier who moved too quickly (phase 0 §2.2 picker; Overclocked
    # family factor 0.05). May find no one: b and c are then unavailable.
    immediate = {
        hidden_effect = {
            eotg_aug_pick_victim_effect = { NAME = eotg_victim  FAMILY_FACTOR = 0.05 }
        }
    }

    # "Control yourself" — resist, gain stress, reduce risk
    option = {
        name = eotg_aug_tier3.001.a
        stress_impact = { base = medium_stress_impact_gain }
        eotg_add_fracture_risk = { AMOUNT = -3 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = patient }
            modifier = { add = 10  has_trait = content }
        }
    }

    # "Strike first" — wound a courtier, add dread and risk
    option = {
        name = eotg_aug_tier3.001.b
        trigger = { exists = scope:eotg_victim }
        scope:eotg_victim = {
            increase_wounds_effect = { REASON = fight }
            add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
        }
        add_dread = 15
        eotg_add_fracture_risk = { AMOUNT = 5 }
        eotg_aug_stress_wound_effect = yes
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = wrathful }
            modifier = { add = 10  has_trait = callous }
        }
    }

    # "Kill them" — execute a courtier, heavy dread and risk
    option = {
        name = eotg_aug_tier3.001.c
        trigger = { exists = scope:eotg_victim }
        scope:eotg_victim = {
            death = { death_reason = death_murder  killer = root }
        }
        add_dread = 30
        add_tyranny = 10
        eotg_add_fracture_risk = { AMOUNT = 15 }
        eotg_aug_stress_murder_effect = yes
        ai_chance = {
            base = 20
            modifier = { add = 30  has_trait = callous }
            modifier = { add = -30  has_trait = compassionate }
        }
    }

    # [calm] "Breathe. Count. Walk out." - you leave mid-audience
    option = {
        name = eotg_aug_tier3.001.d
        trigger = { has_trait = calm }
        trait = calm
        add_prestige = -50
        eotg_add_fracture_risk = { AMOUNT = -6 }
        stress_impact = {
            calm = medium_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = calm }
            modifier = { add = 10  has_trait = patient }
        }
    }

    # [wrathful] "Take it out on the practice posts."

# ── eotg_aug_tier3.006 ────────────────────────────────────────────────────────
# The Delegation
# Senior vassals arrive together to question fitness to rule.

eotg_aug_tier3.006 = {
    type = character_event
    title = eotg_aug_tier3.006.t
    desc = eotg_aug_tier3.006.desc
    theme = realm

    trigger = {
        any_vassal = {
            is_alive = yes
            opinion = { target = root  value < -20 }
        }
    }

    left_portrait = {
        character = root
        animation = stress
    }
    right_portrait = {
        character = scope:eotg_delegation_leader
        animation = anger
    }

    # The delegation's spokesperson: the vassal with the lowest opinion of root.
    immediate = {
        hidden_effect = {
            ordered_vassal = {
                limit = {
                    is_alive = yes
                    opinion = { target = root  value < -20 }
                }
                order_by = {
                    value = 0
                    subtract = "opinion(root)"
                }
                save_scope_as = eotg_delegation_leader
            }
        }
    }

    # "Have them arrested." — imprison the spokesperson; dread, tyranny
    option = {
        name = eotg_aug_tier3.006.a
        imprison_character_effect = {    # vanilla 00_prison_effects.txt
            IMPRISONER = root
            TARGET = scope:eotg_delegation_leader
        }
        add_dread = 30
        add_tyranny = 15
        every_vassal = {
            limit = {
                is_alive = yes
                opinion = { target = root  value < -20 }
            }
            add_opinion = { modifier = eotg_opinion_aug_fear  target = root  years = 10 }
        }
        eotg_aug_stress_tyranny_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = callous }
            modifier = { add = 20  has_trait = wrathful }
        }
    }

    # "Hear them out. Set conditions." — prestige cost; clears fear/unease, adds reassured

        name = eotg_aug_tier3.006.b

                is_alive = yes

            remove_opinion = { target = root  modifier = eotg_opinion_aug_unease }
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
        }
        stress_impact = { base = minor_stress_impact_gain }
        ai_chance = {
            base = 50
            modifier = { add = 20  has_trait = patient }
            modifier = { add = 10  has_trait = humble }

    }

    # "Demonstrate what I am still capable of." — mass intimidation, fracture risk

        add_dread = 50

        }
        eotg_add_fracture_risk = { AMOUNT = 12 }

            base = 20
