# ── eotg_aug_tier3.001 ────────────────────────────────────────────────────────
# Violent Impulse
# The overclocked system registers a threat and the body responds before
# the mind can intervene.

eotg_aug_tier3.001 = {
    type = character_event
    title = eotg_aug_tier3.001.t
    desc = eotg_aug_tier3.001.desc
    theme = intrigue

    left_portrait = {
        character = root
        animation = stress
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_oc_event_cooldown  months = 6 }
        }
    }

    # "Control yourself" — resist, gain stress, reduce risk
    option = {
        name = eotg_aug_tier3.001.a
        stress_impact = { base = medium_stress_impact_gain }
        eotg_add_fracture_risk = { AMOUNT = -3 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = patient }
            modifier = { add = 10  has_trait = content }
        }
    }

    # "Strike first" — wound a courtier, add dread and risk
    option = {
        name = eotg_aug_tier3.001.b
        trigger = { any_courtier = { is_alive = yes  NOT = { is_imprisoned = yes } } }
        random_courtier = {
            limit = { is_alive = yes  NOT = { is_imprisoned = yes } }
            add_trait = wounded
        }
        add_dread = 15
        eotg_add_fracture_risk = { AMOUNT = 5 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = wrathful }
            modifier = { add = 10  has_trait = callous }
        }
    }

    # "Kill them" — execute a courtier, heavy dread and risk
    option = {
        name = eotg_aug_tier3.001.c
        trigger = { any_courtier = { is_alive = yes  NOT = { is_imprisoned = yes } } }
        random_courtier = {
            limit = { is_alive = yes  NOT = { is_imprisoned = yes } }
            death = { death_reason = death_murder }
        }
        add_dread = 30
        add_tyranny = 10
        eotg_add_fracture_risk = { AMOUNT = 15 }
        ai_chance = {
            base = 20
            modifier = { add = 30  has_trait = callous }
            modifier = { add = -30  has_trait = compassionate }
        }
    }
}

# ── eotg_aug_tier3.007 ────────────────────────────────────────────────────────
# Two Machines
# Another overclocked character exists in the same orbit.

eotg_aug_tier3.007 = {
    type = character_event
    title = eotg_aug_tier3.007.t
    desc = eotg_aug_tier3.007.desc
    theme = intrigue

    trigger = {
        OR = {
            any_vassal = { is_alive = yes  has_trait = eotg_overclocked }
            any_courtier = { is_alive = yes  has_trait = eotg_overclocked }
        }
    }

    left_portrait = {
        character = root
        animation = personality_bold
    }
    right_portrait = {
        character = scope:other_machine
        animation = personality_bold
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_oc_event_cooldown  months = 6 }
            if = {
                limit = { any_vassal = { is_alive = yes  has_trait = eotg_overclocked } }
                random_vassal = {
                    limit = { is_alive = yes  has_trait = eotg_overclocked }
                    save_scope_as = other_machine
                }
            }
            else = {
                random_courtier = {
                    limit = { is_alive = yes  has_trait = eotg_overclocked }
                    save_scope_as = other_machine
                }
            }
        }
    }

    # "Acknowledge the kinship." — mutual positive opinion
    option = {
        name = eotg_aug_tier3.007.a
        scope:other_machine = {
            add_opinion = { modifier = eotg_opinion_aug_admiration  target = root }
        }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = gregarious }
        }
    }

    # "The other is a threat." — intrigue gain, they fear you
    option = {
        name = eotg_aug_tier3.007.b
        add_intrigue_skill = 1
        scope:other_machine = {
            add_opinion = { modifier = eotg_opinion_aug_fear  target = root }
        }
        eotg_add_fracture_risk = { AMOUNT = 5 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = paranoid }
            modifier = { add = 10  has_trait = suspicious }
        }
    }

    # "Compare calibrations." — both gain prowess, both gain fracture risk
    option = {
        name = eotg_aug_tier3.007.c
        add_prowess_skill = 2
        eotg_add_fracture_risk = { AMOUNT = 10 }
        scope:other_machine = {
            add_prowess_skill = 2
            eotg_add_fracture_risk = { AMOUNT = 10 }
        }
        ai_chance = {
            base = 20
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  has_trait = brave }
        }
    }
}
