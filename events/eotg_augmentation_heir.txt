# -- eotg_augmentation_heir.txt -----------------------------------------------
# The Heir's Arc (thread T3; docs/specs/cybernetics_v2_phase3.md §4).
# The dynasty watches the ruler break.
#
# eotg_aug_heir.001 -- Concern                 (story stage 0)
#   (stage 1 is eotg_fracture.005 What the Heir Saw)
# eotg_aug_heir.003 -- The Heir's Warning      (story stage 2)
# eotg_aug_heir.004 -- The Heir's Choice       (story stage 3; ally / usurp / kill)
# eotg_aug_heir.005 -- What Must Be Done       (to the heir, after the kill)
# eotg_aug_heir.006 -- (hidden) the kill       (on the ruler, from heir.004)
# eotg_aug_heir.007 -- The Next in Line        (round 2: the dormant story
#                                                finds a later heir)
#
# Round 2 (docs/specs/cybernetics_v2_new_beats.md §5.1): round 1's heir.004
#   parks the story at a dormant stage 5; for the next primary heir the story
#   fires heir.007, then heir.003 and heir.004 with successor descs. Round 2's
#   heir.004 ends the story. Never a third round.
#
# Fires from: the story cycle eotg_story_aug_heir_arc
#   (common/story_cycles/eotg_augmentation_stories.txt), created by
#   eotg_on_yearly_aug_neurofractured_check and (balance spec §5.8)
#   eotg_on_yearly_aug_overclocked_check; heir.006 from heir.004's
#   immediate (kill, success); heir.005 from heir.006, after succession;
#   heir.007 from the story's dormant entry.
# Cooldown: the story's effect_group timing (index §1 rule 5). No flags.
# Each immediate saves the story as scope:eotg_heir_story and its eotg_heir
# variable as scope:eotg_heir (heir.003, .004 and .007 also its
# eotg_prior_heir as scope:eotg_prior_heir, for round 2's loc). Options move
# the story's eotg_heir_dread, which weights heir.004.
# Resource: every option moves eotg_fracture_risk on the ruler, except
# heir.005 (the heir's event), which couples through d (tier change).
#
# event_quality_v1 batch B7 (W5 re-gating, W6 portraits; §13): at most one
# root personality gate per event; the others re-axed to an education gate
# (skill icon, education_*_3/_4/_5 AI weights) or a duchy tier gate (no icon,
# AI base 10 plus counter-trait negatives); kept gates paired with a stress
# row (§13.1 step 5). heir.004's two gates sit on exclusive outcome branches
# but still count, so ally_b is re-axed. heir.005 is a W7 set piece: its
# reactions are the W7 addendum's (§6.5) and the witness portrait is W6's;
# window, effect, widgets and desc text are left to W7.
# -----------------------------------------------------------------------------

# Concern. The heir comes to you about what they have seen.
# Caller: eotg_story_aug_heir_arc, stage 0.
# Balance spec §5.8: the arc can now start at Overclocked; then
# desc_overclocked replaces the base desc (first triggered_desc of the
# first_valid). A child the machine saved (eotg_flag_aug_child_patient on the
# heir) adds desc_patient. scope:eotg_heir is saved in immediate.

# eotg_lint: allow L015 W2 keep list (event_quality_v1 §14.2): family/love/friendly scene stays on the theme's room until art
eotg_aug_heir.001 = {
    type = character_event
    title = eotg_aug_heir.001.t
    desc = {
        first_valid = {
            # §12.5 S3 (b): a Seamless owner gets the log register.
            triggered_desc = {
                trigger = { has_trait = eotg_total_integration }
                desc = eotg_aug_heir.001.desc_seamless
            }
            triggered_desc = {
                trigger = { eotg_is_aug_tier3 = yes }
                desc = eotg_aug_heir.001.desc_overclocked
            }
            desc = eotg_aug_heir.001.desc
        }
        triggered_desc = {
            trigger = {
                exists = scope:eotg_heir
                scope:eotg_heir = { has_character_flag = eotg_flag_aug_child_patient }
                # §12.5 S3: the fragment says "asking me", so first person
                NOT = { has_trait = eotg_total_integration }
            }
            desc = eotg_aug_heir.001.desc_patient
        }
    }
    theme = family

    trigger = {
        any_owned_story = {
            story_type = eotg_story_aug_heir_arc
            exists = var:eotg_heir
            var:eotg_heir = { is_alive = yes }
        }
    }

    left_portrait = {
        character = root
        triggered_animation = {    # §12.5 / rule 7: a Seamless owner stays still
            trigger = { has_trait = eotg_total_integration }
            animation = idle
        }
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = paranoid }
            animation = paranoia
        }
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { OR = { has_trait = arrogant  has_trait = callous } }
            animation = dismissal
        }
        triggered_animation = {    # W6 (event_quality_v1)
            trigger = { has_trait = compassionate }
            animation = sadness
        }
        animation = worry
    }
    right_portrait = {
        trigger = { exists = scope:eotg_heir }
        character = scope:eotg_heir
        animation = worry
    }

    immediate = {
        hidden_effect = {
            random_owned_story = {
                limit = { story_type = eotg_story_aug_heir_arc }
                save_scope_as = eotg_heir_story
                var:eotg_heir = { save_scope_as = eotg_heir }
            }
        }
    }

    # "Reassure them."
    option = {
        name = eotg_aug_heir.001.a
        scope:eotg_heir = {
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
        }
        scope:eotg_heir_story = { change_variable = { name = eotg_heir_dread  add = -1 } }
        eotg_add_fracture_risk = { AMOUNT = -3 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = compassionate }
            modifier = { add = 10  has_trait = gregarious }
        }
    }

    # "This is not your concern."
    option = {
        name = eotg_aug_heir.001.b
        scope:eotg_heir = {
            add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
        }
        scope:eotg_heir_story = { change_variable = { name = eotg_heir_dread  add = 1 } }
        eotg_add_fracture_risk = { AMOUNT = 3 }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = arrogant }
            modifier = { add = 20  has_trait = shy }
        }
    }

    # "Then learn to keep me, if it comes to that." The keeper flag weighs
    # heir.004 toward ally and lets heir.003.c fall back on it.
    option = {
        name = {    # §12.5 S3 (b): Seamless register variant
            trigger = { has_trait = eotg_total_integration }
            text = eotg_aug_heir.001.c_seamless
        }
        name = eotg_aug_heir.001.c
        scope:eotg_heir = {
            add_character_flag = eotg_flag_aug_heir_keeper
            add_opinion = { modifier = eotg_opinion_aug_admiration  target = root  years = 5 }
        }
        eotg_add_fracture_risk = { AMOUNT = -5 }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = humble }
            modifier = { add = 20  has_trait = trusting }
            modifier = { add = -20  has_trait = paranoid }
        }
    }

    # [honest] "Tell them the truth." The one personality gate (honest 9
    # against paranoid 14); W5 step 5 adds its stress row.
    option = {
        name = eotg_aug_heir.001.d
        trigger = { has_trait = honest }
        trait = honest
        scope:eotg_heir = {
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
            add_character_modifier = { modifier = eotg_mod_aug_lesson_learning  years = 5 }
        }
        scope:eotg_heir_story = { change_variable = { name = eotg_heir_dread  add = -1 } }
        eotg_add_fracture_risk = { AMOUNT = -3 }
        stress_impact = {
            honest = minor_stress_impact_loss    # W5 step 5
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = honest }
            modifier = { add = 10  has_trait = compassionate }
        }
    }

    # [education_intrigue] "They want the throne." W5 (event_quality_v1
    # §13.1 3a): was [paranoid]; it moves the heir's opinion, the story's
    # dread (+1) and risk (+5), so it cannot fold. Reading an heir's concern
    # as a move on the seat is a spymaster's read of motive, so intrigue.
    # paranoid had no stress row; it keeps an AI weight here. d (honest) is
    # the kept personality gate.
    option = {
        name = eotg_aug_heir.001.e
        trigger = {
            has_trait = education_intrigue
            # The old personality gate kept a Seamless owner out (no
            # personality traits); keep that population (§12.1 rule 7).
            NOT = { has_trait = eotg_total_integration }
        }
        skill = intrigue
        scope:eotg_heir = {
            add_opinion = { modifier = eotg_opinion_aug_fear  target = root  years = 10 }
        }
        scope:eotg_heir_story = { change_variable = { name = eotg_heir_dread  add = 1 } }
        eotg_add_fracture_risk = { AMOUNT = 5 }
        eotg_aug_stress_cruelty_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = education_intrigue_3 }
            modifier = { add = 30  has_trait = education_intrigue_4 }
            modifier = { add = 30  has_trait = education_intrigue_5 }
            modifier = { add = 20  has_trait = paranoid }
            modifier = { add = 10  has_trait = vengeful }
            modifier = { add = -20  has_trait = trusting }
        }
    }
}
