# Ruling R8 (default (orchestrator, 2026-11-03), pending owner): after the
# Hollowing, exposure is pinned at 100 and the stage's resource is
# eotg_void_hollowing_depth (0-100; single writer
# eotg_void_change_hollowing_depth_effect). Every option here moves depth.
# Depth drifts +4 a year (the on_action), tilts the 0031/0032 pool, and shows

namespace = eotg_void

# -- 0030 The Hollowing ----------------------------------------------------------
eotg_void.0030 = {
    type = character_event
    title = eotg_void.0030.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { eotg_void_is_cult_faith = yes }
                desc = eotg_void.0030.desc_cult
            }
            desc = eotg_void.0030.desc
        }
    }
    theme = corruption
    override_background = { reference = eotg_bg_council }
    left_portrait = {
        character = root
        animation = shock
    }

    immediate = {
        eotg_void_become_hollowed_effect = yes
    }

    # Announce it as ascension (cult; inert, ruling R3).
    option = {
        name = eotg_void.0030.a
        trigger = { eotg_void_is_cult_faith = yes }
        add_piety = 400
        add_prestige = 150
        every_courtier = {
            limit = { eotg_void_is_cult_faith = yes }
            add_opinion = { modifier = eotg_opinion_void_kinship  target = root }
        }
        eotg_void_change_hollowing_depth_effect = { AMOUNT = 10 }
        ai_chance = { base = 90 }
    }

    # [education: intrigue] Say nothing; rule.
    option = {
        name = eotg_void.0030.b
        trigger = { has_trait = education_intrigue }
        skill = intrigue
        add_intrigue_skill = 2
        if = {
            limit = { eotg_void_has_any_insight = no }    # one Insight per character
            add_character_modifier = { modifier = eotg_mod_void_unseen_step }
        }
        eotg_void_change_hollowing_depth_effect = { AMOUNT = 5 }
        ai_chance = { base = 70 }
    }

    # Step down while the decision is still mine. Made real (ruling R9):
    # v1's "Abdicate" did not abdicate.
    option = {
        name = eotg_void.0030.c
        trigger = {
            is_ruler = yes    # fires up to 10 days late; land may be gone
            exists = primary_heir
        }
        add_piety = 300
        primary_heir = {
            add_opinion = { modifier = eotg_opinion_void_defiance_respect  target = root }
        }
        stress_impact = { base = major_stress_impact_loss }
        eotg_void_change_hollowing_depth_effect = { AMOUNT = -15 }
        eotg_void_step_down_effect = yes
        # AI never opts to abdicate (vanilla:
        # events/stress_events/stress_threshold_events.txt:16778, base 0).
        ai_chance = { base = 0 }
    }

    # Let them see it.
    option = {
        name = eotg_void.0030.d
        add_dread = 50
        add_prestige = 100
        eotg_void_change_hollowing_depth_effect = { AMOUNT = 15 }
        ai_chance = {
            base = 40
            modifier = { add = 35  has_trait = callous }
            modifier = { add = 30  has_trait = arrogant }
            modifier = { add = 25  has_trait = ambitious }
        }
    }
}
