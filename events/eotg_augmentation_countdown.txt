        name = eotg_aug_countdown.001.d

# ── eotg_aug_countdown.002 ────────────────────────────────────────────────────
# Lost Time. Days, not hours. [voice] at voice >= 2: the logs kept the days
# you did not.
# Caller: eotg_story_aug_countdown (stage 2).
# d [paranoid]: the suspect comes from the victim picker (family factor 0.05,
# Overclocked) and is arrested through vanilla imprison_character_effect
# (00_prison_effects.txt:1644; QA round 2 M5, as tier2.015.d): house arrest
# for family, else the dungeon, with tyranny and the family's opinion.

eotg_aug_countdown.002 = {
    type = character_event
    title = eotg_aug_countdown.002.t
    desc = {
        desc = eotg_aug_countdown.002.desc
        triggered_desc = {
            trigger = {
                has_variable = eotg_aug_voice
                var:eotg_aug_voice >= 2
            }
            desc = eotg_aug_countdown.002.desc_voice
        }
    }
    theme = mental_health
    override_background = { reference = eotg_bg_corridor_night }
    override_effect_2d = { reference = fog }

    trigger = { eotg_is_aug_tier3 = yes }

    left_portrait = {
        character = root
        animation = shock
    }

    immediate = {
        hidden_effect = {
            if = {
                limit = { has_trait = paranoid }
                eotg_aug_pick_victim_effect = { NAME = eotg_suspect  FAMILY_FACTOR = 0.05 }
            }
        }
    }

    # "Reconstruct the days."
    option = {
        name = eotg_aug_countdown.002.a
        add_character_modifier = { modifier = eotg_mod_aug_lesson_intrigue  years = 5 }
        eotg_add_fracture_risk = { AMOUNT = -3 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = diligent }
            modifier = { add = 10  has_trait = patient }
        }
    }

    # "Let them go."
    option = {
        name = eotg_aug_countdown.002.b
        # Rule 6 (final QA): a beat b outright; letting go is a relief.
        add_stress = minor_stress_loss
        eotg_add_fracture_risk = { AMOUNT = 5 }
        eotg_aug_stress_neglect_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = lazy }
            modifier = { add = 10  has_trait = content }
        }
    }

    # "An archivist logs my every hour."
    option = {
        name = eotg_aug_countdown.002.c
        add_prestige = -50
        eotg_add_fracture_risk = { AMOUNT = -5 }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = humble }
            modifier = { add = 10  has_trait = craven }
        }
    }

    # [paranoid] "Someone took them from me."
    option = {
        name = eotg_aug_countdown.002.d
        trigger = {
            has_trait = paranoid
            exists = scope:eotg_suspect
        }
        trait = paranoid
        if = {
            limit = { scope:eotg_suspect = { is_imprisoned = no } }
            imprison_character_effect = {
                TARGET = scope:eotg_suspect
                IMPRISONER = root
            }
        }
        scope:eotg_suspect = {
            add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
        }
        eotg_add_fracture_risk = { AMOUNT = 8 }
        eotg_aug_stress_tyranny_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = paranoid }
            modifier = { add = -20  has_trait = just }
        }
    }

    # [calm] "Breathe."
    option = {
        name = eotg_aug_countdown.002.e
        trigger = { has_trait = calm }
        trait = calm
        eotg_add_fracture_risk = { AMOUNT = -6 }
        stress_impact = {
            calm = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = calm }
            modifier = { add = 10  has_trait = patient }
        }
    }
}

# ── eotg_aug_countdown.004 ────────────────────────────────────────────────────

# Dead/hurt portrait guard: vanilla events/activities/activity_system_events.txt:89-92.

eotg_aug_countdown.004 = {
    type = character_event
    title = eotg_aug_countdown.004.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:eotg_victim }
                desc = eotg_aug_countdown.004.desc

            desc = eotg_aug_countdown.004.desc_none
        }
        triggered_desc = {
            trigger = {
                has_variable = eotg_aug_voice
                var:eotg_aug_voice >= 2

            desc = eotg_aug_countdown.004.desc_voice
        }
    }
    theme = dread

        character = root
        animation = rage
    }
    # The wound can kill (increase_wounds_effect): dead-portrait shape as in
    # eotg_fracture.002 (vanilla chariot_ongoing_events_jp.txt:2544).

        trigger = { exists = scope:eotg_victim }
        character = scope:eotg_victim

            trigger = { is_alive = no }

        name = eotg_aug_countdown.004.a

        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = generous }
            modifier = { add = 10  has_trait = just }
