# ── eotg_aug_tier2.001 ────────────────────────────────────────────────────────
# Emotional Delay
# The enhanced character notices their emotional responses have slowed.

eotg_aug_tier2.001 = {
    type = character_event
    title = eotg_aug_tier2.001.t
    desc = eotg_aug_tier2.001.desc
    theme = mental_health

    left_portrait = {
        character = root
        animation = personality_rational
    }

    # "Emotion clouds judgment" — lean into the detachment
    option = {
        name = eotg_aug_tier2.001.a
        add_character_modifier = { modifier = eotg_mod_aug_affect_dampened  years = 3 }
        # QA round 2 M14: leaning in has a price (spec §1 rule 6)
        eotg_add_fracture_risk = { AMOUNT = 3 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = callous }
            modifier = { add = 10  has_trait = ambitious }
        }
    }

    # "I do not like this"
    option = {
        name = eotg_aug_tier2.001.b
        stress_impact = { base = minor_stress_impact_gain }
        eotg_aug_stress_reject_effect = yes
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = compassionate }
            modifier = { add = 20  has_trait = content }
        }
    }

    # "Increase neural responsiveness" — push the system, add fracture risk
    option = {
        name = eotg_aug_tier2.001.c
        trigger = { gold >= minor_gold_value }
        remove_short_term_gold = minor_gold_value
        eotg_add_fracture_risk = { AMOUNT = 8 }
        ai_chance = {
            base = 20
            modifier = { add = 20  has_trait = brave }
            modifier = { add = 10  has_trait = ambitious }
        }
    }

    # [compassionate] "Then I will sit with them until I feel it." - the family
    # at court is reassured; the effort costs as much as it relieves.
    option = {
        name = eotg_aug_tier2.001.d
        trigger = { has_trait = compassionate }
        trait = compassionate
        every_spouse = {
            limit = { is_courtier_of = root }
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
        }
        every_child = {
            limit = {
                is_alive = yes
                is_courtier_of = root
            }
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
        }
        stress_impact = {
            base = minor_stress_impact_gain
            compassionate = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = compassionate }
            modifier = { add = 10  has_trait = patient }
        }
    }

    # [callous] "Good." - a longer dampening, and the implant runs the affect
    # it has taken over (risk +4)
    option = {
        name = eotg_aug_tier2.001.e
        trigger = { has_trait = callous }
        trait = callous
        add_character_modifier = { modifier = eotg_mod_aug_affect_dampened  years = 5 }
        eotg_add_fracture_risk = { AMOUNT = 4 }
        stress_impact = {
            callous = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = callous }
            modifier = { add = 10  has_trait = ambitious }
        }
    }
}

# ── eotg_aug_tier2.003 ────────────────────────────────────────────────────────
# The Next Stage — Progression Gate
# Enhanced → Overclocked transition event.
# Callers: eotg_on_yearly_aug_tier2_check (The Next Stage branch,
# unsolicited), eotg_decision_aug_pursue_next_stage (sought; balance spec
# §5.1). Sought (eotg_flag_aug_sought_upgrade, 60 days): the sought desc
# replaces the base line, the AI follows through (declines b, e x0.1), and the
# `after` block clears the flag whichever option is taken.

eotg_aug_tier2.003 = {
    type = character_event
    title = eotg_aug_tier2.003.t
    # E-12 (Phase 4b): what was replaced first (thread T7) and, once the
    # second self exists, "it wants this too" (thread T1).
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { has_character_flag = eotg_flag_aug_sought_upgrade }
                desc = eotg_aug_tier2.003.desc_sought
            }
            desc = eotg_aug_tier2.003.desc
        }
        first_valid = {
            triggered_desc = {
                trigger = { var:eotg_aug_focus ?= flag:limbs }
                desc = eotg_aug_tier2.003.desc_limbs
            }
            triggered_desc = {
                trigger = { var:eotg_aug_focus ?= flag:senses }
                desc = eotg_aug_tier2.003.desc_senses
            }
            triggered_desc = {
                trigger = { var:eotg_aug_focus ?= flag:nerves }
                desc = eotg_aug_tier2.003.desc_nerves
            }
        }
        triggered_desc = {
            trigger = {
                has_variable = eotg_aug_voice
                var:eotg_aug_voice >= 1
            }
            desc = eotg_aug_tier2.003.desc_voice
        }
    }
    theme = physical_health

    # World-state tier guard (mirrors the on_action branch): a regression
    # between roll and firing must not let the upgrade skip a tier.
    trigger = { eotg_is_aug_tier2 = yes }

    left_portrait = {
        character = root
        animation = personality_bold
    }

    # "Proceed" — pay and upgrade
    option = {
        name = eotg_aug_tier2.003.a
        trigger = { gold >= major_gold_value }
        remove_short_term_gold = major_gold_value
        eotg_aug_set_integration_effect = { XP = 100 }
        stress_impact = {
            base = minor_stress_impact_gain
        }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  has_trait = brave }
            modifier = { add = -50  has_trait = zealous }
            modifier = { add = -20  has_trait = content }
        }
    }

    # "The risks are unacceptable"
    option = {
        name = eotg_aug_tier2.003.b
        stress_impact = {
            base = 0
            ambitious = medium_stress_impact_gain
        }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = content }
            modifier = { add = 20  has_trait = patient }
            modifier = { add = 20  gold < major_gold_value }
            modifier = {    # balance spec §5.1: the ruler asked for this
                factor = 0.1
                has_character_flag = eotg_flag_aug_sought_upgrade
            }
        }
    }

    # "Test it on prisoners first" — callous path with fracture risk
    option = {
        name = eotg_aug_tier2.003.c
        # One step below option a's major_gold_value: the prisoners carry the

        trigger = {
            gold >= medium_gold_value

            limit = { is_alive = yes }

        eotg_aug_set_integration_effect = { XP = 100 }

        }

    # [ambitious] "Whatever it costs." - as a, leaning in (risk +5)

        name = eotg_aug_tier2.003.d
        trigger = {
            has_trait = ambitious
            gold >= major_gold_value
        }
        trait = ambitious
        remove_short_term_gold = major_gold_value
        eotg_aug_set_integration_effect = { XP = 100 }
        eotg_add_fracture_risk = { AMOUNT = 5 }

            base = minor_stress_impact_gain
            ambitious = minor_stress_impact_loss
        }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = -30  has_trait = zealous }
        }
    }

    # [content] "This is enough. I am enough." - as b, and the question is

    option = {
        name = eotg_aug_tier2.003.e
        trigger = { has_trait = content }
        trait = content
        add_character_flag = { flag = eotg_flag_suppress_progression  years = 5 }
        stress_impact = {
            content = medium_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = content }

            modifier = {    # balance spec §5.1: the ruler asked for this

# ── eotg_aug_tier2.004 ────────────────────────────────────────────────────────
# The Space Between Us
# A spouse confronts the Enhanced character about the widening emotional distance.
# Phase 4b (E-09): stage 1 of 3. Each option sets eotg_aug_marriage_strain
# (a 0, b +1, c +2, d 0) and fires .017 in 180-365 days; .017's trigger
# checks the marriage still stands. .018 removes the variable.

eotg_aug_tier2.004 = {
    type = character_event
    title = eotg_aug_tier2.004.t
    desc = eotg_aug_tier2.004.desc
    theme = family

    trigger = { is_married = yes }

    left_portrait = {
        character = root
        animation = stress
    }
    right_portrait = {
        character = scope:eotg_aug_spouse
        animation = worry
    }

    immediate = {
        hidden_effect = {
            random_spouse = {
                save_scope_as = eotg_aug_spouse
            }
        }
    }

    # "It is the implant. I am still here." — honest, some trust maintained
    option = {
        name = eotg_aug_tier2.004.a
        scope:eotg_aug_spouse = {
            add_opinion = { modifier = eotg_opinion_aug_unease  target = root  years = 5 }
        }
        eotg_add_fracture_risk = { AMOUNT = -3 }
        set_variable = { name = eotg_aug_marriage_strain  value = 0 }
        trigger_event = { id = eotg_aug_tier2.017  days = { 180 365 } }
        stress_impact = { base = minor_stress_impact_loss }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = honest }
            modifier = { add = 10  has_trait = compassionate }
        }
    }

    # "I am the same person I have always been." — denial, now a gamble
    # (dominated fix, audit §2.5): the spouse believes it, or sees through it.
    # Balance spec §5.6: the gamble is a diplomacy contest against the spouse
    # (vanilla duel with a target, events/activities/chariot_race_activity/
    # chariot_ongoing_events_jp.txt:195).
    option = {
        name = eotg_aug_tier2.004.b
        set_variable = { name = eotg_aug_marriage_strain  value = 1 }
        trigger_event = { id = eotg_aug_tier2.017  days = { 180 365 } }
        duel = {
            skill = diplomacy
            target = scope:eotg_aug_spouse
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = 3.5
                    min = -49
                }
                desc = eotg_aug_tier2.004.b.success
                scope:eotg_aug_spouse = {
                    add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
                }
            }
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = -3.5
                    min = -49
                }
                desc = eotg_aug_tier2.004.b.failure
                scope:eotg_aug_spouse = {
                    add_opinion = { modifier = eotg_opinion_aug_disgust  target = root  years = 8 }
                }
            }
        }
        stress_impact = {
            compassionate = minor_stress_impact_gain
        }
        eotg_aug_stress_lie_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = arrogant }
            modifier = { add = -20  has_trait = honest }
        }
    }

    # "Then perhaps we are no longer suited." — rupture
    option = {
        name = eotg_aug_tier2.004.c
        scope:eotg_aug_spouse = {
            add_opinion = { modifier = eotg_opinion_aug_disgust  target = root  years = 8 }
        }
        eotg_add_fracture_risk = { AMOUNT = 5 }
        set_variable = { name = eotg_aug_marriage_strain  value = 2 }
        trigger_event = { id = eotg_aug_tier2.017  days = { 180 365 } }
        stress_impact = { base = medium_stress_impact_gain }
        ai_chance = {
            base = 30
            modifier = { add = 20  has_trait = callous }
            modifier = { add = -30  has_trait = compassionate }
        }
    }

    # [deceitful] "Perform the old warmth." - it works; performing emotion is
    # work for the implant (risk +3)
    option = {
        name = eotg_aug_tier2.004.d
        trigger = { has_trait = deceitful }
        trait = deceitful
        scope:eotg_aug_spouse = {
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
        }
        eotg_add_fracture_risk = { AMOUNT = 3 }
        set_variable = { name = eotg_aug_marriage_strain  value = 0 }
