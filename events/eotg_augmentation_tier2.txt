# ── eotg_aug_tier2.001 ────────────────────────────────────────────────────────
# Emotional Delay
# The enhanced character notices their emotional responses have slowed.

eotg_aug_tier2.001 = {
    type = character_event
    title = eotg_aug_tier2.001.t
    desc = eotg_aug_tier2.001.desc
    theme = mental_health

    left_portrait = {
        character = root
        animation = personality_rational
    }

    # "Emotion clouds judgment" — lean into the detachment
    option = {
        name = eotg_aug_tier2.001.a
        add_character_modifier = { modifier = eotg_mod_aug_affect_dampened  years = 3 }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = callous }
            modifier = { add = 10  has_trait = ambitious }
        }
    }

    # "I do not like this"
    option = {
        name = eotg_aug_tier2.001.b
        stress_impact = { base = minor_stress_impact_gain }
        eotg_aug_stress_reject_effect = yes
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = compassionate }
            modifier = { add = 20  has_trait = content }
        }
    }

    # "Increase neural responsiveness" — push the system, add fracture risk
    option = {
        name = eotg_aug_tier2.001.c
        trigger = { gold >= minor_gold_value }
        remove_short_term_gold = minor_gold_value
        eotg_add_fracture_risk = { AMOUNT = 8 }
        ai_chance = {
            base = 20
            modifier = { add = 20  has_trait = brave }
            modifier = { add = 10  has_trait = ambitious }
        }
    }

    # [compassionate] "Then I will sit with them until I feel it." - the family
    # at court is reassured; the effort costs as much as it relieves.
    option = {
        name = eotg_aug_tier2.001.d
        trigger = { has_trait = compassionate }
        trait = compassionate
        every_spouse = {
            limit = { is_courtier_of = root }
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
        }
        every_child = {
            limit = {
                is_alive = yes
                is_courtier_of = root
            }
            add_opinion = { modifier = eotg_opinion_aug_reassured  target = root  years = 5 }
        }
        stress_impact = {
            base = minor_stress_impact_gain
            compassionate = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = compassionate }
            modifier = { add = 10  has_trait = patient }
        }
    }

    # [callous] "Good." - a longer dampening, and the implant runs the affect
    # it has taken over (risk +4)
    option = {
        name = eotg_aug_tier2.001.e
        trigger = { has_trait = callous }
        trait = callous
        add_character_modifier = { modifier = eotg_mod_aug_affect_dampened  years = 5 }
        eotg_add_fracture_risk = { AMOUNT = 4 }
        stress_impact = {
            callous = minor_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = callous }
            modifier = { add = 10  has_trait = ambitious }
        }
    }
}

# ── eotg_aug_tier2.003 ────────────────────────────────────────────────────────
# The Next Stage — Progression Gate
# Enhanced → Overclocked transition event.

eotg_aug_tier2.003 = {
    type = character_event
    title = eotg_aug_tier2.003.t
    desc = eotg_aug_tier2.003.desc
    theme = physical_health

    # World-state tier guard (mirrors the on_action branch): a regression
    # between roll and firing must not let the upgrade skip a tier.
    trigger = { eotg_is_aug_tier2 = yes }

    left_portrait = {
        character = root
        animation = personality_bold
    }

    # "Proceed" — pay and upgrade
    option = {
        name = eotg_aug_tier2.003.a
        trigger = { gold >= major_gold_value }
        remove_short_term_gold = major_gold_value
        eotg_aug_set_integration_effect = { XP = 100 }
        stress_impact = {
            base = minor_stress_impact_gain
        }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  has_trait = brave }
            modifier = { add = -50  has_trait = zealous }
            modifier = { add = -20  has_trait = content }
        }
    }

    # "The risks are unacceptable"
    option = {
        name = eotg_aug_tier2.003.b
        stress_impact = {
            base = 0
            ambitious = medium_stress_impact_gain
        }
        ai_chance = {
            base = 40
            modifier = { add = 20  has_trait = content }
            modifier = { add = 20  has_trait = patient }
            modifier = { add = 20  gold < major_gold_value }
        }
    }

    # "Test it on prisoners first" — callous path with fracture risk
    option = {
        name = eotg_aug_tier2.003.c
        # One step below option a's major_gold_value: the prisoners carry the
        # risk, not the purse (phase 0 §1.5 mapping).
        trigger = {
            gold >= medium_gold_value
            any_prisoner = { is_alive = yes }
        }
        remove_short_term_gold = medium_gold_value
        random_prisoner = {
            limit = { is_alive = yes }
            increase_wounds_effect = { REASON = treatment }
        }
        add_dread = 15
        eotg_add_fracture_risk = { AMOUNT = 5 }
        eotg_aug_set_integration_effect = { XP = 100 }
        eotg_aug_stress_wound_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = callous }
            modifier = { add = 10  has_trait = arbitrary }
        }
    }

    # [ambitious] "Whatever it costs." - as a, leaning in (risk +5)
    option = {
        name = eotg_aug_tier2.003.d
        trigger = {
            has_trait = ambitious
            gold >= major_gold_value
        }
        trait = ambitious
        remove_short_term_gold = major_gold_value
        eotg_aug_set_integration_effect = { XP = 100 }
        eotg_add_fracture_risk = { AMOUNT = 5 }
        # One helper only (Phase 1 QA): the surgery helper already gives
        # ambitious minor relief; the gating trait gets one more minor line.
        stress_impact = {
            base = minor_stress_impact_gain
            ambitious = minor_stress_impact_loss
        }
        eotg_aug_stress_surgery_effect = yes
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = -30  has_trait = zealous }
        }
    }

    # [content] "This is enough. I am enough." - as b, and the question is
    # closed for five years
    option = {
        name = eotg_aug_tier2.003.e
        trigger = { has_trait = content }
        trait = content
        add_character_flag = { flag = eotg_flag_suppress_progression  years = 5 }
        stress_impact = {
            content = medium_stress_impact_loss
        }
        ai_chance = {
            base = 30
            modifier = { add = 30  has_trait = content }
            modifier = { add = 10  has_trait = humble }
        }
    }
}
