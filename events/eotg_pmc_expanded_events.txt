# eotg_pmc_flavor.0001 — A New Contract Offer Arrives
# A foreign power offers payment for military services.
eotg_pmc_flavor.0001 = {
	type = character_event
	title = eotg_pmc_flavor_0001_title
	desc = eotg_pmc_flavor_0001_desc
	theme = stewardship

	left_portrait = { character = root  animation = personality_bold }

	trigger = {
		eotg_st_is_contractor_general = yes
		is_at_war = no
		NOT = { has_character_modifier = eotg_pmc_active_contract }
		NOT = { has_character_flag = eotg_pmc_0001_recent }
	}

	option = {
		name = eotg_pmc_flavor_0001_opt_accept
		add_character_modifier = { modifier = eotg_pmc_active_contract  years = 4 }
		add_gold = 60
		add_character_flag = { flag = eotg_pmc_0001_recent  years = 3 }
		ai_chance = { base = 50  modifier = { add = 20  has_trait = ambitious } }
	}
	option = {
		name = eotg_pmc_flavor_0001_opt_negotiate
		add_diplomacy_skill = 1
		add_character_flag = { flag = eotg_pmc_0001_recent  years = 3 }
		ai_chance = { base = 30  modifier = { add = 15  has_trait = gregarious } }
	}
	option = {
		name = eotg_pmc_flavor_0001_opt_reject
		add_prestige = 10
		every_vassal = {
			limit = { eotg_st_is_executive_commander = yes }
			add_opinion = { target = root  modifier = eotg_fringe_fear_rule_resentment }
		}
		add_character_flag = { flag = eotg_pmc_0001_recent  years = 3 }
		ai_chance = { base = 20  modifier = { add = 10  has_trait = lazy } }
	}
}
