# TEST ONLY: not mod content. EotG Test: Race Framework (docs/test_submods/races/README.md).
# Fills the trigger half of the race registry by REDEFINING the main mod's keys
# from common/scripted_triggers/eotg_race_registry_triggers.txt (key-level
# override, differently named file; see zz_eotg_test_race_registry_effects.txt).
# Registered: test_a, test_b (both grouped) and the half-race test_half_ab
# (no heritage group).

eotg_race_framework_active = {
    always = yes
}

# Every race trait carries group eotg_race (common/traits/eotg_test_race_traits.txt).
eotg_race_has_any_race = {
    has_trait = eotg_race
}

# The umbrella parameter on both test heritages (eotg_test_race_pillars.txt).
eotg_race_culture_is_grouped = {
    has_cultural_parameter = eotg_heritage_grouped
}

eotg_race_same_as = {
    OR = {
        AND = { has_trait = eotg_race_test_a  $TARGET$ = { has_trait = eotg_race_test_a } }
        AND = { has_trait = eotg_race_test_b  $TARGET$ = { has_trait = eotg_race_test_b } }
        AND = { has_trait = eotg_race_test_half_ab  $TARGET$ = { has_trait = eotg_race_test_half_ab } }
    }
}

# Grouped races only (the same list as eotg_race_for_each_grouped_race_effect).
# test_half_ab has no heritage pillar, so eotg_heritage_group_test_half_ab is
# defined nowhere and must not be read.
eotg_race_cultures_in_different_groups = {
    OR = {
        AND = { $A$ = { eotg_is_heritage_group = { RACE = test_a } }  $B$ = { NOT = { eotg_is_heritage_group = { RACE = test_a } } } }
        AND = { $A$ = { eotg_is_heritage_group = { RACE = test_b } }  $B$ = { NOT = { eotg_is_heritage_group = { RACE = test_b } } } }
    }
}
