# -- eotg_observer_on_actions.txt ---------------------------------------------
# THROWAWAY observer sub-mod for the 50-year run (balance spec
# docs/specs/cybernetics_v2_balance.md §9.2). Kept in docs/tools/observer/; never loaded by the main mod.
#
# Logging only. Nothing here changes gameplay: every effect is debug_log,
# a global counter (eotg_obs_tick) or, in the overlay files, an observer-only
# flag (eotg_obs_sought / eotg_obs_pursued) that only the observer reads.
#
# Vanilla hooks extended additively (on_actions = { }), as the main mod does:
#   yearly_global_pulse   - every Jan 1, no root (yearly_on_actions.txt:1-3):
#                           year marker; census every 10th pulse
#   on_game_start_after_lobby - once (game_start.txt): census at year 0
#   yearly_playable_pulse - count+ rulers (yearly_on_actions.txt:973):
#                           gold-gate sample (§9.2 item 7)
#   on_death              - (death.txt:7): Neurofractured / Seamless deaths
# debug_log takes a loc key resolved with THIS / ROOT (PX script_docs
# effects.log: "the message can be a localization string with ROOT, SCOPE
# and PREV available"; vanilla precedent: create_noble_family_effect,
# common/scripted_effects/07_dlc_ep3_scripted_effects.txt:3248, whose loc
# reads [THIS.GetCharacter...]).
# -----------------------------------------------------------------------------

yearly_global_pulse = {
    on_actions = {
        eotg_obs_yearly_global
    }
}

on_game_start_after_lobby = {
    on_actions = {
        eotg_obs_game_start
    }
}

yearly_playable_pulse = {
    on_actions = {
        eotg_obs_yearly_ruler
    }
}

on_death = {
    on_actions = {
        eotg_obs_on_death
    }
}

eotg_obs_game_start = {
    effect = {
        set_global_variable = { name = eotg_obs_tick  value = 0 }
        debug_log = eotg_obs_start
        eotg_obs_census_effect = yes
    }
}

# Year marker every Jan 1 (the parser times years 40-50 from these lines'
# real-time stamps, §9.2 item 12); census every 10th pulse (items 1, 5, 8, 9).
eotg_obs_yearly_global = {
    effect = {
        debug_log = eotg_obs_year
        change_global_variable = { name = eotg_obs_tick  add = 1 }
        if = {
            limit = { global_var:eotg_obs_tick >= 10 }
            set_global_variable = { name = eotg_obs_tick  value = 0 }
            eotg_obs_census_effect = yes
        }
    }
}

# §9.2 item 7: tier-1 and tier-2 rulers past settling and past the dev gate
# (the main mod's eotg_can_progress_to_* triggers): can they pay for Pursue?
# The test mirrors the decision's ai_potential gold lines.
eotg_obs_yearly_ruler = {
    trigger = {
        has_trait = eotg_cybernetics
    }
    effect = {
        if = {
            limit = { eotg_can_progress_to_enhanced = yes }
            if = {
                limit = { short_term_gold >= medium_gold_value }
                debug_log = eotg_obs_gold_t1_ok
            }
            else = {
                debug_log = eotg_obs_gold_t1_fail
            }
        }
        else_if = {
            limit = { eotg_can_progress_to_overclocked = yes }
            if = {
                limit = { short_term_gold >= major_gold_value }
                debug_log = eotg_obs_gold_t2_ok
            }
            else = {
                debug_log = eotg_obs_gold_t2_fail
            }
        }
    }
}

# §9.2 item 9: the end of a Neurofractured (or Seamless) life, whatever the
# cause. The terminal effects log their own kind first (overlay).
eotg_obs_on_death = {
    trigger = {
        OR = {
            has_trait = eotg_neurofractured
            has_trait = eotg_total_integration
        }
    }
    effect = {
        if = {
            limit = { has_trait = eotg_neurofractured }
            debug_log = eotg_obs_death_nf
        }
        else = {
            debug_log = eotg_obs_death_ti
        }
    }
}
