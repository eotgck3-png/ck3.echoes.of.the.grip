# ── eotg_legacy_on_actions.txt ───────────────────────────────────────────────
# Event hooks for the Confluence Legacies system.
# Vanilla hooks are extended additively (on_actions sub-blocks merge across
# files) — never with effect = {} blocks.
# ─────────────────────────────────────────────────────────────────────────────

on_yearly_playable = {
    on_actions = { eotg_on_yearly_legacy_pulse }
}

on_title_gain = {
    on_actions = { eotg_on_title_gain_legacy_check }
}

on_game_start_after_lobby = {
    on_actions = { eotg_legacy_seed_start }
}

# ── Yearly pulse ─────────────────────────────────────────────────────────────
# Attention decay, the Exodus tie-in, and the shared aftermath pool.
# At most one aftermath event per ruler per two years; each site also carries
# its own cooldown so the same county stays quiet after a major event.
eotg_on_yearly_legacy_pulse = {
    effect = {
        if = {
            limit = {
                is_landed = yes
                is_adult = yes
            }

            # Attention fades when nothing feeds it
            every_sub_realm_county = {
                limit = {
                    holder = root
                    eotg_st_legacy_known_site = yes
                    var:eotg_legacy_attention > 0
                }
                eotg_se_legacy_add_attention = { AMOUNT = -2 }
            }

            # Titan Exodus tie-in: sheltered refugees recognize a fragment of
            # their lost homeworld buried in the ruler's domain.
            if = {
                limit = {
                    has_character_modifier = eotg_exodus_haven_reputation
                    eotg_st_holds_hidden_legacy = yes
                }
                random = {
                    chance = 10
                    random_sub_realm_county = {
                        limit = {
                            holder = root
                            eotg_st_legacy_hidden_site = yes
                        }
                        eotg_se_legacy_fire_discovery = yes
                    }
                }
            }

            # ── Aftermath dispatch ───────────────────────────────────────────
            if = {
                limit = {
                    NOT = { has_character_flag = eotg_legacy_aftermath_recent }
                }

                # The Rival's Archaeologist — fame invites claimants
                if = {
                    limit = {
                        any_sub_realm_county = {
                            holder = root
                            eotg_st_legacy_known_site = yes
                            eotg_st_legacy_attention_at_least = { VALUE = 50 }
                            eotg_st_legacy_ready = yes
                        }
                    }
                    random = {
                        chance = 25
                        add_character_flag = { flag = eotg_legacy_aftermath_recent  years = 2 }
                        random_sub_realm_county = {
                            limit = {
                                holder = root
                                eotg_st_legacy_known_site = yes
                                eotg_st_legacy_attention_at_least = { VALUE = 50 }
                                eotg_st_legacy_ready = yes
                            }
                            save_scope_as = eotg_legacy_county
                            eotg_se_legacy_cooldown = { YEARS = 4 }
                        }
                        trigger_event = { id = eotg_legacy.0101  days = { 10 40 } }
                    }
                }

                # The Concession's Beneficiaries — extraction enriches someone
                else_if = {
                    limit = {
                        any_vassal = { is_powerful_vassal = yes }
                        any_sub_realm_county = {
                            holder = root
                            eotg_st_legacy_known_site = yes
                            eotg_st_legacy_condition = { CONDITION = 4 }
                            eotg_st_legacy_ready = yes
                        }
                    }
                    random = {
                        chance = 20
                        add_character_flag = { flag = eotg_legacy_aftermath_recent  years = 2 }
                        random_sub_realm_county = {
                            limit = {
                                holder = root
                                eotg_st_legacy_known_site = yes
                                eotg_st_legacy_condition = { CONDITION = 4 }
                                eotg_st_legacy_ready = yes
                            }
                            save_scope_as = eotg_legacy_county
                            eotg_se_legacy_cooldown = { YEARS = 5 }
                        }
                        trigger_event = { id = eotg_legacy.0102  days = { 10 40 } }
                    }
                }

                # The Cult Beneath the Monument — damaged quiet sites attract worship
                else_if = {
                    limit = {
                        any_sub_realm_county = {
                            holder = root
                            eotg_st_legacy_known_site = yes
                            eotg_st_legacy_integrity_at_most = { VALUE = 50 }
                            eotg_st_legacy_ready = yes
                            OR = {
                                eotg_st_legacy_condition = { CONDITION = 3 }
                                eotg_st_legacy_condition = { CONDITION = 5 }
                            }
                        }
                    }
                    random = {
                        chance = 15
                        add_character_flag = { flag = eotg_legacy_aftermath_recent  years = 2 }
                        random_sub_realm_county = {
                            limit = {
                                holder = root
                                eotg_st_legacy_known_site = yes
                                eotg_st_legacy_integrity_at_most = { VALUE = 50 }
                                eotg_st_legacy_ready = yes
                                OR = {
                                    eotg_st_legacy_condition = { CONDITION = 3 }
                                    eotg_st_legacy_condition = { CONDITION = 5 }
                                }
                            }
                            save_scope_as = eotg_legacy_county
                            eotg_se_legacy_cooldown = { YEARS = 5 }
                        }
                        trigger_event = { id = eotg_legacy.0103  days = { 10 40 } }
                    }
                }
            }
        }
    }
}

# ── County transfer ──────────────────────────────────────────────────────────
# A Legacy remembers: the new holder inherits the site, its condition, its
# attention — and the politics. root = new holder, scope:title = gained title.
eotg_on_title_gain_legacy_check = {
    effect = {
        if = {
            limit = {
                scope:title = {
                    tier = tier_county
                    eotg_st_legacy_known_site = yes
                    eotg_st_legacy_ready = yes
                }
            }
            scope:title = {
                save_scope_as = eotg_legacy_county
                eotg_se_legacy_cooldown = { YEARS = 3 }
            }
            trigger_event = { id = eotg_legacy.0105  days = { 20 60 } }
        }
    }
}

# ── Game-start seeding ───────────────────────────────────────────────────────
# Famous hand-placed sites at the bookmark powers' capitals, plus a scattering
# of hidden sites across independent realms. Guards make every placement a
# no-op until the map pass gives characters real capitals (Gate 1).
eotg_legacy_seed_start = {
    effect = {
        # The Cauldron itself — New Cauldron stands on a Second Era capital
        eotg_se_legacy_seed_capital = { CHAR = eotg_char_20001  FAMILY = 1 }
        # Caer Myr — the Prince of Myr sits atop a dead court's records
        eotg_se_legacy_seed_capital = { CHAR = eotg_char_10002  FAMILY = 2 }
        # The Trade League's lead prince built on working Second Era industry
        eotg_se_legacy_seed_capital = { CHAR = eotg_char_30001  FAMILY = 3 }
        # Coldiron space holds a sealed fragment of an infernal layer
        eotg_se_legacy_seed_capital = { CHAR = eotg_char_40001  FAMILY = 4 }
        # Clayd's Helix remnant lives above a buried metropolis of their own people
        eotg_se_legacy_seed_capital = { CHAR = eotg_char_90002  FAMILY = 1 }
        # Vaelorin's seat at Xerxes keeps an archive fragment
        eotg_se_legacy_seed_capital = { CHAR = eotg_char_90001  FAMILY = 2 }
        # The 7th Legion garrisons a relic works
        eotg_se_legacy_seed_capital = { CHAR = eotg_char_90003  FAMILY = 3 }

        # Hidden sites: roughly a third of independent realms hold one, buried
        every_independent_ruler = {
            limit = { is_landed = yes }
            random = {
                chance = 30
                random_sub_realm_county = {
                    limit = {
                        NOT = { has_variable = eotg_legacy_family }
                    }
                    eotg_se_legacy_seed_random = yes
                }
            }
        }
    }
}
