# Script values for the New Cauldron Democracy government.
# ─────────────────────────────────────────────────────────────────────────────

# Campaign vote weight for a presidential candidate.
# Base: stewardship + gold/20. Bonuses for promise, former president, rally.
eotg_sv_nc_vote_weight = {
	value = stewardship
	add = {
		value = gold
		divide = 20
	}
	if = {
		limit = { has_character_flag = eotg_nc_promise_made }
		add = 5
	}
	if = {
		limit = { has_character_modifier = eotg_former_president }
		add = 10
	}
	if = {
		limit = { has_variable = eotg_nc_rally_bonus }
		add = var:eotg_nc_rally_bonus
	}
	if = {
		limit = { has_character_flag = eotg_nc_rivals_record_damaged }
		add = -8
	}
	min = 0
}

# Count of senators and states with positive opinion of the President.
eotg_sv_nc_senate_support_count = {
	value = 0
	every_vassal = {
		limit = {
			OR = {
				eotg_st_is_nc_state = yes
				AND = {
					eotg_st_is_nc_senator = yes
					is_powerful_vassal = yes
				}
			}
			opinion = { target = root  value >= 0 }
		}
		add = 1
	}
}

# Total count of senators and states — denominator for support calculation.
eotg_sv_nc_senate_total = {
	value = 0
	every_vassal = {
		limit = {
			OR = {
				eotg_st_is_nc_state = yes
				AND = {
					eotg_st_is_nc_senator = yes
					is_powerful_vassal = yes
				}
			}
		}
		add = 1
	}
}

# Cost to Rally the Districts. Increases per prior rally (diminishing returns).
eotg_sv_nc_rally_cost = {
	value = 100
	if = {
		limit = { has_variable = eotg_nc_rally_bonus }
		add = {
			value = var:eotg_nc_rally_bonus
			multiply = 50
		}
	}
}

# Cost for the President to buy Senate support — flat 200 gold.
eotg_sv_nc_senate_buy_cost = {
	value = 200
}
