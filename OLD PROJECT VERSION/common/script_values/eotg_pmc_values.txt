# ── eotg_pmc_values.txt ──────────────────────────────────────────────────────
# Script values for the PMC government.
# ─────────────────────────────────────────────────────────────────────────────

# Contract gold payouts
eotg_sv_pmc_contract_minor_gold = 150
eotg_sv_pmc_contract_standard_gold = 350
eotg_sv_pmc_contract_exclusive_gold = 750

# Contract prestige payouts
eotg_sv_pmc_contract_minor_prestige = 50
eotg_sv_pmc_contract_standard_prestige = 100
eotg_sv_pmc_contract_exclusive_prestige = 200

# Count of powerful vassals who have withdrawn forces
eotg_sv_pmc_withdrawal_count = {
	value = 0
	every_vassal = {
		limit = {
			is_powerful_vassal = yes
			has_character_flag = eotg_pmc_withdrew_forces
		}
		add = 1
	}
}

# Number of powerful vassals needed to trigger Loss of Command
eotg_sv_pmc_withdrawal_threshold = 3

# Gold cost to buy back vassal loyalty — scales with number of extensions purchased
eotg_sv_pmc_loyalty_buyback_cost = {
	value = 200
	if = {
		limit = { has_variable = eotg_pmc_extension_count }
		multiply = var:eotg_pmc_extension_count
	}
}
