# Logistics Officer council position — PMC government only.
# Replaces the functional role of Court Chaplain for PMC rulers.
# Uses Stewardship skill. Three tasks: Supply Lines, Resources, Audit.
# ─────────────────────────────────────────────────────────────────────────────

eotg_pmc_logistics_officer = {
	skill = stewardship

	name = {
		desc = eotg_pmc_logistics_officer_name
	}

	valid_position = {
		has_government = eotg_pmc_government
	}

	tooltip = {
		desc = eotg_pmc_logistics_officer_tooltip
	}

	auto_fill = {
	}

	valid_character = {
		is_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	on_get_position = {
		got_council_position_effect = yes
	}

	on_fired_from_position = {
		fired_from_council_position_effect = yes
	}

	portrait_animation = steward

	barbershop_data = {
		position = { 918.0 380.0 }
		click_to_front = no
	}
}
