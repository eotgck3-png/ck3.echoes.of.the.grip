# ── eotg_story_cycle_866_opening.txt ──────────────────────────────────────────
# Story Cycle: The Weight of 866
# Created for each player ruler at game start (eotg_char_arc_actions.txt).
# Introduces the era's three defining tensions via a chained framing sequence:
#   eotg_orrins_grip.0001 — the memory of Orrin's Grip (866 years old)
#   eotg_orrins_grip.0002 — the Titan Exodus is still ongoing
#   eotg_orrins_grip.0003 — the Myr Cluster Wars have just ignited
# The chain advances via each event's `after` block; the cycle ends itself
# once the sequence has had time to play out.
# ─────────────────────────────────────────────────────────────────────────────

eotg_story_866_opening = {
    on_setup = {
        story_owner = {
            trigger_event = { id = eotg_orrins_grip.0001  days = 5 }
        }
    }

    on_end = {
        story_owner = {
            add_prestige = 100
        }
    }

    on_owner_death = {
        end_story = yes
    }

    # The framing chain finishes within ~2 months; close the story after a year.
    effect_group = {
        days = { 365 400 }

        triggered_effect = {
            trigger = { always = yes }
            effect = {
                end_story = yes
            }
        }
    }
}
