# ── eotg_corporation_triggers.txt ─────────────────────────────────────────────
# Scripted triggers for the Corporation government.
# ─────────────────────────────────────────────────────────────────────────────

# Root has the Corporation government
eotg_st_is_corporation_ruler = {
	has_government = eotg_corporation_government
	is_landed = yes
}

# Account Director — county or duchy tier under Corporation
eotg_st_is_account_director = {
	eotg_st_is_corporation_ruler = yes
	highest_held_title_tier >= tier_county
	highest_held_title_tier < tier_kingdom
}

# Operations Chief — duchy tier
eotg_st_is_operations_chief = {
	eotg_st_is_corporation_ruler = yes
	highest_held_title_tier = tier_duchy
}

# Regional Director — kingdom tier. Board-eligible.
eotg_st_is_regional_director = {
	eotg_st_is_corporation_ruler = yes
	highest_held_title_tier = tier_kingdom
}

# CEO — empire tier. Supreme authority.
eotg_st_is_ceo = {
	eotg_st_is_corporation_ruler = yes
	highest_held_title_tier = tier_empire
}

# Character can initiate a Sabotage Rival scheme against target
# Note: eotg_st_is_account_director covers duchy tier (Operations Chiefs) too
eotg_st_corp_can_sabotage = {
	eotg_st_is_account_director = yes
	scope:target = {
		eotg_st_is_account_director = yes
		top_liege = { this = scope:owner.top_liege }
		this != scope:owner
	}
}

# CEO has critically low board influence (no-confidence threshold)
eotg_st_corp_board_influence_critical = {
	eotg_st_is_ceo = yes
	trigger_if = {
		limit = { has_variable = eotg_board_influence }
		var:eotg_board_influence <= 20
	}
	trigger_else = {
		always = yes
	}
}

# Valid initiator for Corporate Vote of No Confidence
eotg_st_corp_vote_valid = {
	eotg_st_is_regional_director = yes
	liege = { eotg_st_corp_board_influence_critical = yes }
	NOT = { has_character_flag = eotg_corp_voted }
}

# CEO has already used Corporate Interference this election cycle
eotg_st_corp_interference_used = {
	has_character_flag = eotg_corp_interference_active
}
