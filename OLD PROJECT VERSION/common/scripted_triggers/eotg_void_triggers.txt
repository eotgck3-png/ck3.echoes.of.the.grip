# -- eotg_void_triggers.txt ---------------------------------------------------
# Reusable triggers for the Void Corruption system.
# Prefix: eotg_st_void_
#
# LORE BASIS (docs/SETTING LORE):
#   The Void is the prison of ORRIN THE EYE -- sealed, unable to act directly,
#   but his influence grows as the seals weaken. The Grip-era rift sites have
#   been dormant for centuries and are re-activating across 850-866 AG; the
#   Grip scars never fully healed. Corruption zones are expanding around them.
#   MALVRICK (Darkness, Void, Secrets) is the god whose VOIDWALKER sect alone
#   understood how to ward these places -- and that sect was hunted to near-
#   extinction between 200 and 700 AG. Its remnant cults are re-emerging in
#   secret at exactly the moment their knowledge is needed again.
#
#   So: exposure is proximity to a reactivating scar, and the only reliable
#   ward comes from a hunted heretic. That tension is the system.
#
# NOTE: eotg_st_is_void_touched lives in eotg_scripted_triggers.txt -- do not
# redefine it here.
#
# Exposure bands (variable eotg_void_exposure, 0-100):
#     0-24   Latent      -- nothing fires
#    25-49   Whispers    -- stage A events, still refusable
#    50-74   Pressure    -- the Bargain becomes available
#    75-99   Saturation  -- the Void takes what was not given
#      100   Hollowing   -- terminal transformation
# -----------------------------------------------------------------------------

# -- Susceptibility ------------------------------------------------------------

# eotg_st_void_susceptible
# TRUE if the character can accrue Void exposure at all. Orrin cannot act
# directly; his reach needs a thin place. Bloodline, worship, proximity to
# someone already taken, or having stood at a reactivating rift scar.
eotg_st_void_susceptible = {
    is_landed = yes
    is_alive = yes
    NOT = { has_trait = eotg_void_hollowed }
    OR = {
        has_trait = eotg_grip_survivor
        has_trait = eotg_void_touched
        eotg_st_void_venerates = yes
        eotg_st_void_near_touched = yes
        has_character_flag = eotg_flag_void_seeded
    }
}

# eotg_st_void_venerates
# TRUE if the character's faith treats the Void as sacred rather than hostile.
eotg_st_void_venerates = {
    faith = { religion_tag = eotg_religion_void_reverence }
}

# eotg_st_void_near_touched
# TRUE if someone in the character's immediate orbit is already Void-touched.
eotg_st_void_near_touched = {
    OR = {
        any_courtier = { is_alive = yes  has_trait = eotg_void_touched }
        any_vassal = { is_alive = yes  has_trait = eotg_void_touched }
        any_spouse = { is_alive = yes  has_trait = eotg_void_touched }
    }
}

# eotg_st_is_void_hollowed
# TRUE if the character has passed the terminal stage.
eotg_st_is_void_hollowed = {
    has_trait = eotg_void_hollowed
}

# eotg_st_void_has_voidwalker
# TRUE if a surviving Voidwalker -- a Malvrick heretic -- is sheltering in the
# character's court. They are the only people alive who still know the wards,
# and harbouring one is a standing political liability.
eotg_st_void_has_voidwalker = {
    any_courtier = {
        is_alive = yes
        has_character_flag = eotg_flag_void_voidwalker
    }
}

# eotg_st_void_rift_contact
# TRUE if the character has stood at a reactivating Grip-era rift scar.
eotg_st_void_rift_contact = {
    has_character_flag = eotg_flag_void_seeded
}

# -- Exposure bands ------------------------------------------------------------

# eotg_st_void_has_exposure
# TRUE if the exposure variable has been initialised at all.
eotg_st_void_has_exposure = {
    has_variable = eotg_void_exposure
}

# eotg_st_void_whisper_stage
# TRUE at 25+ exposure with no trait yet -- the refusable window.
eotg_st_void_whisper_stage = {
    eotg_st_void_has_exposure = yes
    var:eotg_void_exposure >= 25
    NOT = { has_trait = eotg_void_touched }
    NOT = { has_trait = eotg_void_hollowed }
}

# eotg_st_void_bargain_ready
# TRUE when the Void is ready to make its offer. A character who already
# refused it once (Defiant) is not asked again until the ward lapses.
eotg_st_void_bargain_ready = {
    eotg_st_void_has_exposure = yes
    var:eotg_void_exposure >= 50
    NOT = { has_trait = eotg_void_touched }
    NOT = { has_trait = eotg_void_hollowed }
    NOT = { has_trait = eotg_void_defiant }
}

# eotg_st_void_saturated
# TRUE at 75+ with no trait -- past the point where refusal still holds.
eotg_st_void_saturated = {
    eotg_st_void_has_exposure = yes
    var:eotg_void_exposure >= 75
    NOT = { has_trait = eotg_void_touched }
    NOT = { has_trait = eotg_void_hollowed }
}

# eotg_st_void_hollowing_ready
# TRUE when a Void-touched character has reached the terminal threshold.
eotg_st_void_hollowing_ready = {
    has_trait = eotg_void_touched
    eotg_st_void_has_exposure = yes
    var:eotg_void_exposure >= 100
}

# -- Resistance ----------------------------------------------------------------

# eotg_st_void_resists
# TRUE if the character is actively warded against the Void.
eotg_st_void_resists = {
    OR = {
        has_trait = eotg_void_defiant
        has_character_modifier = eotg_mod_void_warded
    }
}

# eotg_st_void_can_be_warded
# Gate for the Warding Rite decision -- you cannot ward what has already taken
# root past saturation, and the Hollowed are beyond it entirely.
eotg_st_void_can_be_warded = {
    eotg_st_void_has_exposure = yes
    var:eotg_void_exposure >= 10
    NOT = { has_trait = eotg_void_hollowed }
    NOT = { has_character_modifier = eotg_mod_void_warded }
}

# -- Event pacing --------------------------------------------------------------

# eotg_st_void_event_ready
# Shared cooldown gate so the Void does not crowd out other content.
eotg_st_void_event_ready = {
    NOT = { has_character_flag = eotg_flag_void_event_cooldown }
}
