# ── eotg_fringe_triggers.txt ─────────────────────────────────────────────────
# Scripted triggers for the Fringe Government.
# ─────────────────────────────────────────────────────────────────────────────

# True for any character holding the Fringe government.
eotg_st_is_fringe_ruler = {
	has_government = eotg_fringe_government
	OR = {
		highest_held_title_tier = tier_kingdom
		highest_held_title_tier = tier_empire
		highest_held_title_tier = tier_duchy
	}
}

# True for the top-level Fringe ruler (kingdom or empire tier).
eotg_st_is_fringe_overlord = {
	has_government = eotg_fringe_government
	OR = {
		highest_held_title_tier = tier_kingdom
		highest_held_title_tier = tier_empire
	}
}

# True for Fringe captains — vassals with duchy or lower holdings.
eotg_st_is_fringe_captain = {
	has_government = eotg_fringe_government
	OR = {
		highest_held_title_tier = tier_duchy
		highest_held_title_tier = tier_county
	}
	is_powerful_vassal = yes
}

# Grip state checks — require variable to exist.
eotg_st_fringe_grip_high = {
	has_variable = eotg_fringe_grip
	var:eotg_fringe_grip >= 80
}

eotg_st_fringe_grip_stable = {
	has_variable = eotg_fringe_grip
	var:eotg_fringe_grip >= 40
	var:eotg_fringe_grip < 80
}

eotg_st_fringe_grip_low = {
	has_variable = eotg_fringe_grip
	var:eotg_fringe_grip < 40
}

eotg_st_fringe_grip_critical = {
	has_variable = eotg_fringe_grip
	var:eotg_fringe_grip <= 10
}

# True when the realm is large enough to trigger overextension.
eotg_st_fringe_overextended = {
	has_government = eotg_fringe_government
	realm_size >= 30
}

# True when the ruler has been at peace long enough for decay to begin.
eotg_st_fringe_peace_decaying = {
	has_variable = eotg_fringe_peace_years
	var:eotg_fringe_peace_years >= 2
}

# True when reform prerequisites are met for any reform path.
eotg_st_fringe_reform_ready = {
	eotg_st_is_fringe_overlord = yes
	is_imprisoned = no
	prestige >= 200
	gold >= 100
}
