# ── eotg_pmc_triggers.txt ────────────────────────────────────────────────────
# Scripted triggers for the PMC government.
# ─────────────────────────────────────────────────────────────────────────────

eotg_st_is_pmc_ruler = {
	has_government = eotg_pmc_government
}

# Contractor: county-tier PMC vassal
eotg_st_is_contractor = {
	has_government = eotg_pmc_government
	is_landed = yes
	highest_held_title_tier = tier_county
}

# Field Director: duchy-tier PMC vassal
eotg_st_is_field_director = {
	has_government = eotg_pmc_government
	highest_held_title_tier = tier_duchy
}

# Executive Commander: kingdom-tier PMC vassal
eotg_st_is_executive_commander = {
	has_government = eotg_pmc_government
	highest_held_title_tier = tier_kingdom
}

# Contractor-General: empire-tier PMC ruler
eotg_st_is_contractor_general = {
	has_government = eotg_pmc_government
	highest_held_title_tier = tier_empire
}

# Contractor-General can accept a new contract (not at war, no active cooldown or exclusive lock)
eotg_st_pmc_contract_available = {
	eotg_st_is_contractor_general = yes
	is_at_war = no
	NOT = { has_character_flag = eotg_pmc_contract_cooldown }
	NOT = { has_character_flag = eotg_pmc_exclusive_contract_active }
}

# PMC has met all requirements to evolve into a Corporation
eotg_st_pmc_evolution_valid = {
	eotg_st_is_contractor_general = yes
	gold >= 1000
	prestige >= 500
	realm_size >= 10
	any_held_title = { tier = tier_kingdom }
}

# Loss of Command sequence is currently active on this Contractor-General
eotg_st_pmc_loss_of_command_active = {
	has_character_flag = eotg_pmc_loss_of_command_flag
}

# Executive Commander can declare a force withdrawal (CG is vulnerable)
eotg_st_pmc_can_withdraw_forces = {
	eotg_st_is_executive_commander = yes
	NOT = { has_character_flag = eotg_pmc_withdrew_forces }
	liege = {
		eotg_st_is_contractor_general = yes
		OR = {
			prestige < 100
			has_character_modifier = eotg_pmc_controversy_exclusive
		}
	}
}
