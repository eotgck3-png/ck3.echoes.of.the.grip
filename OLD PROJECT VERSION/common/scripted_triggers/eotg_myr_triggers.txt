# ── eotg_myr_triggers.txt ────────────────────────────────────────────────────
# Scripted triggers for the Myr Cluster Wars struggle.
# ─────────────────────────────────────────────────────────────────────────────

# Is this character an involved participant in the Myr Cluster Wars?
eotg_st_myr_involved = {
    any_character_struggle = {
        is_struggle_type = eotg_myr_struggle
        involvement = involved
    }
}

# Is the Myr struggle currently in the given phase? ($PHASE$ = phase key)
eotg_st_myr_phase = {
    any_character_struggle = {
        is_struggle_type = eotg_myr_struggle
        is_struggle_phase = $PHASE$
    }
}
