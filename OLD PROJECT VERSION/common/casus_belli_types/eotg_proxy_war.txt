# ── eotg_proxy_war.txt ────────────────────────────────────────────────────────
# Proxy War CB — Corporation government hostile acquisition escalation.
# Declared by a Regional Director whose Hostile Acquisition offer was refused.
# Both sides drain gold each month instead of fielding full levies.
# Winner takes the disputed kingdom-tier title.
# ─────────────────────────────────────────────────────────────────────────────

eotg_proxy_war = {
	icon = claim_cb
	group = event

	combine_into_one = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		eotg_st_is_regional_director = yes
		has_character_flag = eotg_hostile_acquisition_refused_by
	}

	allowed_against_character = {
		eotg_st_is_regional_director = yes
		has_character_flag = eotg_hostile_acquisition_refused
	}

	target_titles = all
	target_title_tier = kingdom

	cost = {
		prestige = 100
	}

	should_invalidate = {
		OR = {
			scope:attacker = {
				NOT = { has_character_flag = eotg_hostile_acquisition_refused_by }
			}
			scope:defender = {
				NOT = { eotg_st_is_regional_director = yes }
			}
		}
	}

	on_invalidated_desc = { desc = eotg_proxy_war_invalidated_desc }
	on_invalidated = { }

	on_declaration = {
		on_declared_war = yes
		# Monthly gold drain begins (tracked via hidden on_monthly effect in story cycle or hidden event)
		scope:attacker = {
			add_character_flag = eotg_proxy_war_active
		}
		scope:defender = {
			add_character_flag = eotg_proxy_war_active
		}
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = eotg_proxy_war_victory_attacker_desc
			}
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = eotg_proxy_war_victory_defender_desc
			}
			desc = eotg_proxy_war_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			change_title_holder_include_vassals = {
				holder = scope:attacker
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change

		scope:attacker = { remove_character_flag = eotg_hostile_acquisition_refused_by }
		scope:defender = { remove_character_flag = eotg_hostile_acquisition_refused }

		add_truce_attacker_victory_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = eotg_proxy_war_defeat_attacker_desc
			}
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = eotg_proxy_war_defeat_defender_desc
			}
			desc = eotg_proxy_war_defeat_desc
		}
	}

	on_defeat = {
		scope:attacker = { remove_character_flag = eotg_hostile_acquisition_refused_by }
		scope:defender = { remove_character_flag = eotg_hostile_acquisition_refused }
		add_legitimacy_attacker_defeat_effect = yes
		add_truce_attacker_defeat_effect = yes
	}

	on_white_peace_desc = { desc = eotg_proxy_war_white_peace_desc }
	on_white_peace = {
		scope:attacker = { remove_character_flag = eotg_hostile_acquisition_refused_by }
		scope:defender = { remove_character_flag = eotg_hostile_acquisition_refused }
		add_truce_white_peace_effect = yes
	}

	war_name = "EOTG_PROXY_WAR_NAME"
	cb_name_no_target = "EOTG_PROXY_WAR_NAME_FALLBACK"

	interface_priority = 45
	attacker_wargoal_percentage = 0.75
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	transfer_behavior = transfer
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
}
