# Update isolationism tier modifier based on current title tier.
# Removes old tier modifier and applies the correct one.
# Called on Elven Monarchy ruler scope.
eotg_se_elven_update_isolationism = {
	remove_character_modifier = eotg_elven_isolationism_county
	remove_character_modifier = eotg_elven_isolationism_duchy
	remove_character_modifier = eotg_elven_isolationism_kingdom
	remove_character_modifier = eotg_elven_isolationism_empire
	if = {
		limit = { highest_held_title_tier = tier_county }
		add_character_modifier = { modifier = eotg_elven_isolationism_county }
	}
	else_if = {
		limit = { highest_held_title_tier = tier_duchy }
		add_character_modifier = { modifier = eotg_elven_isolationism_duchy }
	}
	else_if = {
		limit = { highest_held_title_tier = tier_kingdom }
		add_character_modifier = { modifier = eotg_elven_isolationism_kingdom }
	}
	else_if = {
		limit = { highest_held_title_tier = tier_empire }
		add_character_modifier = { modifier = eotg_elven_isolationism_empire }
	}
}

# Remove all isolationism tier modifiers (cleanup on government change).
eotg_se_elven_clear_isolationism = {
	remove_character_modifier = eotg_elven_isolationism_county
	remove_character_modifier = eotg_elven_isolationism_duchy
	remove_character_modifier = eotg_elven_isolationism_kingdom
	remove_character_modifier = eotg_elven_isolationism_empire
}

# Apply Eternal Court confirmation bonus to a new ruler.
# Called on the new ruler scope after strong court ratification.
eotg_se_elven_court_confirmed_bonus = {
	add_prestige = eotg_sv_elven_confirmation_prestige
	add_legitimacy = eotg_sv_elven_confirmation_bonus_legitimacy
	add_character_modifier = {
		modifier = eotg_elven_court_confirmed
		years = 10
	}
}

# Apply Eternal Court confirmation with legitimacy penalty.
# Called on the new ruler scope after weak court ratification.
eotg_se_elven_court_confirmed_penalty = {
	add_prestige = 50
	add_legitimacy = -15
	add_character_modifier = {
		modifier = eotg_elven_court_contested
		years = 5
	}
}

# Dismiss all non-elven characters from council positions.
# Called on the ruler scope.
eotg_se_elven_purge_council = {
	every_councillor = {
		limit = { eotg_st_is_non_elven = yes }
		save_scope_as = eotg_temp_councillor
		employer = {
			fire_councillor = scope:eotg_temp_councillor
		}
	}
}

# Issue Cultural Appeal — send diplomatic message to all elven rulers in realm.
# Called on Starborn Sovereign scope.
eotg_se_elven_cultural_appeal = {
	add_character_flag = { flag = eotg_elven_appeal_cooldown  years = 5 }
	every_vassal = {
		limit = {
			eotg_st_is_elven_monarch = yes
			is_powerful_vassal = yes
		}
		trigger_event = { id = eotg_elven_culture.0001  days = 3 }
	}
	# Notify independent elven rulers nearby
	every_independent_ruler = {
		limit = {
			eotg_st_is_elven_monarch = yes
			NOT = { this = root }
		}
		trigger_event = { id = eotg_elven_culture.0002  days = 5 }
	}
}
