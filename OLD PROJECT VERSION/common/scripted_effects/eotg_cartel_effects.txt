# ── eotg_cartel_effects.txt ───────────────────────────────────────────────────
# Scripted effects for the Cartel government.
# ─────────────────────────────────────────────────────────────────────────────

# Grand Underlord demands loyalty tax — fires event to all eligible Bosses.
# Called on Grand Underlord scope.
eotg_se_cartel_demand_loyalty_tax = {
	add_character_flag = { flag = eotg_cartel_tax_demanded  years = 2 }
	every_vassal = {
		limit = { eotg_st_cartel_tax_eligible = yes }
		trigger_event = { id = eotg_cartel_tax.0001  days = 3 }
	}
}

# Boss complies with loyalty tax demand. Called on Boss scope.
eotg_se_cartel_boss_pays_tax = {
	add_character_flag = { flag = eotg_cartel_tax_paid  years = 2 }
	remove_character_modifier = eotg_cartel_defiant_boss
	add_gold = {
		value = eotg_sv_cartel_loyalty_tax_demand
		multiply = -1
	}
	liege = {
		add_dread = eotg_sv_cartel_tax_payment_dread_gain
		add_prestige = 25
	}
}

# Boss refuses loyalty tax demand. Called on Boss scope.
eotg_se_cartel_boss_refuses_tax = {
	add_character_modifier = {
		modifier = eotg_cartel_defiant_boss
		years = 2
	}
	liege = {
		add_prestige = {
			value = eotg_sv_cartel_tax_refusal_prestige_loss
			multiply = -1
		}
		# Apply defiance pressure — accelerates dread decay while active
		if = {
			limit = { NOT = { has_character_modifier = eotg_cartel_defiance_pressure } }
			add_character_modifier = {
				modifier = eotg_cartel_defiance_pressure
				months = 6
			}
		}
		trigger_event = { id = eotg_cartel_tax.0002  days = 1 }
	}
}

# Grand Underlord gains dread from a successful raid. Called on Grand Underlord scope.
eotg_se_cartel_raid_dread_gain = {
	add_dread = eotg_sv_cartel_raid_dread_gain
	eotg_se_cartel_clear_gl_paranoia = yes
}

# Apply Grand Underlord paranoia modifier (vulnerability state). Called on GL scope.
eotg_se_cartel_apply_gl_paranoia = {
	if = {
		limit = { NOT = { has_character_modifier = eotg_cartel_gl_paranoia } }
		add_character_modifier = {
			modifier = eotg_cartel_gl_paranoia
			years = 2
		}
	}
}

# Remove Grand Underlord paranoia modifier. Called on GL scope.
eotg_se_cartel_clear_gl_paranoia = {
	if = {
		limit = { has_character_modifier = eotg_cartel_gl_paranoia }
		remove_character_modifier = eotg_cartel_gl_paranoia
	}
}

# Boss joins Shakedown faction. Called on Boss scope.
eotg_se_cartel_join_shakedown = {
	add_character_flag = { flag = eotg_cartel_faction_committed  years = 5 }
	add_character_flag = { flag = eotg_cartel_shakedown_member  years = 5 }
}

# Boss joins Hostile Takeover faction. Called on Boss scope.
eotg_se_cartel_join_hostile_takeover = {
	add_character_flag = { flag = eotg_cartel_faction_committed  years = 5 }
	add_character_flag = { flag = eotg_cartel_takeover_member  years = 5 }
}

# Settle Shakedown standoff — Grand Underlord pays out to defusing the faction.
# Called on Grand Underlord scope.
eotg_se_cartel_settle_shakedown = {
	add_gold = {
		value = eotg_sv_cartel_shakedown_payout
		multiply = -1
	}
	remove_character_modifier = eotg_cartel_shakedown_standoff
	every_vassal = {
		limit = { has_character_flag = eotg_cartel_shakedown_member }
		add_gold = 50
		remove_character_flag = eotg_cartel_faction_committed
		remove_character_flag = eotg_cartel_shakedown_member
	}
}

# Reset Loyalty Tax cycle flags (called on Grand Underlord scope at cycle start).
eotg_se_cartel_reset_tax_cycle = {
	remove_character_flag = eotg_cartel_tax_demanded
	every_vassal = {
		limit = { eotg_st_is_boss = yes }
		remove_character_flag = eotg_cartel_tax_paid
		remove_character_modifier = eotg_cartel_defiant_boss
	}
}

# Yearly paranoia check — apply GL paranoia if dread is low. Called on GL scope.
eotg_se_cartel_yearly_paranoia_check = {
	if = {
		limit = { eotg_st_cartel_gl_vulnerable = yes }
		eotg_se_cartel_apply_gl_paranoia = yes
	}
	else = {
		eotg_se_cartel_clear_gl_paranoia = yes
	}
}
