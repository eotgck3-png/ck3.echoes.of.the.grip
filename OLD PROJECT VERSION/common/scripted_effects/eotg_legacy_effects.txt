# ── eotg_legacy_effects.txt ──────────────────────────────────────────────────
# Scripted effects for the Confluence Legacies system.
# All site state lives in variables on the county title — it survives the
# holder's death and the county changing hands (pitch rule: a Legacy remembers).
# ─────────────────────────────────────────────────────────────────────────────

# Seed a site on a county. $FAMILY$ = 1–4. Starts dormant, undiscovered,
# at 80 integrity and no attention. County scope.
eotg_se_legacy_seed = {
    set_variable = { name = eotg_legacy_family  value = $FAMILY$ }
    set_variable = { name = eotg_legacy_condition  value = 1 }
    set_variable = { name = eotg_legacy_integrity  value = 80 }
    set_variable = { name = eotg_legacy_attention  value = 0 }
}

# Mark a site discovered. County scope.
eotg_se_legacy_make_known = {
    set_variable = { name = eotg_legacy_known  value = 1 }
    if = {
        limit = { var:eotg_legacy_condition = 1 }
        set_variable = { name = eotg_legacy_condition  value = 2 }
    }
}

# Change a site's condition and refresh its county modifier. County scope.
# $CONDITION$ = 1 dormant | 2 exposed | 3 governed | 4 extracted
#               5 sealed | 6 unstable
eotg_se_legacy_set_condition = {
    set_variable = { name = eotg_legacy_condition  value = $CONDITION$ }
    eotg_se_legacy_refresh_modifier = yes
}

# Reapply the persistent county modifier matching family + condition.
# Governed sites get a family-specific identity modifier; extracted, sealed
# and unstable sites share generic ones. Dormant/exposed carry none. County scope.
eotg_se_legacy_refresh_modifier = {
    # Clear whatever legacy modifier is currently applied
    if = { limit = { has_county_modifier = eotg_legacy_metropolis_governed }  remove_county_modifier = eotg_legacy_metropolis_governed }
    if = { limit = { has_county_modifier = eotg_legacy_archive_governed }  remove_county_modifier = eotg_legacy_archive_governed }
    if = { limit = { has_county_modifier = eotg_legacy_works_governed }  remove_county_modifier = eotg_legacy_works_governed }
    if = { limit = { has_county_modifier = eotg_legacy_hellpocket_warded }  remove_county_modifier = eotg_legacy_hellpocket_warded }
    if = { limit = { has_county_modifier = eotg_legacy_extracted_site }  remove_county_modifier = eotg_legacy_extracted_site }
    if = { limit = { has_county_modifier = eotg_legacy_sealed_site }  remove_county_modifier = eotg_legacy_sealed_site }
    if = { limit = { has_county_modifier = eotg_legacy_unstable_site }  remove_county_modifier = eotg_legacy_unstable_site }

    # Governed: family identity
    if = {
        limit = { var:eotg_legacy_condition = 3 }
        if = {
            limit = { var:eotg_legacy_family = 1 }
            add_county_modifier = eotg_legacy_metropolis_governed
        }
        else_if = {
            limit = { var:eotg_legacy_family = 2 }
            add_county_modifier = eotg_legacy_archive_governed
        }
        else_if = {
            limit = { var:eotg_legacy_family = 3 }
            add_county_modifier = eotg_legacy_works_governed
        }
        else_if = {
            limit = { var:eotg_legacy_family = 4 }
            add_county_modifier = eotg_legacy_hellpocket_warded
        }
    }
    # Extracted / sealed / unstable: shared identities
    else_if = {
        limit = { var:eotg_legacy_condition = 4 }
        add_county_modifier = eotg_legacy_extracted_site
    }
    else_if = {
        limit = { var:eotg_legacy_condition = 5 }
        add_county_modifier = eotg_legacy_sealed_site
    }
    else_if = {
        limit = { var:eotg_legacy_condition = 6 }
        add_county_modifier = eotg_legacy_unstable_site
    }
}

# Adjust attention, clamped 0–100. County scope. $AMOUNT$ may be negative.
eotg_se_legacy_add_attention = {
    change_variable = { name = eotg_legacy_attention  add = $AMOUNT$ }
    if = {
        limit = { var:eotg_legacy_attention < 0 }
        set_variable = { name = eotg_legacy_attention  value = 0 }
    }
    if = {
        limit = { var:eotg_legacy_attention > 100 }
        set_variable = { name = eotg_legacy_attention  value = 100 }
    }
}

# Adjust integrity, clamped 0–100. At zero the site destabilizes and the
# holder faces The Exhausted Legacy. County scope. $AMOUNT$ may be negative.
eotg_se_legacy_add_integrity = {
    change_variable = { name = eotg_legacy_integrity  add = $AMOUNT$ }
    if = {
        limit = { var:eotg_legacy_integrity > 100 }
        set_variable = { name = eotg_legacy_integrity  value = 100 }
    }
    if = {
        limit = { var:eotg_legacy_integrity <= 0 }
        set_variable = { name = eotg_legacy_integrity  value = 0 }
        eotg_se_legacy_set_condition = { CONDITION = 6 }
        save_scope_as = eotg_legacy_county
        if = {
            limit = { exists = holder }
            holder = {
                trigger_event = { id = eotg_legacy.0104  days = { 10 30 } }
            }
        }
    }
}

# Put the site on cooldown for $YEARS$ years. County scope.
eotg_se_legacy_cooldown = {
    set_variable = { name = eotg_legacy_cooldown  value = 1  years = $YEARS$ }
}

# Discover the site here and fire the matching family discovery event on the
# county holder. County scope; requires exists = holder.
eotg_se_legacy_fire_discovery = {
    eotg_se_legacy_make_known = yes
    save_scope_as = eotg_legacy_county
    eotg_se_legacy_cooldown = { YEARS = 2 }
    if = {
        limit = { var:eotg_legacy_family = 1 }
        holder = { trigger_event = { id = eotg_legacy.0010  days = { 5 15 } } }
    }
    else_if = {
        limit = { var:eotg_legacy_family = 2 }
        holder = { trigger_event = { id = eotg_legacy.0020  days = { 5 15 } } }
    }
    else_if = {
        limit = { var:eotg_legacy_family = 3 }
        holder = { trigger_event = { id = eotg_legacy.0030  days = { 5 15 } } }
    }
    else_if = {
        limit = { var:eotg_legacy_family = 4 }
        holder = { trigger_event = { id = eotg_legacy.0040  days = { 5 15 } } }
    }
}

# Hand-place a famous (known, dormant) site at a history character's capital.
# $CHAR$ = character history key, $FAMILY$ = 1–4. No-op until the character
# is alive and landed with a real capital (post-map-pass).
eotg_se_legacy_seed_capital = {
    if = {
        limit = {
            exists = character:$CHAR$
            character:$CHAR$ = {
                is_alive = yes
                is_landed = yes
                exists = capital_county
            }
            character:$CHAR$.capital_county = {
                NOT = { has_variable = eotg_legacy_family }
            }
        }
        character:$CHAR$.capital_county = {
            eotg_se_legacy_seed = { FAMILY = $FAMILY$ }
            eotg_se_legacy_make_known = yes
        }
    }
}

# Seed a hidden site of a random family, weighted toward the commoner kinds.
# County scope.
eotg_se_legacy_seed_random = {
    random_list = {
        30 = { eotg_se_legacy_seed = { FAMILY = 1 } }
        30 = { eotg_se_legacy_seed = { FAMILY = 2 } }
        25 = { eotg_se_legacy_seed = { FAMILY = 3 } }
        15 = { eotg_se_legacy_seed = { FAMILY = 4 } }
    }
}
