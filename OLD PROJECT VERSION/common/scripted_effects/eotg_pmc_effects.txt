# Accept a Minor Contract — small payout, minor controversy, 3-year cooldown.
# Called on Contractor-General scope.
eotg_se_pmc_apply_minor_contract = {
	add_gold = 150
	add_prestige = 50
	add_character_flag = { flag = eotg_pmc_contract_cooldown  years = 3 }
	add_character_flag = { flag = eotg_pmc_active_contract  years = 3 }
	add_character_modifier = {
		modifier = eotg_pmc_controversy_minor
		years = 3
	}
	var:eotg_pmc_contract_client = {
		trigger_event = { id = eotg_pmc_contract.0003  days = 5 }
	}
}

# Trigger the Loss of Command sequence on a Contractor-General whose vassals withdrew.
# Called on Contractor-General scope.
eotg_se_pmc_trigger_loss_of_command = {
	add_character_flag = { flag = eotg_pmc_loss_of_command_flag  years = 1 }
	set_variable = { name = eotg_pmc_extension_count  value = 1 }
	trigger_event = { id = eotg_pmc_command.0001  days = 3 }
}

# Buy back vassal loyalty — removes all withdrawal flags, restores opinion.
# Called on Contractor-General scope.
eotg_se_pmc_buyback_loyalty = {
	every_vassal = {
		limit = {
			is_powerful_vassal = yes
			has_character_flag = eotg_pmc_withdrew_forces
		}
		remove_character_flag = eotg_pmc_withdrew_forces
		add_opinion = {
			target = root
			modifier = eotg_pmc_loyalty_restored
		}
	}
	remove_character_flag = eotg_pmc_loss_of_command_flag
	remove_variable = eotg_pmc_extension_count
}

# Fire the Evolution to Corporation narrative event.
# The government change happens in the event option effect.
# Called on Contractor-General scope.
eotg_se_pmc_evolve_to_corporation = {
	trigger_event = { id = eotg_pmc_command.0005  days = 3 }
}
