	obligation_levels = {

			default = yes

			ai_subject_desire = 5
			score = 2
		}
		eotg_cartel_tax_standard = {
			parent = eotg_cartel_tax_token
			position = { 1 0 }
			tax = 0.25
			ai_liege_desire = 3
			ai_subject_desire = 3
			score = 0
		}
		eotg_cartel_tax_heavy = {
			parent = eotg_cartel_tax_standard
			position = { 2 0 }

			subject_opinion = -15

				monthly_income_mult = 0.05
			}
			ai_liege_desire = 5
			ai_subject_desire = 1
			score = -2
		}
	}
}
