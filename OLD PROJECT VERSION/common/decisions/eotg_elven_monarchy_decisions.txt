# Starborn Sovereign issues a Cultural Appeal — calls on elven vassals and
# nearby independent elven rulers to affirm solidarity.
eotg_elven_cultural_appeal = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = major

	desc = eotg_elven_cultural_appeal_desc
	selection_tooltip = eotg_elven_cultural_appeal_tooltip

	is_shown = {
		eotg_st_elven_appeal_valid = yes
	}

	is_valid = {
		eotg_st_elven_appeal_valid = yes
		is_imprisoned = no
		prestige >= 100
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		prestige >= 100
	}

	effect = {
		add_prestige = -100
		eotg_se_elven_cultural_appeal = yes
		custom_tooltip = eotg_elven_cultural_appeal_effect_tooltip
	}

	ai_check_interval = 36
	ai_will_do = {
		base = 30
		modifier = { add = 20  has_trait = gregarious }
		modifier = { add = 20  prestige > 500 }
		modifier = { add = -20  prestige < 150 }
	}
}

# Decree of Court Purity — formally expel non-elven characters from council.
eotg_elven_decree_court_purity = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = major

	desc = eotg_elven_decree_court_purity_desc
	selection_tooltip = eotg_elven_decree_court_purity_tooltip

	is_shown = {
		eotg_st_is_elven_monarch = yes
		any_councillor = { eotg_st_is_non_elven = yes }
	}

	is_valid = {
		eotg_st_is_elven_monarch = yes
		is_imprisoned = no
		any_councillor = { eotg_st_is_non_elven = yes }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
	}

	effect = {
		eotg_se_elven_purge_council = yes
		add_character_modifier = {
			modifier = eotg_elven_court_purity_active
			years = 5
		}
		custom_tooltip = eotg_elven_decree_court_purity_effect_tooltip
	}

	cooldown = { years = 10 }

	ai_check_interval = 60
	ai_will_do = {
		base = 20
		modifier = { add = 30  has_character_modifier = eotg_elven_isolationism_empire }
		modifier = { add = 20  has_character_modifier = eotg_elven_isolationism_kingdom }
		modifier = { add = -50  has_trait = gregarious }
	}
}
