# Corporate Vote of No Confidence
# Available to Regional Directors when the CEO's board influence hits zero.
# Triggers a board confrontation — CEO steps down or refuses (proxy war risk).
eotg_corporation_vote_no_confidence = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = major
	ai_check_interval = 12

	is_shown = {
		eotg_st_corp_vote_valid = yes
	}

	is_valid = {
		eotg_st_corp_vote_valid = yes
		NOT = { has_character_flag = eotg_corp_voted }
	}

	is_valid_showing_failures_only = {
		eotg_st_corp_vote_valid = yes
		NOT = { has_character_flag = eotg_corp_voted }
	}

	effect = {
		hidden_effect = {
			add_character_flag = { flag = eotg_corp_voted  years = 10 }
		}
		liege = {
			trigger_event = { id = eotg_corp_board.0005  days = 3 }
		}
		trigger_event = { id = eotg_corp_board.0006 }
		custom_tooltip = eotg_corporation_vote_no_confidence_tooltip
	}

	ai_potential = {
		is_ai = yes
		NOT = { has_character_flag = eotg_corp_voted }
	}

	ai_will_do = {
		base = 5
		modifier = {
			add = 20
			opinion = { target = liege  value < -20 }
		}
		modifier = {
			add = 15
			liege = { eotg_st_corp_board_influence_critical = yes }
		}
		modifier = {
			add = -30
			opinion = { target = liege  value >= 20 }
		}
	}
}

# Corporate Interference — CEO's pre-election manipulation
# Suppresses one Regional Director's election vote weight.
# Discoverable via the target's intrigue-based check.
eotg_corporate_interference = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	ai_check_interval = 24

	is_shown = {
		eotg_st_is_ceo = yes
		NOT = { eotg_st_corp_interference_used = yes }
		any_vassal = {
			eotg_st_is_regional_director = yes
			NOT = { has_character_flag = eotg_corp_interference_target }
		}
	}

	is_valid = {
		eotg_st_is_ceo = yes
		NOT = { eotg_st_corp_interference_used = yes }
		intrigue >= 10
		gold >= 50
	}

	is_valid_showing_failures_only = {
		intrigue >= 10
		gold >= 50
	}

	cost = {
		gold = 50
	}

	effect = {
		add_character_flag = { flag = eotg_corp_interference_active  years = 10 }
		# Player chooses the target via event
		trigger_event = { id = eotg_corp_board.0007 }
		custom_tooltip = eotg_corporate_interference_tooltip
	}

	ai_will_do = {
		base = 5
		modifier = {
			add = 20
			intrigue >= 15
		}
	}
}
