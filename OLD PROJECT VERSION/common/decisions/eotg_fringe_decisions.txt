# Distribute the Spoils — share war gains with captains to buy loyalty.
eotg_fringe_distribute_spoils = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = major

	desc = eotg_fringe_distribute_spoils_desc
	selection_tooltip = eotg_fringe_distribute_spoils_tooltip

	is_shown = {
		eotg_st_is_fringe_overlord = yes
	}

	is_valid = {
		eotg_st_is_fringe_overlord = yes
		is_imprisoned = no
		gold >= eotg_sv_fringe_spoils_cost
		OR = {
			has_character_flag = eotg_fringe_was_at_war
			is_at_war = yes
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		gold >= eotg_sv_fringe_spoils_cost
	}

	effect = {
		remove_short_term_gold = eotg_sv_fringe_spoils_cost
		eotg_se_fringe_gain_grip = { AMOUNT = 10 }
		every_vassal = {
			limit = { eotg_st_is_fringe_captain = yes }
			add_opinion = {
				target = root
				modifier = eotg_fringe_spoils_distributed
			}
		}
		add_character_flag = { flag = eotg_fringe_captain_paid  years = 2 }
		custom_tooltip = eotg_fringe_distribute_spoils_effect_tooltip
	}

	cooldown = { years = 2 }

	ai_will_do = {
		base = 50
		modifier = { add = 30  gold > 500 }
		modifier = { add = -20  gold < 150 }
	}
}

# Forge a Unified Fringe — major decision for late-game temporary stabilization.

		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}
	decision_group_type = major

	selection_tooltip = eotg_fringe_forge_unified_tooltip

	is_shown = {
		eotg_st_is_fringe_overlord = yes

		prestige >= 500
		dread >= 30

		}

				target = root

	ai_will_do = {
		base = 20
		modifier = { add = 30  prestige > 1000 }
		modifier = { add = 20  dread > 50 }
	}
}
