# eotg_cartel_flavor.0001 — Loyalty Bought with Blood
# Subordinates become loyal after witnessing brutal punishment.
eotg_cartel_flavor.0001 = {
	type = character_event
	title = eotg_cartel_flavor_0001_title
	desc = eotg_cartel_flavor_0001_desc
	theme = intrigue

	left_portrait = { character = root  animation = personality_bold }

	trigger = {
		eotg_st_is_grand_underlord = yes
		dread >= 30
	}

	option = {
		name = eotg_cartel_flavor_0001_opt_display
		add_dread = 20
		every_vassal = {
			limit = { eotg_st_is_boss = yes }
			add_opinion = { target = root  modifier = eotg_fringe_bought_loyalty }
		}
		add_character_modifier = { modifier = eotg_cartel_rule_of_fear  years = 3 }
		add_character_flag = { flag = eotg_cartel_0001_recent  years = 3 }
		ai_chance = { base = 40  modifier = { add = 20  has_trait = callous } }
	}
	option = {
		name = eotg_cartel_flavor_0001_opt_quiet
		add_dread = 10
		add_character_flag = { flag = eotg_cartel_0001_recent  years = 3 }
		ai_chance = { base = 35  modifier = { add = 15  has_trait = patient } }
	}
	option = {
		name = eotg_cartel_flavor_0001_opt_spare
		add_prestige = 15
		add_character_flag = { flag = eotg_cartel_0001_recent  years = 3 }
		ai_chance = { base = 25  modifier = { add = 15  has_trait = compassionate } }
	}
}
