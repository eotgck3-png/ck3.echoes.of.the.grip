# ── eotg_exodus_events.txt ───────────────────────────────────────────────────
# The Titan Exodus — refugee fleet arrival events (866 AG bookmark priority #3).
# Fired from eotg_on_yearly_exodus_pulse in common/on_action/eotg_on_actions.txt.
#
# eotg_exodus.0001 — Ragged Survivors (a battered fleet arrives)
# eotg_exodus.0002 — Roots in New Soil (follow-up: sheltered refugees integrate)
# eotg_exodus.0003 — The Spurned Turn Feral (follow-up: turned-away refugees raid)
# eotg_exodus.0004 — A Master of the Old Arts (skilled specialist among refugees)
# eotg_exodus.0005 — An Organized Remnant (exiled petty court arrives in force)
# eotg_exodus.0006 — The Remnant's Grudge (follow-up: driven-off remnant raids)
# eotg_exodus.0007 — Pilgrims of the Firstlight (religious order seeks sanctuary)
# eotg_exodus.0008 — The Fever Ships (refugees carrying sickness)
# eotg_exodus.0009 — The Last Fleets (one-time ~1000 AG: the Exodus is ending)
# eotg_exodus.0010 — Stragglers of a Dead World (rare post-1000 arrival)
#
# TIMELINE: the Titanworld empties over the centuries. From 1000 AG the arrival
# rate drops to roughly a fifth (see eotg_on_yearly_exodus_pulse), each ruler
# sees the one-time eotg_exodus.0009 transition event, and only the rare
# eotg_exodus.0010 stragglers keep arriving afterward.
#
# Localization is deferred — each event carries a "Loc intent" comment block
# describing what its title/desc/option keys should eventually say.
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_exodus

# A battered refugee fleet limps into the ruler's home systems.
#
# Loc intent:
#   title    "Ragged Survivors"
#   desc     A scorch-hulled civilian fleet drops out of the void at the edge of
#            [capital county]'s systems — half its ships derelict, its holds full
#            of starving Titanworld families begging for landfall.
#   shelter  "Make landfall. We will find room." (coin now, strain, growth later)
#   refuse   "Turn them back into the void." (cold; the desperate may turn raider)
#   press    "Their passage has a price: service." (levies now, resentment later)
eotg_exodus.0001 = {
	type = character_event
	title = eotg_exodus_0001_title
	theme = realm

	desc = eotg_exodus_0001_desc

	left_portrait = {
		character = root
		animation = worry
	}

	immediate = {
		capital_county = { save_scope_as = eotg_exodus_county }
	}

	# Shelter them — pay for their settlement, strain now, growth later.
	option = {
		name = eotg_exodus_0001_opt_shelter
		remove_short_term_gold = 50
		scope:eotg_exodus_county = {
			add_county_modifier = {
				modifier = eotg_exodus_refugee_strain
				years = 3
			}
		}
		if = {
			limit = { has_trait = compassionate }
			add_stress = -15
		}
		hidden_effect = {
			trigger_event = { id = eotg_exodus.0002  days = { 360 720 } }
		}
		ai_chance = {
			base = 40
			modifier = { add = 20  gold > 300 }
			modifier = { add = 20  has_trait = compassionate }
			modifier = { add = -20  has_trait = greedy }
		}
	}

	# Turn them away — no cost, but the desperate do desperate things.
	option = {
		name = eotg_exodus_0001_opt_refuse
		add_prestige = -25
		if = {
			limit = { has_trait = compassionate }
			add_stress = 20
		}
		hidden_effect = {
			random = {
				chance = 35
				trigger_event = { id = eotg_exodus.0003  days = { 90 180 } }
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20  has_trait = callous }
			modifier = { add = -20  has_trait = compassionate }
		}
	}

	# Press them into service — manpower now, resentment later.
	option = {
		name = eotg_exodus_0001_opt_press
		add_character_modifier = {
			modifier = eotg_exodus_pressed_levies
			years = 5
		}
		add_dread = 10
		if = {
			limit = {
				OR = {
					has_trait = compassionate
					has_trait = just
				}
			}
			add_stress = 20
		}
		ai_chance = {
			base = 30
			modifier = { add = 20  has_trait = sadistic }
			modifier = { add = 10  has_trait = ambitious }
			modifier = { add = -20  has_trait = just }
		}
	}
}

# Follow-up: sheltered refugees have put down roots and enriched the county.
#
# Loc intent:
#   title    "Roots in New Soil"
#   desc     A year or two on, the refugee quarter of [county] has become a
#            district — workshops open, fields cleared, children born under a
#            new sky. The gamble of mercy has paid out.
#   good     "They are our people now." (development +1, long integration boon)
eotg_exodus.0002 = {
	type = character_event
	title = eotg_exodus_0002_title
	theme = stewardship

	desc = eotg_exodus_0002_desc

	trigger = {
		exists = scope:eotg_exodus_county
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {
		scope:eotg_exodus_county = {
			remove_county_modifier = eotg_exodus_refugee_strain
			add_county_modifier = {
				modifier = eotg_exodus_integrated_refugees
				years = 10
			}
			change_development_level = 1
		}
	}

	option = {
		name = eotg_exodus_0002_opt_good
		add_prestige = 100
	}
}

# Follow-up: refugees turned away have turned to void-piracy on the fringes.
#
# Loc intent:
#   title    "The Spurned Turn Feral"
#   desc     The fleet you refused did not sail on to die quietly. Its hulks now
#            haunt the shipping lanes of [county], striking convoys and vanishing
#            into the drift — refugees no longer, raiders now.
#   hunt     "Hunt them down." (coin for a punitive sweep; order restored, respect won)
#   pay      "Buy them off." (expensive and humiliating, but the raids stop)
#   ignore   "Let them burn themselves out." (the county bleeds until they scatter)
eotg_exodus.0003 = {
	type = character_event
	title = eotg_exodus_0003_title
	theme = martial

	desc = eotg_exodus_0003_desc

	trigger = {
		exists = scope:eotg_exodus_county
	}

	left_portrait = {
		character = root
		animation = anger
	}

	immediate = {
		scope:eotg_exodus_county = {
			add_county_modifier = {
				modifier = eotg_exodus_raider_scourge
				years = 5
			}
		}
	}

	# Hunt them down — costs coin, restores order, earns respect.
	option = {
		name = eotg_exodus_0003_opt_hunt
		remove_short_term_gold = 50
		scope:eotg_exodus_county = {
			remove_county_modifier = eotg_exodus_raider_scourge
		}
		add_prestige = 50
		ai_chance = {
			base = 50
			modifier = { add = 20  martial >= 12 }
			modifier = { add = -20  gold < 100 }
		}
	}

	# Buy them off — expensive, humiliating, effective.
	option = {
		name = eotg_exodus_0003_opt_pay
		remove_short_term_gold = 100
		scope:eotg_exodus_county = {
			remove_county_modifier = eotg_exodus_raider_scourge
		}
		add_prestige = -25
		ai_chance = {
			base = 20
			modifier = { add = 20  has_trait = craven }
			modifier = { add = -10  has_trait = brave }
		}
	}

	# Let them burn themselves out — the county suffers.
	option = {
		name = eotg_exodus_0003_opt_ignore
		ai_chance = {
			base = 20
			modifier = { add = 15  gold < 50 }
		}
	}
}

# A specialist carrying knowledge of the old Titanworld arts is among the arrivals.
#

#   title    "A Master of the Old Arts"
#   desc     Among the latest arrivals walks one the other refugees defer to — a
#            grey-haired artisan-engineer who studied under the last guilds of the
#            Titanworld, carrying knowledge that predates the Grip itself.
#   recruit  "Such knowledge belongs at my court." (gain the learned courtier)
#   fund     "Fund their onward passage." (small cost; the fleets remember kindness)
#   pass     "Let them drift on with the fleet." (nothing gained, nothing risked)
eotg_exodus.0004 = {
	type = character_event
	title = eotg_exodus_0004_title
	theme = learning

	desc = eotg_exodus_0004_desc

		character = root

		character = scope:eotg_exodus_master
		animation = personality_bold
		trigger = { exists = scope:eotg_exodus_master }
	}

	immediate = {
		create_character = {
			location = root.capital_province

		name = eotg_exodus_0004_opt_pass
		ai_chance = { base = 20 }

# ── The Exodus wanes ─────────────────────────────────────────────────────────

# One-time transition event, fired once per ruler from the yearly pulse when

# Pure narrative beat — no branching, small commemorative reward.
#

#   title    "The Last Fleets"
#   desc     The harbormasters' ledgers tell the story: the fleets have thinned,
#            year on year, and this season only a handful of sails came out of
#            the dark. Eight centuries after Mikey the Great's generation fled,
#            the Titanworld has little left to give. The Exodus — the great
#            migration that shaped every nation of settled space — is ending.
#   ok       "An age ends. We remain." (reflective close; a moment of history)
eotg_exodus.0009 = {
	type = character_event
	title = eotg_exodus_0009_title
	theme = realm

	desc = eotg_exodus_0009_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}

	option = {
		name = eotg_exodus_0009_opt_ok
		add_prestige = 100
	}
}
