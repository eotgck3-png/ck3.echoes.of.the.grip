# ── eotg_void_touched_events.txt ─────────────────────────────────────────────
# Void Corruption, stage C: living with the eotg_void_touched trait.
#
# eotg_void.0020 — The Cold Hand        (at war; martial/brave branches)
# eotg_void.0021 — What the Children See (has a young child; age-aware)
# eotg_void.0022 — The Confessor's Verdict (zealous courtier; faith inversion)
# eotg_void.0023 — The Deep Route        (learning/stewardship; the payoff path)
# eotg_void.0024 — Long Service          (age 60+; a life spent at the boundary)
# eotg_void.0025 — The Vassal's Ultimatum (disaffected vassal forces the issue)
#
# Fires from: common/on_action/eotg_void_on_actions.txt
#
# These are the events the audit asked for: every one of them moves the
# signature resource (eotg_void_exposure), rather than paying out flat
# prestige and leaving the mechanic to drift on its own.
#
# Every event keeps at least one un-triggered option.
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_void


# ── eotg_void.0020 ────────────────────────────────────────────────────────────
# The Cold Hand
# In the field, the thing in you is useful. That is the problem.

eotg_void.0020 = {
    type = character_event
    title = eotg_void.0020.t
    desc = eotg_void.0020.desc
    theme = war

    trigger = {
        has_trait = eotg_void_touched
        is_at_war = yes
    }

    left_portrait = {
        character = root
        animation = personality_bold
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_void_event_cooldown  years = 3 }
        }
    }

    # "Use it." — martial or brave: real advantage, real cost
    option = {
        name = eotg_void.0020.a
        trigger = {
            OR = {
                has_education_martial_trigger = yes
                has_trait = brave
            }
        }
        add_prowess_skill = 1
        add_dread = 25
        eotg_se_void_add_exposure = { AMOUNT = 12 }
        add_prestige = 100
        ai_chance = {
            base = 60
            modifier = { add = 30  has_trait = brave }
            modifier = { add = 25  has_trait = ambitious }
            modifier = { add = -30  has_trait = craven }
        }
    }

    # "Hold it back." — costs the battle, holds the line inside
    option = {
        name = eotg_void.0020.b
        eotg_se_void_add_exposure = { AMOUNT = -8 }
        add_piety = 75
        stress_impact = { base = medium_stress_impact_gain }
        ai_chance = {
            base = 40
            modifier = { add = 30  has_trait = zealous }
            modifier = { add = 20  has_trait = temperate }
            modifier = { add = -25  has_trait = wrathful }
        }
    }

    # "Let the knights see it." — dread strategy; the court pays for it
    option = {
        name = eotg_void.0020.c
        add_dread = 40
        eotg_se_void_add_exposure = { AMOUNT = 6 }
        every_knight = {
            limit = { is_alive = yes }
            add_opinion = { modifier = eotg_opinion_void_unease  target = root }
        }
        ai_chance = {
            base = 25
            modifier = { add = 35  has_trait = callous }
            modifier = { add = 25  has_trait = arrogant }
        }
    }
}


# ── eotg_void.0021 ────────────────────────────────────────────────────────────
# What the Children See
# A child says the quiet thing. How the ruler answers depends on whether they
# still remember being small.

eotg_void.0021 = {
    type = character_event
    title = eotg_void.0021.t
    desc = eotg_void.0021.desc
    theme = family

    trigger = {
        has_trait = eotg_void_touched
        any_child = {
            is_alive = yes
            age < 12
        }
    }

    left_portrait = {
        character = root
        animation = worry
    }
    right_portrait = {
        character = scope:void_child
        animation = fear
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_void_event_cooldown  years = 4 }
            random_child = {
                limit = {
                    is_alive = yes
                    age < 12
                }
                save_scope_as = void_child
            }
        }
    }

    # "Tell them the truth." — compassionate/honest; costs nothing but comfort
    option = {
        name = eotg_void.0021.a
        trigger = {
            OR = {
                has_trait = honest
                has_trait = compassionate
                has_education_diplomacy_trigger = yes
            }
        }
        scope:void_child = {
            add_opinion = { modifier = eotg_opinion_void_unease  target = root }
            add_character_modifier = { modifier = eotg_mod_void_warded  years = 5 }
        }
        eotg_se_void_add_exposure = { AMOUNT = -5 }
        stress_impact = { base = minor_stress_impact_gain }
        ai_chance = {
            base = 60
            modifier = { add = 30  has_trait = honest }
            modifier = { add = 25  has_trait = compassionate }
        }
    }

    # "Tell them they imagined it." — the kind lie, which does not hold
    option = {
        name = eotg_void.0021.b
        eotg_se_void_add_exposure = { AMOUNT = 5 }
        stress_impact = {
            base = 0
            honest = medium_stress_impact_gain
        }
        ai_chance = {
            base = 40
            modifier = { add = 25  has_trait = deceitful }
            modifier = { add = -30  has_trait = honest }
        }
    }

    # "Send them away from court." — protects the child, costs the bond
    option = {
        name = eotg_void.0021.c
        scope:void_child = {
            add_opinion = { modifier = eotg_opinion_void_unease  target = root }
        }
        eotg_se_void_add_exposure = { AMOUNT = -3 }
        stress_impact = { base = medium_stress_impact_gain }
        ai_chance = {
            base = 25
            modifier = { add = 25  has_trait = paranoid }
            modifier = { add = -25  has_trait = compassionate }
        }
    }
}


# ── eotg_void.0022 ────────────────────────────────────────────────────────────
# The Confessor's Verdict
# A zealous courtier names the condition out loud. Under Void Reverence the
# same speech is a blessing rather than an accusation — the description and
# the option set both invert on faith.

eotg_void.0022 = {
    type = character_event
    title = eotg_void.0022.t
    theme = faith

    trigger = {
        has_trait = eotg_void_touched
        any_courtier = {
            is_alive = yes
            has_trait = zealous
        }
    }

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { eotg_st_void_venerates = yes }
                desc = eotg_void.0022.desc_cult
            }
            desc = eotg_void.0022.desc
        }
    }

    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        character = scope:void_confessor
        animation = anger
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_void_event_cooldown  years = 3 }
            random_courtier = {
                limit = {
                    is_alive = yes
                    has_trait = zealous
                }
                save_scope_as = void_confessor
            }
        }
    }

    # "Submit to the rite." — cleansing, available only outside the Void Cult
    option = {
        name = eotg_void.0022.a
        trigger = { eotg_st_void_venerates = no }
        add_piety = 150
        eotg_se_void_add_exposure = { AMOUNT = -15 }
        add_character_modifier = { modifier = eotg_mod_void_warded  years = 6 }
        scope:void_confessor = {
            add_opinion = { modifier = eotg_opinion_void_defiance_respect  target = root }
        }
        stress_impact = { base = minor_stress_impact_loss }
        ai_chance = {
            base = 50
            modifier = { add = 40  has_trait = zealous }
            modifier = { add = 20  has_trait = humble }
        }
    }

    # "Accept the blessing." — Void Cult inversion of the same scene
    option = {
        name = eotg_void.0022.b
        trigger = { eotg_st_void_venerates = yes }
        add_piety = 200
        eotg_se_void_add_exposure = { AMOUNT = 10 }
        scope:void_confessor = {
            add_opinion = { modifier = eotg_opinion_void_kinship  target = root }
        }
        ai_chance = { base = 80 }
    }

    # "Have them silenced." — always available
    option = {
        name = eotg_void.0022.c
        add_dread = 25
        add_piety = -100
        eotg_se_void_add_exposure = { AMOUNT = 8 }
        scope:void_confessor = {
            add_opinion = { modifier = eotg_opinion_void_unease  target = root }
        }
        stress_impact = {
            base = 0
            compassionate = minor_stress_impact_gain
            just = minor_stress_impact_gain
        }
        ai_chance = {
            base = 20
            modifier = { add = 35  has_trait = callous }
            modifier = { add = 30  has_trait = paranoid }
            modifier = { add = -40  has_trait = just }
        }
    }

    # "Say nothing and let them wonder." — always available
    option = {
        name = eotg_void.0022.d
        eotg_se_void_add_exposure = { AMOUNT = 3 }
        stress_impact = { base = minor_stress_impact_gain }
        ai_chance = {
            base = 25
            modifier = { add = 25  has_trait = shy }
        }
    }
}

# ── eotg_void.0024 ────────────────────────────────────────────────────────────
# Long Service
# For a ruler who has carried this since middle age, the question is no longer
# what the Void wants. It is what is left to hand on.

eotg_void.0024 = {
    type = character_event
    title = eotg_void.0024.t
    desc = eotg_void.0024.desc
    theme = mental_health

    trigger = {
        has_trait = eotg_void_touched
        age >= 60
    }

    left_portrait = {
        character = root
        animation = sadness
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_void_event_cooldown  years = 5 }
        }
    }

    # "Write it down for whoever comes next." — the heir inherits knowledge
    option = {
        name = eotg_void.0024.a
        trigger = {
            exists = primary_heir
            primary_heir = { is_alive = yes }
        }
        add_learning_skill = 1
        add_prestige = 100
        primary_heir = {
            eotg_se_void_seed = yes
            add_opinion = { modifier = eotg_opinion_void_defiance_respect  target = root }
        }
        eotg_se_void_add_exposure = { AMOUNT = -5 }
        ai_chance = {
            base = 60
            modifier = { add = 30  has_trait = diligent }
            modifier = { add = 25  has_education_learning_trigger = yes }
        }
    }

    # "Spend what is left." — a last surge, paid for in years
    option = {
        name = eotg_void.0024.b
        eotg_se_void_edu_insight = yes
        eotg_se_void_add_exposure = { AMOUNT = 20 }
        add_prestige = 150
        ai_chance = {
            base = 30
            modifier = { add = 40  has_trait = ambitious }
            modifier = { add = -30  has_trait = content }
        }
    }

    # "Make your peace." — always available; the graceful exit
    option = {
        name = eotg_void.0024.c
        add_piety = 200
        eotg_se_void_add_exposure = { AMOUNT = -15 }
        stress_impact = { base = major_stress_impact_loss }
        ai_chance = {
            base = 45
            modifier = { add = 35  has_trait = content }
            modifier = { add = 30  has_trait = zealous }
            modifier = { add = 20  has_trait = humble }
        }
    }
}
