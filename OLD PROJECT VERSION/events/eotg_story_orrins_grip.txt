# The Weight of 866 — the three framing events that open the 866 AG bookmark.
# Fired in sequence from the eotg_story_866_opening story cycle:
#
# eotg_orrins_grip.0001 — The Grip Remembered (866 years since the cataclysm)
# eotg_orrins_grip.0002 — The Fleets Still Come (the Titan Exodus is ongoing)
# eotg_orrins_grip.0003 — The Clusters Burn (the Myr Cluster Wars just ignited)
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_orrins_grip

# 866 years since Orrin's Grip — living history, not myth.
eotg_orrins_grip.0001 = {
	type = character_event
	title = eotg_orrins_grip_0001_title
	theme = faith

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_trait = eotg_grip_survivor }
				desc = eotg_orrins_grip_0001_desc_survivor
			}
			desc = eotg_orrins_grip_0001_desc
		}
	}

	left_portrait = {
		character = root
		animation = worry
	}

	# Reflect on what was lost — the weight settles.
	option = {
		name = eotg_orrins_grip_0001_opt_reflect
		add_piety = 50
		if = {
			limit = { has_trait = eotg_grip_survivor }
			add_stress = -15
		}
		ai_chance = {
			base = 50
			modifier = { add = 20  has_trait = zealous }
		}
	}

	# The past is fuel — steel yourself for the era ahead.
	option = {
		name = eotg_orrins_grip_0001_opt_steel
		add_prestige = 50
		ai_chance = {
			base = 50
			modifier = { add = 20  has_trait = ambitious }
		}
	}

	after = {
		hidden_effect = {
			trigger_event = { id = eotg_orrins_grip.0002  days = { 10 20 } }
		}
	}
}

# The Titan Exodus is still ongoing — refugee fleets keep arriving.
eotg_orrins_grip.0002 = {
	type = character_event
	title = eotg_orrins_grip_0002_title
	theme = realm

	desc = eotg_orrins_grip_0002_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Open your ports to the fleets — word will spread.
	option = {
		name = eotg_orrins_grip_0002_opt_open
		add_character_modifier = {
			modifier = eotg_exodus_haven_reputation
			years = 2
		}
		ai_chance = {
			base = 40
			modifier = { add = 20  has_trait = compassionate }
			modifier = { add = 15  has_trait = generous }
		}
	}

	# Harden your border pickets — charity invites chaos.
	option = {
		name = eotg_orrins_grip_0002_opt_harden
		remove_short_term_gold = 25
		add_prestige = 25
		ai_chance = {
			base = 40
			modifier = { add = 20  has_trait = paranoid }
			modifier = { add = 15  has_trait = callous }
		}
	}

	# Watch the skies and wait.
	option = {
		name = eotg_orrins_grip_0002_opt_watch
		ai_chance = { base = 20 }
	}

	after = {
		hidden_effect = {
			trigger_event = { id = eotg_orrins_grip.0003  days = { 10 20 } }
		}
	}
}

# The Myr Cluster Wars ignited sixteen years ago — the fires are spreading.
eotg_orrins_grip.0003 = {
	type = character_event
	title = eotg_orrins_grip_0003_title
	theme = war

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { eotg_st_myr_involved = yes }
				desc = eotg_orrins_grip_0003_desc_involved
			}
			desc = eotg_orrins_grip_0003_desc
		}
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}

	# The Cluster is opportunity — sharpen your blades.
	option = {
		name = eotg_orrins_grip_0003_opt_opportunity
		add_prestige = 25
		if = {
			limit = { eotg_st_myr_involved = yes }
			eotg_se_myr_catalyst = {
				CATALYST = eotg_catalyst_myr_aggression
				CHARACTER = root
			}
		}
		ai_chance = {
			base = 40
			modifier = { add = 20  has_trait = ambitious }
		}
	}

	# The peace must hold — someone has to keep their head.
	option = {
		name = eotg_orrins_grip_0003_opt_peace
		add_piety = 25
		if = {
			limit = { eotg_st_myr_involved = yes }
			eotg_se_myr_catalyst = {
				CATALYST = eotg_catalyst_myr_diplomacy
				CHARACTER = root
			}
		}
		ai_chance = {
			base = 40
			modifier = { add = 20  has_trait = calm }
		}
	}

	# Distant fires warm no one — tend your own realm.
	option = {
		name = eotg_orrins_grip_0003_opt_distance
		add_stress = -10
		ai_chance = { base = 20 }
	}
}
