# ── eotg_new_cauldron_senate_events.txt ──────────────────────────────────────
# Senate events for the New Cauldron Democracy.
#
# eotg_nc_senate.0001 — Senate veto warning (fires to President)
# eotg_nc_senate.0002 — War vote result (fires to President)
# eotg_nc_senate.0003 — Emergency powers declared (fires to Senators)
# eotg_nc_senate.0004 — Scandal — political exposure (fires to offending character)
# ─────────────────────────────────────────────────────────────────────────────

namespace = eotg_nc_senate

# Senate registers concern — hostile Senate threatens a veto on upcoming decisions.
eotg_nc_senate.0001 = {
	type = character_event
	title = eotg_nc_senate_0001_title
	theme = diplomacy

	desc = eotg_nc_senate_0001_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Acknowledge and work to rebuild support.
	option = {
		name = eotg_nc_senate_0001_opt_address
		add_prestige = -50
		if = {
			limit = { has_variable = eotg_nc_senate_support }
			change_variable = { name = eotg_nc_senate_support  add = 10 }
		}
		stress_impact = {
			base = minor_stress_impact_gain
		}
		ai_chance = { base = 70 }
	}

	# Hold firm — the Senate does not rule here.
	option = {
		name = eotg_nc_senate_0001_opt_hold
		add_legitimacy = -10
		ai_chance = { base = 30 }
	}
}

# Senate war vote result — fires to President after calling a vote.
eotg_nc_senate.0002 = {
	type = character_event
	title = eotg_nc_senate_0002_title
	theme = martial

	desc = {
		triggered_desc = {
			trigger = {
				has_variable = eotg_nc_senate_support
				var:eotg_nc_senate_support >= 40
			}
			desc = eotg_nc_senate_0002_desc_approved
		}
		triggered_desc = {
			trigger = {
				OR = {
					NOT = { has_variable = eotg_nc_senate_support }
					var:eotg_nc_senate_support < 40
				}
			}
			desc = eotg_nc_senate_0002_desc_denied
		}
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Senate approved — war mandate in hand.
	option = {
		name = eotg_nc_senate_0002_opt_approved
		trigger = {
			has_variable = eotg_nc_senate_support
			var:eotg_nc_senate_support >= 40
		}
		add_character_flag = { flag = eotg_nc_senate_war_approved  years = 1 }
		add_prestige = 25
		ai_chance = { base = 100 }
	}

	# Senate denied — stand down or invoke Emergency Powers.
	option = {
		name = eotg_nc_senate_0002_opt_denied
		trigger = {
			OR = {
				NOT = { has_variable = eotg_nc_senate_support }
				var:eotg_nc_senate_support < 40
			}
		}
		add_prestige = -25
		ai_chance = { base = 100 }
	}
}

# Emergency powers declared — notification fires to all senators.
eotg_nc_senate.0003 = {
	type = character_event
	title = eotg_nc_senate_0003_title
	theme = martial
	hidden = yes

	desc = eotg_nc_senate_0003_desc

	option = {
		name = eotg_nc_senate_0003_opt_protest
		add_opinion = {
			target = liege
			modifier = eotg_nc_emergency_powers_anger
		}
		ai_chance = { base = 100 }
	}
}

# Political scandal — exposure of underhanded campaign or Senate tactics.
eotg_nc_senate.0004 = {
	type = character_event
	title = eotg_nc_senate_0004_title
	theme = intrigue

	desc = eotg_nc_senate_0004_desc

	left_portrait = {
		character = root
		animation = stress
	}

	# Weather the exposure.
	option = {
		name = eotg_nc_senate_0004_opt_endure
		add_prestige = -100
		add_legitimacy = -10
		if = {
			limit = { has_variable = eotg_nc_rally_bonus }
			change_variable = { name = eotg_nc_rally_bonus  add = -3 }
		}
		stress_impact = {
			base = minor_stress_impact_gain
			paranoid = medium_stress_impact_gain
		}
		ai_chance = { base = 100 }
	}
}
