	title = eotg_legacy_0002_title
	theme = learning

	desc = eotg_legacy_0002_desc

		exists = scope:eotg_legacy_county
	}

		animation = personality_rational
	}

	option = {

				add_prestige = 25

			base = 50
			modifier = { add = 20  learning >= 14 }

	option = {
		name = eotg_legacy_0002_opt_sell

# An expedition falls silent below an exposed seam.
eotg_legacy.0003 = {
	type = character_event
	title = eotg_legacy_0003_title
	theme = martial

	desc = eotg_legacy_0003_desc

	trigger = {
		exists = scope:eotg_legacy_county
	}

	left_portrait = {
		character = root
		animation = worry
	}

	# [Brave / Martial education 2+] Lead a relief force yourself.
	option = {
		name = eotg_legacy_0003_opt_relief
		trigger = {
			OR = {
				has_trait = brave
				eotg_st_edu_martial_2 = yes
			}
		}
		random_list = {
			60 = {
				add_gold = 50
				add_prestige = 50
			}
			25 = {
				create_character = {
					location = root.capital_province
					age = 32
					culture = root.culture
					faith = root.faith
					save_scope_as = eotg_legacy_survivor
				}
				add_courtier = scope:eotg_legacy_survivor
				add_prestige = 25
			}
			15 = {
				add_trait = wounded_1
			}
		}
		ai_chance = {
			base = 40
			modifier = { add = 20  has_trait = brave }
		}
	}

	# [Compassionate] Pay for a careful recovery.
	option = {
		name = eotg_legacy_0003_opt_recover
		trigger = { has_trait = compassionate }
		remove_short_term_gold = 50
		add_piety = 50
		ai_chance = { base = 50 }
	}

	# [Callous] Mark them lost and continue.
	option = {
		name = eotg_legacy_0003_opt_abandon
		trigger = { has_trait = callous }
		random = {
			chance = 35
			scope:eotg_legacy_county = {
				add_county_modifier = {
					modifier = eotg_legacy_grim_rumors
					years = 3
				}
			}
		}
		ai_chance = { base = 40 }
	}

	# [Intrigue education 3+] Learn why they stopped.
	option = {
		name = eotg_legacy_0003_opt_investigate
		trigger = { eotg_st_edu_intrigue_3 = yes }
		if = {
			limit = {
				any_vassal = { is_adult = yes }
			}
			random_vassal = {
				limit = { is_adult = yes }
				save_scope_as = eotg_legacy_saboteur
			}
			add_hook = {
				type = manipulation_hook
				target = scope:eotg_legacy_saboteur
			}
		}
		else = {
			add_gold = 50
		}
		ai_chance = {
			base = 40
			modifier = { add = 20  has_trait = paranoid }
		}
	}

	# Send a second expedition and wait — the broad option.
	option = {
		name = eotg_legacy_0003_opt_wait
		remove_short_term_gold = 25
		random = {
			chance = 40
			scope:eotg_legacy_county = {
				eotg_se_legacy_seed_random = yes
				eotg_se_legacy_fire_discovery = yes
			}
		}
		ai_chance = { base = 30 }
	}
}
