# -- eotg_void_whispers.txt ---------------------------------------------------
# Void Corruption, stages A and B: contact, whispers, and the Bargain.
#
# eotg_void.0001 -- The Scar Reopens        (contact / seeding, education-gated)
# eotg_void.0002 -- The Sound Under the Floor (whisper, one option per education)
# eotg_void.0003 -- The Star That Is Not There (whisper, age-gated)
# eotg_void.0004 -- The Heretic at the Gate  (Voidwalker arrival)
# eotg_void.0010 -- The Bargain              (pivotal: touched, or defiant)
# eotg_void.0011 -- Taken Anyway             (saturation, refusal has lapsed)
#
# Fires from: common/on_action/eotg_void_on_actions.txt
#
# LORE: the Void is Orrin the Eye's prison. He cannot act, only leak. The
# Grip-era rift scars are re-activating across 850-866 AG as the seals weaken.
#
# EVERY event here keeps at least one un-triggered option so it can never
# soft-lock a character who matches none of the gated branches.
# -----------------------------------------------------------------------------

namespace = eotg_void


# -- eotg_void.0001 -----------------------------------------------------------
# The Scar Reopens
# A dormant Grip-era rift site on the character's land wakes up. This is the
# entry point for anyone with no bloodline or cult connection to the Void.

eotg_void.0001 = {
    type = character_event
    title = eotg_void.0001.t
    desc = eotg_void.0001.desc
    theme = disaster

    left_portrait = {
        character = root
        animation = disbelief
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_void_event_cooldown  years = 3 }
        }
    }

    # "Survey it properly." -- learning: the scholar's answer, and the safest
    option = {
        name = eotg_void.0001.a
        trigger = { has_education_learning_trigger = yes }
        add_learning_skill = 1
        add_character_modifier = { modifier = eotg_mod_void_warded  years = 5 }
        eotg_se_void_seed = yes
        eotg_se_void_add_exposure = { AMOUNT = 5 }
        ai_chance = {
            base = 60
            modifier = { add = 20  has_trait = curious }
        }
    }

    # "Wall it off and post a guard." -- martial/stewardship: contain, do not study
    option = {
        name = eotg_void.0001.b
        trigger = {
            OR = {
                has_education_martial_trigger = yes
                has_education_stewardship_trigger = yes
            }
        }
        remove_short_term_gold = 50
        add_prestige = 40
        eotg_se_void_seed = yes
        ai_chance = {
            base = 60
            modifier = { add = 20  has_trait = paranoid }
            modifier = { add = -30  gold < 50 }
        }
    }

    # "Go in." -- the reckless read. Big exposure, and Orrin notices you early.
    option = {
        name = eotg_void.0001.c
        trigger = {
            OR = {
                has_trait = brave
                eotg_st_void_venerates = yes
                has_education_intrigue_trigger = yes
            }
        }
        eotg_se_void_seed = yes
        eotg_se_void_add_exposure = { AMOUNT = 20 }
        add_prestige = 75
        stress_impact = {
            base = minor_stress_impact_gain
            craven = medium_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 40  eotg_st_void_venerates = yes }
            modifier = { add = 25  has_trait = brave }
            modifier = { add = -40  has_trait = craven }
        }
    }

    # "Pretend it is not there." -- always available fallback
    option = {
        name = eotg_void.0001.d
        eotg_se_void_seed = yes
        eotg_se_void_add_exposure = { AMOUNT = 10 }
        stress_impact = { base = minor_stress_impact_gain }
        ai_chance = {
            base = 20
            modifier = { add = 30  has_trait = craven }
            modifier = { add = 20  has_trait = lazy }
        }
    }
}


# -- eotg_void.0002 -----------------------------------------------------------
# The Sound Under the Floor
# The whisper stage. This is the education showcase event: five mutually
# exclusive readings of the same phenomenon, plus a universal fallback.

eotg_void.0002 = {
    type = character_event
    title = eotg_void.0002.t
    theme = corruption

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { eotg_st_void_venerates = yes }
                desc = eotg_void.0002.desc_cult
            }
            triggered_desc = {
                trigger = { has_trait = eotg_grip_survivor }
                desc = eotg_void.0002.desc_bloodline
            }
            desc = eotg_void.0002.desc
        }
    }

    left_portrait = {
        character = root
        animation = worry
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_void_event_cooldown  years = 3 }
        }
    }

    # Learning -- "Record it. Measure it. Name it."
    option = {
        name = eotg_void.0002.a
        trigger = { has_education_learning_trigger = yes }
        add_learning_skill = 1
        add_piety = 50
        eotg_se_void_add_exposure = { AMOUNT = -5 }
        ai_chance = { base = 100 }
    }

    # Martial -- "Drill until it is drowned out."
    option = {
        name = eotg_void.0002.b
        trigger = { has_education_martial_trigger = yes }
        add_martial_skill = 1
        add_prestige = 50
        stress_impact = { base = minor_stress_impact_gain }
        ai_chance = { base = 100 }
    }

    # Intrigue -- "Find out who else hears it."
    option = {
        name = eotg_void.0002.c
        trigger = { has_education_intrigue_trigger = yes }
        add_intrigue_skill = 1
        eotg_se_void_add_exposure = { AMOUNT = 5 }
        random_courtier = {
            limit = { is_alive = yes }
            add_opinion = { modifier = eotg_opinion_void_unease  target = root }
        }
        ai_chance = { base = 100 }
    }

    # Stewardship -- "Cost it out. Everything has a cause and a price."
    option = {
        name = eotg_void.0002.d
        trigger = { has_education_stewardship_trigger = yes }
        add_stewardship_skill = 1
        add_gold = 50
        ai_chance = { base = 100 }
    }

    # Diplomacy -- "Ask, carefully, whether anyone else has noticed."
    option = {
        name = eotg_void.0002.e
        trigger = { has_education_diplomacy_trigger = yes }
        add_diplomacy_skill = 1
        eotg_se_void_add_exposure = { AMOUNT = -3 }
        stress_impact = { base = minor_stress_impact_loss }
        ai_chance = { base = 100 }
    }

    # Universal fallback -- no education, or none of the above fired
    option = {
        name = eotg_void.0002.f
        eotg_se_void_add_exposure = { AMOUNT = 5 }
        stress_impact = { base = minor_stress_impact_gain }
        ai_chance = { base = 10 }
    }
}

    left_portrait = {
        character = root

        hidden_effect = {

        }
        ai_chance = {
            base = 10
            modifier = { add = 30  has_trait = callous }
            modifier = { add = 25  has_trait = wrathful }
            modifier = { add = -40  has_trait = compassionate }
        }
    }
}


# -- eotg_void.0004 -----------------------------------------------------------
# The Heretic at the Gate
# A surviving Voidwalker -- Malvrick's sect, hunted to near-extinction between
# 200 and 700 AG -- asks for shelter. They are the only people alive who still
# know the wards. Sheltering one is the strongest defence in the system and a
# permanent liability with the devout.

eotg_void.0004 = {
    type = character_event
    title = eotg_void.0004.t
    desc = eotg_void.0004.desc
    theme = secret

    left_portrait = {
        character = root
        animation = scheme
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_void_event_cooldown  years = 5 }
        }
    }

    # "Shelter them." -- the wards, and the risk
    option = {
        name = eotg_void.0004.a
        eotg_se_void_shelter_voidwalker = yes
        eotg_se_void_add_exposure = { AMOUNT = -10 }
        add_piety = -100

            limit = {
                is_alive = yes

            modifier = { add = 20  has_trait = just }
            modifier = { add = -50  eotg_st_void_venerates = yes }

        name = eotg_void.0004.d
        add_dread = 20
        add_piety = 75
        eotg_se_void_add_exposure = { AMOUNT = 8 }

            modifier = { add = 30  has_trait = sadistic }
            modifier = { add = -50  has_trait = compassionate }
        }
    }
}


# -- eotg_void.0010 -----------------------------------------------------------
# The Bargain
# The pivotal event of the system. At 50+ exposure Orrin makes an offer through
# the weakening seal. Accepting grants the education-differentiated insight and
# the eotg_void_touched trait. Refusing costs a great deal now and wards later.

eotg_void.0010 = {
    type = character_event
    title = eotg_void.0010.t
    theme = corruption

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { eotg_st_void_venerates = yes }
                desc = eotg_void.0010.desc_cult
            }
            triggered_desc = {
                trigger = { has_trait = eotg_grip_survivor }
                desc = eotg_void.0010.desc_bloodline
            }
            desc = eotg_void.0010.desc
        }
    }

    left_portrait = {
        character = root
        animation = paranoia
    }

    immediate = {
        hidden_effect = {
            add_character_flag = { flag = eotg_flag_void_event_cooldown  years = 5 }
            add_character_flag = eotg_flag_void_bargain_offered
        }
    }

    # "Accept." -- become Void-touched, gain insight matched to your education
    option = {
        name = eotg_void.0010.a
        eotg_se_void_become_touched = yes
        eotg_se_void_edu_insight = yes
        add_dread = 20
        every_courtier = {
            limit = {
                is_alive = yes
                NOT = { eotg_st_void_venerates = yes }
            }
            add_opinion = { modifier = eotg_opinion_void_unease  target = root }
        }
        every_courtier = {
            limit = {
                is_alive = yes
                eotg_st_void_venerates = yes
            }
            add_opinion = { modifier = eotg_opinion_void_kinship  target = root }
        }
        stress_impact = {
            base = 0
            zealous = major_stress_impact_gain
            humble = medium_stress_impact_gain
        }
        ai_chance = {
            base = 30
            modifier = { add = 50  eotg_st_void_venerates = yes }
            modifier = { add = 30  has_trait = ambitious }
            modifier = { add = 20  has_trait = brave }
            modifier = { add = -60  has_trait = zealous }
            modifier = { add = -25  has_trait = content }
        }
    }

    # "Refuse, and mean it." -- become Defiant. Expensive, and it holds.
    option = {
        name = eotg_void.0010.b
        eotg_se_void_become_defiant = yes
        add_piety = 200
        stress_impact = { base = major_stress_impact_gain }
        every_courtier = {
            limit = {
                is_alive = yes
                has_trait = zealous
            }
            add_opinion = { modifier = eotg_opinion_void_defiance_respect  target = root }
        }
        ai_chance = {
            base = 40
            modifier = { add = 60  has_trait = zealous }
            modifier = { add = 30  has_trait = brave }
            modifier = { add = 20  has_trait = just }
            modifier = { add = -50  eotg_st_void_venerates = yes }
        }
    }

    # "Refuse -- but keep what it showed you." -- intrigue/learning only.
    # Take the insight without the trait, and pay in exposure for the theft.
    option = {
        name = eotg_void.0010.c
        trigger = {
            OR = {
                has_education_intrigue_trigger = yes
                has_education_learning_trigger = yes
            }
        }
        eotg_se_void_edu_insight = yes
        eotg_se_void_add_exposure = { AMOUNT = 15 }
        add_piety = -50
        stress_impact = { base = medium_stress_impact_gain }
        ai_chance = {
            base = 35
            modifier = { add = 30  has_trait = deceitful }
            modifier = { add = 25  has_trait = ambitious }
            modifier = { add = -30  has_trait = honest }
        }
    }

    # "Say nothing at all." -- always available; defers, does not resolve
    option = {
        name = eotg_void.0010.d
        eotg_se_void_add_exposure = { AMOUNT = 10 }
        stress_impact = { base = medium_stress_impact_gain }
        ai_chance = {
            base = 10
            modifier = { add = 25  has_trait = craven }
            modifier = { add = 20  has_trait = shy }
        }
    }
}
