		}
	}

		character = scope:eotg_election_winner

	}

		name = eotg_corp_board_0002_opt_acknowledge

}

# CEO receives the board's no-confidence demand — step down or refuse
eotg_corp_board.0005 = {
	type = character_event
	title = eotg_corp_board_0005_title
	theme = intrigue

	desc = eotg_corp_board_0005_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Step down gracefully
	option = {
		name = eotg_corp_board_0005_opt_step_down
		add_prestige = 200
		add_gold = 150
		add_trait = eotg_honorable_exit
		trigger_event = { id = eotg_corp_board.0008  days = 1 }
		ai_chance = {
			base = 30
			modifier = { add = 30  has_trait = content }
			modifier = { add = -20  has_trait = ambitious }
		}
	}

	# Refuse — legitimizes proxy war declarations
	option = {
		name = eotg_corp_board_0005_opt_refuse
		add_internal_flag = dangerous
		add_prestige = -100
		add_character_flag = eotg_corp_refused_exit
		every_vassal = {
			limit = { eotg_st_is_regional_director = yes }
			send_interface_message = {
				title = eotg_corp_no_confidence_refused_title
				desc = eotg_corp_no_confidence_refused_desc
				left_icon = root
			}
		}
		ai_chance = {
			base = 70
			modifier = { add = 20  has_trait = ambitious }
			modifier = { add = -20  has_trait = content }
		}
	}
}

# CEO picks the target for Corporate Interference (choose event)
eotg_corp_board.0007 = {
	type = character_event
	title = eotg_corp_board_0007_title
	theme = intrigue

	desc = eotg_corp_board_0007_desc

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# One option per eligible Regional Director (up to 5 shown via triggered options)
	# In production this would use a find_character / random_vassal approach;
	# for now we fire a targeted suppress on the single highest-weight rival
	option = {
		name = eotg_corp_board_0007_opt_target_rival
		ordered_vassal = {
			order_by = var:eotg_election_weight
			limit = {
				eotg_st_is_regional_director = yes
				NOT = { has_character_flag = eotg_corp_interference_target }
			}
			position = 0
			max = 1
			save_scope_as = eotg_interference_target
		}
		if = {
			limit = { exists = scope:eotg_interference_target }
			scope:eotg_interference_target = {
				add_character_flag = { flag = eotg_corp_interference_target  years = 10 }
				add_character_modifier = {
					modifier = eotg_corp_interference_suppressed
					years = 10
				}
				# Discovery chance: high intrigue target has a chance to spot the manipulation
				random = {
					chance = {
						value = scope:eotg_interference_target.intrigue
						multiply = 2
					}
					trigger_event = { id = eotg_corp_board.0009 }
				}
			}
		}
		ai_chance = { base = 100 }
	}
}
