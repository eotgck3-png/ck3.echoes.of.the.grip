posteffect_values = {
	name = standard
	lut = "gfx/map/post_effects/colorcorrection_neutral.tga"

	# MOD(eotg) darker ambience. Everything else in this file inherits from standard, directly
	# or through zoom_step_00, so one value here moves every zoom level together.
	#
	# Done here rather than in the shaders on purpose: EOTG_TERRAIN_EMISSIVE would darken the
	# land and leave the water, the nebula and the markers where they were, and the balance
	# between those took a long time to settle. value_scale is applied after everything is
	# composited, so the relationship between them is preserved.
	value_scale = 0.82
}
