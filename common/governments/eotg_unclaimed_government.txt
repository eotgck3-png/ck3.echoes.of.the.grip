# -- eotg_unclaimed_government.txt --------------------------------------------
# Unclaimed Regions (docs/specs/frontier_unclaimed_regions.md §4): the passive
# placeholder government. It stands for the scattered people of a Region that
# no ruler has claimed; it is not a ruler. One placeholder holds one county.
#
# Shape: vanilla herder_government (game/common/governments/
# 00_government_types.txt:733-792), without its mechanic_type and fertility
# (spec §4.1, research R§1: they bring herd behaviour, and the no-MPO
# conversion at game_start.txt:1249-1263 matches herder_government by key).
# Fields not set on purpose: mechanic_type, is_mechanic_type_default,
# uses_county_fertility, replenishes_county_fertility, fallback,
# vassal_contract, royal_court.
#
# Tiger 1.17 and PX don't validate 1.20 government fields (spec §4.1, R risk
# 8): the local session checks this block BY HAND against
# _governments.info and herder_government.
# UNVERIFIED-VANILLA: every field below exists in 1.20.0.3 with this meaning
#   (government_rules: inherit_from_dynastic_government, use_title_tier_modifiers,
#   deny_powerful_vassal, buildings, allow_accolades, dynasty_named_realms,
#   legitimacy; color; supply_limit_mult_for_others; the ai block keys).
# UNVERIFIED-VANILLA (V-U10): an unset `fallback` (0) does not make this a
#   fallback government, and can_get_government on the trait does not block
#   the scripted change_government.
# UNVERIFIED-VANILLA (V-U1): a custom government with no mechanic_type is
#   PLAYABLE (spec §4.5 inference); the probe decision answers it.
# -----------------------------------------------------------------------------

eotg_unclaimed_government = {
    generated_character_template = eotg_unclaimed_holder_template

    government_rules = {
        court_generate_spouses = no
        council = no
        create_cadet_branches = no
        rulers_should_have_dynasty = no
        # Fenced off, as holy orders and mercenaries are
        # (00_government_types.txt:327-400)
        inherit_from_dynastic_government = no
        legitimacy = no
        use_title_tier_modifiers = no
        deny_powerful_vassal = yes
        buildings = no
        allow_accolades = no
        dynasty_named_realms = no
    }
    court_generate_commanders = no

    character_modifier = {
        knight_limit = -100          # herder
        county_opinion_add = 100     # herder: no revolts against nobody
        monthly_income_mult = -10    # herder: no treasury, so no men-at-arms
        levy_size = -1               # (spec, new) no levies in military-strength sums
    }

    primary_holding = castle_holding
    # Every settled holding type is legal, so whatever capital history gives a
    # Region works (spec §4.1)
    valid_holdings = { city_holding church_holding tribal_holding }

    # Only placeholders (the template and the creation effects add the trait
    # BEFORE change_government; spec §4.2, the holy order guard at
    # 00_government_types.txt:436-441)
    can_get_government = {
        has_trait = eotg_unclaimed_folk
    }

    ai = {
        use_lifestyle = no
        arrange_marriage = no
        use_goals = no
        use_decisions = no
        use_scripted_guis = no
        use_legends = no
        perform_religious_reformation = no
        use_great_projects = no
    }

    flags = {
        # The identity flag: eotg_is_unclaimed_holder and the war override read it
        eotg_government_is_unclaimed
        # Never a liege or vassal (mercenary and holy order use it,
        # 00_government_types.txt:348, 388, 431)
        cannot_be_vassal_or_liege
    }

    # Attrition (spec §7.3, owner UQ7): armies of any other government lose
    # half their supply limit here; vanilla tribal and wanua use -0.5
    # (00_government_types.txt:199, 272).
    # UNVERIFIED-VANILLA (V-U3): it applies to an army standing in this
    #   government's county (the .info says "army owners of different
    #   government type"); fallback: a supply_limit_mult line on
    #   eotg_unclaimed_mod_unclaimed.
    supply_limit_mult_for_others = -0.5

    # The government map mode; the same neutral slate as the realm colour
    # (spec §7.1, owner UQ8)
    color = { 88 92 100 }
}
