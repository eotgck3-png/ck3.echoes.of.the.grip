# -- eotg_cartel_government.txt ------------------------------------------------
# Governments v2, Phase G1 (docs/specs/governments_v2.md §5.0, §5.4).
# Cartel: an organized syndicate grown into a state, ruling through fear and
# expanding by tribute, not raids. Knights read "Brutes" (KnightCulture
# override, §10). Signature resource (G2): vanilla dread.
#
# Map-agnostic (owner gate ruling 2026-10-08): no title, province, culture,
# faith or character key. History data (the test sub-mod) assigns it.
#
# Shape: vanilla feudal_government (G/common/governments/00_government_types.txt
# :5-62) for the castle holdings, rules and flags; tribute rules from
# mandala_government :922-1099; extra rules from admin :450-558.
# Not set on purpose (§5.0, §5.4): fallback, primary_heritages,
# preferred_religions, can_get_government (§5.6), mechanic_type,
# use_as_base_on_rank_up, government_is_feudal, deny_powerful_vassal (v1
# design ruling), government_can_raid_rule (v1 1.1).
# legitimacy = yes on purpose (deviation from v1 `no`, robustness plan F1b).
#
# Tiger 1.17 and PX don't validate 1.20 government fields: QA checks this block
# BY HAND against _governments.info and the cited vanilla lines (§13 item 2).
# G2 C2 (docs/specs/governments_v2_g2_cartel.md §5.3, OQ1 owner 2026-10-20):
#   the government_rules entry that allows asking for tribute is REMOVED
#   (spec §13 item 6: the token must not appear here). It only enabled
#   vanilla Embrace as Tributary
#   (00_tributary_interactions.txt:4464, piety cost, "devout tributaries"
#   notification) and the tributarization map mode (_governments.info
#   :260-266). The Cartel's one tribute verb is now
#   eotg_cartel_put_on_schedule_interaction (eotg_cartel_interactions.txt),
#   gated on the government flag; start_tributary does not need the rule.
#   count_tributaries_for_title_requirements stays (V-C4).
#   SAVE COMPATIBILITY (spec §10): G1 payers made through Embrace stay in
#   vanilla tributary_subjugated and lapse at death as before; new payers
#   go to eotg_cartel_tributary (eotg_cartel_contract_groups.txt).
# -----------------------------------------------------------------------------

eotg_cartel_government = {
    government_rules = {
        create_cadet_branches = yes         # feudal :7
        rulers_should_have_dynasty = yes    # feudal :8 (deviation from v1 `no`)
        dynasty_named_realms = no           # admin :454
        legitimacy = yes                    # feudal :10
        allow_accolades = yes               # feudal :11
        use_as_base_on_landed = yes         # admin :458; .info:73-77
        sticky_government = yes             # admin :462; .info:148-151
        count_tributaries_for_title_requirements = yes # mandala :938; .info:284-292
    }

    royal_court = any                       # feudal :14 (V-G15)

    # Castle shape, verbatim from feudal :18-20 (§5.0)
    primary_holding = castle_holding
    valid_holdings = { temple_citadel_holding }
    required_county_holdings = { castle_holding city_holding church_holding }

    vassal_contract_group = eotg_cartel_vassal # §6; feudal :22

    # G2 C4 (docs/specs/governments_v2_g2_cartel.md §7): the feudal list,
    # verbatim (G/common/governments/00_government_types.txt:40; same at admin
    # :553, celestial :915, meritocratic :1354). is_grant_government_shown
    # offers only the faith's own theocratic type
    # (G/common/scripted_rules/01_grant_title_rules.txt:37-46), so no faith
    # key is needed. feudal_government is deliberately NOT listed (it would
    # be an erosion path out of the Cartel). grant_vassal_ai_will_do is not
    # set: no government lists the Cartel as a grant type, so it would be
    # dead (spec §7; default 0, _governments.info:738-756).
    possible_grant_vassal_governments = { theocracy_government ecclesiastical_government }

    character_modifier = {
        intrigue = 2                        # v1
        diplomacy = -2                      # v1
        dread_gain_mult = 0.25
        # G2 C1 sign fix (docs/specs/governments_v2_g2_cartel.md §2.4): a
        # NEGATIVE decay mult is the bonus (slower fade), as vanilla
        # champion aptitude -0.02..-0.3 (00_court_positions.txt:6850-6870) and
        # tradition_ruling_caste -0.3 (00_realm_traditions.txt:2260, 2279).
        # G1's +0.25 made dread fade faster. 0.5/month (00_defines.txt:122)
        # becomes 0.375/month. Value change only: not save-breaking.
        dread_decay_mult = -0.25
    }

    ai = {
        use_legends = yes                   # feudal :24-26
    }

    flags = {
        eotg_government_is_cartel           # identity flag (§3.1)
        government_is_settled               # feudal :29-35
        government_uses_domain_limit
        government_uses_crown_authority     # load-bearing: crown authority law group,
                                            # and crown authority 1 grants can_have_tributaries (§5.4)
    }

    # Crimson; v1's colour duplicated tribal's dark red (§5.4, V-G18)
    color = hsv{ 0.92 0.80 0.55 }
    realm_mask_offset = { 0.0 0.01 }        # feudal :60-61
    realm_mask_scale = { 1 1 }
}
