# ── eotg_augmentation_secrets.txt ────────────────────────────────────────────
# Secret types for the Cybernetic Augmentation system.
# Spec: docs/specs/investigation_rulings_2026-10-18.md §4.4 (I2a, R-A3).
#
# eotg_secret_illegal_augmentation: undiscovered unlawful implant work. The
#   owner is the ANSWERABLE party (R-A2: the sponsor when they are the
#   patient's liege or above, else the patient); the target, when set, is the
#   patient. Vanilla shape: secret_embezzler (game/common/secret_types/
#   00_secret_types.txt:620-656; realm-law crime, is_shunned = not criminal,
#   on_expose). Fields: _secret_types.info.
# Added only through eotg_aug_give_illegal_work_secret_effect (vanilla NOTE,
#   00_secret_types.txt:8), called from eotg_aug_procedure_roll_effect.
# Vanilla then provides the rest: Find Secrets and the hostile scheme
#   (00_councillor_triggers.txt:813-826, 876-880 read is_criminal_for /
#   is_blackmailable_secret_trigger with no type whitelist), blackmail, Expose
#   Secret, and once exposed a lawful reason to imprison (game_concept_crime).
# Category civil: vanilla icon gfx/interface/icons/secret_categories/civil.dds
#   and loc secret_category_civil; no art.
# Lore (Q1, I5): the evidence is physical, in a body; never a log or record.
# ─────────────────────────────────────────────────────────────────────────────

eotg_secret_illegal_augmentation = {
    category = civil

    # Valid while the hardware it is about is still in a body. A full removal
    # or Excision dissolves it: the evidence is physical (lore I5).
    is_valid = {
        trigger_if = {
            limit = { exists = scope:secret_target }
            scope:secret_target = {
                is_alive = yes
                eotg_is_augmented_any = yes
            }
        }
        trigger_else = {
            scope:secret_owner = { eotg_is_augmented_any = yes }
        }
    }

    # A crime only to the owner's liege or above, under that realm's law.
    is_criminal = {
        eotg_aug_illegal_work_is_criminal = { OWNER = scope:secret_owner  VIEWER = scope:target }
    }

    # Otherwise a scandal inside a realm that bans it (secret_embezzler shape).
    is_shunned = {
        NOT = { eotg_aug_illegal_work_is_criminal = { OWNER = scope:secret_owner  VIEWER = scope:target } }
        scope:target.top_liege ?= scope:secret_owner.top_liege
        scope:target = { eotg_aug_under_policy = { LAW = ban } }
    }

    on_expose = {
        save_scope_as = secret
        # Known hardware is now known unlawful hardware: under a License,
        # refusing Demand Removal becomes a crime (eotg_aug_refusal_is_crime).
        if = {
            limit = { exists = scope:secret_target }
            scope:secret_target = {
                if = {
                    limit = { NOT = { has_character_modifier = eotg_mod_illegal_implants } }
                    add_character_modifier = eotg_mod_illegal_implants
                }
            }
        }
        else = {
            scope:secret_owner = {
                if = {
                    limit = { NOT = { has_character_modifier = eotg_mod_illegal_implants } }
                    add_character_modifier = eotg_mod_illegal_implants
                }
            }
        }
        # Unknown types fall to the feed-message branch
        # (00_secret_effects.txt:1023-1025): no new notification event.
        secret_exposed_notification_effect = yes
    }
    # on_discover, on_owner_death: not set (engine default; vanilla
    # secret_deviant, secret_homosexual)
}
