# Scripted effects for the Myr Cluster Wars struggle.

# Award struggle catalyst points from a character scope.
# $CATALYST$ = catalyst key, $CHARACTER$ = character credited with the act.
eotg_se_myr_catalyst = {
    random_character_struggle = {
        limit = { is_struggle_type = eotg_myr_struggle }
        activate_struggle_catalyst = {
            catalyst = $CATALYST$
            character = $CHARACTER$
        }
    }
}
