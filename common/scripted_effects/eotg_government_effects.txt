# -- eotg_government_effects.txt ----------------------------------------------
# Governments v2, Phase G1 (docs/specs/governments_v2.md §7.2).
# eotg_gov_ensure_succession_law_effect (character scope): a ruler of one of
# the four governments that lacks its own succession law gets it. (Vanilla
# single heir is never valid for these governments, so it is not exempted:
# can_have_single_heir_succession_law_trigger, G/common/scripted_triggers/
# 00_law_triggers.txt:371-397, needs feudal/clan/admin/japan_feudal.) Shape: vanilla forces a law after setup with
# add_realm_law_skip_effects (G/common/on_action/game_start.txt:2503).
# No cooldown, no event. Runs at every tier (§7.2 tier gating).
# Callers: eotg_gov_on_game_start, eotg_gov_on_government_change
# (common/on_action/eotg_government_on_actions.txt).
# -----------------------------------------------------------------------------

eotg_gov_ensure_succession_law_effect = {
    if = {
        limit = {
            government_has_flag = eotg_government_is_pmc
            NOR = {
                has_realm_law = eotg_pmc_succession_law
            }
        }
        add_realm_law_skip_effects = eotg_pmc_succession_law
    }
    else_if = {
        limit = {
            government_has_flag = eotg_government_is_corporation
            NOR = {
                has_realm_law = eotg_corporation_succession_law
            }
        }
        add_realm_law_skip_effects = eotg_corporation_succession_law
    }
    else_if = {
        limit = {
            government_has_flag = eotg_government_is_gobcorp
            NOR = {
                has_realm_law = eotg_gobcorp_succession_law
            }
        }
        add_realm_law_skip_effects = eotg_gobcorp_succession_law
    }
    else_if = {
        limit = {
            government_has_flag = eotg_government_is_cartel
            NOR = {
                has_realm_law = eotg_cartel_succession_law
            }
        }
        add_realm_law_skip_effects = eotg_cartel_succession_law
    }
}
