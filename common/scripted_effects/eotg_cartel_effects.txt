# -- eotg_cartel_effects.txt --------------------------------------------------
# Governments v2, Phase G2, Cartel batch C1 (docs/specs/governments_v2_g2_cartel.md
# §2.2-§2.6, §3.1). All character scope.
#
# AUTHORITY (spec §2.2, §13 item 3): eotg_cartel_change_dread_effect is the ONLY
# place Cartel content calls add_dread. Vanilla content and other mod systems
# still write dread directly; the rule covers Cartel files only.
# eotg_cartel_refresh_schedule_effect is the only writer of
# var:eotg_cartel_schedule_scale. eotg_cartel_sync_story_effect is the only
# writer of the story's display mirror eotg_cartel_dread_shown (on_setup
# seeds it once, see eotg_cartel_story.txt).
#
# Batch C2 adds eotg_cartel_start_schedule_effect (interaction on_accept) and
# eotg_cartel_walk_off_effect (story yearly tick). C3/C4 callers of the change
# effect: eotg_cartel_decisions.txt (Tighten, Lean, Go Legitimate).
# -----------------------------------------------------------------------------

# Change dread by $AMOUNT$: a number or a NAMED script value. Not an inline
# { } block: a block as a scripted-effect argument is rejected (Tiger
# error(macro); QA C2, 2026-10-20). Negative spends therefore have named
# values (eotg_cartel_schedule_spend_value, eotg_cartel_tighten_spend_value,
# eotg_cartel_exit_dread_loss_value in eotg_cartel_values.txt).
# Callers gate spends in is_valid (dread >= cost); this effect never checks
# affordability (spec §2.3, vanilla's literal-gated dread option,
# G/common/character_interactions/00_tributary_interactions.txt:5598-5604).
# Gains are multiplied by the engine (dread_gain_mult); the clamp works on the
# pre-multiplier amount and the engine re-clamps at the cap (accepted, §2.3).
# save_scope_value_as form: G/common/scripted_effects/
# 00_administrative_effects.txt:101-104.
eotg_cartel_change_dread_effect = {
    save_scope_value_as = {
        name = eotg_cartel_dread_amount
        value = $AMOUNT$
    }
    add_dread = eotg_cartel_dread_clamped_value
    # Refresh the story counter now, not at the monthly tick
    eotg_cartel_sync_story_effect = yes
}

# Copy the owner's dread onto the schedule story's display variable (spec
# §2.7 layer 4). Display only: no script reads eotg_cartel_dread_shown as the
# resource. Story lookup form: eotg_augmentation_effects.txt:1044-1047;
# `.dread` as a value: G/common/character_interactions/
# 09_mpo_interactions.txt:4392.
eotg_cartel_sync_story_effect = {
    save_temporary_scope_as = eotg_cartel_sync_owner
    random_owned_story = {
        limit = { story_type = eotg_story_cartel_schedule }
        set_variable = {
            name = eotg_cartel_dread_shown
            value = scope:eotg_cartel_sync_owner.dread
        }
    }
}

# Rewrite the baseline mirror: min(4 x payers, 40) (spec §2.2, OQ6 numbers
# approved 2026-10-20). The schedule modifier scales by it.
eotg_cartel_refresh_schedule_effect = {
    set_variable = {
        name = eotg_cartel_schedule_scale
        value = {
            value = eotg_cartel_payer_count_value
            multiply = 4
            max = 40
        }
    }
}

# Idempotent setup (spec §2.6). Called from eotg_cartel_on_game_start,
# eotg_cartel_on_government_change (new government is Cartel) and
# eotg_cartel_on_title_gain (Cartel with no story: heirs, newly landed
# courtiers via use_as_base_on_landed). Modifier added without duration as
# G/common/on_action/travel_on_actions.txt:1432.
eotg_cartel_init_effect = {
    if = {
        limit = {
            NOT = {
                any_owned_story = { story_type = eotg_story_cartel_schedule }
            }
        }
        create_story = eotg_story_cartel_schedule
    }
    if = {
        limit = {
            NOT = { has_character_modifier = eotg_cartel_schedule_modifier }
        }
        add_character_modifier = eotg_cartel_schedule_modifier
    }
    eotg_cartel_refresh_schedule_effect = yes
    eotg_cartel_sync_story_effect = yes
}

# Undo eotg_cartel_init_effect (spec §2.6). Does NOT touch dread (a universal
# currency: the fear lingers). Payers are ended by the exit decision
# (eotg_decision_cartel_go_legitimate) and
# by the contract group's validity check (eotg_cartel_tributary
# is_valid_tributary_contract, eotg_cartel_contract_groups.txt). Called from
# eotg_cartel_on_government_change (Cartel -> other) and the story's monthly
# tick when the owner is no longer a landed Cartel ruler.
eotg_cartel_teardown_effect = {
    if = {
        limit = {
            any_owned_story = { story_type = eotg_story_cartel_schedule }
        }
        random_owned_story = {
            limit = { story_type = eotg_story_cartel_schedule }
            end_story = yes
        }
    }
    if = {
        limit = { has_character_modifier = eotg_cartel_schedule_modifier }
        remove_character_modifier = eotg_cartel_schedule_modifier
    }
    if = {
        limit = { has_variable = eotg_cartel_schedule_scale }
        remove_variable = eotg_cartel_schedule_scale
    }
    if = {
        limit = { has_variable = eotg_cartel_succession_grace }
        remove_variable = eotg_cartel_succession_grace
    }
}

# Put the recipient on the Cartel schedule (spec §5.3). Interaction scope:
# scope:actor = the suzerain, scope:recipient = the new payer; called from
# eotg_cartel_put_on_schedule_interaction on_accept.
# Copies only the pieces of vanilla start_tributary_interaction_effect it
# needs: start_tributary (form G/common/scripted_effects/
# 00_interaction_effects.txt:2982-2985) with the OWN group, the low-obligation
# branch (:2988-2995, always low so Tighten (C3) has something to raise),
# its tooltips (:2996-2997, :3010-3011), then the vanilla truces (:3016).
# Deliberately NOT called (spec §5.3): mandala_embrace_tributary_effect
# (gratitude opinion, letter 0362 with subjugated-terms tooltip) and
# consume_all_criminal_reasons_effect (a Cartel does not forgive when paid).
eotg_cartel_start_schedule_effect = {
    scope:recipient = {
        start_tributary = {
            contract_group = eotg_cartel_tributary
            suzerain = scope:actor
        }
        if = {
            limit = { is_tributary = yes } # suppress error logging (vanilla :2990)
            hidden_effect = {
                tributary_contract_set_obligation_level = { type = default_tributary_taxes level = 1 }
                tributary_contract_set_obligation_level = { type = default_tributary_prestige level = 0 }
            }
        }
        custom_tooltip = low_subjugated_tribute_taxes
        custom_tooltip = none_subjugated_tribute_prestige
        custom_tooltip = suzerain_guarantee
        custom_tooltip = no_tributary_war_obligations
        # A tributary leaves its confederation, as vanilla Embrace does
        # (00_tributary_interactions.txt:4630-4639)
        if = {
            limit = {
                is_confederation_member = yes
                confederation ?= { is_house_based = no }
            }
            confederation = { remove_confederation_member = scope:recipient }
        }
    }
    setup_tributary_truces_effect = { SUZERAIN = scope:actor TRIBUTARY = scope:recipient }
    scope:actor = { eotg_cartel_refresh_schedule_effect = yes }
}

# The dread walk-off (spec §5.4). Character scope: the schedule's owner.
# Called ONLY from eotg_story_cartel_schedule's yearly effect_group, which is
# the walk-off's single cadence authority (invariant 4); this effect checks no
# cooldown and no threshold. One payer leaves, weighted towards the lowest
# opinion of the owner (101 - opinion keeps every weight >= 1; opinion form
# G/common/decisions/00_dynasty_decisions.txt:259). end_tributary is vanilla
# (00_tributary_interactions.txt:2104); toast shape :2176-2183.
eotg_cartel_walk_off_effect = {
    save_scope_as = eotg_cartel_suzerain
    random_tributary = {
        limit = { has_subject_contract_group = eotg_cartel_tributary }
        weight = {
            base = 1
            modifier = {
                add = {
                    value = 100
                    subtract = "opinion(scope:eotg_cartel_suzerain)"
                }
            }
        }
        save_scope_as = eotg_cartel_walker
        end_tributary = yes
    }
    if = {
        limit = { exists = scope:eotg_cartel_walker }
        send_interface_toast = {
            type = event_toast_effect_bad
            title = eotg_cartel_walk_off_toast
            left_icon = scope:eotg_cartel_walker
            right_icon = scope:eotg_cartel_suzerain
            custom_tooltip = eotg_cartel_walk_off_toast_desc
        }
    }
    eotg_cartel_refresh_schedule_effect = yes
}
