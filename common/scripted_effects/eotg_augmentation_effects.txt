# Clear, then apply the current tier's synergy by highest_skill (martial fallback).
# Applies nothing to Neurofractured or unaugmented characters.

    eotg_aug_clear_synergy_effect = yes

        limit = { eotg_is_aug_tier3 = yes }
        if = {
            limit = { highest_skill = martial }
            add_character_modifier = { modifier = eotg_mod_oc_martial_bonus }
        }
        else_if = {
            limit = { highest_skill = stewardship }
            add_character_modifier = { modifier = eotg_mod_oc_steward_bonus }
        }
        else_if = {
            limit = { highest_skill = intrigue }
            add_character_modifier = { modifier = eotg_mod_oc_intrigue_bonus }
        }
        else_if = {
            limit = { highest_skill = learning }
            add_character_modifier = { modifier = eotg_mod_oc_learning_bonus }
        }

            add_character_modifier = { modifier = eotg_mod_oc_martial_bonus }

        }
        else_if = {
            limit = { highest_skill = intrigue }
            add_character_modifier = { modifier = eotg_mod_enh_intrigue_bonus }
        }
        else_if = {
            limit = { highest_skill = learning }
            add_character_modifier = { modifier = eotg_mod_enh_learning_bonus }

# Initiation: add the track trait (engine default XP 0 = Augmented) and its synergy.
# Used for rulers and knights. Callers: eotg_aug_init.001-.005 (accept options,
# incl. the Phase 1 trait options init.001.d/e, .002.d, .003.d, .004.e, .005.d),
# eotg_aug_init.002 (c, knight), eotg_aug_tier1.004 (a, d: knight).
# Phase 2: eotg_aug_init.006-.018 install options, init.017.c (knight),
# init.020.d, eotg_aug_child_install_effect, eotg_aug_retainer_install_effect.
# Final QA: a character coming (back) into the system starts at the install
# value, risk 0, so stale risk from an earlier augmentation (e.g. one the
# Patron's debt kept moving) cannot carry into the new one.
# Balance spec §5.1: a newcomer settles for 5 years before the next stage
# (eotg_flag_aug_settling, read by eotg_can_progress_to_* and the Pursue
# decision). Only in the "was unaugmented" branch, so a knight re-initiated
# while augmented does not reset it.
eotg_aug_initiate_effect = {
    if = {
        limit = { eotg_is_augmented_any = no }
        set_variable = { name = eotg_fracture_risk  value = 0 }
        add_character_flag = { flag = eotg_flag_aug_settling  years = 5 }
    }
    add_trait = eotg_cybernetics
    eotg_aug_refresh_synergy_effect = yes
}

# THE ONLY place add_trait_xp on eotg_cybernetics may appear (spec §2.2 rule 2).
# Sets XP to exactly $XP$ (0, 50 or 100) and refreshes the synergy modifier.
# Callers: eotg_aug_tier1.022 (a, b) XP 50 (Phase 4a: tier1.002 now pays and
#          fires the First Upgrade chain); eotg_aug_tier2.003 (a, c, d, f, g) XP 100;
#          eotg_aug_removal_perform_effect (eotg_aug_proc.001, booked by
#          eotg_decision_partial_removal: XP 0; eotg_decision_overclock_regression:
#          XP 50).
# Balance spec §5.1: every tier change, regressions included, starts a fresh
# 5-year eotg_flag_aug_settling (relapse is slower too). The Pursue decision
# reaches this only through tier1.022 / tier2.003; it never sets XP itself.
eotg_aug_set_integration_effect = {
    if = {
        limit = { has_trait = eotg_cybernetics }
        add_trait_xp = {
            trait = eotg_cybernetics
            value = {
                value = $XP$
                subtract = "has_trait_xp(eotg_cybernetics)"
            }
        }
        eotg_aug_refresh_synergy_effect = yes
        add_character_flag = { flag = eotg_flag_aug_settling  years = 5 }
    }
}

# Increment fracture risk variable — expects $AMOUNT$ parameter.
# A Seamless character (eotg_total_integration, Phase 3b) has no risk: the
# variable was removed with the seam, and nothing may recreate it (e.g. the
# Heir's Arc options, which can reach a Seamless owner).
# QA round 2 M3: nor may anything recreate it on someone who has left the
# system (delayed chain stages, e.g. tier1.017, tier2.017/.018/.021, can land
# after Remove Implants or Excision): the character must be augmented. Every
# install path adds the trait before it touches risk, so no caller loses a
# move it meant to make.
eotg_add_fracture_risk = {
    if = {
        limit = {
            eotg_is_augmented_any = yes
            NOT = { has_trait = eotg_total_integration }
        }
        if = {
            limit = { NOT = { has_variable = eotg_fracture_risk } }
            set_variable = { name = eotg_fracture_risk  value = 0 }
        }
        change_variable = { name = eotg_fracture_risk  add = $AMOUNT$ }
        clamp_variable = { name = eotg_fracture_risk  min = 0  max = 100 }
    }
}

# Trigger the Neural Cascade (fracture transformation)
eotg_trigger_neurofracture = {
    if = { limit = { has_trait = eotg_cybernetics }  remove_trait = eotg_cybernetics }    # discards the track XP
    add_trait = eotg_neurofractured
    eotg_apply_neurofracture_distortion = yes
    set_variable = { name = eotg_fracture_risk  value = 0 }
    # Open the Neurofractured cooldown here so eotg_on_yearly_aug_neurofractured_check,
    # which runs later in the same yearly_playable_pulse, cannot roll fracture.002-.007
    # days after the cascade. Same duration the on_action uses (11 months, B4: expires
    # before the next pulse, so no same-day race); no event trigger reads it.
    add_character_flag = { flag = eotg_flag_nf_event_cooldown  months = 11 }
    # QA round 2 M1: the cascade year is not also a Neurofractured year. The
    # NF check runs later in the same pulse; this 1-day flag makes it skip
    # the drift and the Heir's Arc start. Read only in
    # eotg_on_yearly_aug_neurofractured_check (no event reads it).
    add_character_flag = { flag = eotg_flag_aug_cascaded_this_pulse  days = 1 }
    # Phase 5: the cascade is the Countdown's ending (phase5 §1.1)
    eotg_aug_end_countdown_effect = yes
    # Balance spec §5.6 (B2 QA): a cascade ends any open chain, so its
    # consumer (e.g. tier3.024, which needs eotg_is_aug_tier3) may never
    # fire to clear the edge/strain flag; clear it here instead
    eotg_aug_clear_chain_flags_effect = yes
    trigger_event = { id = eotg_fracture.0001  days = 1 }
}

# Heal the scoped character's wound completely: the implant replaces the damaged
# tissue. `wounded` is a trait GROUP (wounded_1/2/3); add_trait/remove_trait do
# not accept a group, which is why v1's `remove_trait = wounded` did nothing.
# Pattern copied from vanilla events/activities/pilgrimage_activity/
# pilgrimage_events.txt (remove whichever rank is held; a character holds one).
# Callers: eotg_aug_init.001 (a, d, e), eotg_aug_init.004 (a, e).
eotg_aug_heal_wounds_effect = {
    if = {
        limit = { has_trait = wounded_1 }
        remove_trait = wounded_1
    }
    else_if = {
        limit = { has_trait = wounded_2 }
        remove_trait = wounded_2
    }
    else_if = {
        limit = { has_trait = wounded_3 }
        remove_trait = wounded_3
    }
}

# Reduce the scoped character's wound by one rank (wounded_3 -> 2 -> 1 -> healed).
# Vanilla pattern: common/scripted_effects/20_health_effects.txt, risky treatment
# critical success (remove the rank, add the one below, so the loss is visible).
# Caller: eotg_aug_init.001 (b), inside a 50% random.
eotg_aug_ease_wound_effect = {
    if = {
        limit = { has_trait = wounded_1 }
        remove_trait = wounded_1
    }
    else_if = {
        limit = { has_trait = wounded_2 }
        remove_trait = wounded_2
        add_trait = wounded_1
    }
    else_if = {
        limit = { has_trait = wounded_3 }
        remove_trait = wounded_3
        add_trait = wounded_2
    }
}

# ── Victim picker (phase 0 §2.2, audit §1.5) ────────────────────────────────
# Run in the ruler's scope. Picks one courtier and saves them as scope:$NAME$.
# If no candidate exists nothing is saved: callers guard with
# `exists = scope:$NAME$` and their loc carries a "no one was near" variant.
# Callers flag anyone they hurt:
#   add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
# FAMILY_FACTOR: 0.05 Overclocked / Flicker / Fracture band; 0.25 Storm band and
# Court Massacre; 0 where family must never be hit.
# The ruler is saved as a temporary scope so the family test does not depend on
# root (a caller may run this on a saved scope, not the event root).
# Weighted random_courtier shape: vanilla `weight = { base  modifier = { factor } }`.
# Kinslayer policy (QA round 2 M4): an option where the ruler CHOOSES to kill
# a picked victim (fracture.002.c, .019.c, tier3.001.c, .015.e, .017.b,
# heir.004 kill_c, nr.006.d) calls vanilla add_kinslayer_trait_or_nothing_effect
# (00_secret_effects.txt:433) before the death. Involuntary episode deaths

    remove_variable = eotg_aug_focus

    remove_variable = eotg_aug_sedation_courses

    remove_variable = eotg_aug_nr_battle_bonus

            }
        }
        if = {

            death = { death_reason = death_treatment }
        }
    }

# Apply the saved outcome (spec §4.2 table, last column). No params; run in
# the patient's scope, in the reveal event's immediate, or silently by
# eotg_aug_procedure_effect for a plain reputable outcome. Risk moves are
# hidden. A severe outcome sets eotg_flag_aug_procedure_injury (30 days)
# first, so eotg_on_trait_gained_aug_repair does not offer to repair what the
# table just did. Severe shapes: vanilla apply_maimed_trait_and_modifier_effect
# (20_health_effects.txt:1398), maimed_in_battle_effect's
# add_trait_force_tooltip (:1355-1395), the guarded blind
# (events/activities/coronation_activity/coronation_events_1.txt:8719-8726),
# and the no-death wound step (:1101), so death odds stay exactly the death
# weight.
# Callers: eotg_aug_procedure_effect, eotg_aug_init.011, eotg_aug_tier1.022,
# eotg_aug_proc.002 (immediate).
eotg_aug_procedure_apply_effect = {
    # Clean: only a repair has anything to do
    if = {
        limit = { scope:eotg_proc_outcome ?= flag:clean }
        if = {
            limit = { scope:eotg_proc_kind ?= flag:repair }
            eotg_aug_repair_injury_effect = yes
        }
    }
    else_if = {
        limit = { scope:eotg_proc_outcome ?= flag:excellent }
        if = {
            limit = { scope:eotg_proc_kind ?= flag:repair }
            eotg_aug_repair_injury_effect = yes
            add_character_modifier = { modifier = eotg_mod_aug_clean_install  years = 5 }
        }
        else_if = {
            limit = { scope:eotg_proc_kind ?= flag:removal }
            if = {
                limit = { has_character_modifier = eotg_mod_aug_removal_withdrawal }
                remove_character_modifier = eotg_mod_aug_removal_withdrawal
            }
        }
        else = {
            add_character_modifier = { modifier = eotg_mod_aug_clean_install  years = 10 }
        }
    }
    else_if = {
        limit = { scope:eotg_proc_outcome ?= flag:infection }
        if = {
            limit = { scope:eotg_proc_kind ?= flag:install }
            add_character_modifier = { modifier = eotg_mod_aug_infection  years = 2 }
        }
        else = {
            add_character_modifier = { modifier = eotg_mod_aug_infection  years = 1 }
        }
        if = {
            limit = { scope:eotg_proc_kind ?= flag:repair }
            eotg_aug_repair_injury_effect = yes
        }
    }
    else_if = {
        limit = { scope:eotg_proc_outcome ?= flag:flaw }
        hidden_effect = { add_character_flag = eotg_flag_aug_hidden_flaw }
        if = {
            limit = { scope:eotg_proc_kind ?= flag:repair }
            eotg_aug_repair_injury_effect = yes
        }
    }
    else_if = {
        limit = { scope:eotg_proc_outcome ?= flag:rejection }
        # Install: nothing here; init.011 d/e lead to init.012
        if = {
            limit = { scope:eotg_proc_kind ?= flag:upgrade }
            hidden_effect = { eotg_add_fracture_risk = { AMOUNT = 10 } }
            add_character_modifier = { modifier = eotg_mod_aug_infection  years = 1 }
        }
        else_if = {
            limit = { scope:eotg_proc_kind ?= flag:removal }
            add_character_modifier = { modifier = eotg_mod_aug_fragments  years = 5 }
        }
        else_if = {
            limit = { scope:eotg_proc_kind ?= flag:repair }
            # The repair failed: the injury stays
            hidden_effect = {
                eotg_add_fracture_risk = { AMOUNT = 5 }
                remove_variable = eotg_aug_repair_injury
            }
        }
    }
    else_if = {
        limit = { scope:eotg_proc_outcome ?= flag:discovered }
        add_character_modifier = eotg_mod_illegal_implants
        # Realm spec §4.2: a crime under the patient's realm's Ban or License
        eotg_aug_contraband_effect = yes
        if = {
            limit = { scope:eotg_proc_kind ?= flag:repair }
            eotg_aug_repair_injury_effect = yes
        }
    }
    else_if = {
        limit = { scope:eotg_proc_outcome ?= flag:maimed }
        hidden_effect = { add_character_flag = { flag = eotg_flag_aug_procedure_injury  days = 30 } }
        if = {
            limit = { NOT = { has_trait = maimed } }
            apply_maimed_trait_and_modifier_effect = yes
        }
        increase_wounds_no_death_effect = { REASON = treatment }
    }
    else_if = {
        limit = { scope:eotg_proc_outcome ?= flag:one_eyed }
        hidden_effect = { add_character_flag = { flag = eotg_flag_aug_procedure_injury  days = 30 } }
        add_trait_force_tooltip = one_eyed
        increase_wounds_no_death_effect = { REASON = treatment }
    }
    else_if = {
        limit = { scope:eotg_proc_outcome ?= flag:blind }
        hidden_effect = { add_character_flag = { flag = eotg_flag_aug_procedure_injury  days = 30 } }
        if = {
            limit = { has_trait = one_eyed }
            remove_trait = one_eyed
        }
        if = {
            limit = { NOT = { has_trait = blind } }
            add_trait = blind
        }
        increase_wounds_no_death_effect = { REASON = treatment }
    }
}

# The public entry point (spec §4.1): roll, then report or apply.
#   - death: nothing more (applied in the roll);
#   - a repair always reports (eotg_aug_proc.002): the result is the payoff;
#   - clean or a hidden flaw is applied silently (a flaw is never reported);
#   - an install with anything else is revealed by eotg_aug_init.011;
#   - everything else by eotg_aug_proc.002.
# Callers wrap it in hidden_effect.
# Callers: eotg_aug_proc.001 (a-c, e, f), eotg_aug_proc.003 (a-c, e, f),
# eotg_aug_tier2.003 (a, c, d, f, g), and the ruler install offers
# init.001 (a, d, e), .002 (a, d), .003 (a, d), .004 (a, e), .005 (a, d),
# .006 (a, c, e, f), .009 (a, d), .016 (a, e), .017 (a, d), .018 (a, c),
# .019 (a, d). Interactions spec: eotg_aug_offer_install_effect,
# eotg_aug_demand_removal_effect, eotg_aug_salvage_effect, eotg_aug_int.001
# (b), eotg_aug_int.002 (b), eotg_aug_tamper.004 (a, b, c).
eotg_aug_procedure_effect = {
    eotg_aug_procedure_roll_effect = { PATIENT = $PATIENT$  PROVIDER = $PROVIDER$  PROCEDURE = $PROCEDURE$  SURGEON = $SURGEON$ }
    if = {
        limit = { NOT = { scope:eotg_proc_outcome = flag:death } }
        if = {
            limit = { scope:eotg_proc_kind = flag:repair }
            $PATIENT$ = { trigger_event = { id = eotg_aug_proc.002  days = { 14 45 } } }
        }
        else_if = {
            limit = {
                OR = {
                    scope:eotg_proc_outcome = flag:clean
                    scope:eotg_proc_outcome = flag:flaw
                }
            }
            $PATIENT$ = { eotg_aug_procedure_apply_effect = yes }
        }
        else_if = {
            limit = { scope:eotg_proc_kind = flag:install }
            $PATIENT$ = { trigger_event = { id = eotg_aug_init.011  days = { 30 90 } } }
        }
        else = {
            $PATIENT$ = { trigger_event = { id = eotg_aug_proc.002  days = { 14 45 } } }
        }
    }
}

# A removal's body (spec §4.4, proc.001): the three removal decisions' old
# effect bodies, moved unchanged into one ladder on var:eotg_aug_removal_kind
# (set by the decision). Each carries its decision's original stress: the
# reject helper for a full removal, medium_stress_impact_loss for partial and
# downgrade, so proc.001's paid options run no surgery helper.
# The tooltips are proc.001's own (eotg_aug_proc.001.tt_partial /
# .tt_downgrade), not the decisions' keys: those end by asking who performs
# it, which is already chosen here (procedures QA).
# Caller: eotg_aug_proc.001 (a-c, e, f).
eotg_aug_removal_perform_effect = {
    if = {
        limit = { var:eotg_aug_removal_kind ?= flag:partial }
        eotg_aug_set_integration_effect = { XP = 0 }
        # Balance spec §5.9: a masked congenital trait shows again
        eotg_aug_unmask_congenital_effect = yes
        stress_impact = { base = medium_stress_impact_loss }
        custom_tooltip = eotg_aug_proc.001.tt_partial
    }
    else_if = {
        limit = { var:eotg_aug_removal_kind ?= flag:downgrade }
        # Some of the damage remains: risk is halved, not cleared (B3). Halved
        # before the tier change so the Enhanced character carries it forward.
        if = {
            limit = { has_variable = eotg_fracture_risk }
            change_variable = { name = eotg_fracture_risk  multiply = 0.5 }
        }
        eotg_aug_set_integration_effect = { XP = 50 }
        # Phase 5 §1.1: a running Countdown stops if the halved risk is < 50
        # (eotg_aug_countdown.006 The Quiet ends the story). At 50+ the story
        # sees the lost tier at its next tick and fires The Quiet itself.
        if = {
            limit = {
                eotg_aug_has_countdown = yes
                OR = {

        # Balance spec §5.9: a masked congenital trait shows again

# Offer Augmentation's install (spec §4.1), mirroring init.018's three paths
# with PATIENT = scope:recipient. PROVIDER: clinic / physician / backstreet.
# Price: minor_gold_value (the price for installing on someone else,
# init.020), x eotg_aug_price_mult_physician for the physician;
# tiny_gold_value in the back streets (mirror: the interaction's
# is_valid_showing_failures_only). The offer doubles as a prosthetic: one
# physical loss is restored (The Prosthetic, init.006). The reveal goes to the
# recipient (init.011 or silent). Willing: gratitude; hook-forced: Forced
# Procedure.
# Caller: eotg_aug_offer_augmentation_interaction (on_accept).
eotg_aug_offer_install_effect = {
    save_temporary_scope_value_as = { name = eotg_offer_provider  value = flag:$PROVIDER$ }
    # 1. The actor pays
    scope:actor = {
        if = {
            limit = { scope:eotg_offer_provider = flag:physician }
            eotg_aug_pay_procedure_effect = { BASE = minor_gold_value  MULT = eotg_aug_price_mult_physician }
        }
        else_if = {
            limit = { scope:eotg_offer_provider = flag:backstreet }
            eotg_aug_pay_procedure_effect = { BASE = tiny_gold_value  MULT = 1 }
        }
        else = {
            eotg_aug_pay_procedure_effect = { BASE = minor_gold_value  MULT = 1 }
        }
    }
    # 2. Install, and restore one loss
    scope:recipient = {
        eotg_aug_initiate_effect = yes
        if = {
            limit = { eotg_has_physical_loss = yes }
            eotg_aug_restore_loss_effect = yes
        }
    }
    # 4. The back streets (init.018.b): risk 5, roll, the reveal at init.011
    if = {
        limit = { scope:eotg_offer_provider = flag:backstreet }
        scope:recipient = {
            hidden_effect = {
                set_variable = { name = eotg_fracture_risk  value = 5 }
            }
        }
        custom_tooltip = eotg_aug_proc.install_tt
        hidden_effect = {
            eotg_aug_procedure_roll_effect = { PATIENT = scope:recipient  PROVIDER = backstreet  PROCEDURE = install  SURGEON = scope:recipient }
        }
        if = {
            limit = { NOT = { scope:eotg_proc_outcome = flag:death } }
            scope:recipient = {
                trigger_event = { id = eotg_aug_init.011  days = { 180 540 } }
            }
        }
    }
    # 3. Clinic or physician (init.018 a / c)
    else = {
        scope:recipient = {
            if = {
                limit = { scope:eotg_offer_provider = flag:physician }
                add_character_modifier = { modifier = eotg_mod_implant_calibrated  years = 3 }
            }
            else = {
                add_character_modifier = { modifier = eotg_mod_implant_calibrated  years = 2 }
            }
        }
        custom_tooltip = eotg_aug_proc.install_tt
        if = {
            limit = { scope:eotg_offer_provider = flag:physician }
            hidden_effect = {
                scope:actor = { eotg_aug_save_surgeon_effect = yes }
                eotg_aug_procedure_effect = { PATIENT = scope:recipient  PROVIDER = physician  PROCEDURE = install  SURGEON = scope:eotg_proc_surgeon }
            }
        }
        else = {
            hidden_effect = {
                eotg_aug_procedure_effect = { PATIENT = scope:recipient  PROVIDER = clinic  PROCEDURE = install  SURGEON = scope:recipient }
            }
        }
    }
    # 5. Opinion (the willing shape: eotg_aug_retainer_install_effect)
    scope:recipient = {
        if = {
            limit = { is_alive = yes }
            if = {
                limit = { scope:hook ?= yes }
                add_opinion = { modifier = eotg_opinion_aug_forced_procedure  target = scope:actor }
            }
            else = {
                add_opinion = { modifier = eotg_opinion_aug_grateful_patient  target = scope:actor  years = 10 }
            }
        }
    }
}

# Demand Implant Removal (spec §4.2). Same order as proc.001: pay, perform,
# roll. The full-removal body (eotg_aug_removal_perform_effect: remove-all,
# 3-year withdrawal, the reject stress; former flag:removed) runs on the
# recipient through a 30-day booking that is spent at once. The interaction's
# is_valid_showing_failures_only refuses while the recipient has a booking of
# their own (the procedures QA booking guard), so none is overwritten.
# PROVIDER: clinic / physician / backstreet.
# Caller: eotg_aug_demand_removal_interaction (on_accept).
eotg_aug_demand_removal_effect = {
    save_temporary_scope_value_as = { name = eotg_demand_provider  value = flag:$PROVIDER$ }
    # 1. The actor pays (full-removal clinic price, medium_gold_value)
    scope:actor = {
        if = {
            limit = { scope:eotg_demand_provider = flag:physician }
            eotg_aug_pay_procedure_effect = { BASE = medium_gold_value  MULT = eotg_aug_price_mult_physician }
        }
        else_if = {
            limit = { scope:eotg_demand_provider = flag:backstreet }
            eotg_aug_pay_procedure_effect = { BASE = medium_gold_value  MULT = eotg_aug_price_mult_backstreet_removal }
        }
        else = {
            eotg_aug_pay_procedure_effect = { BASE = medium_gold_value  MULT = 1 }
        }
    }
    # 2. The actor's surgeon
    if = {
        limit = { scope:eotg_demand_provider = flag:physician }
        scope:actor = { eotg_aug_save_surgeon_effect = yes }
    }
    # 3. Perform, on the recipient
    scope:recipient = {
        hidden_effect = {
            set_variable = { name = eotg_aug_removal_kind  value = flag:full  days = 30 }
        }
        eotg_aug_removal_perform_effect = yes
    }
    # 4. Roll
    custom_tooltip = eotg_aug_demand_removal_effect_tt
    if = {
        limit = { scope:eotg_demand_provider = flag:physician }
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = scope:recipient  PROVIDER = physician  PROCEDURE = removal  SURGEON = scope:eotg_proc_surgeon }
        }
    }
    else_if = {
        limit = { scope:eotg_demand_provider = flag:backstreet }
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = scope:recipient  PROVIDER = backstreet  PROCEDURE = removal  SURGEON = scope:recipient }
        }
    }
    else = {
        hidden_effect = {
            eotg_aug_procedure_effect = { PATIENT = scope:recipient  PROVIDER = clinic  PROCEDURE = removal  SURGEON = scope:recipient }
        }
    }
    # 5. The booking is spent
    scope:recipient = {
        hidden_effect = { remove_variable = eotg_aug_removal_kind }
    }
    # 6. Opinion: forced, or a willing zealot reassured; otherwise none
    scope:recipient = {
        if = {
            limit = { is_alive = yes }
            if = {
                limit = { scope:hook ?= yes }
                add_opinion = { modifier = eotg_opinion_aug_forced_procedure  target = scope:actor }
            }
            else_if = {
                limit = { has_trait = zealous }
                add_opinion = { modifier = eotg_opinion_aug_reassured  target = scope:actor  years = 5 }
            }
        }
    }
    # 7. The actor's stress: only a hook-forced demand
    if = {
        limit = { scope:hook ?= yes }
        scope:actor = { eotg_aug_stress_tyranny_effect = yes }
    }
}

# A Fracturing Inheritance — docs/specs/cybernetics_v2_fracturing_inheritance.md
# The ruler is sound and the primary heir is Neurofractured. Every effect here
# runs in the RULER's scope unless stated, and reads the people through saved
# scopes (scope:eotg_inh_ruler, scope:eotg_inh_heir, ...) rather than root:
# the start effect can run from the non-ruler cascade, whose root is the
# cascading champion. Chain state lives in eotg_inh_* variables and
# eotg_flag_aug_inh_* flags on the ruler (spec §3.3); the close effect clears
# them. Outcome mechanics live in the leaf effects, called by the tree's leaf
# events AND by the hidden eotg_aug_inherit.028 (owner Q5), so both paths

# Entry (spec §5.1). SOURCE = report / cascade. The only place the cooldown
# is set; it is read only in eotg_aug_inherit_can_start (lesson 5).
# Callers: eotg_on_yearly_aug_inherit_check, eotg_aug_nr_cascade_effect.
# Fires eotg_aug_inherit.001 (player ruler) or eotg_aug_inherit.028 (AI).
eotg_aug_inherit_start_effect = {
    save_scope_as = eotg_inh_ruler
    add_character_flag = { flag = eotg_flag_aug_inh_cooldown  years = 3 }
    add_character_flag = { flag = eotg_flag_aug_inh_active  years = 6 }
    primary_heir = {
        add_character_flag = eotg_flag_aug_inh_seen
        save_scope_as = eotg_inh_heir
    }
    set_variable = { name = eotg_inh_heir  value = scope:eotg_inh_heir }
    set_variable = { name = eotg_inh_source  value = flag:$SOURCE$ }

    # The profile (§5A.1): how the Neurofracture shows. Weights from the
    # heir's vanilla traits and the Neurofractured distortion modifier
    # (eotg_apply_neurofracture_distortion). Flag-variable shape: the
    # Patron's terms (eotg_aug_patron_accept_effect).
    scope:eotg_inh_heir = {
        random_list = {
            20 = {    # Running Hot
                modifier = { add = 25  has_trait = wrathful }
                modifier = { add = 10  has_trait = impatient }
                modifier = { add = 15  has_character_modifier = eotg_mod_nf_martial_distortion }
                modifier = { add = 10  prowess >= 12 }
                modifier = { add = 10  eotg_aug_pressure_storm = yes }
                modifier = { factor = 0.5  has_trait = calm }
                scope:eotg_inh_ruler = { set_variable = { name = eotg_inh_profile  value = flag:rage } }
            }
            20 = {    # The Ledger
                modifier = { add = 25  has_trait = paranoid }
                modifier = { add = 10  has_trait = deceitful }
                modifier = { add = 15  has_character_modifier = eotg_mod_nf_intrigue_distortion }
                modifier = { factor = 0.5  has_trait = trusting }
                scope:eotg_inh_ruler = { set_variable = { name = eotg_inh_profile  value = flag:ledger } }
            }
            20 = {    # Flatline
                modifier = { add = 20  has_trait = callous }
                modifier = { add = 10  has_trait = shy }
                modifier = { add = 10  has_trait = content }
                modifier = {
                    add = 15
                    OR = {
                        has_character_modifier = eotg_mod_nf_learning_distortion
                        has_character_modifier = eotg_mod_nf_steward_distortion
                    }
                }
                modifier = { factor = 0.5  has_trait = compassionate }
                scope:eotg_inh_ruler = { set_variable = { name = eotg_inh_profile  value = flag:cold } }
            }
            20 = {    # Certainty
                modifier = { add = 20  has_trait = arrogant }
                modifier = { add = 20  has_trait = ambitious }
                modifier = { add = 15  has_character_modifier = eotg_mod_nf_diplomacy_distortion }
                modifier = { factor = 0.5  has_trait = humble }
                scope:eotg_inh_ruler = { set_variable = { name = eotg_inh_profile  value = flag:certain } }
            }
        }
    }

    # The bond (§5A.2): first valid, read once.
    if = {
        limit = {
            scope:eotg_inh_heir = {
                OR = {
                    has_relation_rival = scope:eotg_inh_ruler
                    has_relation_potential_rival = scope:eotg_inh_ruler
                }
            }
        }
        set_variable = { name = eotg_inh_bond  value = flag:rival }
    }
    else_if = {
        limit = {
            scope:eotg_inh_heir = {
                opinion = { target = scope:eotg_inh_ruler  value <= -25 }
            }
        }
        set_variable = { name = eotg_inh_bond  value = flag:estranged }
    }
    else_if = {
        limit = {
            scope:eotg_inh_heir = {
                has_trait = ambitious
                opinion = { target = scope:eotg_inh_ruler  value < 20 }
            }
        }
        set_variable = { name = eotg_inh_bond  value = flag:claimant }
    }
    else_if = {
        limit = {
            OR = {
                has_relation_friend = scope:eotg_inh_heir
                has_relation_best_friend = scope:eotg_inh_heir
                has_relation_soulmate = scope:eotg_inh_heir
                opinion = { target = scope:eotg_inh_heir  value >= 40 }
                scope:eotg_inh_heir = {
                    opinion = { target = scope:eotg_inh_ruler  value >= 50 }
                }
            }
        }
        set_variable = { name = eotg_inh_bond  value = flag:favourite }
    }
    else = {
        set_variable = { name = eotg_inh_bond  value = flag:dutiful }
    }

    if = {
        limit = { eotg_aug_inherit_tree_runs = yes }
        # The witness, first valid: the court physician (endgame.txt:98
        # shape), the spouse, an adult courtier (a non-violent pick of a
        # speaker, index §6 item 6). May stay unset.
        if = {
            limit = { employs_court_position = court_physician_court_position }
            court_position:court_physician_court_position = {
                if = {
                    limit = { NOT = { this = scope:eotg_inh_heir } }
                    save_scope_as = eotg_inh_witness
                }
            }
        }
        if = {
            limit = {
                NOT = { exists = scope:eotg_inh_witness }
                exists = primary_spouse
                NOT = { primary_spouse = scope:eotg_inh_heir }
            }
            primary_spouse = { save_scope_as = eotg_inh_witness }
        }
        if = {
            limit = { NOT = { exists = scope:eotg_inh_witness } }
            random_courtier = {
                limit = {
                    is_adult = yes
                    NOT = { this = scope:eotg_inh_heir }
                }
                save_scope_as = eotg_inh_witness
            }
        }
        if = {
            limit = { exists = scope:eotg_inh_witness }
            set_variable = { name = eotg_inh_witness  value = scope:eotg_inh_witness }
        }
        trigger_event = { id = eotg_aug_inherit.001  days = { 3 10 } }
    }
    else = {
        # The months the tree would have taken (spec §5.5)
        trigger_event = { id = eotg_aug_inherit.028  days = { 30 180 } }
    }
}

# L14 (required): the massacre (spec §5.4.2). From the ward: release first.

    if = {
        limit = { scope:eotg_inh_massacre_n >= 3 }
        eotg_aug_inherit_strike_effect = { NAME = eotg_inh_dead_3 }
    }
    if = {
        limit = { scope:eotg_inh_massacre_n >= 4 }
        eotg_aug_inherit_strike_effect = { NAME = eotg_inh_dead_4 }
    }
    eotg_aug_inherit_strike_effect = { NAME = eotg_inh_wounded }
    if = {
        limit = { exists = scope:eotg_inh_dead_1 }
        scope:eotg_inh_dead_1 = { death = { death_reason = death_murder_known  killer = scope:eotg_inh_heir } }
        change_variable = { name = eotg_inh_dead_count  add = 1 }
    }
    if = {
        limit = { exists = scope:eotg_inh_dead_2 }
        scope:eotg_inh_dead_2 = { death = { death_reason = death_murder_known  killer = scope:eotg_inh_heir } }
        change_variable = { name = eotg_inh_dead_count  add = 1 }
    }
    if = {
        limit = { exists = scope:eotg_inh_dead_3 }
        scope:eotg_inh_dead_3 = { death = { death_reason = death_murder_known  killer = scope:eotg_inh_heir } }
        change_variable = { name = eotg_inh_dead_count  add = 1 }

        limit = { exists = scope:eotg_inh_dead_4 }
        scope:eotg_inh_dead_4 = { death = { death_reason = death_murder_known  killer = scope:eotg_inh_heir } }

    }
    if = {
        limit = { NOT = { exists = scope:eotg_inh_dead_1 } }
        increase_wounds_effect = { REASON = attacked }
    }
    add_character_modifier = { modifier = eotg_mod_aug_inh_emptied_court  years = 5 }
    set_variable = { name = eotg_inh_outcome  value = flag:l14 }

# L15 (required): the heir comes for the ruler. MOD is the option's roll
# modifier (a number or a script value). Success: kinslayer on the heir
# (VICTIM the ruler), H-10, the chain closes, and the ruler dies
# (death_murder_known, killer the heir; ep3_contract_events.txt:8779). The
# event to the heir is queued before the death (heir.006's order). Failure:
# the ruler is wounded, H+10, and NOTIFY sends .024. The AI default after a
# failure (.024 a) is the caller's.
# Callers: .018 (a, b, c; NOTIFY = yes), .028 (attempt; NOTIFY = no).
eotg_aug_inherit_leaf_attempt_effect = {
    save_scope_value_as = { name = eotg_inh_attempt_mod  value = $MOD$ }
    eotg_aug_inherit_roll_effect = { CHANCE = eotg_aug_inherit_attempt_chance  NAME = eotg_inh_attempt_ok }
    if = {
        limit = { exists = scope:eotg_inh_attempt_ok }
        save_scope_as = eotg_inh_old_ruler_s
        save_scope_value_as = { name = eotg_inh_profile_s  value = var:eotg_inh_profile }
        set_variable = { name = eotg_inh_outcome  value = flag:l15 }
        scope:eotg_inh_heir = {
            eotg_add_fracture_risk = { AMOUNT = -10 }
            if = {
                limit = { always = $NOTIFY$ }
                trigger_event = { id = eotg_aug_inherit.025  days = 3 }
            }
            add_kinslayer_trait_or_nothing_effect = { VICTIM = scope:eotg_inh_old_ruler_s }
        }
        eotg_aug_inherit_close_effect = yes
        death = {
            death_reason = death_murder_known
            killer = scope:eotg_inh_heir
        }
    }
    else = {
        increase_wounds_effect = { REASON = attacked }
        scope:eotg_inh_heir = { eotg_add_fracture_risk = { AMOUNT = 10 } }
        if = {
            limit = { always = $NOTIFY$ }
            send_interface_toast = {
                title = eotg_aug_inherit.018.fail_tt
                left_icon = scope:eotg_inh_heir
            }
            trigger_event = { id = eotg_aug_inherit.024  days = 1 }
        }
    }
}
