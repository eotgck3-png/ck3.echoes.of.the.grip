# Remove all 15 tier synergy modifiers.
# CB-27 follow-up (tooltip): every removal is guarded on the modifier being
# held, so an option tooltip lists only what the character actually loses.
# Vanilla pattern: common/scripted_effects/00_prison_effects.txt:2572-2575

# Clear, then apply the current tier's synergy by highest_skill (martial fallback).
# Applies nothing to Neurofractured or unaugmented characters.
eotg_aug_refresh_synergy_effect = {
    eotg_aug_clear_synergy_effect = yes
    if = {
        limit = { eotg_is_aug_tier3 = yes }
        if = {
            limit = { highest_skill = martial }
            add_character_modifier = { modifier = eotg_mod_oc_martial_bonus }
        }
        else_if = {
            limit = { highest_skill = stewardship }
            add_character_modifier = { modifier = eotg_mod_oc_steward_bonus }
        }
        else_if = {
            limit = { highest_skill = intrigue }
            add_character_modifier = { modifier = eotg_mod_oc_intrigue_bonus }
        }
        else_if = {
            limit = { highest_skill = learning }
            add_character_modifier = { modifier = eotg_mod_oc_learning_bonus }
        }
        else_if = {
            limit = { highest_skill = diplomacy }
            add_character_modifier = { modifier = eotg_mod_oc_diplomacy_bonus }
        }
        else = {
            add_character_modifier = { modifier = eotg_mod_oc_martial_bonus }
        }
    }
    else_if = {
        limit = { eotg_is_aug_tier2 = yes }
        if = {
            limit = { highest_skill = martial }
            add_character_modifier = { modifier = eotg_mod_enh_martial_bonus }
        }
        else_if = {
            limit = { highest_skill = stewardship }
            add_character_modifier = { modifier = eotg_mod_enh_steward_bonus }
        }
        else_if = {
            limit = { highest_skill = intrigue }
            add_character_modifier = { modifier = eotg_mod_enh_intrigue_bonus }
        }
        else_if = {
            limit = { highest_skill = learning }
            add_character_modifier = { modifier = eotg_mod_enh_learning_bonus }
        }
        else_if = {
            limit = { highest_skill = diplomacy }
            add_character_modifier = { modifier = eotg_mod_enh_diplomacy_bonus }
        }
        else = {
            add_character_modifier = { modifier = eotg_mod_enh_martial_bonus }
        }
    }
    else_if = {
        limit = { eotg_is_aug_tier1 = yes }
        if = {
            limit = { highest_skill = martial }
            add_character_modifier = { modifier = eotg_mod_aug_martial_bonus }

        else_if = {
            limit = { highest_skill = stewardship }

            limit = { highest_skill = learning }

# Initiation: add the track trait (engine default XP 0 = Augmented) and its synergy.
# Used for rulers and knights. Callers: eotg_aug_init.001-.005 (accept options,
# incl. the Phase 1 trait options init.001.d/e, .002.d, .003.d, .004.e, .005.d),
# eotg_aug_init.002 (c, knight), eotg_aug_tier1.004 (a, d: knight).
# Phase 2: eotg_aug_init.006-.018 install options, init.017.c (knight),
# init.020.d, eotg_aug_child_install_effect, eotg_aug_retainer_install_effect.
# Final QA: a character coming (back) into the system starts at the install
# value, risk 0, so stale risk from an earlier augmentation (e.g. one the
# Patron's debt kept moving) cannot carry into the new one.
# Balance spec §5.1: a newcomer settles for 5 years before the next stage
# (eotg_flag_aug_settling, read by eotg_can_progress_to_* and the Pursue
# decision). Only in the "was unaugmented" branch, so a knight re-initiated
# while augmented does not reset it.
eotg_aug_initiate_effect = {
    if = {
        limit = { eotg_is_augmented_any = no }
        set_variable = { name = eotg_fracture_risk  value = 0 }
        add_character_flag = { flag = eotg_flag_aug_settling  years = 5 }
    }
    add_trait = eotg_cybernetics
    eotg_aug_refresh_synergy_effect = yes
}

# THE ONLY place add_trait_xp on eotg_cybernetics may appear (spec §2.2 rule 2).
# Sets XP to exactly $XP$ (0, 50 or 100) and refreshes the synergy modifier.
# Callers: eotg_aug_tier1.022 (a, b) XP 50 (Phase 4a: tier1.002 now pays and
#          fires the First Upgrade chain); eotg_aug_tier2.003 (a, c, d) XP 100;
#          eotg_decision_partial_removal XP 0; eotg_decision_overclock_regression XP 50.
# Balance spec §5.1: every tier change, regressions included, starts a fresh
# 5-year eotg_flag_aug_settling (relapse is slower too). The Pursue decision
# reaches this only through tier1.022 / tier2.003; it never sets XP itself.
eotg_aug_set_integration_effect = {
    if = {
        limit = { has_trait = eotg_cybernetics }
        add_trait_xp = {
            trait = eotg_cybernetics
            value = {
                value = $XP$
                subtract = "has_trait_xp(eotg_cybernetics)"
            }
        }
        eotg_aug_refresh_synergy_effect = yes
        add_character_flag = { flag = eotg_flag_aug_settling  years = 5 }
    }
}

# Increment fracture risk variable — expects $AMOUNT$ parameter.
# A Seamless character (eotg_total_integration, Phase 3b) has no risk: the
# variable was removed with the seam, and nothing may recreate it (e.g. the
# Heir's Arc options, which can reach a Seamless owner).
# QA round 2 M3: nor may anything recreate it on someone who has left the
# system (delayed chain stages, e.g. tier1.017, tier2.017/.018/.021, can land
# after Remove Implants or Excision): the character must be augmented. Every
# install path adds the trait before it touches risk, so no caller loses a
# move it meant to make.
eotg_add_fracture_risk = {
    if = {
        limit = {
            eotg_is_augmented_any = yes
            NOT = { has_trait = eotg_total_integration }
        }
        if = {
            limit = { NOT = { has_variable = eotg_fracture_risk } }
            set_variable = { name = eotg_fracture_risk  value = 0 }
        }
        change_variable = { name = eotg_fracture_risk  add = $AMOUNT$ }
        clamp_variable = { name = eotg_fracture_risk  min = 0  max = 100 }
    }
}

# Trigger the Neural Cascade (fracture transformation)
eotg_trigger_neurofracture = {
    if = { limit = { has_trait = eotg_cybernetics }  remove_trait = eotg_cybernetics }    # discards the track XP
    add_trait = eotg_neurofractured
    eotg_apply_neurofracture_distortion = yes
    set_variable = { name = eotg_fracture_risk  value = 0 }
    # Open the Neurofractured cooldown here so eotg_on_yearly_aug_neurofractured_check,
    # which runs later in the same yearly_playable_pulse, cannot roll fracture.002-.007
    # days after the cascade. Same duration the on_action uses (11 months, B4: expires
    # before the next pulse, so no same-day race); no event trigger reads it.
    add_character_flag = { flag = eotg_flag_nf_event_cooldown  months = 11 }
    # QA round 2 M1: the cascade year is not also a Neurofractured year. The
    # NF check runs later in the same pulse; this 1-day flag makes it skip
    # the drift and the Heir's Arc start. Read only in
    # eotg_on_yearly_aug_neurofractured_check (no event reads it).
    add_character_flag = { flag = eotg_flag_aug_cascaded_this_pulse  days = 1 }
    # Phase 5: the cascade is the Countdown's ending (phase5 §1.1)
    eotg_aug_end_countdown_effect = yes
    # Balance spec §5.6 (B2 QA): a cascade ends any open chain, so its
    # consumer (e.g. tier3.024, which needs eotg_is_aug_tier3) may never
    # fire to clear the edge/strain flag; clear it here instead
    eotg_aug_clear_chain_flags_effect = yes
    trigger_event = { id = eotg_fracture.0001  days = 1 }
}

# Heal the scoped character's wound completely: the implant replaces the damaged
# tissue. `wounded` is a trait GROUP (wounded_1/2/3); add_trait/remove_trait do
# not accept a group, which is why v1's `remove_trait = wounded` did nothing.
# Pattern copied from vanilla events/activities/pilgrimage_activity/
# pilgrimage_events.txt (remove whichever rank is held; a character holds one).
# Callers: eotg_aug_init.001 (a, d, e), eotg_aug_init.004 (a, e).
eotg_aug_heal_wounds_effect = {
    if = {
        limit = { has_trait = wounded_1 }
        remove_trait = wounded_1
    }
    else_if = {
        limit = { has_trait = wounded_2 }
        remove_trait = wounded_2
    }
    else_if = {
        limit = { has_trait = wounded_3 }
        remove_trait = wounded_3
    }
}

# Reduce the scoped character's wound by one rank (wounded_3 -> 2 -> 1 -> healed).
# Vanilla pattern: common/scripted_effects/20_health_effects.txt, risky treatment
# critical success (remove the rank, add the one below, so the loss is visible).
# Caller: eotg_aug_init.001 (b), inside a 50% random.
eotg_aug_ease_wound_effect = {
    if = {
        limit = { has_trait = wounded_1 }
        remove_trait = wounded_1
    }
    else_if = {
        limit = { has_trait = wounded_2 }
        remove_trait = wounded_2
        add_trait = wounded_1
    }
    else_if = {
        limit = { has_trait = wounded_3 }
        remove_trait = wounded_3
        add_trait = wounded_2
    }
}

# ── Victim picker (phase 0 §2.2, audit §1.5) ────────────────────────────────
# Run in the ruler's scope. Picks one courtier and saves them as scope:$NAME$.
# If no candidate exists nothing is saved: callers guard with
# `exists = scope:$NAME$` and their loc carries a "no one was near" variant.
# Callers flag anyone they hurt:
#   add_character_flag = { flag = eotg_flag_aug_breach_victim  days = 30 }
# FAMILY_FACTOR: 0.05 Overclocked / Flicker / Fracture band; 0.25 Storm band and
# Court Massacre; 0 where family must never be hit.
# The ruler is saved as a temporary scope so the family test does not depend on
# root (a caller may run this on a saved scope, not the event root).
# Weighted random_courtier shape: vanilla `weight = { base  modifier = { factor } }`.
# Kinslayer policy (QA round 2 M4): an option where the ruler CHOOSES to kill
# a picked victim (fracture.002.c, .019.c, tier3.001.c, .015.e, .017.b,
# heir.004 kill_c, nr.006.d) calls vanilla add_kinslayer_trait_or_nothing_effect
# (00_secret_effects.txt:433) before the death. Involuntary episode deaths
# (the cascade, Containment Failure, the Court Massacre, Blood on the Sleeve,
# The Missing Courtier) carry none: the implant acted, not the ruler.
eotg_aug_pick_victim_effect = {
    save_temporary_scope_as = eotg_aug_victim_owner
    random_courtier = {
        limit = { eotg_aug_victim_candidate = yes }
        weight = {
            base = 100
            # Guards and knights stand closest
            modifier = {
                factor = 1.5
                is_knight = yes
            }
            modifier = {
                factor = $FAMILY_FACTOR$
                OR = {
                    is_close_family_of = scope:eotg_aug_victim_owner
                    is_spouse_of = scope:eotg_aug_victim_owner
                }
            }
            modifier = {

# The Warrant made real (audit §1.4). Run in the ruler's scope; the caller
# saves scope:eotg_warrant_vassal. Shape copied from vanilla
# prelude_events_0010_screw_over_guest_effect
# (events/activities/coronation_activity/prelude_events.txt:236), including its
# `NOT = { exists = faction_war }` guard: discontent on a faction already at war
# does nothing. Liberty first ("bind the mad ruler"), independence as fallback;
# the vanilla factions and CBs run any war. No title is referenced.
# QA round 2: create_faction never conscripts a player (multiplayer) vassal:
# the vassal must be is_ai = yes, as vanilla filters the guest whose grudge
# raises the faction (prelude_events.txt:209-211,
# prelude_events_0010_appropriate_guest). A player vassal falls through to
# "the warrant circulates".
# Saves scope:eotg_warrant_result = flag:discontent / flag:faction /
# flag:circulated so the caller's desc can say what happened.
# Callers: eotg_fracture.006 (immediate), eotg_fracture.025 (b, 30%),
# eotg_aug_heir.004 (usurp, landed heir).
eotg_aug_raise_warrant_effect = {
    if = {
        limit = {
            any_targeting_faction = { NOT = { exists = faction_war } }
        }
        every_targeting_faction = {
            limit = { NOT = { exists = faction_war } }
            add_faction_discontent = major_discontent_gain
        }
        save_scope_value_as = { name = eotg_warrant_result  value = flag:discontent }
    }
    else_if = {
        limit = {
            scope:eotg_warrant_vassal = {
                is_ai = yes
                can_create_faction = { type = liberty_faction  target = root }
            }
        }
        scope:eotg_warrant_vassal = {
            create_faction = { type = liberty_faction  target = root }
        }
        save_scope_value_as = { name = eotg_warrant_result  value = flag:faction }
    }
    else_if = {
        limit = {
            scope:eotg_warrant_vassal = {
                is_ai = yes
                can_create_faction = { type = independence_faction  target = root }
            }
        }
        scope:eotg_warrant_vassal = {
            create_faction = { type = independence_faction  target = root }
        }
        save_scope_value_as = { name = eotg_warrant_result  value = flag:faction }
    }
    else = {
        # The warrant circulates
        every_vassal = {
            limit = { opinion = { target = root  value < 0 } }
            add_opinion = { modifier = eotg_opinion_aug_disgust  target = root  years = 5 }
        }
        save_scope_value_as = { name = eotg_warrant_result  value = flag:circulated }
    }
}

            limit = { exists = scope:eotg_misled }

                limit = { scope:eotg_letter_kind = flag:confession }
                send_interface_toast = {
                    title = eotg_fracture.015.b.reveal_confession
