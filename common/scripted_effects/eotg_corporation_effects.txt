# -- eotg_corporation_effects.txt ------------------------------------------------
# Governments v2, Phase G2, Corporation batch CO1 (resource core).
# Spec: docs/specs/governments_v2_g2_corporation.md. Pattern: the shipped Cartel batch C1
# (eotg_cartel_*.txt), whose shapes were validated locally; this file copies
# them. UNVALIDATED cloud build (no Tiger, PX or game run).
# All character scope.
# AUTHORITY (spec §2.2): eotg_corporation_change_board_influence_effect is the ONLY place any script
# writes var:eotg_corp_board_influence. Init (start value), inheritance (on_death)
# and, in later batches, every decision call it. Teardown does NOT touch the
# variable (it stays dormant, spec §2.6), so there is no second writer.
# eotg_corporation_sync_story_effect is the only writer of the story mirror
# eotg_corporation_board_shown (on_setup seeds it at 0 once).
# Map-agnostic: no title, culture, faith, province or character key.
# Identity is read through the government flag, never has_government.
# -----------------------------------------------------------------------------

# Change the resource by $AMOUNT$: a number or a NAMED script value (never an
# inline { } block). Callers gate spends in is_valid; this effect never
# checks affordability (Cartel spec §2.3). set_variable shows no tooltip, so
# every visible caller adds its own custom tooltip (spec §2.3).
# save_scope_value_as form: as eotg_cartel_change_dread_effect
# (vanilla 00_administrative_effects.txt:101-104, cited there).
eotg_corporation_change_board_influence_effect = {
    save_scope_value_as = {
        name = eotg_corporation_board_amount
        value = $AMOUNT$
    }
    set_variable = {
        name = eotg_corp_board_influence
        value = eotg_corporation_board_influence_next_value
    }
    # Refresh the story counter now
    eotg_corporation_sync_story_effect = yes
}

# Copy the owner's value onto the story's display variable (spec §2.7 layer
# 4). Display only. Story lookup form: eotg_cartel_sync_story_effect. A
# character without the story (an heir before taking land) is a no-op.
# The direct var: read is safe: every caller runs after the variable is set
# (the change effect writes it first; init writes it before its final sync;
# a story exists only on a character init has already run on).
eotg_corporation_sync_story_effect = {
    save_temporary_scope_as = eotg_corporation_sync_owner
    random_owned_story = {
        limit = { story_type = eotg_story_corporation_board }
        set_variable = {
            name = eotg_corporation_board_shown
            value = scope:eotg_corporation_sync_owner.var:eotg_corp_board_influence
        }
    }
}

# Idempotent setup (spec §2.6). Called from eotg_corporation_on_game_start (also
# after the lobby), eotg_corporation_on_government_change (new government is
# Corporation) and eotg_corporation_on_title_gain (landed Corporation ruler
# without the story: heirs, newly landed courtiers). The start value is
# written only when the variable is unset, so an inherited (eotg_corporation_on_death)
# or dormant value is kept. Story first, so the change effect's sync finds it.
eotg_corporation_init_effect = {
    if = {
        limit = {
            NOT = {
                any_owned_story = { story_type = eotg_story_corporation_board }
            }
        }
        create_story = eotg_story_corporation_board
    }
    if = {
        limit = {
            NOT = { has_variable = eotg_corp_board_influence }
        }
        eotg_corporation_change_board_influence_effect = { AMOUNT = eotg_corporation_board_influence_start_value }
    }
    if = {
        limit = {
            NOT = { has_character_modifier = eotg_corporation_board_modifier }
        }
        add_character_modifier = eotg_corporation_board_modifier
    }
    eotg_corporation_sync_story_effect = yes
}

# Undo eotg_corporation_init_effect (spec §2.6). Does NOT touch var:eotg_corp_board_influence:
# it stays dormant on the character (nothing reads it without the
# government flag), which keeps the change effect the only writer. Called
# from eotg_corporation_on_government_change (Corporation -> other) and the
# story's monthly tick when the owner is no longer a landed Corporation ruler.
eotg_corporation_teardown_effect = {
    if = {
        limit = {
            any_owned_story = { story_type = eotg_story_corporation_board }
        }
        random_owned_story = {
            limit = { story_type = eotg_story_corporation_board }
            end_story = yes
        }
    }
    if = {
        limit = { has_character_modifier = eotg_corporation_board_modifier }
        remove_character_modifier = eotg_corporation_board_modifier
    }
}
