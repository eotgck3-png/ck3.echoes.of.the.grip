# -- eotg_government_succession_laws.txt --------------------------------------
# Governments v2, Phase G1 (docs/specs/governments_v2.md §7.1): one realm
# succession law per government, added to vanilla's succession_order_laws group
# by law_group_type (1.20 splits groups from laws), so the vanilla group file is
# not touched.
#
# G2 succession (SU1, docs/specs/governments_v2_g2_succession.md §6): three of
# the four laws are elective. The Corporation, Trade Oligarchy and Cartel laws
# keep their keys and use order_of_succession = election with their own
# election_type (common/succession_election/eotg_government_elections.txt);
# candidates come from the ruler's dynasty. The PMC law stays single heir
# (spec §11). The default-law notes below apply to all four unchanged.
#
# Why the mod needs them: every vanilla realm succession law is unavailable or
# unsafe here (partition needs feudal/clan, 00_law_triggers.txt:271-283; city
# succession needs republic :592-594; appointment/acclamation need the admin
# mechanic :644-679; single heir needs innovation_primogeniture :346-373 AND a
# feudal/clan/admin/japan_feudal government :374-411, which none of the four
# is, so vanilla single heir is never valid for them).
#
# Shape (PMC): single_heir_succession_law (G/common/laws/00_succession_laws.txt:249-347);
# the elective three: theocratic_elective_succession_law (:727-765).
# index 50-53: vanilla's succession_order_laws run 0-34; the mod has no other
# succession law. Each government's own law is its only valid realm succession
# law (spec §7.1 assumed vanilla single heir stayed available; it does not).
# The default-law pick (_laws.info:305-309) takes, in definition order, the
# first law whose should_start_with is true or absent. Partition is not the
# risk: its should_start_with needs government_is_feudal
# (00_succession_laws.txt:125-135). The real pre-emptors sit before index 50:
# single_heir_succession_law (index 3), whose should_start_with checks only the
# historical-access trigger and culture innovation_primogeniture (:289-296), and
# single_heir_succession_law_youngest (index 5), which has no should_start_with
# at all (:395+). common/on_action/eotg_government_on_actions.txt is the safety
# net (§7.2): on_game_start, on_government_change and on_title_gain (the last
# also covers tier-up inheritance, which copies the dead ruler's laws
# unchecked, _laws.info:302-303; assessment 2026-10-11 G1/G2).
# UNVERIFIED-VANILLA (V-G4): the engine skips a default-law candidate whose
#   can_have fails; if not, the on_game_start net corrects it.
# -----------------------------------------------------------------------------

eotg_pmc_succession_law = {
    law_group_type = succession_order_laws  # G/common/law_groups/00_succession_law_groups.txt:1
    index = 50

    potential = { government_has_flag = eotg_government_is_pmc }
    can_have = { government_has_flag = eotg_government_is_pmc }
    can_keep = { government_has_flag = eotg_government_is_pmc }
    can_pass = {
        can_change_succession_law_trigger = yes  # single_heir :270; 00_legal_triggers.txt:49
    }
    should_start_with = { government_has_flag = eotg_government_is_pmc }

    succession = {
        order_of_succession = inheritance
        traversal_order = children
        rank = oldest
        title_division = single_heir
    }
    flag = advanced_succession_law  # single_heir :303

    pass_cost = {
        prestige = change_succession_law_prestige_cost
    }
    revoke_cost = {
        prestige = change_succession_law_prestige_cost
    }

    ai_will_do = {
        value = 3  # single_heir :330-332
    }
}

    should_start_with = { government_has_flag = eotg_government_is_corporation }

    # Elective (G2 succession SU1, docs/specs/governments_v2_g2_succession.md §6).
    # Shape: theocratic_elective_succession_law (G/common/laws/00_succession_laws.txt:727-765),
    # a realm law in succession_order_laws using election. _laws.info:366-385
    # lists election as "Title only"; vanilla contradicts it, vanilla wins.
    # traversal_order / rank / title_division are invalid with election
    # (_laws.info:220-226) and are dropped. Fallback if V-SU1 fails: spec §3.3
    # (not built).
    can_title_have = { always = yes }  # :737-739; title-level resolution of a realm election
    succession = {
        order_of_succession = election  # :741
        election_type = eotg_corporation_elective  # :742; common/succession_election/eotg_government_elections.txt
    }
    flag = elective_succession_law  # :746 (was advanced_succession_law)
    title_allegiance_opinion = 10   # :747; _laws.info:140-144

    pass_cost = {
        prestige = change_succession_law_prestige_cost
    }
    revoke_cost = {
        prestige = change_succession_law_prestige_cost
    }

    ai_will_do = {
        value = 3  # single_heir :330-332
    }
}
