# ── eotg_market_faction_cycle.txt ─────────────────────────────────────────────
# Shared market faction story cycle for all corporate-government nations.
# Activated on powerful vassals whose liege holds a corporate government.
# Tracks: eotg_market_share, eotg_faction_role (0=independent, 1=member, 2=leader)
#
# Started via eotg_on_actions.txt → on_yearly_playable hook (create_story call).
# Each government type adds its own branch inside eotg_gobcorp_faction.0001.
# eotg_market_faction_active flag is set on the owner for external checks.
# ─────────────────────────────────────────────────────────────────────────────

eotg_market_faction_cycle = {
	on_setup = {
		story_owner = {
			set_variable = { name = eotg_market_share  value = 0 }
			set_variable = { name = eotg_faction_role  value = 0 }
			add_character_flag = eotg_market_faction_active
		}
	}

	on_end = {
		story_owner = {
			remove_variable = eotg_market_share
			remove_variable = eotg_faction_role
			remove_character_flag = eotg_market_faction_active
		}
	}

	on_owner_death = {
		end_story = yes
	}

	effect_group = {
		days = { 25 35 }

		first_valid = {
			# Terminate when conditions no longer apply
			triggered_effect = {
				trigger = {
					story_owner = {
						OR = {
							NOT = { is_alive = yes }
							NOT = { is_powerful_vassal = yes }
							NOT = {
								liege = {
									government_has_flag = eotg_government_is_corporate
								}
							}
						}
					}
				}
				effect = {
					end_story = yes
				}
			}

			# Monthly market faction pulse for corporate vassals
			triggered_effect = {
				trigger = { always = yes }
				effect = {
					story_owner = {
						trigger_event = { id = eotg_gobcorp_faction.0001 }
					}
				}
			}
		}
	}
}
