# Story cycles for the Cybernetic Augmentation system.
#
# eotg_story_aug_heir_arc — The Heir's Arc (thread T3,
#   docs/specs/cybernetics_v2_phase3.md §4). The dynasty watches the ruler
#   break. Shape copied from vanilla
#   game/common/story_cycles/story_cycle_murders_at_court.txt (on_setup
#   variables, on_owner_death, first_valid effect_group).
# Phase 5 (docs/specs/cybernetics_v2_phase5.md), further down:
#   eotg_story_aug_countdown — The Countdown (O-13)
#   eotg_story_aug_patron    — The Patron's debt (thread T2)
#   eotg_story_aug_retinue   — The Iron Retinue Program (thread T4)
#
# Created by: eotg_on_yearly_aug_neurofractured_check (step 2), once per
#   Neurofractured ruler with a primary heir aged 14+; and (balance spec
#   §5.8a) eotg_on_yearly_aug_overclocked_check, once the Countdown runs.
# Fires: eotg_aug_heir.001 (stage 0), eotg_fracture.005 (stage 1),
#   eotg_aug_heir.003 (stage 2), eotg_aug_heir.004 (stage 3, outside the
#   Flicker band). Stages 1-3 wait for a Neurofractured or Seamless owner, so
#   at Overclocked only Concern fires.
# Round 2 (docs/specs/cybernetics_v2_new_beats.md §5.1): heir.004 in round 1
#   parks the story at a dormant stage 5 instead of ending it; the dormant
#   entry below gives the next primary heir (14+, not the round-1 heir)
#   eotg_aug_heir.007 The Next in Line, then heir.003 and heir.004 again.
#   heir.004 in round 2 ends the story: never a third round. Vanilla shape
#   for a story that re-targets a successor instead of ending:
#   game/common/story_cycles/ep3_story_cycle_admin_eunuch.txt:145-182.
#
# Hidden (index §1 rule 3): visible left at its default (no), no
# visualization block, no counters. The story's effect_group timing is the
# arc's cooldown (index §1 rule 5).
#
# Story variables (prefixed per invariant 1; the spec calls them heir / stage /
# heir_dread):
#   eotg_heir        the watching heir (character)
#   eotg_stage       0-4; 5 = dormant after round 1's heir.004
#   eotg_heir_dread  how far the heir has come to fear the ruler; weights
#                    heir.004. Moved by heir.001, fracture.005, heir.003,
#                    fracture.026 (b) and heir.007.
#   eotg_round       1 or 2 (new beats §5.1)
#   eotg_prior_heir  the round-1 heir, copied from eotg_heir at the reprise
#   eotg_prior_choice  flag:ally / usurp / kill_failed / kill_success /
#                    executed: what round 1's heir.004 came to (heir.004,
#                    its kill_c)
# ─────────────────────────────────────────────────────────────────────────────

eotg_story_aug_heir_arc = {
    # Hidden: `visible` defaults to no (game/common/story_cycles/
    # _story_cycles.info). The line is left out on purpose: Tiger 1.17 does not
    # know the field (vanilla uses it, e.g. book_translation_story_cycle.txt)
    # and would report it as unknown. Never add visible = yes (index §1 rule 3).

    on_setup = {
        set_variable = {
            name = eotg_heir
            value = story_owner.primary_heir
        }
        set_variable = { name = eotg_stage  value = 0 }
        set_variable = { name = eotg_heir_dread  value = 0 }
        set_variable = { name = eotg_round  value = 1 }
        # Seeds only shift weights (index §4 T3). The five seeds live in
        # eotg_aug_heir_seed_dread_effect (new beats §3.1: moved verbatim, same
        # order and clamp) so the round-2 reprise below shares them.
        eotg_aug_heir_seed_dread_effect = yes
    }

    on_end = {
    }

    on_owner_death = {
        end_story = yes
    }

    effect_group = {
        years = { 1 2 }
        chance = 70

        first_valid = {
            # QA round 2: a ruler-only arc ends when its owner is deposed or
            # landless (end.030's abdication, a lost war). Vanilla precedent:
            # story_cycle_take_mandate_of_heaven.txt:172-189 (story_owner
            # is_landed = no -> end_story).
            # QA round 2 follow-up: the arc is spent (eotg_flag_aug_heir_arc_done,
            # read by the arc-start guards in the on_actions), so a ruler who
            # regains land does not restart it.
            triggered_effect = {
                trigger = {
                    story_owner = { is_landed = no }
                }
                effect = {
                    story_owner = { add_character_flag = eotg_flag_aug_heir_arc_done }
                    end_story = yes
                }
            }

            # 0. The owner left the system entirely (no augmentation trait):
            # nothing left to watch. Excision (eotg_aug_excision_effect) also
            # ends the story directly; Total Integration keeps it
            # (eotg_is_augmented_any includes eotg_total_integration).
            # Balance spec §5.8(a): an arc started at Overclocked also ends
            # when the owner regresses to Augmented or Enhanced. No done flag,
            # so a later climb can start it again.
            triggered_effect = {
                trigger = {
                    story_owner = {
                        OR = {
                            eotg_is_augmented_any = no
                            eotg_is_aug_tier1 = yes
                            eotg_is_aug_tier2 = yes
                        }
                    }
                }
                effect = {
                    end_story = yes
                }
            }

            # 0b. Dormant (stage 5): round 1's heir.004 has been answered
            # (new beats §5.1 step 2). Matches every dormant tick, so a dormant
            # story never falls through to branch 1. When the primary heir is
            # someone else (the round-1 heir died, was executed, or no longer
            # stands first) and is 14+, round 2 starts: the new heir is seeded
            # like round 1, plus what they watched happen to the last one, and
            # goes straight to the Warning (stage 2) after heir.007. An heir
            # under 14 is waited for. Vanilla precedent: the admin eunuch story
            # re-targets a successor in its tick
            # (game/common/story_cycles/ep3_story_cycle_admin_eunuch.txt:145-182;
            # events/dlc/ep3/ep3_story_cycle_admin_eunuch_events.txt:6021-6097).
            triggered_effect = {
                trigger = { var:eotg_stage = 5 }
                effect = {
                    if = {
                        limit = {
                            story_owner = {
                                primary_heir ?= {
                                    is_alive = yes
                                    age >= 14
                                    # QA S2: round 2 always has a predecessor
                                    # to name (heir.007, heir.003 successor)
                                    exists = scope:story.var:eotg_heir
                                    NOT = { this = scope:story.var:eotg_heir }
                                }
                            }
                        }
                        set_variable = { name = eotg_prior_heir  value = var:eotg_heir }
                        set_variable = {
                            name = eotg_heir
                            value = story_owner.primary_heir
                        }
                        set_variable = { name = eotg_round  value = 2 }
                        set_variable = { name = eotg_heir_dread  value = 0 }
                        eotg_aug_heir_seed_dread_effect = yes
                        # What the new heir watched happen to the last one
                        if = {
                            limit = { var:eotg_prior_choice ?= flag:executed }
                            change_variable = { name = eotg_heir_dread  add = 2 }
                        }
                        else_if = {
                            limit = {
                                OR = {
                                    var:eotg_prior_choice ?= flag:usurp
                                    var:eotg_prior_choice ?= flag:kill_failed
                                }
                            }
                            change_variable = { name = eotg_heir_dread  add = 1 }
                        }
                        clamp_variable = { name = eotg_heir_dread  min = 0  max = 10 }
                        set_variable = { name = eotg_stage  value = 2 }
                        story_owner = {
                            # That ally is gone (eotg_fracture.027 reads it)
                            if = {
                                limit = { has_character_flag = eotg_flag_aug_heir_ally }
                                remove_character_flag = eotg_flag_aug_heir_ally
                            }
                            trigger_event = { id = eotg_aug_heir.007  days = { 1 30 } }
                        }
                    }
                }
            }

            # 1. The heir died or is no longer the primary heir: pass the arc
            # to a new heir of 14+ one stage back, else end it. Never while
            # dormant (new beats §5.1 step 3; 0b already shadows it). Inside
            # round 2 this still hands the arc to a third heir one stage back:
            # a hand-off within the round, not a third round.
            triggered_effect = {
                trigger = {
                    var:eotg_stage < 5
                    OR = {
                        NOT = { exists = var:eotg_heir }
                        var:eotg_heir = { is_alive = no }
                        NOT = {
                            story_owner = { primary_heir ?= scope:story.var:eotg_heir }
                        }
                    }
                }
                effect = {
                    if = {
                        limit = {
                            story_owner = {
                                primary_heir ?= {
                                    is_alive = yes
                                    age >= 14
                                }
                            }
                        }
                        set_variable = {
                            name = eotg_heir
                            value = story_owner.primary_heir
                        }
                        change_variable = { name = eotg_stage  add = -1 }
                        clamp_variable = { name = eotg_stage  min = 0  max = 4 }
                    }
                    else = {
                        end_story = yes
                    }
                }
            }

            # 2. Concern
            triggered_effect = {
                trigger = { var:eotg_stage = 0 }
                effect = {
                    set_variable = { name = eotg_stage  value = 1 }
                    story_owner = {
                        trigger_event = { id = eotg_aug_heir.001  days = { 1 30 } }
                    }
                }
            }

            # 3. What the Heir Saw (existing event, re-homed from the random pool)
            # Balance spec §5.8(a): stages 3-5 wait for the break
            # (Neurofractured or Seamless); at Overclocked the arc holds here.
            triggered_effect = {
                trigger = {
                    var:eotg_stage = 1
                    story_owner = {
                        OR = {
                            has_trait = eotg_neurofractured
                            has_trait = eotg_total_integration
                        }
                    }
                }
                effect = {
                    set_variable = { name = eotg_stage  value = 2 }
                    story_owner = {
                        trigger_event = { id = eotg_fracture.005  days = { 1 30 } }
                    }
                }
            }

            # 4. The Heir's Warning
            triggered_effect = {
                trigger = {
                    var:eotg_stage = 2
                    story_owner = {
                        OR = {
                            has_trait = eotg_neurofractured
                            has_trait = eotg_total_integration
                        }
                    }
                }
                effect = {
                    set_variable = { name = eotg_stage  value = 3 }
                    story_owner = {
                        trigger_event = { id = eotg_aug_heir.003  days = { 1 30 } }
                    }
                }
            }

            # 5. The Heir's Choice — not while the owner is in the Flicker band
            # (the heir acts when the ruler is visibly breaking). A Seamless
            # owner has no eotg_fracture_risk (reads as Flicker), and
            # eotg_aug_total_integration_effect sets stage 3: the heir must
            # decide about the machine, so Seamless always qualifies.
            triggered_effect = {
                trigger = {
                    var:eotg_stage = 3
                    story_owner = {
                        OR = {
                            has_trait = eotg_neurofractured
                            has_trait = eotg_total_integration
                        }
                    }
                    story_owner = {
                        OR = {
                            has_trait = eotg_total_integration
                            eotg_aug_pressure_flicker = no
                        }
                    }
                }
                effect = {
                    set_variable = { name = eotg_stage  value = 4 }
                    story_owner = {
                        trigger_event = { id = eotg_aug_heir.004  days = { 1 30 } }
                    }
                }
            }
        }
    }
}

# ── eotg_story_aug_countdown — The Countdown (O-13, §1) ──────────────────────
# Created by: eotg_aug_try_start_countdown_effect (OC on_action at risk >= 50,
#   or >= 40 with eotg_flag_aug_hidden_flaw; eotg_aug_patron_betray_effect).
# Ended by: eotg_aug_end_countdown_effect (the cascade, full removal, The
#   Quiet), the first two branches below, or the owner's death.
# Fires: eotg_aug_countdown.001-.006 and eotg_aug_tier3.020 (stage 5).
# Stages are chosen BY eotg_fracture_risk, never shown as a number. Each fires
# at most once (eotg_last_stage); stages can skip.
# Story variable: eotg_last_stage 0-5 (spec: last_stage).
eotg_story_aug_countdown = {
    on_setup = {
        set_variable = { name = eotg_last_stage  value = 0 }
    }

    on_end = {
    }

    on_owner_death = {
        end_story = yes
    }

    effect_group = {
        months = { 3 5 }

        first_valid = {
            # 1. No longer Overclocked (regressed, removed): The Quiet
            triggered_effect = {
                trigger = {
                    story_owner = { eotg_is_aug_tier3 = no }
                }
                effect = {
                    story_owner = {
                        trigger_event = { id = eotg_aug_countdown.006  days = { 1 10 } }
                    }
                    end_story = yes
                }
            }
            # 2. Maintenance got ahead of it: The Quiet
            triggered_effect = {
                trigger = {
                    story_owner = {
                        OR = {
                            NOT = { has_variable = eotg_fracture_risk }
                            var:eotg_fracture_risk < 35
                        }
                    }
                }
                effect = {
                    story_owner = {
                        trigger_event = { id = eotg_aug_countdown.006  days = { 1 10 } }
                    }
                    end_story = yes
                }
            }
            # 3. Minor Anomalies (risk 50-59)
            triggered_effect = {
                trigger = {
                    var:eotg_last_stage < 1
                    story_owner = {
                        var:eotg_fracture_risk >= 50
                        var:eotg_fracture_risk < 60
                    }
                }
                effect = {
                    set_variable = { name = eotg_last_stage  value = 1 }
                    story_owner = {
                        trigger_event = { id = eotg_aug_countdown.001  days = { 1 20 } }
                    }
                }
            }
            # Balance spec §5.10(b): stage windows 60-67 / 68-76 / >= 77, so
            # calm accrual (12-15 a year) lands in stage 3's window from more
            # start points. .001 stays 50-59; eotg_last_stage is unchanged.
            # 4. Lost Time (60-67)
            triggered_effect = {
                trigger = {
                    var:eotg_last_stage < 2
                    story_owner = {
                        var:eotg_fracture_risk >= 60
                        var:eotg_fracture_risk < 68
                    }
                }
                effect = {
                    set_variable = { name = eotg_last_stage  value = 2 }
                    story_owner = {
                        trigger_event = { id = eotg_aug_countdown.002  days = { 1 20 } }
                    }
                }
            }
            # 5. Contradictory Memories (68-76)
            triggered_effect = {
                trigger = {
                    var:eotg_last_stage < 3
                    story_owner = {
                        var:eotg_fracture_risk >= 68
                        var:eotg_fracture_risk < 77
                    }
                }
                effect = {
                    set_variable = { name = eotg_last_stage  value = 3 }
                    story_owner = {
                        trigger_event = { id = eotg_aug_countdown.003  days = { 1 20 } }
                    }
                }
            }
            # 6. Violence (77+)
            triggered_effect = {
                trigger = {
                    var:eotg_last_stage < 4
                    story_owner = { var:eotg_fracture_risk >= 77 }
                }
                effect = {
                    set_variable = { name = eotg_last_stage  value = 4 }
                    story_owner = {
                        trigger_event = { id = eotg_aug_countdown.004  days = { 1 20 } }
                    }
                }
            }
            # 7a. After Violence, having refused the Intervention: No One Asks
            triggered_effect = {
                trigger = {
                    var:eotg_last_stage = 4
                    story_owner = { has_character_flag = eotg_flag_aug_intervention_refused }
                }
                effect = {
                    set_variable = { name = eotg_last_stage  value = 5 }
                    story_owner = {
                        trigger_event = { id = eotg_aug_countdown.005  days = { 1 20 } }
                    }
                }
            }
            # 7b. After Violence: someone intervenes (Phase 4c tier3.020)
            triggered_effect = {
                trigger = {
                    var:eotg_last_stage = 4
                    story_owner = {
                        OR = {
                            exists = primary_spouse
                            primary_heir ?= { is_adult = yes }
                        }
                    }
                }
                effect = {
                    set_variable = { name = eotg_last_stage  value = 5 }
                    story_owner = {
                        trigger_event = { id = eotg_aug_tier3.020  days = { 1 20 } }
                    }
                }
            }
            # 8. Otherwise nothing: the dread is in the silence.
        }
    }
}


# ── eotg_story_aug_patron — The Patron's debt (I-05, thread T2, §2) ──────────
# Created by: eotg_aug_patron_accept_effect (patron.001 a, c, d).
# Fires: eotg_aug_patron.002-.009. Ended by: patron.006 (every option,
#   unless the paper was sold), patron.008 (a, b, e success), patron.009
#   (every option), or the owner's death (inherited debt is deferred, §5).
# The syndicate is never named (index §5.7): "the syndicate", "the syndicate
#   envoy".
# Story variables (spec: envoy / demand / grievance / terms):
#   eotg_envoy      the syndicate envoy at court (character)
#   eotg_demand     demands made, 0-4
#   eotg_grievance  refusals; >= 3 brings the Final Demand early (betrayal tone)
#   eotg_terms      flag:standard / flag:read / flag:greedy (patron.001)
#   eotg_paper_served  bare, one-shot: set by the tick when it fires
#                   patron.008 The Paper (new beats §5.2)
#   eotg_syndicate  the Patron's syndicate (flag), copied from the owner's
#                   eotg_patron_syndicate_passthrough in on_setup (CB-42)
#   eotg_paper_sold    bare, one-shot: set by eotg_aug_patron_betray_effect
#                   when the owner has no implants (reprisal spec §5.2); the
#                   story parks after .006 and the tick fires patron.009
#                   The Collector
eotg_story_aug_patron = {
    on_setup = {
        set_variable = {
            name = eotg_envoy
            value = story_owner.var:eotg_patron_envoy_passthrough
        }
        set_variable = {
            name = eotg_terms
            value = story_owner.var:eotg_patron_terms_passthrough
        }
        # CB-42 (seller names §5.4): the syndicate named at the offer, kept for
        # the whole story (vanilla var-to-var copy, 00_animal_effects.txt:395-398)
        if = {
            limit = { exists = story_owner.var:eotg_patron_syndicate_passthrough }
            set_variable = {
                name = eotg_syndicate
                value = story_owner.var:eotg_patron_syndicate_passthrough
            }
        }
        set_variable = { name = eotg_demand  value = 0 }
        set_variable = { name = eotg_grievance  value = 0 }
        story_owner = {
            remove_variable = eotg_patron_envoy_passthrough
            remove_variable = eotg_patron_terms_passthrough
            remove_variable = eotg_patron_syndicate_passthrough    # CB-42
            # New beats §5.2 "Once per life": permanent; read by the offer
            # sites only (the initiation check's Patron entry, init.018 desc
            # and e). Vanilla precedent: story_cycle_murders_at_court.txt:9.
            add_character_flag = eotg_flag_aug_had_patron
        }
    }

    # The open-ended exclusivity clause lasts as long as the debt
    # (patron.003 a/d). The final clause (patron.006 a) is a separate key.
    # QA round 2 M12: the envoy leaves with the debt. A living envoy who is
    # free and holds no land goes to the pool (move_to_pool, vanilla
    # events/activities/hold_court_activity/hold_court_events_general.txt:508),
    # so each patron.007 replacement does not leave a permanent courtier.
    # Not tied to the owner's court: on the owner's death the envoy would
    # otherwise pass to the heir's.
    on_end = {
        story_owner = {
            remove_character_modifier = eotg_mod_aug_patron_clause
        }
        if = {
            limit = {
                exists = var:eotg_envoy
                var:eotg_envoy = {
                    is_alive = yes
                    is_landed = no
                    is_imprisoned = no
                }
            }
            var:eotg_envoy = { move_to_pool = yes }
        }
    }

    on_owner_death = {
        end_story = yes
    }

    # Two cadences: read terms are slower (phase5 §2.3)
    # Balance spec §5.8: both ticks open (after the landless ending) with the
    # Neurofractured write-off.
    # New beats §5.2: then the paper (patron.008), once per story, for an
    # owner with no implants while the envoy is at court and the Final Demand
    # is not yet due.
    # The tick (inlined in both cadences so the event graph can follow it;
    # keep the two copies identical). Order, phase5 §2.3 with 1 and 2
    # swapped (deviation): grievance >= 3 or demand >= 4 -> .006 first, so
    # refusing every new envoy (.007 c) ends in the Final Demand; then the
    # envoy dead or gone -> .007; then by demand 0-3 -> .002-.005. The
    # Errand (.004) needs someone to name; with no one the syndicate moves on
    # to the Family Clause (demand 3, .005) instead of stalling.
    effect_group = {
        years = { 2 3 }
        trigger = { NOT = { var:eotg_terms = flag:read } }
        first_valid = {
            # QA round 2: a ruler-only arc ends when its owner is deposed or
            # landless (end.030's abdication, a lost war). Vanilla precedent:
            # story_cycle_take_mandate_of_heaven.txt:172-189 (story_owner
            # is_landed = no -> end_story).
            triggered_effect = {
                trigger = {
                    story_owner = { is_landed = no }
                }
                effect = {
                    end_story = yes
                }
            }
            # Reprisal spec §5.3: the paper has been sold (a betrayal by an
            # owner with no implants, eotg_aug_patron_betray_effect). The
            # collector comes in person. Second, after the landless ending,
            # which must still win; it shadows every later entry, so a parked
            # story never re-fires .006 (grievance >= 3 or demand >= 4 still
            # hold) or .007 (the envoy is dead). The tick is the cooldown
            # authority (index §1 rule 5); patron.009 checks no flag. Every
            # .009 option ends the story. Vanilla shape:
            # story_cycle_pledge_loyalty_to_liege_overdue.txt:188-207.
            triggered_effect = {
                trigger = { has_variable = eotg_paper_sold }
                effect = {
                    story_owner = { trigger_event = { id = eotg_aug_patron.009  days = { 1 30 } } }
                }
            }
            # Balance spec §5.8: the syndicate writes off a Neurofractured
            # owner. The Final Demand comes now (demand to 4); the timed
            # owner flag eotg_flag_aug_patron_writeoff (60 days) gives
            # patron.006 its write-off desc. After the landless ending, which
            # must still win.
            triggered_effect = {
                trigger = {
                    story_owner = { has_trait = eotg_neurofractured }
                    var:eotg_demand < 4
                }
                effect = {
                    set_variable = { name = eotg_demand  value = 4 }
                    story_owner = {
                        add_character_flag = { flag = eotg_flag_aug_patron_writeoff  days = 60 }
                        trigger_event = { id = eotg_aug_patron.006  days = { 1 30 } }
                    }
                }
            }
            # New beats §5.2: The Paper. The owner has no implants and the debt
            # is still live: the syndicate re-prices it, once per story. The
            # tick is the cooldown authority (index §1 rule 5): the one-shot
            # story variable is set here and patron.008 checks no flag.
            # Yields to .006 when the Final Demand is due, and to .007 when
            # the envoy is dead or gone (the paper then comes the next tick).
