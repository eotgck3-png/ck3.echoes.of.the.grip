# -- eotg_cartel_story.txt ----------------------------------------------------
# eotg_story_cartel_schedule: "The Payment Schedule" (docs/specs/
# governments_v2_g2_cartel.md §2.7 layer 4, §3.2). Batch C1.
#
# Created by: eotg_cartel_init_effect (on_actions in eotg_cartel_on_actions.txt).
# Ended by: eotg_cartel_teardown_effect (government change away from Cartel, or
#   this story's monthly tick when the owner is no longer a landed Cartel
#   ruler), or on_owner_death (the heir gets a fresh story through
#   eotg_cartel_on_title_gain).
# Fires: no event (spec §11: E1-E5 are not approved). Calls
#   eotg_cartel_walk_off_effect from the yearly effect_group (batch C2).
#
# VISIBLE (OQ3, owner 2026-10-20): the mod's second visible story, after the
# kingpin. It shows a basic_counter of dread mirrored on the story
# (eotg_cartel_dread_shown, written by eotg_cartel_sync_story_effect from the
# change effect and the monthly tick, so it can lag vanilla dread changes by
# up to a month; the tooltip says so) and the schedule modifier.
# Shape: G/common/story_cycles/_story_cycles.info (visible, icon, background,
# visualization, basic_counter :92-100, effect_group :119-170) and
# book_translation_story_cycle.txt:1-34. Icon and background are vanilla files
# as in eotg_aug_kingpin_story.txt (no art debt). Tiger 1.17 flags
# visible/icon/background/visualization as unknown fields: benign (CLAUDE.md).
#
# Story variables: eotg_cartel_dread_shown (display only; never read as the
# resource).
# Two ticks: monthly (C1, self-heal and refresh) and yearly (C2, the dread
# walk-off; spec §3.2, the walk-off's single cadence authority).
# -----------------------------------------------------------------------------

eotg_story_cartel_schedule = {
    visible = yes

    icon = {
        reference = "gfx/interface/icons/story_cycles/story_icon_palace_politics.dds"
    }
    background = {
        reference = "gfx/interface/illustrations/event_scenes/alley.dds"
    }

    visualization = {
        custom_string_key = "eotg_story_cartel_schedule_info"
        basic_counter = {
            variable_name = "eotg_cartel_dread_shown"
            min = 0
            max = 100
            label = "eotg_story_cartel_schedule_dread_label"
            min_label = "eotg_story_cartel_schedule_min_label"
            max_label = "eotg_story_cartel_schedule_max_label"
            tooltip = "eotg_story_cartel_schedule_tt"
        }
        modifiers = { eotg_cartel_schedule_modifier }
    }

    # Seed the counter. Written directly (not via eotg_cartel_sync_story_effect)
    # because that effect finds the story through random_owned_story, which
    # may not see a story still in setup; init calls the sync effect right
    # after create_story anyway.
    on_setup = {
        set_variable = {
            name = eotg_cartel_dread_shown
            value = story_owner.dread
        }
    }

    # Idempotent with eotg_cartel_teardown_effect
    on_end = {
        story_owner = {
            if = {
                limit = { has_character_modifier = eotg_cartel_schedule_modifier }
                remove_character_modifier = eotg_cartel_schedule_modifier
            }
            if = {
                limit = { has_variable = eotg_cartel_schedule_scale }
                remove_variable = eotg_cartel_schedule_scale
            }
        }
    }

    on_owner_death = {
        end_story = yes
    }

    # Monthly tick (spec §3.2): self-heal or refresh. No cooldown; no event.
    effect_group = {
        months = 1
        first_valid = {
            # (a) The owner is no longer a landed Cartel ruler: tear down
            triggered_effect = {
                trigger = {
                    story_owner = {
                        NOT = { eotg_cartel_is_landed_cartel = yes }
                    }
                }
                effect = {
                    story_owner = { eotg_cartel_teardown_effect = yes }
                }
            }
            # (b) Otherwise: refresh the baseline, then the counter
            triggered_effect = {
                trigger = {
                    always = yes
                }
                effect = {
                    story_owner = {
                        eotg_cartel_refresh_schedule_effect = yes
                        eotg_cartel_sync_story_effect = yes
                    }
                }
            }
        }
    }

    # Yearly tick (spec §3.2, §5.4): the dread walk-off. THIS is the
    # walk-off's only cadence authority (invariant 4, v1 lesson 5): the
    # effect itself checks nothing. At most one payer leaves a year, and
    # none while the heir's succession grace (eotg_cartel_on_death, 3 years)
    # runs. low_dread = 20 (G/common/script_values/00_basic_values.txt:993).
    effect_group = {
        years = 1
        trigger = {
            story_owner = {
                eotg_cartel_role_top = yes
                eotg_cartel_payer_count_value > 0
                dread < low_dread
                NOT = { has_variable = eotg_cartel_succession_grace }
            }
        }
        triggered_effect = {
            trigger = { always = yes }
            effect = {
                story_owner = { eotg_cartel_walk_off_effect = yes }
            }
        }
    }
}
