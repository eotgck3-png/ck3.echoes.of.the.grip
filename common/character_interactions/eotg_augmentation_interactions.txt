# ── eotg_augmentation_interactions.txt ───────────────────────────────────────
# Character interactions for the Cybernetic Augmentation system
# (docs/specs/cybernetics_v2_interactions.md §4).
#
# eotg_aug_offer_augmentation_interaction — Offer Augmentation (G4)
# eotg_aug_demand_removal_interaction     — Demand Implant Removal (G4)
# eotg_aug_examine_interaction            — Have Them Examined (G4) -> eotg_aug_int.001
# eotg_aug_start_tamper_interaction       — Tamper with Implants (G5): starts
#                                           the scheme eotg_aug_tamper
# eotg_aug_salvage_interaction            — Salvage Implants (G8) -> eotg_aug_int.002
#
# Shared rules (spec §4.0):
#   - scope:actor / scope:recipient, the shape of
#     ask_for_conversion_courtier_interaction (vanilla
#     00_religious_interactions.txt:158-505). Puppet actors
#     (scope:puppet_or_actor) are not supported (spec §10).
#   - Every procedure roll runs in on_accept (the click), inside hidden_effect,
#     followed by a custom_tooltip. No tooltip quantifies odds.
#   - The actor pays in on_accept (eotg_aug_pay_procedure_effect), never with
#     `cost` (charged on send, _character_interactions.info:484-487, so a
#     player's refusal would still cost the actor).
#   - Cooldowns: cooldown_against_recipient only (index §1 rule 5).
#   - Provider send options (Offer, Demand). The spec asks for exclusive
#     provider options AND a hook send option. send_options_exclusive is
#     interaction-wide (_character_interactions.info:372), so with it the hook
#     would deselect the provider. These two interactions are therefore
#     non-exclusive (as the hook + influence pair in vanilla
#     ask_for_conversion_courtier_interaction), and
#     is_valid_showing_failures_only requires exactly one provider
#     (eotg_aug_send_one_provider_tt); ai_will_do zeroes every other
#     combination. Salvage has no hook and stays exclusive.
#   - "My own physician" in an interaction is the ACTOR's surgeon operating on
#     someone else (eotg_aug_save_surgeon_effect in scope:actor), so a
#     lifestyle_physician actor qualifies (spec §4.0). It is invalid when the
#     recipient is the actor's own court physician, who would then operate on
#     themself at the physician's price without the self penalty (the
#     exploit the procedures QA closed with eotg_aug_has_court_physician).
# ─────────────────────────────────────────────────────────────────────────────


# ── Offer Augmentation (spec §4.1) ────────────────────────────────────────────
# A gift: the actor pays for the recipient's first implants and chooses the
# hands. eotg_aug_offer_install_effect mirrors init.018's three paths.
eotg_aug_offer_augmentation_interaction = {
    category = interaction_category_friendly
    icon = icon_personal          # placeholder (spec §3.1, art debt)
    interface_priority = 40
    desc = eotg_aug_offer_augmentation_interaction_desc
    notification_text = eotg_aug_offer_augmentation_interaction_notification

    # Actor-only state belongs here (_character_interactions.info:570-573);
    # the AI half as vanilla 00_artifact_interactions.txt:2828-2847
    is_available = {
        is_adult = yes
        is_imprisoned = no
        trigger_if = {
            limit = { is_ai = yes }
            OR = {
                eotg_is_augmented_any = yes
                has_trait = ambitious
                has_trait = cynical
            }
            NOT = { has_trait = zealous }
        }
    }

    is_shown = {
        NOT = { scope:recipient = scope:actor }
        scope:recipient = {
            eotg_aug_in_charge_of = { ACTOR = scope:actor }
            eotg_is_augmented_any = no
            is_imprisoned = no
        }
    }

    is_valid_showing_failures_only = {
        scope:recipient = { eotg_aug_offer_candidate = yes }
        custom_description = {
            text = eotg_aug_send_one_provider_tt
            # Exactly one: an explicit comparator pair (vanilla never writes
            # a bare `amount =`)
            calc_true_if = {
                amount >= 1
                scope:eotg_aug_provider_clinic ?= yes
                scope:eotg_aug_provider_physician ?= yes
                scope:eotg_aug_provider_backstreet ?= yes
            }
            calc_true_if = {
                amount <= 1
                scope:eotg_aug_provider_clinic ?= yes
                scope:eotg_aug_provider_physician ?= yes
                scope:eotg_aug_provider_backstreet ?= yes
            }
        }
        # The price by provider (spec §4.1 step 1; mirror:
        # eotg_aug_offer_install_effect)
        trigger_if = {
            limit = { scope:eotg_aug_provider_clinic ?= yes }
            scope:actor = { eotg_aug_can_afford_procedure = { BASE = minor_gold_value  MULT = 1 } }
        }
        trigger_if = {
            limit = { scope:eotg_aug_provider_physician ?= yes }
            scope:actor = { eotg_aug_can_afford_procedure = { BASE = minor_gold_value  MULT = eotg_aug_price_mult_physician } }

            limit = { scope:eotg_aug_provider_backstreet ?= yes }
            scope:actor = { eotg_aug_can_afford_procedure = { BASE = tiny_gold_value  MULT = 1 } }

    cooldown_against_recipient = { years = 5 }

    ai_min_reply_days = 1
    ai_max_reply_days = 5

    # Strong hook: auto-accept (vanilla 00_religious_interactions.txt:236-245)

        custom_description = {
            text = "spending_hook"
            subject = scope:actor
            object = scope:recipient
            scope:hook = yes
            scope:actor = { has_strong_hook = scope:recipient }
        }
    }

    options_heading = eotg_aug_send_provider_heading
    send_option = {
        flag = eotg_aug_provider_clinic
        localization = eotg_aug_send_clinic
        current_description = eotg_aug_send_clinic_tt
        # Matches is_valid: not preselected where the recipient's realm bans it
        starts_enabled = { scope:recipient = { eotg_aug_clinic_open = yes } }
        # Realm spec §3.2: the patient's realm decides (LAW AT 866)
        is_valid = {
            custom_description = {
                text = eotg_aug_clinic_closed_tt
                scope:recipient = { eotg_aug_clinic_open = yes }
            }
        }
    }
    send_option = {
        flag = eotg_aug_provider_physician
        localization = eotg_aug_send_physician
        current_description = eotg_aug_send_physician_tt
        # One requirement line (Tiger: an OR is too complex to tooltip in a
        # failures-only trigger). Realm spec §3.2: any surgeon of the actor's
        # (physician or Implant Technician, own or borrowed) who is not the
        # recipient.
        is_valid = {
            custom_description = {
                text = eotg_aug_send_physician_valid_tt

            }
        }
    }
    send_option = {
        flag = eotg_aug_provider_backstreet
        localization = eotg_aug_send_backstreet
        current_description = eotg_aug_send_backstreet_tt
    }
    # Use hook (vanilla 00_religious_interactions.txt:362-378)
    send_option = {
        is_valid = {
            scope:actor = { has_usable_hook = scope:recipient }
        }
        flag = hook
        localization = SCHEME_HOOK
    }
    should_use_extra_icon = {
        scope:actor = { has_usable_hook = scope:recipient }
    }
    extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"
    send_options_exclusive = no

    on_accept = {
        if = {
            limit = { scope:eotg_aug_provider_physician ?= yes }
            eotg_aug_offer_install_effect = { PROVIDER = physician }
        }
        else_if = {
            limit = { scope:eotg_aug_provider_backstreet ?= yes }
            eotg_aug_offer_install_effect = { PROVIDER = backstreet }
        }
        else = {
            eotg_aug_offer_install_effect = { PROVIDER = clinic }
        }

            limit = { scope:hook ?= yes }
            scope:actor = {

                    limit = { has_usable_hook = scope:recipient }
                    use_hook = scope:recipient
                }
            }
        }
        scope:actor = {
            send_interface_toast = {
                type = event_toast_effect_good
                title = eotg_aug_offer_accepted_toast
                left_icon = scope:recipient
            }
        }
    }

    # No opinion penalty: it was a gift (spec §4.1)
    on_decline = {
        scope:actor = {
            send_interface_toast = {
                type = event_toast_effect_neutral
                title = eotg_aug_offer_declined_toast
                left_icon = scope:recipient
            }
        }
    }

    # Placeholder (the spec's `learning` has no file; as Examine)
    icon = plague
    desc = eotg_aug_borrow_technician_interaction_desc
    notification_text = eotg_aug_borrow_technician_interaction_notification

    is_shown = {
        NOT = { scope:recipient = scope:actor }
        scope:recipient = { employs_court_position = eotg_aug_implant_technician_court_position }
        scope:actor = {
            eotg_is_augmented_any = yes
            NOT = { employs_court_position = eotg_aug_implant_technician_court_position }
        }
        OR = {
            scope:actor = { is_allied_to = scope:recipient }
            scope:actor = { is_vassal_of = scope:recipient }
            scope:recipient = { is_vassal_of = scope:actor }
            scope:actor = { is_close_family_of = scope:recipient }
            scope:actor = { is_spouse_of = scope:recipient }
            scope:actor = { has_relation_friend = scope:recipient }
        }
    }

    is_valid_showing_failures_only = {
        scope:actor = {
            custom_description = {
                text = eotg_aug_borrow_none_held_tt
                eotg_aug_borrowed_technician_valid = no
            }
            short_term_gold >= minor_gold_value
            NOT = { is_at_war_with = scope:recipient }
            # The work would happen in the actor's realm
            custom_description = {
                text = eotg_aug_clinic_closed_tt
                eotg_aug_clinic_open = yes
            }
        }
        scope:recipient = {
            court_position:eotg_aug_implant_technician_court_position ?= {
                is_imprisoned = no
                NOT = { has_trait = blind }
            }
        }
    }

    cooldown_against_recipient = { years = 3 }

    ai_min_reply_days = 1
    ai_max_reply_days = 5

    on_accept = {
        scope:actor = {
            pay_short_term_gold = {
                target = scope:recipient
                gold = minor_gold_value
            }
        }
        scope:recipient.court_position:eotg_aug_implant_technician_court_position = {
            save_scope_as = eotg_lent_technician
        }
        scope:actor = {
            set_variable = {
                name = eotg_aug_borrowed_technician
                value = scope:eotg_lent_technician
                years = 1
            }
            send_interface_toast = {
                type = event_toast_effect_good
                title = eotg_aug_borrow_accepted_toast
                left_icon = scope:recipient
                right_icon = scope:eotg_lent_technician
                custom_tooltip = eotg_aug_borrow_accepted_toast_desc
            }
        }
    }

    # No opinion: it was a request
    on_decline = {
        scope:actor = {
            send_interface_toast = {
                type = event_toast_effect_neutral
                title = eotg_aug_borrow_declined_toast
                left_icon = scope:recipient
            }
        }
    }

    # The lender's view (spec §4.6 table)
    ai_accept = {
        base = 10
        opinion_modifier = {
            who = scope:recipient
            opinion_target = scope:actor
            multiplier = 0.5
            desc = AI_OPINION_REASON
        }
        modifier = {
            add = 20
            desc = EOTG_AUG_AI_ALLIED
            scope:recipient = { is_allied_to = scope:actor }
        }
        modifier = {
            add = 25
            desc = EOTG_AUG_AI_LIEGE
            scope:recipient = { is_vassal_of = scope:actor }
        }
        modifier = {
            add = 15
            desc = EOTG_AUG_AI_FAMILY
            scope:recipient = {
                OR = {
                    is_close_family_of = scope:actor
                    is_spouse_of = scope:actor
                }
            }
        }
        modifier = {
            add = 15
            desc = EOTG_AUG_AI_GREEDY
            scope:recipient = { has_trait = greedy }
        }
        modifier = {
            add = 10
            desc = EOTG_AUG_AI_GENEROUS
            scope:recipient = { has_trait = generous }
        }
        modifier = {
            add = -20
            desc = EOTG_AUG_AI_PARANOID
            scope:recipient = { has_trait = paranoid }
        }
        modifier = {
            add = -30
            desc = EOTG_AUG_AI_NEEDS_TECHNICIAN
            scope:recipient = {
                OR = {
                    eotg_is_aug_tier3 = yes
                    has_trait = eotg_neurofractured
                }
            }
        }
    }

    ai_target_quick_trigger = {
        adult = yes
    }
    ai_targets = {
        ai_recipients = war_allies
    }
    ai_targets = {
        ai_recipients = family
        max = 5
    }
    ai_targets = {
        ai_recipients = liege
    }
    ai_targets = {
        ai_recipients = vassals
        max = 10
    }
    ai_frequency_by_tier = {
        barony = 0
        county = 0
        duchy = 48
        kingdom = 36
        empire = 36
        hegemony = 36
    }
    ai_will_do = {
        base = 0
        modifier = {
            add = 40
            scope:actor = {
                OR = {
                    eotg_is_aug_tier3 = yes
                    has_trait = eotg_neurofractured
                }
            }
        }
        modifier = {
            add = 20
            scope:actor = {
                OR = {
                    has_character_modifier = eotg_mod_aug_infection
                    has_character_modifier = eotg_mod_aug_fragments
                    has_character_modifier = eotg_mod_aug_tampered
                }
            }
        }
        modifier = {
            add = 20
            scope:actor = { has_variable = eotg_aug_repair_injury }
        }
        modifier = {
            factor = 0.5
            scope:actor = { eotg_has_physician_access = yes }
        }
        modifier = {
            factor = 0
            scope:actor = { short_term_gold < minor_gold_value }
        }
    }
}
