# -- eotg_unclaimed_interactions.txt ------------------------------------------
# Unclaimed Regions (docs/specs/frontier_unclaimed_regions.md §8).
#
# eotg_unclaimed_claim_interaction: "Raise Your Colours". The recipient is an
# Unsworn placeholder; the target is its one county
# (scope:recipient.capital_county). Precedent: vanilla
# mpo_retrieve_land_from_herder_interaction (09_mpo_interactions.txt:7426),
# vanilla's own "take land from someone who holds it without ruling it":
# neighbour reach, a prestige cost, auto_accept, AI neighboring_rulers.
# Deviations (spec §8.1): our cost, our cooldown, our AI throttle.
# frontier_v4.md A7.3: the Raise Your Colours decision
# (eotg_unclaimed_decisions.txt) is the second path to the same claim. Both
# share one cooldown, the actor variable eotg_unclaimed_colours_cd set in
# eotg_unclaimed_claim_effect; exactly one AI path is ever live (see ai_*
# below).
#
# Rules (the mod's interaction rules, eotg_augmentation_interactions.txt
# header): the actor pays in on_accept, never with `cost`; affordability is
# checked in is_valid_showing_failures_only.
# Cooldown: per actor, never cooldown_against_recipient (each claim vanishes
# its recipient, so a recipient cooldown would never bind; spec §8.3). Since
# frontier_v4.md A7.3 it is the variable eotg_unclaimed_colours_cd, not the
# interaction's own `cooldown`, so the decision shares it.
# LAW AT 866: the text never says a claim is filed, registered, recognized or
# licensed, or that others must respect it.
# -----------------------------------------------------------------------------

eotg_unclaimed_claim_interaction = {
    category = interaction_category_diplomacy
    # Art wanted (spec §18); the mod's existing placeholder icon until then
    icon = icon_personal
    interface_priority = 60
    desc = eotg_unclaimed_claim_interaction_desc

    # Actor-only state (_character_interactions.info:556); the AI throttle
    # (spec §8.4)
    is_available = {
        trigger_if = {
            limit = { is_ai = yes }
            eotg_unclaimed_ai_may_claim = yes
        }
    }

    is_shown = {
        NOT = { scope:recipient = scope:actor }
        scope:actor = {
            eotg_is_unclaimed_holder = no
            eotg_is_unclaimed_folk = no
        }
        scope:recipient = {
            eotg_is_unclaimed_holder = yes
            capital_county ?= { eotg_unclaimed_is_unclaimed = yes }
        }
    }

    is_valid_showing_failures_only = {
        scope:actor = { eotg_unclaimed_can_claim = yes }
        custom_description = {
            text = eotg_unclaimed_claim_reach_tt
            scope:recipient.capital_county = {
                eotg_unclaimed_in_reach = { ACTOR = scope:actor }
            }
        }
        custom_description = {
            text = eotg_unclaimed_claim_gold_tt
            scope:actor = { gold >= eotg_unclaimed_claim_gold_value }
        }
        custom_description = {
            text = eotg_unclaimed_claim_prestige_tt
            scope:actor = { prestige >= eotg_unclaimed_claim_prestige_value }
        }
        # frontier_v4.md A7.3: the one cooldown both paths share
        custom_description = {
            text = eotg_unclaimed_claim_cooldown_tt
            scope:actor = { eotg_unclaimed_colours_ready = yes }
        }
    }

    # Nobody answers for the Unsworn: the placeholder cannot refuse
    auto_accept = yes

    on_accept = {
        scope:actor = {
            custom_tooltip = eotg_unclaimed_claim_effect_tt
            custom_tooltip = eotg_unclaimed_claim_returns_tt
            # Sets the cooldown and the worldwide AI throttle (frontier_v4.md
            # A7.3: moved into the effect so every path throttles)
            eotg_unclaimed_claim_effect = { TARGET = scope:recipient.capital_county }
        }
    }

    # AI (spec §8.4). The distance reach (b) is for players: the AI only looks
    # at neighbouring rulers.
    # UNVERIFIED-VANILLA (V-U14): placeholders count as neighboring_rulers.
    # frontier_v4.md A7.3, the V-U14 switch: this is the live AI path while
    # V-U14 is open or passes. If V-U14 FAILS, set every tier of
    # ai_frequency_by_tier below to 0 and turn on the decision's AI
    # (eotg_decision_unclaimed_raise_colours: the lines marked V-U14 there).
    # Exactly one AI path is ever live.
    ai_target_quick_trigger = {
        adult = yes
    }
    ai_targets = {
        ai_recipients = neighboring_rulers
        max = 5
    }
    ai_frequency_by_tier = {
        barony = 0
        county = 36
        duchy = 24
        kingdom = 24
        empire = 24
        hegemony = 24
    }
    ai_will_do = {
        base = 10
        modifier = {
            add = 10
            scope:actor = { has_trait = ambitious }
        }
        modifier = {
            add = 10
            scope:actor = { stewardship >= 12 }
        }
        modifier = {
            add = -10
            scope:actor = { has_trait = content }
        }
        modifier = {
            add = -10
            scope:actor = { has_trait = lazy }
        }
        modifier = {
            factor = 0
            scope:actor = { is_at_war = yes }
        }
    }
}
