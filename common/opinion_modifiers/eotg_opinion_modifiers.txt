# eotg_sabotage_caught_opinion
# Applied when a sabotage scheme is discovered or Corporate Interference exposed.
# Decaying negative opinion — the wound heals slowly over time.
eotg_sabotage_caught_opinion = {
	opinion = -30
	months = 60
	decaying = yes
}

# eotg_cartel_crackdown_opinion
# Applied to Bosses when the Grand Underlord crushes the Shakedown faction by force.
# Long-lived, non-decaying — Bosses remember who broke them.
eotg_cartel_crackdown_opinion = {
	opinion = -25
	months = 48
	decaying = yes
}
