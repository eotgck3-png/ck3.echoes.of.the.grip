# ── eotg_myr_struggle.txt ────────────────────────────────────────────────────
# The Myr Cluster Wars — the defining conflict of the 866 AG bookmark.
# Ignited 850 AG; historically ends ~1300 AG. Four great powers contest the
# Cluster: the Elrossi Imperium (via the Prince of Myr), New Cauldron's
# mercenary legions, the Gob-Ogre Trade League, and the Coldiron Caliphate —
# fought over the homes of the Helix remnant locals.
#
# Phases:
#   ignition   → skirmishes and posturing (start phase, 866 AG state)
#   escalation → open war across the Cluster
#   attrition  → the long grind; exhaustion sets in
#   settlement → negotiation window; ending decision available
#   conclusion → ending phase (the wars are over)
#
# Illustration is a vanilla placeholder — custom art comes later.
# ─────────────────────────────────────────────────────────────────────────────

eotg_myr_struggle = {
	illustration = "gfx/interface/illustrations/event_story/fp2_compromise.dds"    # Placeholder

	##################################################

		eotg_culture_solvari

		eotg_culture_khadrashi

		eotg_culture_uvreki
		eotg_culture_nyssari

		eotg_faith_secular_contract
		eotg_faith_commerce

	}
	# MAP-PASS TODO: once baronies have province ids (and title keys are

	# and reference it here so region-based involvement and county effects apply.
	regions = {}

	# The Cluster is a contested frontier — powers with half their counties
	# inside get pulled into the wars (inert until a region is defined above).
	involvement_prerequisite_percentage = 0.5

	transition_state_duration = { months = 3 }

	on_start = {
		every_involved_ruler = {
			limit = { is_alive = yes }
			trigger_event = { id = eotg_myr.0001  days = 5 }
		}
	}

	on_change_phase = {
		every_involved_ruler = {
			limit = { is_alive = yes }
			trigger_event = { id = eotg_myr.0002  days = 3 }
		}
	}

	# Root = the character who has just become involved/interloper
	on_join = {
		trigger_event = { id = eotg_myr.0003  days = 3 }
	}

	##################################################
	# Phases
	##################################################
	start_phase = struggle_eotg_myr_phase_ignition

	phase_list = {

		# ── Ignition: skirmishes, raids, and posturing ───────────────────────
		struggle_eotg_myr_phase_ignition = {
			duration = { points = 250 }

			future_phases = {
				struggle_eotg_myr_phase_escalation = {
					default = yes
					catalysts = {
						eotg_catalyst_myr_time = minor_struggle_catalyst_over_time_gain

						eotg_catalyst_myr_ruler_slain = medium_struggle_catalyst_gain
					}
				}
				struggle_eotg_myr_phase_settlement = {
					catalysts = {
						eotg_catalyst_myr_diplomacy = medium_struggle_catalyst_gain

					}
				}
			}

			war_effects = {
				name = eotg_myr_ignition_war_effects

				involved_parameters = {
					eotg_myr_wars_active = yes
					eotg_myr_skirmish_raiding = yes
				}

				involved_character_modifier = {
					ai_war_chance = 2
					mercenary_hire_cost_mult = -0.15
				}
			}

			other_effects = {
				name = eotg_myr_ignition_other_effects

				involved_character_modifier = {
					monthly_prestige = 0.1
				}
			}
		}

		# ── Escalation: open war across the Cluster ──────────────────────────
		struggle_eotg_myr_phase_escalation = {
			duration = { points = 500 }

			future_phases = {
				struggle_eotg_myr_phase_attrition = {
					default = yes
					catalysts = {
						eotg_catalyst_myr_time = medium_struggle_catalyst_over_time_gain
						eotg_catalyst_myr_war_concluded = medium_struggle_catalyst_gain
						eotg_catalyst_myr_ruler_slain = major_struggle_catalyst_gain
					}
				}
				struggle_eotg_myr_phase_settlement = {
					catalysts = {
						eotg_catalyst_myr_diplomacy = minor_struggle_catalyst_gain
					}
				}
			}

			war_effects = {
				name = eotg_myr_escalation_war_effects

				involved_parameters = {
					eotg_myr_wars_active = yes
					eotg_myr_mercenary_boom = yes
				}

				involved_character_modifier = {
					ai_war_chance = 10
					ai_war_cooldown = -0.5
					mercenary_hire_cost_mult = -0.3
					levy_reinforcement_rate = 0.2
				}

				all_county_modifier = {
					development_growth_factor = -0.10
				}
			}

			other_effects = {
				name = eotg_myr_escalation_other_effects

				involved_character_modifier = {
					monthly_prestige = 0.2
				}
			}
		}

		# ── Attrition: the long grind ────────────────────────────────────────
		struggle_eotg_myr_phase_attrition = {
			duration = { points = 400 }

			future_phases = {
				struggle_eotg_myr_phase_settlement = {
					default = yes
					catalysts = {
						eotg_catalyst_myr_time = medium_struggle_catalyst_over_time_gain
						eotg_catalyst_myr_diplomacy = major_struggle_catalyst_gain
						eotg_catalyst_myr_war_concluded = medium_struggle_catalyst_gain
					}
				}
				struggle_eotg_myr_phase_escalation = {
					catalysts = {
						eotg_catalyst_myr_war_declared = major_struggle_catalyst_gain
						eotg_catalyst_myr_aggression = medium_struggle_catalyst_gain

			}

			other_effects = {
				name = eotg_myr_attrition_other_effects

					stress_gain_mult = 0.1

		}

		struggle_eotg_myr_phase_settlement = {
			duration = { points = 300 }

			future_phases = {
				struggle_eotg_myr_phase_escalation = {
					default = yes
					catalysts = {
						eotg_catalyst_myr_war_declared = massive_struggle_catalyst_gain
						eotg_catalyst_myr_aggression = major_struggle_catalyst_gain
						eotg_catalyst_myr_ruler_slain = medium_struggle_catalyst_gain
					}

			}

			war_effects = {
				name = eotg_myr_settlement_war_effects

				involved_parameters = {
					eotg_myr_peace_talks = yes
				}

				involved_character_modifier = {
					ai_war_chance = -10
					monthly_prestige = 0.1
				}

				all_county_modifier = {
					development_growth_factor = 0.10
					monthly_county_control_growth_factor = 0.10
				}
			}

			ending_decisions = {
				eotg_decision_myr_settlement
			}
		}

		# ── Conclusion: the wars are over ────────────────────────────────────
		struggle_eotg_myr_phase_conclusion = {
			on_start = {
				every_involved_ruler = {
					limit = { is_alive = yes }
					trigger_event = { id = eotg_myr.0090  days = 5 }
				}
			}
