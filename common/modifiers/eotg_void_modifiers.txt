# ── eotg_void_modifiers.txt ──────────────────────────────────────────────────
# Modifiers for the Void Corruption system.
# Band modifiers (whispers/pressure/saturation) surface the hidden
# eotg_void_exposure variable on the character sheet — they are swapped by
# eotg_se_void_refresh_band and must never be applied with a duration.
# Insight modifiers are the permanent education-differentiated payoff from
# eotg_se_void_edu_insight. eotg_mod_void_aura lives in eotg_modifiers.txt.
# Icons are vanilla placeholders — custom art comes later.
# ─────────────────────────────────────────────────────────────────────────────

# ── Exposure bands (visible state of eotg_void_exposure) ─────────────────────

# 25–49: something at the edge of hearing.
# Loc intent: "Void Whispers" — sleep comes shallow; prayer comes hard.
eotg_mod_void_whispers = {
    icon = stress_negative    # Placeholder
    stress_gain_mult = 0.05
    monthly_piety_gain_mult = -0.05
}

# 50–74: the whispers have opinions now.
# Loc intent: "Void Pressure" — it is easier to keep secrets than company.
eotg_mod_void_pressure = {
    icon = stress_negative    # Placeholder
    stress_gain_mult = 0.1
    diplomacy = -1
    intrigue = 1
}

# 75–99: past the point where refusal still holds.
# Loc intent: "Void Saturation" — the boundary runs through you now.
eotg_mod_void_saturation = {
    icon = stress_negative    # Placeholder
    stress_gain_mult = 0.2
    diplomacy = -2
    intrigue = 2
    health = -0.25
}

# ── Warding ──────────────────────────────────────────────────────────────────

# An active Voidwalker ward — exposure decays while it holds.
# Loc intent: "Warded Against the Void" — heretic sigils, properly drawn.
eotg_mod_void_warded = {
    icon = health_positive    # Placeholder
    stress_gain_mult = -0.1
    monthly_piety_gain_mult = 0.1
}

# ── Insight (permanent education-differentiated payoff) ──────────────────────
