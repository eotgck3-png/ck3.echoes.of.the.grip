# -- §5.5(a) Neurofractured drift --------------------------------------------
# Yearly risk drift for a Neurofractured ruler (cybernetics_v2 B1 rows kept:
# 8, +4 per stress level). Treatments slow it but never pause it: the floor is
# 2 a year. Hidden: read only by eotg_on_yearly_aug_neurofractured_check
# (step 1), never shown. Scope: the ruler.
eotg_aug_nf_drift_value = {
    value = 8
    if = {
        limit = { stress_level >= 3 }
        add = 12
    }
    else_if = {
        limit = { stress_level = 2 }
        add = 8
    }
    else_if = {
        limit = { stress_level = 1 }
        add = 4
    }
    if = {
        limit = { has_character_flag = eotg_flag_aug_sedated }
        add = eotg_aug_sedation_relief_value
    }
    if = {
        limit = { has_character_flag = eotg_flag_aug_restrained }
        add = -4
    }
    min = 2
}

# Sedation tolerance: each course works less well than the last. The course
# counter (eotg_aug_sedation_courses) is incremented by
# eotg_decision_aug_sedation before the flag is set, so the running course is
# 1, 2, 3... A ruler sedated before the counter existed counts as course 1.
# -6 on the first course, -3 on the second, 0 from the third on.
eotg_aug_sedation_relief_value = {
    value = 0
    if = {
        limit = {
            OR = {
                NOT = { has_variable = eotg_aug_sedation_courses }
                var:eotg_aug_sedation_courses <= 1
            }
        }
        add = -6
    }
    else_if = {
        limit = { var:eotg_aug_sedation_courses = 2 }
        add = -3
    }
}

# -- §5.4 Neurofractured hidden overrides -------------------------------------
# The % chance that a calm choice in a Fracture- or Storm-band event is
# replaced by the harsher order (the converted options of fracture.006, .016,
# .019, .023, .024, .025). Fracture 25, Storm 40; +15 once access was granted
# (voice 4, fracture.017.a); -10 under Sedation. Clamped 0-60. Flicker stays
# at 0 whatever the voice: Flicker lies (the misleading descs), it does not
# seize. Hidden: read only as `random = { chance = ... }` inside
# hidden_effect (vanilla script-value chance: events/activities/hunt_activity/
# hunt_events.txt:4196). Scope: the Neurofractured ruler.
eotg_aug_override_chance = {
    value = 0
    if = {
        limit = { eotg_aug_pressure_flicker = no }
        if = {
            limit = { eotg_aug_pressure_storm = yes }
            add = 40
        }
        else = {
            add = 25
        }
        if = {
            limit = {
                has_variable = eotg_aug_voice
                var:eotg_aug_voice >= 4
            }
            add = 15
        }
        if = {
            limit = { has_character_flag = eotg_flag_aug_sedated }
            add = -10
        }
    }
    min = 0
    max = 60
}

# -- §5.7 Non-ruler progression ----------------------------------------------
# Yearly % chance that an augmented knight or courtier (tier 1 or 2) moves up
# a stage, in eotg_on_yearly_aug_nonruler_check step 2. Base 10; Iron Retinue
# knight +5, +5 more with the permanent flag (on the knight or the liege, as
# Phase 6 words it); a child patient (init.014/.015) +5; plus the battle bonus
# (eotg_aug_nr_battle_bonus, 0-10, +2 per battle from
# eotg_on_combat_aug_knights, cleared on progression). Scope: the non-ruler.
eotg_aug_nr_progress_chance = {
    value = 10
    if = {
        limit = { eotg_aug_is_retinue_knight = yes }
        add = 5
        if = {
            limit = {
                OR = {
                    has_character_flag = eotg_flag_aug_retinue_permanent
                    employer ?= { has_character_flag = eotg_flag_aug_retinue_permanent }
                }
            }
            add = 5
        }
    }
    if = {
        limit = { has_character_flag = eotg_flag_aug_child_patient }
        add = 5
    }
    if = {
        limit = { has_variable = eotg_aug_nr_battle_bonus }
        add = var:eotg_aug_nr_battle_bonus
    }
}

# -- Procedures (docs/specs/cybernetics_v2_procedures.md §4.2, §4.6) ----------
# The shared roll's "bad outcome" factor (eotg_aug_procedure_roll_effect:
# every bad entry takes `modifier = { factor = eotg_aug_proc_bad_factor }`).
# Evaluated in the patient's scope inside the roll, after it has saved
# scope:eotg_proc_provider / _kind / _surgeon. Stepped surgeon-skill factors
# in the shape of vanilla risky_wound_treatment_effect
# (common/scripted_effects/20_health_effects.txt:2368-2420; ratings
# common/script_values/00_basic_values.txt:474-476). A learning-16 physician
# with the physician lifestyle runs at about 4% bad (8% x 0.6 x 0.8).
# The place a later technician or licence line is added (spec §10).
eotg_aug_proc_bad_factor = {
    value = 1
    if = {
        limit = {
            OR = {
                scope:eotg_proc_provider ?= flag:physician
                scope:eotg_proc_provider ?= flag:self
            }
        }
        if = {
            limit = { scope:eotg_proc_surgeon ?= { learning < medium_skill_rating } }
            multiply = 1.25
        }
        else_if = {
            limit = { scope:eotg_proc_surgeon ?= { learning >= high_skill_rating } }
            multiply = 0.6
        }
        else_if = {
            limit = { scope:eotg_proc_surgeon ?= { learning >= decent_skill_rating } }
            multiply = 0.8
        }
        if = {
            limit = { scope:eotg_proc_surgeon ?= { has_trait = lifestyle_physician } }
            multiply = 0.8
        }
        # Realm spec §3.2 (G10): an Implant Technician as the surgeon (own or
        # borrowed), by aptitude; on top of the learning rows
        if = {
            limit = {
                scope:eotg_proc_surgeon ?= {
                    has_court_position = eotg_aug_implant_technician_court_position
                }
            }
            if = {
                limit = {
                    scope:eotg_proc_surgeon = {
                        aptitude = { court_position = eotg_aug_implant_technician_court_position  value >= 4 }
                    }
                }
                multiply = 0.5
            }
            else_if = {
                limit = {
                    scope:eotg_proc_surgeon = {
                        aptitude = { court_position = eotg_aug_implant_technician_court_position  value >= 3 }
                    }
                }
                multiply = 0.6
            }
            else_if = {
                limit = {
                    scope:eotg_proc_surgeon = {
                        aptitude = { court_position = eotg_aug_implant_technician_court_position  value >= 2 }
                    }
                }
                multiply = 0.75
            }
            else = {
                multiply = 0.9
            }
        }
    }
    # Realm spec §3.2 (G7): a License vets the sanctioned clinics
    if = {
        limit = {
            scope:eotg_proc_provider ?= flag:clinic
            eotg_aug_under_policy = { LAW = license }
        }
        multiply = 0.8
    }
    # Operating on yourself
    if = {
        limit = { scope:eotg_proc_provider ?= flag:self }
        multiply = 1.5
    }
    # A chain head's edge (replaces the old "clean +15", balance spec §5.6)
    if = {
        limit = { has_character_flag = eotg_flag_aug_chain_edge }
        multiply = 0.6
    }
    # The Bidding War's safe vendor (tier2.013), on an upgrade
    if = {
        limit = {
            scope:eotg_proc_kind ?= flag:upgrade
            has_character_flag = eotg_flag_aug_vendor_safe
        }
        multiply = 0.75
    }
    # Unstable hardware
    if = {
        limit = { has_trait = eotg_neurofractured }
        multiply = 1.5
    }
    # Sabotaged hardware (interactions spec §3.2; set by
    # eotg_aug_sabotage_apply_effect, spent by the roll, cleared by a fault
    # repair). The reserved extension point (procedures §10).
    if = {
        limit = { has_character_flag = eotg_flag_aug_sabotaged }
        multiply = 1.5
    }
    min = 0.2
}

# The clinic price of a removal, by eotg_aug_removal_kind (set by the three
# removal decisions; read in eotg_aug_proc.001): partial and full
# medium_gold_value, downgrade major_gold_value; x0.75 for partial and
# downgrade at stress level 2+ or with the Intervention discount (B3, as the
# decisions' old cost blocks).
# Before the booking exists (the decisions' is_valid / ai_potential run before
# their effect sets the variable) the kind is read from the tier instead: the
# three decisions are tier-gated one to one (partial = Enhanced, downgrade =
# Overclocked, full = Augmented), as proc.001's trigger mirrors. This keeps
# the decisions' gold floor in the spec's own form (eotg_aug_can_afford_procedure
# with this BASE) and the discount flag out of their is_valid tooltips.
eotg_aug_removal_base_value = {
    value = medium_gold_value
    if = {
        limit = {
            OR = {
                var:eotg_aug_removal_kind ?= flag:downgrade
                AND = {
                    NOT = { has_variable = eotg_aug_removal_kind }
                    eotg_is_aug_tier3 = yes
                }
            }
        }
        value = major_gold_value
    }
    if = {
        limit = {
            OR = {
                var:eotg_aug_removal_kind ?= flag:partial
                var:eotg_aug_removal_kind ?= flag:downgrade
                AND = {
                    NOT = { has_variable = eotg_aug_removal_kind }
                    OR = {
                        eotg_is_aug_tier2 = yes
                        eotg_is_aug_tier3 = yes
                    }
                }
            }
            OR = {
                stress_level >= 2
                has_character_flag = eotg_flag_aug_intervention_discount
            }
        }
        multiply = 0.75
    }
}
