# Governments v2, Phase G2, Cartel batch C1 (docs/specs/governments_v2_g2_cartel.md
# §2.2, §3.1). All character scope. Magnitudes and thresholds are vanilla
# (miniscule..massive_dread_gain/_loss, low/medium/high_dread:
# G/common/script_values/00_basic_values.txt:725-734, 993-995).
# C2 adds eotg_cartel_schedule_cost_value. The C3-C4 values (tighten cost,
# exit legitimacy) come with their batches.
# -----------------------------------------------------------------------------

# The real change eotg_cartel_change_dread_effect applies: the requested
# amount (scope:eotg_cartel_dread_amount) clamped so the result stays inside
# [0, 100], i.e. the change is clamped to [-dread, 100 - dread], and the
# tooltip shows the change that actually happens (spec §2.2).
# 100 = MAX_DREAD (G/common/defines/00_defines.txt:120); the Cartel does not
# set max_dread (OQ5, owner 2026-10-20: no change yet). `add = dread` on a
# character is vanilla (G/common/on_action/dlc/mpo/mpo_on_actions_2.txt:440).
eotg_cartel_dread_clamped_value = {
    value = scope:eotg_cartel_dread_amount
    add = dread             # now: the resulting dread
    min = 0
    max = 100
    subtract = dread        # back to a change
}

# Payers on the Cartel schedule (spec §2.2): every tributary in the
# eotg_cartel_tributary contract group (batch C2; the C1 stub returned 0).
# Shape: every_tributary in a value, G/common/script_values/
# 09_mpo_values.txt:2210-2214; has_subject_contract_group on the tributary,
# G/common/character_interactions/00_tributary_interactions.txt:5367-5375.
# G1 payers in vanilla tributary_subjugated are not counted (save compat,
# spec §4.10 note: they stay in that group and lapse at death as before).
eotg_cartel_payer_count_value = {
    value = 0
    every_tributary = {
        limit = { has_subject_contract_group = eotg_cartel_tributary }
        add = 1
    }
}

# The schedule modifier's scale (spec §2.7): var:eotg_cartel_schedule_scale,
# 0 when unset. Shape: prepare_travels_modifier_value
# (G/common/script_values/00_court_position_values.txt:2602-2611).
eotg_cartel_schedule_scale_value = {
    value = 0
    if = {
        limit = {
            has_variable = eotg_cartel_schedule_scale
        }
        value = var:eotg_cartel_schedule_scale
    }
}

# Dread price of Put on the Schedule (spec §5.2, OQ6 numbers approved
# 2026-10-20): by the recipient's highest held tier, county 10, duchy 15,
# kingdom 25, empire 35 (hegemons are not valid targets); +10 under the
# compel send option. Scopes: scope:recipient, scope:eotg_cartel_compel (the
# interaction; `?=` because the AI evaluates every option combination,
# _character_interactions.info:611). Read by the interaction's is_valid,
# on_accept (through eotg_cartel_change_dread_effect) and ai_will_do.
eotg_cartel_schedule_cost_value = {
    value = 10
    if = {
        limit = { scope:recipient.highest_held_title_tier = tier_duchy }
        value = 15
    }
    else_if = {
        limit = { scope:recipient.highest_held_title_tier = tier_kingdom }
        value = 25
    }
    else_if = {
        limit = { scope:recipient.highest_held_title_tier >= tier_empire }
        value = 35
    }
    if = {
        limit = { scope:eotg_cartel_compel ?= yes }
        add = 10
    }
}

# The same price as a negative change, for eotg_cartel_change_dread_effect
# (AMOUNT takes a value name, not an inline block). Interaction on_accept only.
eotg_cartel_schedule_spend_value = {
    value = eotg_cartel_schedule_cost_value
    multiply = -1
}
