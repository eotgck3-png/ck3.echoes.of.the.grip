# -- eotg_frontier_values.txt -------------------------------------------------
# Script values for Frontier Systems v1 (docs/specs/frontier_v1.md §3.4, §5).
# Vanilla folder (game/common/script_values/), no replace_path. Shapes as
# common/script_values/eotg_augmentation_values.txt (a `value =` block,
# conditional adds, a trailing clamp).
#
# The tuning levers: the type rows below are the project-type "data" (§5).
# Adding a type = one more row in eotg_frontier_base_rate_value,
# eotg_frontier_min_dev_value and eotg_frontier_min_control_value.
#
# Values with a COUNTY root read the county title's own variables; the gold
# values are always read in a CHARACTER scope (spec §16 V22).
# -----------------------------------------------------------------------------

# -- Costs (character scope; vanilla realm-relative gold values) ---------------
eotg_frontier_establish_cost_value = {
    value = medium_gold_value
}
eotg_frontier_invest_cost_value = {
    value = minor_gold_value
}
# Paid yearly by the sponsor (character scope: the sponsor)
eotg_frontier_sponsor_payment_value = {
    value = minor_gold_value
}

# -- Type rows (county scope) --------------------------------------------------
# Base Establishment gained per year (spec §5.1)
eotg_frontier_base_rate_value = {
    value = 10
    if = {
        limit = { var:eotg_frontier_type ?= flag:settlement }
        value = 12
    }
    else_if = {
        limit = {
            OR = {
                var:eotg_frontier_type ?= flag:trade
                var:eotg_frontier_type ?= flag:mining
            }
        }
        value = 11
    }
    else_if = {
        limit = { var:eotg_frontier_type ?= flag:research }
        value = 9
    }
    # military, religious, administrative: 10
}

# Completion floor: development_level (spec §5.2), RELATIVE (verification Q4,
# orchestrator ruling): min(type floor, the development this project can still
# reach). eotg_frontier_dev_reachable is recorded at start: starting
# development + the milestone development not yet granted to this Region.
eotg_frontier_min_dev_value = {
    value = 2
    if = {
        limit = {
            OR = {
                var:eotg_frontier_type ?= flag:trade
                var:eotg_frontier_type ?= flag:research
                var:eotg_frontier_type ?= flag:administrative
            }
        }
        value = 3
    }
    # Phase 2 (frontier_v2.md §5.2): Ancient Infrastructure lowers the floor
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_ancient_infrastructure }
        add = -1
    }
    if = {
        limit = { has_variable = eotg_frontier_dev_reachable }
        max = var:eotg_frontier_dev_reachable
    }
}

# Recorded by eotg_frontier_start_effect (county scope): development now +
# the milestones (+1 each, at most 2) this Region has not been granted yet
# (Q5: once per Region, ever).
eotg_frontier_dev_reachable_value = {
    value = development_level
    add = 2
    if = {
        limit = { has_variable = eotg_frontier_dev_granted }
        subtract = var:eotg_frontier_dev_granted
    }
}

# Current values, for the debug readout (county scope)
eotg_frontier_current_dev_value = {
    value = development_level
}
eotg_frontier_current_control_value = {
    value = county_control
}

# Completion floor: county_control (spec §5.2)
eotg_frontier_min_control_value = {
    value = 50
    if = {
        limit = { var:eotg_frontier_type ?= flag:settlement }
        value = 40
    }
    else_if = {
        limit = { var:eotg_frontier_type ?= flag:military }
        value = 70
    }
    else_if = {
        limit = { var:eotg_frontier_type ?= flag:administrative }
        value = 80
    }
    # Phase 2 (frontier_v2.md §5.2): contested ground needs a firmer hold; a
    # remote Region is held more loosely
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_strategic }
        add = 10
    }
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_remote }
        add = -10
    }
}

# -- Yearly gain inputs (county scope) -----------------------------------------
# development_level and county_control as values in county scope: confirmed
# (03_dlc_fp2_script_values.txt:88-91; 00_county_control_values.txt:127;
# 09_mpo_wars.txt:1335).
eotg_frontier_dev_bonus_value = {
    value = development_level
    divide = 5
    floor = yes
    max = 4
    min = 0
}

eotg_frontier_control_bonus_value = {
    value = 0
    if = {
        limit = { county_control >= 80 }
        value = 2
    }
    else_if = {
        limit = { county_control < 40 }
        value = -3
    }
}

# The founder's skill for the type, / 4, max 5; 0 without a valid founder.
# The validity test is the same trigger the strain check uses.
eotg_frontier_founder_bonus_value = {
    value = 0
    if = {
        limit = { eotg_frontier_has_valid_founder = yes }
        if = {
            limit = { var:eotg_frontier_type ?= flag:military }
            add = var:eotg_frontier_founder.martial
        }
        else_if = {
            limit = {
                OR = {
                    var:eotg_frontier_type ?= flag:religious
                    var:eotg_frontier_type ?= flag:research
                }
            }
            add = var:eotg_frontier_founder.learning
        }
        else = {
            add = var:eotg_frontier_founder.stewardship
        }
        divide = 4
        floor = yes
        max = 5
    }
}

# frontier_v4.md A8.2: 8 pre-pace (was 6), +4 a year after the x0.5 pace.
# Sponsored Settlement about 8.1 years, sponsored Research about 9.8: every
# type in the owner's 8-10 band (not +10, which takes Settlement under it)
eotg_frontier_sponsor_bonus_value = {
    value = 0
    if = {
        limit = { has_variable = eotg_frontier_sponsor_paid }
        value = 8
    }
}

# The resettle head start: 10 per earlier abandonment, max 30 (spec §6.2)
eotg_frontier_trace_bonus_value = {
    value = 0
    if = {
        limit = { has_variable = eotg_frontier_attempts }
        add = var:eotg_frontier_attempts
        multiply = 10
        max = 30
    }
}

# Establishment gained this year (spec §6.1 step 3). Owner decision
# (verification §4): pacing DOUBLED, about 10-16 years unsponsored and 8-10
# sponsored. Every input above keeps its relative weight; the sum is halved
# here (eotg_frontier_pace_value), so retuning the pace is one number.
eotg_frontier_yearly_gain_value = {
    value = eotg_frontier_base_rate_value
    add = eotg_frontier_dev_bonus_value
    add = eotg_frontier_control_bonus_value
    add = eotg_frontier_founder_bonus_value
    add = eotg_frontier_sponsor_bonus_value
    # Phase 2 (frontier_v2.md §4.2, §5.2, §6.2): traits, infrastructure and the
    # type's own input, before the pace multiplier like everything else
    add = eotg_frontier_trait_rate_value
    add = eotg_frontier_infra_rate_value
    add = eotg_frontier_type_input_value
    # Phase 3a hook (frontier_v3.md §7): a future trade network; 0 by default
    add = eotg_frontier_trade_network_value
    multiply = eotg_frontier_pace_value
    min = 0
}

# INT5 (integration_v2.md §6.3): Frontier difficulty scales the pace for
# player-held Shoals only (vanilla has_game_rule in a script value:
# common/script_values/00_basic_values.txt:1875-1896). The legibility bands
# read the post-pace gain, so their words stay accurate.
eotg_frontier_pace_value = {
    value = 0.5
    if = {
        limit = {
            has_game_rule = eotg_frontier_difficulty_easy
            holder ?= { is_ai = no }
        }
        value = 0.65
    }
    else_if = {
        limit = {
            has_game_rule = eotg_frontier_difficulty_hard
            holder ?= { is_ai = no }
        }
        value = 0.4
    }
}

# frontier_v4.md A8.1: years until Establishment reaches Settled at this
# year's pace (county scope; a Frontier only). Read ONLY to choose a worded
# band in eotg_frontier_legibility_tooltip_effect; never printed (frontier_v1
# §R: players see consequences, never numbers). A gain under 1 counts as 1
# here; the stalled band suppresses the time line anyway.
eotg_frontier_years_left_value = {
    value = 100
    subtract = var:eotg_frontier_progress
    divide = {
        value = eotg_frontier_yearly_gain_value
        min = 1
    }
    min = 0
}

# Strain at which the project falters (.005). Doubled with the pacing (0-4
# became 0-8), so a cause held for the same share of the project fails it
# equally often. Desc bands in .002 and the AI gates use half and
# three-quarters of it (4 and 6).
eotg_frontier_strain_fail_value = {
    value = 8
}

# -- AI throttle ----------------------------------------------------------------
# frontier_v4.md A2.2: 8, plus 1 per 20 Unclaimed Regions, at most 24. Reads
# the global variable eotg_unclaimed_count only (A10 seam rule): without
# Unclaimed Regions it is absent and the cap is 8.
eotg_frontier_ai_cap_value = {
    value = 8
    if = {
        limit = { has_global_variable = eotg_unclaimed_count }
        add = {
            value = global_var:eotg_unclaimed_count
            divide = 20
            floor = yes
        }
    }
    max = 24
}

# frontier_v4.md A4.3: the skill a founder brings to the Region's project
# type, the same mapping as eotg_frontier_founder_bonus_value: martial for
# Military, learning for Religious and Research, stewardship otherwise.
# CHARACTER scope (the would-be founder); reads scope:eotg_frontier_county.
eotg_frontier_type_skill_value = {
    value = stewardship
    if = {
        limit = { scope:eotg_frontier_county ?= { var:eotg_frontier_type ?= flag:military } }
        value = martial
    }
    else_if = {
        limit = {
            scope:eotg_frontier_county ?= {
                OR = {
                    var:eotg_frontier_type ?= flag:religious
                    var:eotg_frontier_type ?= flag:research
                }
            }
        }
        value = learning
    }
}

# -- Yearly inputs (county scope, pre-pace) -------------------------------------
# Traits (spec §5.2). Each row: a flat input, plus more for the type it suits.
# Hostile Environment and Remote are countered by an Orbital Station and a
# Navigation Beacon.
eotg_frontier_trait_rate_value = {
    value = 0
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_rich_resources }
        add = 1
        if = { limit = { var:eotg_frontier_type ?= flag:mining } add = 2 }
    }
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_trade_route }
        add = 1
        if = { limit = { var:eotg_frontier_type ?= flag:trade } add = 2 }
    }
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_strategic }
        add = 1
        if = { limit = { var:eotg_frontier_type ?= flag:military } add = 2 }
    }
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_habitable }
        add = 1
        if = { limit = { var:eotg_frontier_type ?= flag:settlement } add = 2 }
    }
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_ancient_ruins }
        if = { limit = { var:eotg_frontier_type ?= flag:research } add = 3 }
        else_if = { limit = { var:eotg_frontier_type ?= flag:religious } add = 2 }
    }
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_ancient_infrastructure }
        add = 2
    }
    if = {
        limit = { has_county_modifier = eotg_frontier_trait_anomalous }
        add = -1
        if = { limit = { var:eotg_frontier_type ?= flag:research } add = 5 }
    }
    if = {
        limit = {
            has_county_modifier = eotg_frontier_trait_hostile
            NOT = { has_county_modifier = eotg_frontier_infra_orbital_station }
        }
        add = -2
    }
    if = {
        limit = {
            has_county_modifier = eotg_frontier_trait_remote
            NOT = { has_county_modifier = eotg_frontier_infra_navigation_beacon }
        }
        add = -2
    }
}

# The type's own input, in normal CK3 terms (spec §6.2). Research has none:
# its inputs are the founder's learning and the research traits.
eotg_frontier_type_input_value = {
    value = 0
    if = {
        limit = {
            var:eotg_frontier_type ?= flag:settlement
            county_control >= 60
        }
        add = 2
    }
    else_if = {
        limit = {
            var:eotg_frontier_type ?= flag:trade
            has_variable = eotg_frontier_sponsor_paid
        }
        add = 3
    }
    else_if = {
        limit = { var:eotg_frontier_type ?= flag:mining }
        add = {
            value = development_level
            divide = 5
            floor = yes
            max = 3
        }
    }
    else_if = {
        limit = { var:eotg_frontier_type ?= flag:military }
        add = {
            value = holder.martial
            divide = 6
            floor = yes
            max = 3
        }
    }
    else_if = {
        # piety_level >= n as a character trigger: confirmed against vanilla
        # 1.20.0.3 (spec §15 item 1)
        limit = { var:eotg_frontier_type ?= flag:religious }
        if = {
            limit = { holder = { piety_level >= 3 } }
            add = 2
        }
        else_if = {
            limit = { holder = { piety_level >= 1 } }
            add = 1
        }
    }
    else_if = {
        limit = { var:eotg_frontier_type ?= flag:administrative }
        add = {
            value = holder.stewardship
            divide = 6
            floor = yes
            max = 3
        }
    }
}

# -- Local Ruler gender (fix I1b F3) -------------------------------------------
# Chance 0-100 that a secular Local Ruler (the settlers' leader or the Trade
# factor, granted a city System) is female. Root is the Region's holder (the
# granter; root of eotg_frontier.004, the only caller of the completion).
# The realm's gender law first: vanilla reads male_only_law / female_only_law
# for a created candidate (01_dynamic_values.txt:1116-1126, in
# random_theocratic_ruler_chance_for_female) and for a grant
# (00_scripted_triggers.txt:332-354, can_be_granted_titles_by). Otherwise the
# faith's dominant gender, the value vanilla's secular templates use
# (00_court_position_templates.txt:77, root_faith_dominant_gender_female_chance),
# which keeps the figure inside can_be_granted_titles_by's faith check
# (00_scripted_triggers.txt:305-331).
eotg_frontier_secular_ruler_female_chance_value = {
    if = {
        limit = { root = { has_realm_law = male_only_law } }
        value = 0
    }
    else_if = {
        limit = { root = { has_realm_law = female_only_law } }
        value = 100
    }
    else = {
        value = root_faith_dominant_gender_female_chance
    }
}
