# ── eotg_corporation_values.txt ───────────────────────────────────────────────
# Script values for the Corporation government mechanics.
# ─────────────────────────────────────────────────────────────────────────────

# Annual board influence decay — CEO loses 10 points per year of inaction
eotg_sv_corp_board_influence_decay = 10

# Election vote weight per Regional Director (called on that character's scope)
# Base: stewardship. Bonus: gold holdings (capped). Board Seat adds flat bonus.
eotg_sv_corp_election_weight = {
	value = 0
	add = { value = stewardship  multiply = 2 }
	add = { value = gold  divide = 50 }
	if = {
		limit = { vassal_contract_has_flag = eotg_corp_board_seat_flag }
		add = 10
	}
	min = 0
	max = 60
}

# Gold cost for a Hostile Acquisition interaction (scales with target realm size)
eotg_sv_corp_hostile_acquisition_cost = {
	value = 100
	add = { value = scope:recipient.realm_size  multiply = 15 }
	min = 100
	max = 2000
}

# Board influence restored when the CEO successfully completes a diplomacy action
eotg_sv_corp_board_influence_diplomacy_gain = 15

# Board influence spent when the CEO declares an offensive war
eotg_sv_corp_board_influence_war_cost = 25

# Board influence spent on major economic decisions
eotg_sv_corp_board_influence_decision_cost = 15

# Monthly market share accumulation for a vassal character (called in story cycle scope)
eotg_sv_corp_market_share_gain = {
	value = 1
	add = { value = stewardship  divide = 5 }
	add = { value = realm_size  divide = 10 }
	min = 1
	max = 50
}
