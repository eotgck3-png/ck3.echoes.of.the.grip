# ── LANGUAGES ─────────────────────────────────────────────────────────────────

eotg_language_elven = {
	type = language
	is_shown = {
		language_is_shown_trigger = {
			LANGUAGE = eotg_language_elven
		}
	}
	ai_will_do = {
		value = 10
		if = {
			limit = { has_cultural_pillar = eotg_language_elven }
			multiply = 10
		}
	}
	color = latin
}
