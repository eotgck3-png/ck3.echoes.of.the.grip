# hostility_doctrine drives cross-family conversion cost and opinion. Only four
# exist in vanilla: abrahamic_, pagan_, eastern_ and sinitic_hostility_doctrine.

# Divine Order — Yu and his sons: Elross, Bohemut, Malvrick, plus the Yu-aligned
# elder deity Frozt. These faiths argue over which son speaks for the sealed
# father; none of them doubts the father is real.

    graphical_faith = "christian_gfx"    # TODO: replace with custom gfx once art exists
    piety_icon_group = "christian"
    hostility_doctrine = abrahamic_hostility_doctrine
    doctrine_background_icon = core_tenet_banner_christian.dds
}
