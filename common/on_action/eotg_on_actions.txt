# ── eotg_on_actions.txt ───────────────────────────────────────────────────────
# Event hooks for Echoes of the Grip.
#
# IMPORTANT: The `events` and `random_events` sub-blocks are additive.
# The `effect = {}` key is NOT additive — it overwrites vanilla.
# Always chain via `on_actions = { eotg_my_action }` instead of
# adding effect = {} directly to a vanilla on_action.
# ─────────────────────────────────────────────────────────────────────────────

# Hook into vanilla yearly character pulse (fires once per year for count+ chars)
on_yearly_playable = {
    on_actions = {
        eotg_on_yearly_void_check
        eotg_on_yearly_corporate_faction_check
        eotg_on_yearly_corp_board_influence_decay
        eotg_on_yearly_cartel_paranoia_check
        eotg_on_yearly_elven_isolationism_update
        eotg_on_yearly_pmc_controversy_check
        eotg_on_yearly_pmc_loss_of_command_check
        eotg_on_yearly_nc_campaign_check
        eotg_on_yearly_nc_election_check
        eotg_on_yearly_nc_senate_check
        eotg_on_yearly_fringe_check
        eotg_on_yearly_fringe_flavor
        eotg_on_yearly_gobcorp_flavor
        eotg_on_yearly_corp_flavor
        eotg_on_yearly_cartel_flavor
        eotg_on_yearly_pmc_flavor
        eotg_on_yearly_elven_flavor
        eotg_on_yearly_nc_flavor
    }
}

# Hook into war declaration to offer contracts to eligible Contractor-Generals
on_war_started = {
    on_actions = { eotg_on_war_started_pmc_check }
}

# Hook into title gain to set/clear the corporate government flag
on_title_gain = {

        eotg_on_title_gain_corp_board_influence

# Hook into character death to fire Corporation election event chain and NC VP succession
on_death = {

        eotg_on_death_elven_monarch
        eotg_on_death_nc_president

    }
}

# Yearly check to start the market faction story cycle on newly-powerful
# corporate vassals who don't have it yet
eotg_on_yearly_corporate_faction_check = {
    effect = {
        if = {
            limit = {
                is_powerful_vassal = yes
                NOT = { has_character_flag = eotg_market_faction_active }
                liege = { government_has_flag = eotg_government_is_corporate }
            }
            create_story = eotg_market_faction_cycle
        }
    }
}

# On title gain: set/clear the corporate government flag used by story cycle
eotg_on_title_gain_corporate_flag = {
    effect = {
        if = {
            limit = {
                government_has_flag = eotg_government_is_corporate
                highest_held_title_tier = tier_empire
            }
            add_character_flag = eotg_flag_corporate_government
        }
        else_if = {
            limit = {
                NOT = { government_has_flag = eotg_government_is_corporate }
                has_character_flag = eotg_flag_corporate_government
            }
            remove_character_flag = eotg_flag_corporate_government
        }
    }
}

# Yearly controversy backlash check — fires eotg_pmc_contract.0004 for active CGs

            limit = {

                    has_character_modifier = eotg_pmc_controversy_exclusive

# Yearly check — if CG Loss of Command flag has expired naturally, fire consequence event

        if = {

            trigger_event = { id = eotg_pmc_command.0003  days = 5 }
        }
    }
}

# On war started: check if any Contractor-General can receive a contract offer
eotg_on_war_started_pmc_check = {
    effect = {
        scope:attacker = { save_scope_as = eotg_pmc_war_client }
        every_independent_ruler = {
            limit = {
                eotg_st_pmc_contract_available = yes
                NOT = { this = scope:eotg_pmc_war_client }
                opinion = { target = scope:eotg_pmc_war_client  value >= 20 }
            }
            set_variable = { name = eotg_pmc_contract_client  value = scope:eotg_pmc_war_client }
            trigger_event = { id = eotg_pmc_contract.0001  days = 5 }
        }
    }
}

# On death — apply succession chaos to the new Fringe Overlord.
# The deceased character is in scope; check their primary title's new holder.
eotg_on_death_fringe_overlord = {
    effect = {
        if = {
            limit = { eotg_st_is_fringe_overlord = yes }
            primary_title = {
                holder = {
                    if = {
                        limit = { eotg_st_is_fringe_overlord = yes }
                        eotg_se_fringe_succession_chaos = yes
                        trigger_event = { id = eotg_fringe.0005  days = 7 }
                        if = {
                            limit = { realm_size >= 30 }
                            trigger_event = { id = eotg_fringe.0006  days = 14 }
                        }
                        # Heir challenge fires if an ambitious captain exists
                        random = {
                            chance = 50
                            trigger_event = { id = eotg_fringe.0018  days = 21 }
                        }
                    }
                }
            }
        }
    }
}

# Yearly flavor event rotation for Fringe Overlords.
# Uses if/limit/random blocks so multiple events can fire in the same year,
# each gated by its own cooldown flag to prevent spam.
eotg_on_yearly_fringe_flavor = {
    effect = {
        if = {
            limit = { eotg_st_is_fringe_overlord = yes }

            # The Grip Weakens — fires when Grip is low (50% chance, 2yr cooldown)
            if = {
                limit = {
                    eotg_st_fringe_grip_low = yes
                    realm_size >= 8
                    NOT = { has_character_flag = eotg_fringe_0014_recent }
                }
                random = {
                    chance = 50
                    add_character_flag = { flag = eotg_fringe_0014_recent  years = 2 }
                    trigger_event = { id = eotg_fringe.0014  days = 3 }
                }
            }

            # The Fringe Hungers — fires after 3+ peace years (40% chance, 2yr cooldown)
            if = {
                limit = {
                    has_variable = eotg_fringe_peace_years
                    var:eotg_fringe_peace_years >= 3
                    NOT = { has_character_flag = eotg_fringe_0016_recent }
                }
                random = {
                    chance = 40
                    add_character_flag = { flag = eotg_fringe_0016_recent  years = 2 }
                    trigger_event = { id = eotg_fringe.0016  days = 5 }
                }
            }

            # Raiders Desert the Cause — 5+ peace years + broke (35%, 2yr cooldown)
            if = {
                limit = {
                    has_variable = eotg_fringe_peace_years
                    var:eotg_fringe_peace_years >= 5
                    gold < 100
                    NOT = { has_character_flag = eotg_fringe_0017_recent }
                }
                random = {
                    chance = 35
                    add_character_flag = { flag = eotg_fringe_0017_recent  years = 2 }
                    trigger_event = { id = eotg_fringe.0017  days = 7 }
                }
            }

            # The Realm Fractures — large realm + low Grip (30%, 3yr cooldown)
            if = {
                limit = {
                    realm_size >= 20
                    eotg_st_fringe_grip_low = yes
                    NOT = { has_character_flag = eotg_fringe_0019_recent }
                }
                random = {
                    chance = 30
                    add_character_flag = { flag = eotg_fringe_0019_recent  years = 3 }
                    trigger_event = { id = eotg_fringe.0019  days = 5 }
                }
            }

            # Refugees Join the Fringe — fires when overlord is at peace (25%, 2yr cooldown)
            if = {
                limit = {
                    NOT = { is_at_war = yes }
                    NOT = { has_character_flag = eotg_fringe_0023_recent }
                }
                random = {
                    chance = 25
                    add_character_flag = { flag = eotg_fringe_0023_recent  years = 2 }
                    trigger_event = { id = eotg_fringe.0023  days = 5 }
                }
            }

            # The Raiders Grow Fat — 3+ peace years + wealthy + sized realm (20%, 3yr cooldown)
            if = {
                limit = {
                    has_variable = eotg_fringe_peace_years
                    var:eotg_fringe_peace_years >= 3
                    gold > 200
                    realm_size >= 8
                    NOT = { has_character_flag = eotg_fringe_0024_recent }
                }
                random = {
                    chance = 20
                    add_character_flag = { flag = eotg_fringe_0024_recent  years = 3 }
                    trigger_event = { id = eotg_fringe.0024  days = 5 }
                }
            }

            # Discipline Over Chaos — high martial (15%, 4yr cooldown)
            if = {
                limit = {
                    martial >= 14
                    realm_size >= 8
                    NOT = { has_character_flag = eotg_fringe_0025_recent }
                }
                random = {
                    chance = 15
                    add_character_flag = { flag = eotg_fringe_0025_recent  years = 4 }
                    trigger_event = { id = eotg_fringe.0025  days = 5 }
                }
            }

            # A Warlord of Legend Emerges — rare (10%, 5yr cooldown)
            if = {
                limit = {
                    any_vassal = {
                        eotg_st_is_fringe_captain = yes
                        prowess >= 16
                    }
                    NOT = { has_character_flag = eotg_fringe_0027_recent }
                }
                random = {
                    chance = 10
                    add_character_flag = { flag = eotg_fringe_0027_recent  years = 5 }
                    trigger_event = { id = eotg_fringe.0027  days = 5 }
                }
            }

            # Songs of the Fringe — prestigious overlord (15%, 4yr cooldown)
            if = {
                limit = {
                    prestige >= 1000
                    NOT = { has_character_flag = eotg_fringe_0028_recent }
                }
                random = {
                    chance = 15
                    add_character_flag = { flag = eotg_fringe_0028_recent  years = 4 }
                    trigger_event = { id = eotg_fringe.0028  days = 7 }
                }
            }
            # Captain Demands Equipment — any captain vassal present (35%, 4yr cooldown)
            if = {
                limit = {
                    any_vassal = { eotg_st_is_fringe_captain = yes }
                    NOT = { has_character_flag = eotg_fringe_0030_recent }
                }
                random = {
                    chance = 35
                    add_character_flag = { flag = eotg_fringe_0030_recent  years = 4 }
                    trigger_event = { id = eotg_fringe.0030  days = 7 }
                }
            }
            # Crew Demands Their Due — gold >= 100 (30%, 4yr cooldown)
            if = {
                limit = {
                    gold >= 100
                    any_vassal = { is_alive = yes }
                    NOT = { has_character_flag = eotg_fringe_0031_recent }
                }
                random = {
                    chance = 30
                    add_character_flag = { flag = eotg_fringe_0031_recent  years = 4 }
                    trigger_event = { id = eotg_fringe.0031  days = 5 }
                }
            }
            # Captain Challenges Overlord — captain with prowess >= 12 (20%, 5yr cooldown)
            if = {
                limit = {
                    any_vassal = {
                        eotg_st_is_fringe_captain = yes
                        prowess >= 12
                    }
                    NOT = { has_character_flag = eotg_fringe_0032_recent }
                }
                random = {
                    chance = 20
                    add_character_flag = { flag = eotg_fringe_0032_recent  years = 5 }
                    trigger_event = { id = eotg_fringe.0032  days = 5 }
                }
            }
        }
    }
}
