# On_actions for Frontier Systems v1 (docs/specs/frontier_v1.md §3.6, §6.1, §7).
#
# Vanilla hooks are extended ADDITIVELY only (CLAUDE.md invariant 4): an
# `on_actions = { }` list in this file, never a top-level effect/trigger on a
# vanilla hook. The yearly hook is yearly_playable_pulse (on_yearly_playable is
# dead: catalog §4 lesson 13).
# -----------------------------------------------------------------------------

yearly_playable_pulse = {
    on_actions = {
        eotg_on_yearly_frontier_tick
    }
}

# The yearly Frontier tick (spec §6.1). Root = a playable character; each of
# its held Regions in the Frontier state is ticked once a year.
# COOLDOWN AUTHORITY: the calendar-year marker eotg_frontier_tick_year (on the
# county), set here and nowhere else: exactly one tick per in-game year, even
# when the Region changes holder (verification Q9, orchestrator ruling). No
# event trigger reads it.
# every_held_title with tier = tier_county, and yearly_playable_pulse reaching
# every county holder: confirmed (verification V13).
# UNVERIFIED-VANILLA: current_year as a value in set_variable / a comparison.
eotg_on_yearly_frontier_tick = {
    effect = {
        every_held_title = {
            limit = {
                tier = tier_county
                eotg_frontier_is_frontier = yes
                OR = {
                    NOT = { has_variable = eotg_frontier_tick_year }
                    var:eotg_frontier_tick_year < current_year
                }
            }
            set_variable = { name = eotg_frontier_tick_year  value = current_year }
            eotg_frontier_tick_effect = yes
        }
    }
}

# Vanilla hook, extended additively only (invariant 4). Root = the dying
# character, still alive inside the hook; vanilla reads primary_heir only here
# (death.txt:31-51), so the sponsor hand-off runs now (verification E2).
on_death = {
    on_actions = {
        eotg_frontier_on_sponsor_death
    }
}

eotg_frontier_on_sponsor_death = {
    trigger = {
        has_variable = eotg_frontier_sponsoring
    }
    effect = {
        eotg_frontier_on_sponsor_death_effect = yes
    }
}
