# -- eotg_unclaimed_on_actions.txt --------------------------------------------
# Unclaimed Regions (docs/specs/frontier_unclaimed_regions.md §9).
#
# Vanilla hooks are extended ADDITIVELY only (CLAUDE.md invariant 4):
# `on_actions = { }` lists here, never a top-level effect/trigger on a vanilla
# hook. on_title_gain and on_death fire for every character in the game, so
# each handler's trigger is the cheap check first (spec §5.5).
# No cooldowns are duplicated in any trigger: the interaction's `cooldown`, the
# AI gap global and the sweep's yearly cadence each own their own pacing (§9.1).
# -----------------------------------------------------------------------------

# Runs before the lobby (game_start.txt:2509), so the split is done before
# anyone picks a character (spec §5.1).
on_game_start = {
    on_actions = {
        eotg_unclaimed_on_game_start
    }
}

eotg_unclaimed_on_game_start = {
    effect = {
        eotg_unclaimed_game_start_effect = yes
    }
}

# The catch-all (spec §8.5): root = the new holder, scope:title,
# scope:previous_holder (title_on_actions.txt:229-233). Whatever moved the
# county (the interaction, a vanilla event, the console, a future system),
# the bookkeeping runs here, once.
# UNVERIFIED-VANILLA (V-U9): on_title_gain fires for scripted transfers
#   (change_title_holder) as well as for the engine's own.
on_title_gain = {
    on_actions = {
        eotg_unclaimed_on_title_gain
    }
}

eotg_unclaimed_on_title_gain = {
    trigger = {
        scope:title = { has_variable = eotg_unclaimed_county }
    }
    effect = {
        if = {
            # The §5.3 fallback: the engine gave the county to a generated
            # successor (folk through the government's template); adopt it.
            # Script never turns a REAL character into a placeholder.
            limit = {
                eotg_is_unclaimed_folk = yes
                eotg_is_unclaimed_holder = no
            }
            change_government = eotg_unclaimed_government
            eotg_unclaimed_track_holder_effect = yes
        }
        else_if = {
            limit = { eotg_is_unclaimed_folk = no }
            scope:title = {
                eotg_unclaimed_on_claimed_effect = {
                    CLAIMANT = root
                    OLD = scope:previous_holder
                }
            }
        }
    }
}

# Root is just about to die (death.txt:1). A dying placeholder's counties get
# new placeholders while it still holds them (spec §5.3).
on_death = {
    on_actions = {
        eotg_unclaimed_on_death
    }
}

eotg_unclaimed_on_death = {
    trigger = {
        eotg_is_unclaimed_holder = yes
    }
    effect = {
        eotg_unclaimed_on_holder_death_effect = yes
    }
}

# Every 1 January, no root (yearly_on_actions.txt:1-3): the safety net.
# UNVERIFIED-VANILLA: yearly_global_pulse is the hook's name in 1.20.0.3
#   (spec §5.3 cites it; pitfalls §12: a hook name from .info prose can be dead)
yearly_global_pulse = {
    on_actions = {
        eotg_unclaimed_on_yearly_sweep
    }
}

eotg_unclaimed_on_yearly_sweep = {
    effect = {
        eotg_unclaimed_sweep_effect = yes
    }
}

# -- The pulse probe (frontier_v4.md A6, V-U1a; debug tooling, no gating) ----
# Vanilla's random yearly event pool (yearly_on_actions.txt:3018-3021) is
# gated by basic_is_valid_for_yearly_events_trigger; the parent's trigger
# gates this child too, so a count above 0 after one game year in
# -debug_mode means placeholders get vanilla yearly events (then build M2).
# The debug readout toasts the count. debug_only as a trigger:
# game_start.txt:7288.
random_yearly_playable_pulse = {
    on_actions = {
        eotg_unclaimed_debug_pulse_probe
    }
}

eotg_unclaimed_debug_pulse_probe = {
    trigger = {
        debug_only = yes
        eotg_is_unclaimed_holder = yes
    }
    effect = {
        if = {
            limit = { NOT = { has_global_variable = eotg_unclaimed_debug_pulsed } }
            set_global_variable = { name = eotg_unclaimed_debug_pulsed  value = 0 }
        }
        change_global_variable = { name = eotg_unclaimed_debug_pulsed  add = 1 }
    }
}

# -- Integration hooks (spec §9.3) -------------------------------------------
# Empty on purpose. Later systems (corporations claiming for a backer,
# exploration revealing claimable space) extend them additively from their
# own files. Fired with trigger_event = { on_action = ... }, as Frontier's.
#   eotg_unclaimed_on_claimed:  root = the claimant;
#     scope:eotg_unclaimed_target (the county), scope:eotg_unclaimed_old_holder
#     (the placeholder, when known)
#   eotg_unclaimed_on_released: root = the new placeholder;
#     scope:eotg_unclaimed_target, plus scope:eotg_unclaimed_old_holder when the
#     county had a real holder

eotg_unclaimed_on_claimed = {
}

eotg_unclaimed_on_released = {
}

# -- Frontier hooks (spec §6; Batch 2) -----------------------------------------
# Frontier's own integration hooks, extended additively from this file (the
# pattern eotg_frontier_on_actions.txt documents). Fired by
# eotg_frontier_abandon_effect / eotg_frontier_settle_effect with root = the
# Region's holder and scope:eotg_frontier_county set (frontier_v1.md §7).
# TEST-IN-GAME (Frontier V12): the saved scopes reach these on_actions.

eotg_frontier_on_abandoned = {
    on_actions = {
        eotg_unclaimed_on_frontier_abandoned
    }
}

eotg_frontier_on_settled = {
    on_actions = {
        eotg_unclaimed_on_frontier_settled
    }
}

# An abandoned Frontier that came out of unclaimed space goes back to it
# (not the holder's capital Region; failed and voluntary alike, owner UQ5).
eotg_unclaimed_on_frontier_abandoned = {
    trigger = {
        exists = scope:eotg_frontier_county
        scope:eotg_frontier_county = { has_variable = eotg_unclaimed_claimed }
    }
    effect = {
        scope:eotg_frontier_county = { eotg_unclaimed_on_abandoned_effect = yes }
    }
}

# A settled Region is ordinary land and never goes back (spec §6.2).
eotg_unclaimed_on_frontier_settled = {
    trigger = {
        exists = scope:eotg_frontier_county
        scope:eotg_frontier_county = { has_variable = eotg_unclaimed_claimed }
    }
    effect = {
        scope:eotg_frontier_county = { remove_variable = eotg_unclaimed_claimed }
    }
}
