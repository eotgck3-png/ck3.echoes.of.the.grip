# ── eotg_augmentation_on_actions.txt ─────────────────────────────────────────
# Yearly hooks for the Cybernetic Augmentation system.
#
# Signature resource: character variable eotg_fracture_risk (0-100, hidden).
# Set at initiation, carried through Augmented/Enhanced, accumulates yearly
# while Overclocked, and triggers the Neural Cascade at the threshold in
# eotg_neurofracture_threshold_met. After the cascade it is reset to 0 and
# then read here as "episode pressure": three hidden bands (Flicker < 30,
# Fracture 30-59, Storm >= 60) each select their own event pool, and >= 95
# fires the terminal event (cybernetics_v2_phase3.md §1).
#
# Tier state: the XP on trait eotg_cybernetics (exactly 0 / 50 / 100), read
# through eotg_is_aug_tier1/2/3 (docs/specs/cybernetics_track.md).
#
# Wired into yearly_playable_pulse (fires once per year for count+ characters,
# on a per-character date). v1 hooked `on_yearly_playable`, a stale name that
# survives only in vanilla's _on_actions.info; the engine never fires it, so
# no augmentation content ever ran in v1.
#
# COOLDOWN AUTHORITY lives here and only here (catalog §4 lesson 5): each
# branch sets the cooldown flag at the moment it fires the event. The events
# never check or set these flags. Flags:
#   eotg_flag_aug_event_cooldown  — initiation + Augmented tier
#   eotg_flag_enh_event_cooldown  — Enhanced tier
#   eotg_flag_oc_event_cooldown   — Overclocked tier
#   eotg_flag_nf_event_cooldown   — Neurofractured tier
#   eotg_flag_aug_terminal_cooldown — The Cascade (eotg_fracture.027), 11 months
#   eotg_flag_aug_scramble_cooldown — Scrambled (eotg_fracture.018), 5 years
#   eotg_flag_aug_last_lucid      — The Last Lucid Moment (eotg_fracture.026), 5 years
#   eotg_flag_tier1_vassal_suspicion_fired — one-time (eotg_aug_tier1.003)
#   eotg_flag_aug_child_offer_cooldown — Sickly Child offer, 5 years (init.014)
#   eotg_flag_aug_birth_cooldown  — Absent at the Birth (tier2.010), 3 years,
#     per parent, set in eotg_on_birth_aug_parent_check
#   eotg_flag_aug_vendor_done     — one-time (The Bidding War, tier2.013)
#   eotg_flag_aug_first_contact_done — one-time (First Contact, tier3.012)
#   eotg_flag_aug_nr_cooldown     — on the LIEGE: one non-ruler event a year
#     (eotg_on_yearly_aug_nonruler_check; nr.006 ignores it)
#   eotg_flag_aug_intervention_refused — not a cooldown: tier3.020's 2-year
#     "refused" state (T3), read here so the Intervention does not repeat
#     while it holds
#   eotg_flag_aug_backalley_retry — not a cooldown: a one-shot pass that lets
#     init.010 fire past eotg_flag_aug_event_cooldown (set by init.012.b)
#   eotg_flag_aug_repair_cooldown — repair offers (proc.003 / .004), 3 years,
#     set in eotg_on_trait_gained_aug_repair (procedures spec §4.9)
#   eotg_flag_aug_seamless_cooldown — the Seamless pool (end.040-.042 and the
#     Heir's Arc start), 2 years, set in eotg_on_yearly_aug_seamless_check
#   eotg_flag_aug_phantom_done — one-shot per exit (Phantom Static,
#     eotg_aug_proc.020), set in eotg_on_yearly_aug_former_check; cleared by
#     eotg_aug_mark_former_effect on the next exit
#
# Each tier rolls one weighted random_list per pulse (weights = v1 chances),
# with a "nothing" branch. v1 used independent ifs whose cooldown was set in
# the event's immediate, days later, so up to seven events could fire in the
# same pulse; random_list removes that and the first-listed bias.
#
# Overclocked risk accrual (cybernetics_v2 B2): base 12/year, + 8 per vanilla
# stress_level (1-3), paranoid +12, at war +10, hidden flaw +4, vendor
# safe -3 / bold +3. Threshold: 80, always (eotg_neurofracture_threshold_met).
# Neurofractured drift (B1; balance spec §5.5): eotg_aug_nf_drift_value, +8/year,
# + 4 per stress_level, sedated -6/-3/0 by course, restrained -4, floor 2.
# Runs every pulse, outside the episode cooldown.
# Cooldowns (B4): Overclocked and Neurofractured 11 months, i.e. expired by the
# next pulse; Court Massacre 23 months. Both are stated as "every pulse /
# every other pulse" without depending on same-day expiry order.
# ─────────────────────────────────────────────────────────────────────────────

# Initiation checks — offer augmentation to eligible rulers.
# One weighted random_list per pulse, the same shape as the tier checks below
# (orchestrator ruling S2, Phase 2 follow-up). The former if/else_if priority
# chain let a persistent condition (physician access, a physical loss,
# infirmity) claim the pulse and then fire nothing, starving every later
# branch. Each entry keeps its old limit as `trigger` and its old chance as the
# weight (docs/specs/cybernetics_v2_phase2.md §1.1 order), and sets
# eotg_flag_aug_event_cooldown inside the entry. Cooldown authority stays here.
# Declined offers (init.006.b, init.007.b set 10-year flags) weigh 0.25: an
# on_action weight, not an event trigger (lesson 5).
# Two blocks sit outside the list:
#   - the back-alley retry (eotg_flag_aug_backalley_retry, set by init.012.b),
#     which skips the cooldown on purpose (spec: "regardless of cooldown");
#   - the Sickly Child offer, which has its own 5-year cooldown and does not
#     require the ruler to be unaugmented (the child must be).
eotg_on_yearly_aug_initiation_check = {
    effect = {
        # Back-Alley retry: a brave ruler who lost the implant looks again.
        # Spends the retry flag; still sets the normal cooldown.
        if = {
            limit = {
                eotg_can_receive_augmented = yes
                has_character_flag = eotg_flag_aug_backalley_retry
            }
            random = {
                chance = 50
                remove_character_flag = eotg_flag_aug_backalley_retry
                add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                trigger_event = { id = eotg_aug_init.010  days = { 1 15 } }
            }
        }
        else_if = {
            limit = {
                eotg_can_receive_augmented = yes
                NOT = { has_character_flag = eotg_flag_aug_event_cooldown }
            }
            random_list = {
                # Desperation Protocol — at war + wounded
                50 = {
                    trigger = {
                        has_trait = wounded
                        is_at_war = yes
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.004  days = { 1 5 } }
                }
                # The Prosthetic — a lost limb, eye or sight. Balance spec §5.9:
                # also a congenital clubfoot or hunchback (init.006.f), at half
                # weight when that is all there is (mirrors init.006's trigger).
                40 = {
                    trigger = {
                        OR = {
                            eotg_has_physical_loss = yes
                            has_trait = clubfooted
                            has_trait = hunchbacked
                        }
                    }
                    modifier = {
                        factor = 0.25
                        has_character_flag = eotg_flag_aug_declined_prosthetic
                    }
                    modifier = {
                        factor = 0.5
                        eotg_has_physical_loss = no
                    }
                    # Procedures spec §4.5 (G3, "What Was Taken"): someone
                    # who has been out of the system before looks again
                    modifier = {
                        factor = 2
                        has_variable = eotg_aug_former
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.006  days = { 1 10 } }
                }
                # Post-wound offer (wounded ruler not at war)
                40 = {
                    trigger = {
                        has_trait = wounded
                        is_at_war = no
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.001  days = { 1 10 } }
                }
                # The Neural Bridge — incapable or infirm
                30 = {
                    trigger = {
                        OR = {
                            has_trait = incapable
                            has_trait = infirm
                        }
                    }
                    modifier = {
                        factor = 0.25
                        has_character_flag = eotg_flag_aug_declined_bridge
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.007  days = { 1 10 } }
                }
                # A Parent's Hardware — set by eotg_on_death_aug_hardware
                60 = {
                    trigger = {
                        has_variable = eotg_parent_hardware
                        exists = var:eotg_parent_hardware    # mirrors init.013
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.013  days = { 1 20 } }
                }
                # The Physician's Proposal
                12 = {
                    trigger = { eotg_has_physician_access = yes }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.017  days = { 1 20 } }
                }
                # The Cynic's Argument
                20 = {
                    trigger = {
                        has_trait = cynical
                        learning >= 12
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.009  days = { 1 20 } }
                }
                # Corporate offer (wealthy landed ruler). B5 scaled gold (phase 0
                # §1.5); inline-value form as vanilla
                # common/character_interactions/00_tribal_interactions.txt
                10 = {
                    trigger = {
                        is_landed = yes
                        gold >= { value = major_gold_value  multiply = 1.5 }
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.002  days = { 1 15 } }
                }
                # The Patron (Phase 5, thread T2): ambitious or greedy, short of
                # money, landed, no debt running
                15 = {
                    trigger = {
                        OR = {
                            has_trait = ambitious
                            has_trait = greedy
                        }
                        gold < medium_gold_value
                        is_landed = yes
                        eotg_aug_has_patron = no
                        # New beats §5.2 "Once per life": the offer sites own
                        # the cooldown (lesson 5); patron.001 does not read it.
                        NOT = { has_character_flag = eotg_flag_aug_had_patron }
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_patron.001  days = { 1 20 } }
                }
                # Back-Alley Surgery — poor, deceitful or at war
                10 = {
                    trigger = {
                        OR = {
                            gold < medium_gold_value
                            has_trait = deceitful
                            is_at_war = yes
                        }
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.010  days = { 1 15 } }
                }
                # The Duel Shame — weak, and someone strong is watching
                15 = {
                    trigger = {
                        prowess <= 6
                        OR = {
                            any_relation = {
                                type = rival
                                prowess >= 12
                            }
                            is_at_war = yes
                        }
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.016  days = { 1 20 } }
                }
                # Aging ruler with low prowess
                15 = {
                    trigger = {
                        age >= 50
                        prowess <= 8
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_init.003  days = { 1 10 } }
                }
                # A Familiar Change — peer already augmented. Balance spec
                # §5.3(a): the contagion branch, halved for the AI (every
                # augmented vassal or courtier raises it).
                20 = {
                    trigger = {
                        OR = {
                            any_vassal = { has_trait = eotg_cybernetics }
                            any_courtier = { is_alive = yes  has_trait = eotg_cybernetics }
                        }
                    }
                    modifier = { factor = 0.5  is_ai = yes }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 3 }
                    trigger_event = { id = eotg_aug_init.005  days = { 1 20 } }
                }
                # The Arms Race (Phase 5) — two augmented knights, you are not.
                # x1.5 with eotg_flag_aug_want_more (Phase 6, nr.002.d).
                20 = {
                    modifier = {
                        factor = 1.5
                        has_character_flag = eotg_flag_aug_want_more
                    }
                    trigger = {
                        eotg_aug_has_retinue = no
                        # QA round 2 H2: the Program ran and ended (permanent)
                        NOT = { has_character_flag = eotg_flag_aug_retinue_done }
                        any_knight = {
                            count >= 2
                            eotg_is_augmented_any = yes
                        }
                    }
                    add_character_flag = { flag = eotg_flag_aug_event_cooldown  years = 3 }
                    trigger_event = { id = eotg_aug_init.019  days = { 1 20 } }
                }
                # Nothing this year. Balance spec §5.3(a): AI throttle by title
                # tier, counts hardest, kings least (vanilla tiered no-event
                # chances, common/on_action/yearly_on_actions.txt:3022-3030;
                # AI-only weight modifiers, common/on_action/dlc/mpo/
                # mpo_on_actions_2.txt:468-474). Milder than vanilla's 95%
                # for counts: this list creates world state, not flavor.
                100 = {
                    # AI count x6 / duke x3 / king+ x2 (first guesses for the observer run)
                    modifier = { factor = 6  is_ai = yes  highest_held_title_tier < tier_duchy }
                    modifier = { factor = 3  is_ai = yes  highest_held_title_tier = tier_duchy }
                    modifier = { factor = 2  is_ai = yes  highest_held_title_tier >= tier_kingdom }
                }
            }
        }

        # The Sickly Child — outside the unaugmented gate: the ruler may be
        # augmented or not; the child must not be. Own cooldown, set here.
        if = {
            limit = {
                NOT = { has_character_flag = eotg_flag_aug_child_offer_cooldown }
                gold >= minor_gold_value
                any_child = { eotg_aug_sick_child_candidate = yes }
            }
            random = {
                chance = 25
                add_character_flag = { flag = eotg_flag_aug_child_offer_cooldown  years = 5 }
                trigger_event = { id = eotg_aug_init.014  days = { 1 10 } }
            }
        }
    }
}

# QA round 2: a character who starts with eotg_cybernetics without the
# install effect (the ruler designer sells the trait; a history file may set
# it) has no risk variable and no tier synergy. Give them what
# eotg_aug_initiate_effect would have: risk 0 and the synergy modifier. The
# trait test comes first, so the one-off sweep is cheap.
eotg_on_game_start_aug_init = {
    effect = {
        every_living_character = {
            limit = { has_trait = eotg_cybernetics }
            if = {
                limit = { NOT = { has_variable = eotg_fracture_risk } }
                set_variable = { name = eotg_fracture_risk  value = 0 }
                eotg_aug_refresh_synergy_effect = yes
            }
            # Balance spec §5.1 item 3 (ruled 2026-10-04): every character
            # who starts augmented settles for 5 years like any newcomer
            # (eotg_aug_initiate_effect), so HQ1's 10-year minimum holds
            # from game start. Outside the variable test: a history file
            # that set the risk variable still starts settling.
            add_character_flag = { flag = eotg_flag_aug_settling  years = 5 }
        }
    }
}

# Inherited hardware (cybernetics_v2_phase2.md §1.2). Root = the dying
# character (game/common/on_action/death.txt:7). Each living, unaugmented child
# is offered the parent's implants for 5 years (read by initiation branch 5,
# eotg_aug_init.013); a Neurofractured parent also flags the children.
eotg_on_death_aug_hardware = {
    trigger = {
        OR = {
            has_trait = eotg_cybernetics
            has_trait = eotg_neurofractured
        }
    }
    effect = {
        every_child = {
            limit = {
                is_alive = yes
                eotg_is_augmented_any = no
            }
            set_variable = { name = eotg_parent_hardware  value = root  years = 5 }
        }
        if = {
            limit = { has_trait = eotg_neurofractured }
            every_child = {
                limit = { is_alive = yes }
                add_character_flag = { flag = eotg_flag_aug_hardware_fractured  years = 5 }
            }
        }
    }
}

# The Late Grief (cybernetics_v2_phase4.md §3). Root = the dying character
# (game/common/on_action/death.txt:7). Every Enhanced close relative carries
# the loss for a year; eotg_on_yearly_aug_tier2_check reads it (tier2.008).
# A killer is remembered for the vengeful option. every_close_family_member:
# PX effects list (character scope).
# QA round 2 performance guard: on_death runs for every character, so a
# trigger rejects the death first on a plain has_trait test (the cheapest
# check, as eotg_on_death_aug_hardware and the non-ruler pulse lead with
# one) before the effect walks the family again with the XP-based tier test.
eotg_on_death_aug_grief = {
    trigger = {
        any_close_family_member = { has_trait = eotg_cybernetics }
    }
    effect = {
        every_close_family_member = {
            limit = { eotg_is_aug_tier2 = yes }
            set_variable = { name = eotg_aug_grief_for  value = root  years = 1 }
            if = {
                limit = { exists = scope:killer }
                set_variable = { name = eotg_aug_grief_killer  value = scope:killer  years = 1 }
            }
            else = {
                remove_variable = eotg_aug_grief_killer
            }
        }
    }
}

# Absent at the Birth (cybernetics_v2_phase4.md §3). Root = the newborn
# (game/common/on_action/child_birth_on_actions.txt:855: scope:mother,
# scope:father). Each Enhanced or Overclocked parent is checked on its own;
# the 3-year per-parent cooldown lives here (lesson 5).
eotg_on_birth_aug_parent_check = {
    effect = {
        scope:father ?= {
            if = {
                limit = {
                    OR = {
                        eotg_is_aug_tier2 = yes
                        eotg_is_aug_tier3 = yes
                    }
                    NOT = { has_character_flag = eotg_flag_aug_birth_cooldown }
                }
                add_character_flag = { flag = eotg_flag_aug_birth_cooldown  years = 3 }
                root = { save_scope_as = eotg_newborn }
                trigger_event = { id = eotg_aug_tier2.010  days = 3 }
            }
        }
        scope:mother ?= {
            if = {
                limit = {
                    OR = {
                        eotg_is_aug_tier2 = yes
                        eotg_is_aug_tier3 = yes
                    }
                    NOT = { has_character_flag = eotg_flag_aug_birth_cooldown }
                }
                add_character_flag = { flag = eotg_flag_aug_birth_cooldown  years = 3 }
                root = { save_scope_as = eotg_newborn }
                trigger_event = { id = eotg_aug_tier2.010  days = 3 }
            }
        }
    }
}

# The non-ruler lifecycle. Root = a knight or courtier below county tier,
# employed by a count+ liege (`employer`, vanilla event target). Hooked on
# random_yearly_everyone_pulse, so the trigger is ordered cheapest first: the
# trait/knight OR rejects almost every character in the world before any
# scope link is walked.
# Steps (spec §1): 1 the offer (unaugmented knight), 2 progression (tier 1-2),
# 3 accrual and the non-ruler cascade (tier 3; eotg_aug_nr_cascade_effect,
# never eotg_trigger_neurofracture), 4 Neurofractured drift, 5 one liege event.
# COOLDOWN: eotg_flag_aug_nr_cooldown lives on the LIEGE and is set only here
# (lesson 5); no eotg_aug_nr event checks it. nr.006 (the cascade) ignores it.
# Phase 5 hooks read: eotg_aug_is_retinue_knight, eotg_flag_aug_retinue_permanent
# (set on the knights by eotg_aug_retinue.005.a; also honoured on the liege,
# as the Phase 6 spec words it).
# Balance spec §5.7 rates: progression eotg_aug_nr_progress_chance (about
# 10% a year, faster for the Retinue, plus battles); Overclocked accrual 10 a
# year (+6 at stress level 2+); battle accrual in eotg_on_combat_aug_knights.
eotg_on_yearly_aug_nonruler_check = {
    trigger = {
        OR = {
            has_trait = eotg_cybernetics
            has_trait = eotg_neurofractured
            is_knight = yes
        }
        highest_held_title_tier < tier_county
        is_alive = yes
        is_adult = yes
        exists = employer
        employer = { highest_held_title_tier >= tier_county }
    }
    effect = {
        employer = { save_scope_as = eotg_nr_liege }

        # 1. NR-01: an unaugmented knight asks (3%, 6% with want_more)
        if = {
            limit = {
                is_knight = yes
                eotg_is_augmented_any = no
                scope:eotg_nr_liege = {
                    NOT = { has_character_flag = eotg_flag_aug_nr_cooldown }
                    OR = {
                        has_trait = eotg_cybernetics
                        has_character_flag = eotg_flag_aug_want_more
                        any_knight = { eotg_is_augmented_any = yes }
                    }
                }
            }
            random = {
                chance = 3
                modifier = {
                    add = 3
                    scope:eotg_nr_liege = { has_character_flag = eotg_flag_aug_want_more }
                }
                scope:eotg_nr_liege = {
                    add_character_flag = { flag = eotg_flag_aug_nr_cooldown  years = 1 }
                }
                save_scope_as = eotg_champion
                scope:eotg_nr_liege = {
                    trigger_event = { id = eotg_aug_nr.001  days = { 1 30 } }
                }
            }
        }
        # 2-4 only apply to the augmented
        else_if = {
            limit = { eotg_is_augmented_any = yes }

            # 2. Progression. Balance spec §5.7: eotg_aug_nr_progress_chance
            # (base 10; Iron Retinue +5, +5 more permanent; child patient +5;
            # plus the battle bonus, 0-10). The battle bonus is spent on
            # progression. Script-value chance: vanilla
            # events/activities/hunt_activity/hunt_events.txt:4196.
            # A knight's progression is reported to the liege by nr.002
            # (final QA: fired here, so it only ever reports a real
            # progression). The liege cooldown is set here and step 5 is
            # skipped for this character this year.
            if = {
                limit = {
                    OR = {
                        eotg_is_aug_tier1 = yes
                        eotg_is_aug_tier2 = yes
                    }
                }
                random = {
                    chance = eotg_aug_nr_progress_chance
                    if = {
                        limit = { eotg_is_aug_tier1 = yes }
                        eotg_aug_set_integration_effect = { XP = 50 }
                    }
                    else = {
                        eotg_aug_set_integration_effect = { XP = 100 }
                    }
                    remove_variable = eotg_aug_nr_battle_bonus
                    # Your Champion's New Edge
                    if = {
                        limit = {
                            is_knight = yes
                            scope:eotg_nr_liege = {
                                NOT = { has_character_flag = eotg_flag_aug_nr_cooldown }
                            }
                        }
                        scope:eotg_nr_liege = {
                            add_character_flag = { flag = eotg_flag_aug_nr_cooldown  years = 1 }
                        }
                        save_scope_value_as = { name = eotg_nr_reported  value = yes }
                        save_scope_as = eotg_champion
                        scope:eotg_nr_liege = {
                            trigger_event = { id = eotg_aug_nr.002  days = { 1 30 } }
                        }
                    }
                }
            }
            # 3. Accrual at Overclocked; the cascade at 80. Balance spec §5.7:
            # base 6 -> 10 (the +6 at stress level 2+ stays); battles add
            # through eotg_on_combat_aug_knights.
            else_if = {
                limit = { eotg_is_aug_tier3 = yes }
                eotg_add_fracture_risk = { AMOUNT = 10 }
                if = {
                    limit = { stress_level >= 2 }
                    eotg_add_fracture_risk = { AMOUNT = 6 }
                }
                if = {
                    limit = { var:eotg_fracture_risk >= 80 }
                    eotg_aug_nr_cascade_effect = yes
                    save_scope_value_as = { name = eotg_nr_cascaded  value = yes }
                }
            }
            # 4. Drift once fractured
            else_if = {
                limit = { has_trait = eotg_neurofractured }
                eotg_add_fracture_risk = { AMOUNT = 6 }
            }

            # 5. One liege event a year (not in a cascade year: nr.006 is it;
            # not when step 2 already reported a progression: nr.002 is it)
            if = {
                limit = {
                    NOT = { exists = scope:eotg_nr_cascaded }
                    NOT = { exists = scope:eotg_nr_reported }
                    scope:eotg_nr_liege = {
                        NOT = { has_character_flag = eotg_flag_aug_nr_cooldown }
                    }
                }
                random_list = {
                    # Something Is Wrong With Them
                    15 = {
                        trigger = {
                            OR = {
                                eotg_is_aug_tier2 = yes
                                eotg_is_aug_tier3 = yes
                            }
                            has_variable = eotg_fracture_risk
                            var:eotg_fracture_risk >= 30
                        }
                        modifier = {
                            factor = 1.5
                            eotg_aug_is_retinue_knight = yes
                        }
                        scope:eotg_nr_liege = {
                            add_character_flag = { flag = eotg_flag_aug_nr_cooldown  years = 1 }
                        }
                        save_scope_as = eotg_champion
                        scope:eotg_nr_liege = {
                            trigger_event = { id = eotg_aug_nr.003  days = { 1 30 } }
                        }
                    }
                    # The Champion's Mistake
                    20 = {
                        trigger = {
                            is_imprisoned = no
                            OR = {
                                has_trait = eotg_neurofractured
                                AND = {
                                    eotg_is_aug_tier3 = yes
                                    has_variable = eotg_fracture_risk
                                    var:eotg_fracture_risk >= 60
                                }
                            }
                        }
                        modifier = {
                            factor = 1.5
                            eotg_aug_is_retinue_knight = yes
                        }
                        scope:eotg_nr_liege = {
                            add_character_flag = { flag = eotg_flag_aug_nr_cooldown  years = 1 }
                        }
                        save_scope_as = eotg_champion
                        scope:eotg_nr_liege = {
                            trigger_event = { id = eotg_aug_nr.004  days = { 1 30 } }
                        }
                    }
                    # The Familiar Change (courtiers)
                    10 = {
                        trigger = {
                            is_knight = no
                            OR = {
                                eotg_is_aug_tier2 = yes
                                eotg_is_aug_tier3 = yes
                                has_trait = eotg_neurofractured
                            }
                        }
                        modifier = {
                            factor = 1.5
                            eotg_aug_is_retinue_knight = yes
                        }
                        scope:eotg_nr_liege = {
                            add_character_flag = { flag = eotg_flag_aug_nr_cooldown  years = 1 }
                        }
                        save_scope_as = eotg_champion
                        scope:eotg_nr_liege = {

# Battle accrual for augmented knights (balance spec §5.7). Root = a combat
# side (on_combat_end_winner / _loser, game/common/on_action/
# combat_on_actions.txt:12-14, :523-525). every_side_knight: vanilla, same
# hook, combat_on_actions.txt:190 (PX effects.log: combat_side scope).
# Non-rulers only (below county tier: rulers accrue +10 a year at war in the
# Overclocked check). Tier 1-2: +2 battle bonus (clamped 0-10), read by
# eotg_aug_nr_progress_chance and spent on progression. Overclocked and
# Neurofractured: risk +2 (eotg_add_fracture_risk; the Seamless are skipped
# there). The limit leads with the trait test: this runs on every battle.
# No event fires, so no cooldown is involved.
eotg_on_combat_aug_knights = {
    effect = {
        every_side_knight = {
            limit = {
                OR = {
                    has_trait = eotg_cybernetics
                    has_trait = eotg_neurofractured
                }
                highest_held_title_tier < tier_county
            }
            if = {
                limit = {
                    OR = {
                        eotg_is_aug_tier1 = yes
                        eotg_is_aug_tier2 = yes
                    }
                }
                change_variable = { name = eotg_aug_nr_battle_bonus  add = 2 }
                clamp_variable = { name = eotg_aug_nr_battle_bonus  min = 0  max = 10 }
            }
            else = {
                eotg_add_fracture_risk = { AMOUNT = 2 }
            }
        }
    }
}

# Repairs (§4.9). Hooked on on_trait_gained (root = the character, scope:trait
# = the trait gained; game/common/on_action/traits_on_actions.txt:3-13).
# A permanent loss or a near-death wound on an augmented ruler (players at any
# tier, the AI at count+) offers a cybernetic repair: eotg_aug_proc.004 for
# the Seamless (the system orders the part itself), else eotg_aug_proc.003.
# wounded_1/2 are excluded: routine battle wounds, which vanilla's own wound
# treatment covers (decide_who_picks_wound_treatment_effect,
# common/scripted_effects/20_health_effects.txt:3730).
# COOLDOWN AUTHORITY: eotg_flag_aug_repair_cooldown (3 years) is set here and
# only here; proc.003 / .004 never read it. eotg_flag_aug_procedure_injury
# (30 days, set by eotg_aug_procedure_apply_effect before a severe outcome's
# trait) keeps the table's own damage from being offered a repair.
# The trait test leads: this runs on every trait gained in the world.
# In-game check (spec §4.9, human): whether change_trait_rank (used by the
# wound effects) fires on_trait_gained for wounded_3, and that characters
# who start with a listed trait get no offer on day 1. No guard is guessed.
eotg_on_trait_gained_aug_repair = {
    trigger = {
        OR = {
            scope:trait = trait:wounded_3
            scope:trait = trait:maimed
            scope:trait = trait:one_legged
            scope:trait = trait:one_eyed
            scope:trait = trait:blind
            scope:trait = trait:disfigured
        }
        is_alive = yes
        eotg_is_augmented_any = yes
        OR = {
            is_ai = no
            highest_held_title_tier >= tier_county
        }
        NOT = { has_character_flag = eotg_flag_aug_repair_cooldown }
        NOT = { has_character_flag = eotg_flag_aug_procedure_injury }
    }
    effect = {
        add_character_flag = { flag = eotg_flag_aug_repair_cooldown  years = 3 }
        # What is to be repaired (read by eotg_aug_has_repair_injury and
        # eotg_aug_repair_injury_effect)
        if = {
            limit = { scope:trait = trait:wounded_3 }
            set_variable = { name = eotg_aug_repair_injury  value = flag:wounded  days = 365 }
        }
        else_if = {
            limit = { scope:trait = trait:maimed }
            set_variable = { name = eotg_aug_repair_injury  value = flag:maimed  days = 365 }
        }
        else_if = {
            limit = { scope:trait = trait:one_legged }
            set_variable = { name = eotg_aug_repair_injury  value = flag:one_legged  days = 365 }
        }
        else_if = {
            limit = { scope:trait = trait:one_eyed }
            set_variable = { name = eotg_aug_repair_injury  value = flag:one_eyed  days = 365 }
        }
        else_if = {
            limit = { scope:trait = trait:blind }
            set_variable = { name = eotg_aug_repair_injury  value = flag:blind  days = 365 }
        }
        else = {
            set_variable = { name = eotg_aug_repair_injury  value = flag:disfigured  days = 365 }
        }
        if = {
            limit = { has_trait = eotg_total_integration }
            trigger_event = { id = eotg_aug_proc.004  days = { 7 30 } }
        }
        else = {
            trigger_event = { id = eotg_aug_proc.003  days = { 7 30 } }
        }
    }
}

# the year of settling is over, someone who has left the system has a 35%
# chance a year of one Phantom Static (eotg_aug_proc.020) per exit.
# COOLDOWN AUTHORITY: eotg_flag_aug_phantom_done is set here and only here

        }

# The Seamless life (§4.11, G2). On yearly_playable_pulse, after the
# Neurofractured check. eotg_fracture_risk is gone at Total Integration, so
# this pool reads the Seamless resource, eotg_aug_residue (0-3, hidden; it
# only falls). One weighted list a pulse, the tier checks' shape.
# COOLDOWN AUTHORITY: eotg_flag_aug_seamless_cooldown (2 years) is set here
# and only here, in the branch that fires; end.040-.042 never read it.
# The Heir's Arc branch starts eotg_story_aug_heir_arc as the Overclocked
# check does (same guards: ruler-only, once, eotg_flag_aug_heir_arc_done) and
# jumps it to stage 3, the same jump eotg_aug_total_integration_effect makes:
# the story's next tick fires heir.004 (the Choice), which reads a Seamless
# owner (desc_seamless). No new event.
# Nothing-weight AI throttle as the Neurofractured pools (balance spec
# §5.10(a): count x2.5, duke x1.75, king+ x1.25).
eotg_on_yearly_aug_seamless_check = {
    trigger = {
        has_trait = eotg_total_integration
    }
    effect = {
        if = {
            limit = { NOT = { has_character_flag = eotg_flag_aug_seamless_cooldown } }
            random_list = {
                # The Ledger Balances
                30 = {
                    add_character_flag = { flag = eotg_flag_aug_seamless_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_end.040  days = { 1 30 } }
                }
                # A Resignation (mirrors end.041's trigger: never the spouse
                # or close family)
                25 = {
                    trigger = {
                        any_councillor = {
                            is_landed = no
                            NOT = { this = root }
                            NOT = { is_spouse_of = root }
                            NOT = { is_close_family_of = root }
                        }
                    }
                    add_character_flag = { flag = eotg_flag_aug_seamless_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_end.041  days = { 1 30 } }
                }
                # A Flicker (mirrors end.042's trigger)
                10 = {
                    trigger = {
                        has_variable = eotg_aug_residue
                        var:eotg_aug_residue >= 1
                    }
                    add_character_flag = { flag = eotg_flag_aug_seamless_cooldown  years = 2 }
                    trigger_event = { id = eotg_aug_end.042  days = { 1 30 } }
                }
                # The heir confronts what is in the chair (the Heir's Arc,
                # straight to the Choice)
                20 = {
                    trigger = {
                        is_landed = yes
                        eotg_aug_has_heir_arc = no
                        NOT = { has_character_flag = eotg_flag_aug_heir_arc_done }
                        primary_heir ?= {
                            is_alive = yes
                            is_adult = yes
                        }
                    }
                    add_character_flag = { flag = eotg_flag_aug_seamless_cooldown  years = 2 }
                    # The new story by saved scope, not a later search
                    # (vanilla scripted_effect
                    # ep3_story_cycle_admin_eunuch_create_story_effect,
                    # events/dlc/ep3/ep3_story_cycle_admin_eunuch_events.txt:82-89)
                    create_story = {
                        type = eotg_story_aug_heir_arc
                        save_scope_as = eotg_seamless_heir_arc
                    }
                    scope:eotg_seamless_heir_arc = {
                        set_variable = { name = eotg_stage  value = 3 }
                    }
                }
                # Nothing this year
                100 = {
                    modifier = { factor = 2.5  is_ai = yes  highest_held_title_tier < tier_duchy }
                    modifier = { factor = 1.75  is_ai = yes  highest_held_title_tier = tier_duchy }
                    modifier = { factor = 1.25  is_ai = yes  highest_held_title_tier >= tier_kingdom }
                }
            }
        }
    }
}
