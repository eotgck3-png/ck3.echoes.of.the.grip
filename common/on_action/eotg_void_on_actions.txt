# ── eotg_void_on_actions.txt ─────────────────────────────────────────────────
# Yearly hooks for the Void Corruption system.
#
# The parent hook eotg_on_yearly_void_check is registered in eotg_on_actions.txt
# (inside on_yearly_playable) and chains to the four on_actions below. It is
# NOT redefined here — redefining it would collide silently.
#
# Ordering matters. Accrual runs first so that every stage gate below reads the
# same year's exposure value, and so the Hollowing threshold is caught in the
# year it is crossed rather than the year after.
#
# All event selection is gated behind eotg_flag_void_event_cooldown (set by each
# event's own immediate block) so the Void never crowds out government content.
# NOTE: no event's own `trigger` re-checks that cooldown flag — doing so is the
# self-cancellation fault that silently killed 40 events in this mod before.
# ─────────────────────────────────────────────────────────────────────────────

# ── 1. Accrual ────────────────────────────────────────────────────────────────

eotg_on_yearly_void_accrual = {
    effect = {
        if = {
            limit = { eotg_st_void_susceptible = yes }
            eotg_se_void_yearly_accrual = yes
        }
    }
}


# ── 2. Contact — seeding the unconnected ──────────────────────────────────────
# The Grip-era rift scars are re-activating across 850–866 AG. This is the only
# route into the system for a ruler with no bloodline, cult or proximity tie.

eotg_on_yearly_void_contact = {
    effect = {
        if = {
            limit = {
                is_landed = yes
                is_alive = yes
                NOT = { has_character_flag = eotg_flag_void_seeded }
                NOT = { has_trait = eotg_void_touched }
                NOT = { has_trait = eotg_void_hollowed }
                eotg_st_void_event_ready = yes
            }
            random = {
                chance = 4
                trigger_event = { id = eotg_void.0001  days = { 5 40 } }
            }
        }
    }
}


# ── 3. Whispers and the Bargain ───────────────────────────────────────────────

eotg_on_yearly_void_whispers = {
    effect = {
        # Saturation — refusal has stopped being a decision. Checked first so it
        # takes precedence over the ordinary whisper events in the same year.
        if = {
            limit = {
                eotg_st_void_saturated = yes
                eotg_st_void_event_ready = yes
            }
            random = {
                chance = 75
                trigger_event = { id = eotg_void.0011  days = { 1 20 } }
            }
        }
        # The Bargain — the pivotal event of the system.
        else_if = {
            limit = {
                eotg_st_void_bargain_ready = yes
                eotg_st_void_event_ready = yes
            }
            random = {
                chance = 50
                trigger_event = { id = eotg_void.0010  days = { 5 30 } }
            }
        }
        # Ordinary whisper-stage flavor.
        else_if = {
            limit = {
                eotg_st_void_whisper_stage = yes
                eotg_st_void_event_ready = yes
            }
            random_list = {
                40 = { trigger_event = { id = eotg_void.0002  days = { 5 40 } } }
                35 = { trigger_event = { id = eotg_void.0003  days = { 5 40 } } }
                25 = { }
            }
        }

        # The Heretic at the Gate — independent of the chain above. A Voidwalker
        # only approaches someone who already has a reason to listen, and only
        # if one is not already sheltered here.
        if = {
            limit = {
                eotg_st_void_has_exposure = yes
                var:eotg_void_exposure >= 20
                eotg_st_void_has_voidwalker = no
                NOT = { has_trait = eotg_void_hollowed }
                eotg_st_void_event_ready = yes
            }
            random = {
                chance = 12
                trigger_event = { id = eotg_void.0004  days = { 10 60 } }
            }
        }
    }
}


# ── 4. Living with it, and the Hollowing ──────────────────────────────────────

eotg_on_yearly_void_touched = {
    effect = {
        if = {
            limit = { has_trait = eotg_void_touched }

            # Terminal threshold — deterministic, and checked before flavor so a
            # ruler cannot draw a flavor event in the year they Hollow.
            if = {
                limit = { eotg_st_void_hollowing_ready = yes }
                trigger_event = { id = eotg_void.0030  days = { 1 10 } }
            }
            else_if = {
                limit = { eotg_st_void_event_ready = yes }
                random_list = {
                    20 = {
                        trigger = { is_at_war = yes }
                        trigger_event = { id = eotg_void.0020  days = { 5 30 } }
                    }
                    15 = {
                        trigger = {
                            any_child = {
                                is_alive = yes
                                age < 12
                            }
                        }
                        trigger_event = { id = eotg_void.0021  days = { 5 30 } }
                    }
                    20 = {
                        trigger = {
                            any_courtier = {
                                is_alive = yes
                                has_trait = zealous
                            }
                        }
                        trigger_event = { id = eotg_void.0022  days = { 5 30 } }
                    }
                    20 = { trigger_event = { id = eotg_void.0023  days = { 5 30 } } }
                    15 = {
                        trigger = { age >= 60 }
                        trigger_event = { id = eotg_void.0024  days = { 5 30 } }
                    }
                    20 = {
                        trigger = {
                            any_vassal = {
                                is_alive = yes
                                opinion = { target = root  value < -20 }
                            }
                        }
                        trigger_event = { id = eotg_void.0025  days = { 5 30 } }
                    }
                    30 = { }
                }
            }
        }
    }
}


# ── 5. After the Hollowing ────────────────────────────────────────────────────

eotg_on_yearly_void_hollowed = {
    effect = {
        if = {
            limit = {
                has_trait = eotg_void_hollowed
                eotg_st_void_event_ready = yes
            }
            random_list = {
                45 = { trigger_event = { id = eotg_void.0031  days = { 5 40 } } }
                15 = { trigger_event = { id = eotg_void.0032  days = { 5 40 } } }
                40 = { }
            }
        }
    }
}
