# -- eotg_void_traits.txt -----------------------------------------------------
# Traits for the Void Corruption system.
#
# The existing eotg_void_touched (eotg_traits.txt) is the middle stage and is
# NOT redefined here. This file adds the two endpoints of the ladder:
#
#   eotg_void_defiant   -- refused Orrin's offer and held. Warded, respected.
#   eotg_void_hollowed  -- terminal. Enormously capable, barely a person.
#
# Icons: gfx/interface/icons/traits/ (see docs/art_needed.txt)
# -----------------------------------------------------------------------------

# eotg_void_defiant
# Awarded only by refusing the Bargain at real cost. The refusal itself is the
# ward -- Voidwalker doctrine holds that Orrin cannot take what is given back.
eotg_void_defiant = {
    # icon = eotg_void_defiant

    martial = 1
    learning = 1
    health = 0.25

    monthly_piety_gain_mult = 0.15
    stress_gain_mult = -0.10
    general_opinion = 5

    desc = eotg_void_defiant_desc

    flag = eotg_flag_void_defiance
}

# eotg_void_hollowed
# The terminal stage. What is left is precise, tireless and unreachable.
# Deliberately double-edged: the strongest stat line in the mod, attached to
# the worst social position in the mod.
eotg_void_hollowed = {
    # icon = eotg_void_hollowed

    intrigue = 4
    learning = 3
    prowess = 4
    diplomacy = -5
    health = -1

    monthly_piety_gain_mult = -0.35
    dread_baseline_add = 25
    general_opinion = -20
    attraction_opinion = -30
    fertility = -0.5
    stress_gain_mult = -0.50

    desc = eotg_void_hollowed_desc

    flag = eotg_flag_void_corruption
}
