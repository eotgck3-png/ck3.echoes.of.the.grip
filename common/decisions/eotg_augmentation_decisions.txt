# ── eotg_augmentation_decisions.txt ─────────────────────────────────────────
# Decisions for the Cybernetic Augmentation system.
#
# eotg_decision_partial_removal       — Enhanced → Augmented (XP to 0) (scaled gold)
# eotg_decision_overclock_regression  — Overclocked → Enhanced (XP to 50), halves fracture risk (scaled gold)
# eotg_decision_maintenance_protocol  — Periodic calibration (all tiers, reduces fracture risk)
#
# Costs are vanilla realm-relative script values (cybernetics_v2 B5). Cost shape
# with a conditional discount copied from vanilla
# common/decisions/00_diarchy_decisions.txt (diarch_attempt_to_overthrow_liege).
# The engine blocks an unaffordable `cost` by itself (skill reference
# decisions/_decisions.info, `minimum_cost` note; vanilla puts no gold test in
# is_valid), so is_valid carries no gold check that could drift from the cost.
#
# Regression is open at any stress (cybernetics_v2 B3): prudence must be possible.
# The discount (stress level 2+, or the Intervention flag from Phase 4c) is 25%.
# ─────────────────────────────────────────────────────────────────────────────

eotg_decision_partial_removal = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/eotg_decision_partial_removal.dds"
    }
    ai_check_interval = 120

    is_shown = {
        eotg_is_aug_tier2 = yes
    }

    cost = {
        gold = {
            value = medium_gold_value
            if = {
                limit = {
                    OR = {
                        stress_level >= 2
                        has_character_flag = eotg_flag_aug_intervention_discount
                    }
                }
                multiply = 0.75
            }
        }
    }

    effect = {
        eotg_aug_set_integration_effect = { XP = 0 }
        stress_impact = { base = medium_stress_impact_loss }
        custom_tooltip = eotg_decision_partial_removal_tooltip
    }

    # AI only regresses when it is hurting
    ai_potential = {
        stress_level >= 2
    }
    ai_will_do = {
        base = 5
        modifier = { add = 20  stress >= 300 }
        modifier = { add = 10  has_trait = content }
        modifier = { add = -15  has_trait = ambitious }
    }
}


eotg_decision_overclock_regression = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/eotg_decision_overclock_regression.dds"
    }
    ai_check_interval = 120

    is_shown = {
        eotg_is_aug_tier3 = yes
    }

    cost = {
        gold = {
            value = major_gold_value
            if = {
                limit = {
                    OR = {
                        stress_level >= 2
                        has_character_flag = eotg_flag_aug_intervention_discount
                    }
                }
                multiply = 0.75
            }
        }
    }

    effect = {
        # Some of the damage remains: risk is halved, not cleared (B3). Halved
        # before the tier change so the Enhanced character carries it forward.
        if = {
            limit = { has_variable = eotg_fracture_risk }
            change_variable = { name = eotg_fracture_risk  multiply = 0.5 }
        }
        eotg_aug_set_integration_effect = { XP = 50 }
        add_character_modifier = { modifier = eotg_mod_oc_regression_recovery  years = 5 }
        remove_character_flag = eotg_flag_aug_intervention_discount
        stress_impact = { base = medium_stress_impact_loss }
        custom_tooltip = eotg_decision_overclock_regression_tooltip
    }

    # AI only regresses when it is hurting
    ai_potential = {
        stress_level >= 2
    }
    ai_will_do = {
        base = 3
        modifier = { add = 20  stress >= 350 }
        modifier = { add = 10  has_trait = content }
        modifier = { add = -20  has_trait = ambitious }
        modifier = { add = -10  has_trait = brave }
    }
}


eotg_decision_maintenance_protocol = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/eotg_decision_maintenance_protocol.dds"
    }
    ai_check_interval = 60

    is_shown = {
        has_trait = eotg_cybernetics
    }

    # Decision-owned cooldown (index §1 rule 5)
    is_valid = {
        NOT = { has_character_flag = eotg_flag_maintenance_cooldown }
    }

    # A physician to hand makes the calibration cheaper (thread T5)
    cost = {
        gold = {
            value = minor_gold_value
            if = {
                limit = { eotg_has_physician_access = yes }
                multiply = 0.75
            }
        }
    }

    effect = {
        add_character_flag = { flag = eotg_flag_maintenance_cooldown  years = 3 }
        add_character_modifier = { modifier = eotg_mod_implant_calibrated  years = 3 }
        # Recalibration replaces black-market hardware (audit §1.7c)
        remove_character_modifier = eotg_mod_illegal_implants
        if = {
            limit = { eotg_is_aug_tier3 = yes }
            eotg_add_fracture_risk = { AMOUNT = -8 }
        }
        stress_impact = { base = minor_stress_impact_loss }
        custom_tooltip = eotg_decision_maintenance_protocol_tooltip
    }

    ai_potential = {
        OR = {
            eotg_is_aug_tier3 = yes
            gold >= medium_gold_value
        }
    }
    ai_will_do = {
        base = 10
        modifier = { add = 30  eotg_is_aug_tier3 = yes }
        modifier = { add = 20  stress >= 200 }
        modifier = { add = -10  has_trait = callous }
    }
}

eotg_decision_consult_physician = {
    # PLACEHOLDER art: vanilla decision_physician.dds
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_physician.dds"
    }
    ai_check_interval = 60

    is_shown = {
        has_trait = eotg_cybernetics
        eotg_has_physician_access = yes
    }

    # Half price when the ruler is the physician
    cost = {
        gold = {
            value = minor_gold_value
            if = {
                limit = { has_trait = lifestyle_physician }
                multiply = 0.5
            }
        }
    }

    cooldown = { years = 3 }

    effect = {
        add_character_modifier = { modifier = eotg_mod_implant_calibrated  years = 5 }
        remove_character_modifier = eotg_mod_aug_infection
        # Deeper than maintenance at Overclocked (maintenance gives -8)
        if = {
            limit = { eotg_is_aug_tier3 = yes }
            eotg_add_fracture_risk = { AMOUNT = -12 }
        }
        if = {
            limit = { employs_court_position = court_physician_court_position }
            court_position:court_physician_court_position = {
                add_character_modifier = { modifier = eotg_mod_aug_lesson_learning  years = 5 }
            }
        }
        custom_tooltip = eotg_decision_consult_physician_tooltip
    }

    ai_will_do = {
        base = 10
        modifier = { add = 30  eotg_is_aug_tier3 = yes }
        modifier = { add = 10  has_trait = diligent }
        modifier = { add = -10  has_trait = lazy }
    }
}
