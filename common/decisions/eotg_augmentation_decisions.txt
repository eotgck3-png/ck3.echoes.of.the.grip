# ── eotg_augmentation_decisions.txt ─────────────────────────────────────────
# Decisions for the Cybernetic Augmentation system.
#
# eotg_decision_partial_removal       — Enhanced → Augmented (XP to 0) (scaled gold)
# eotg_decision_overclock_regression  — Overclocked → Enhanced (XP to 50), halves fracture risk (scaled gold)
# eotg_decision_maintenance_protocol  — Periodic calibration (all tiers, reduces fracture risk)
#
# Costs are vanilla realm-relative script values (cybernetics_v2 B5). Cost shape
# with a conditional discount copied from vanilla
# common/decisions/00_diarchy_decisions.txt (diarch_attempt_to_overthrow_liege).
# The engine blocks an unaffordable `cost` by itself (skill reference
# decisions/_decisions.info, `minimum_cost` note; vanilla puts no gold test in
# is_valid), so is_valid carries no gold check that could drift from the cost.
#
# Regression is open at any stress (cybernetics_v2 B3): prudence must be possible.
# The discount (stress level 2+, or the Intervention flag from Phase 4c) is 25%.
#
# AI layer (balance spec §5.2): every decision uses ai_check_interval_by_tier
# (months; every tier key required, 0 = never checks: skill reference
# decisions/_decisions.info:110; vanilla common/decisions/
# 00_artifact_decisions.txt:616, 00_dynasty_decisions.txt:14,
# 00_lifestyle_decisions.txt:342 for county = 0). Barony is always 0: the
# system serves count-tier rulers and above. ai_potential leads with the cheap
# trait test; ai_will_do is the % chance per check (_decisions.info:143), with
# `factor = 0` vetoes as 00_artifact_decisions.txt:616.
# ─────────────────────────────────────────────────────────────────────────────

eotg_decision_partial_removal = {
    picture = {

        modifier = { add = 15  has_trait = content }
        modifier = { add = 15  has_trait = zealous }
        modifier = { add = -15  has_trait = ambitious }
    }
}


eotg_decision_overclock_regression = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/eotg_decision_overclock_regression.dds"
    }
    ai_check_interval_by_tier = {
        barony = 0
        county = 12
        duchy = 12
        kingdom = 6
        empire = 6
        hegemony = 6
    }

    is_shown = {
        eotg_is_aug_tier3 = yes
    }

    cost = {
        gold = {
            value = major_gold_value
            if = {
                limit = {
                    OR = {
                        stress_level >= 2
                        has_character_flag = eotg_flag_aug_intervention_discount
                    }
                }
                multiply = 0.75
            }
        }
    }

    effect = {
        # Some of the damage remains: risk is halved, not cleared (B3). Halved
        # before the tier change so the Enhanced character carries it forward.
        if = {
            limit = { has_variable = eotg_fracture_risk }
            change_variable = { name = eotg_fracture_risk  multiply = 0.5 }
        }
        eotg_aug_set_integration_effect = { XP = 50 }
        # Phase 5 §1.1: a running Countdown stops if the halved risk is < 50
        # (eotg_aug_countdown.006 The Quiet ends the story). At 50+ the story
        # sees the lost tier at its next tick and fires The Quiet itself.
        if = {
            limit = {
                eotg_aug_has_countdown = yes
                OR = {
                    NOT = { has_variable = eotg_fracture_risk }
                    var:eotg_fracture_risk < 50
                }
            }
            # End the story now (final QA): otherwise its next tick could see
            # the lost tier inside the 3 days and fire The Quiet a second time.
            eotg_aug_end_countdown_effect = yes
            trigger_event = { id = eotg_aug_countdown.006  days = 3 }
        }
        add_character_modifier = { modifier = eotg_mod_oc_regression_recovery  years = 5 }
        remove_character_flag = eotg_flag_aug_intervention_discount
        # Balance spec §5.9: a masked congenital trait shows again
        eotg_aug_unmask_congenital_effect = yes
        stress_impact = { base = medium_stress_impact_loss }
        custom_tooltip = eotg_decision_overclock_regression_tooltip
    }

    # Balance spec §5.2: the AI steps back when it is hurting or the
    # Countdown runs. The AI may read the band; the player never sees it.
    ai_potential = {
        eotg_is_aug_tier3 = yes
        OR = {
            stress_level >= 2
            eotg_aug_has_countdown = yes
            eotg_aug_pressure_storm = yes
        }
    }
    ai_will_do = {
        base = 0
        modifier = {
            add = 50
            eotg_aug_has_countdown = yes
            stress_level >= 2
        }
        modifier = {
            add = 25
            eotg_aug_has_countdown = yes
            stress_level < 2
        }
        modifier = { add = 20  eotg_aug_pressure_storm = yes }
        modifier = { add = 15  has_trait = content }
        modifier = { add = 15  has_trait = craven }
        modifier = { add = 10  has_character_flag = eotg_flag_aug_intervention_discount }
        modifier = { add = -20  has_trait = ambitious }
        modifier = { add = -15  has_trait = arrogant }
        modifier = { add = -15  has_trait = stubborn }
    }
}

        modifier = { add = 10  has_trait = diligent }
        modifier = { add = -10  has_trait = lazy }
        modifier = { add = -10  has_trait = callous }
    }

# (commission_epic_decision: a gold-gated decision that fires an event
# chain, with a cooldown, ai_check_interval_by_tier and a short_term_gold
# ai_potential); the custom_tooltip trigger wrapper as :527; the RHS script
# value in a trigger as 10_religious_decisions.txt:3054.
# No cost: the fired event charges, as eotg_decision_seek_augmentation's hub
# does (a decision cost would charge twice). The gold checks in is_valid
# mirror the events' paying options (eotg_aug_can_afford_upgrade = tier1.002.a;
# major_gold_value = tier2.003.a).
# Cooldown authority: the decision's own cooldown block (index §1 rule 5).
# The 5-year eotg_flag_aug_settling is a world-state gate (minimum time at

# (desc_sought, ai_chance) and cleared by tier1.022 and tier2.003.

eotg_decision_aug_pursue_next_stage = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/eotg_decision_maintenance_protocol.dds"
    }
    ai_check_interval_by_tier = {
        barony = 0
        county = 24
        duchy = 12
        kingdom = 12
        empire = 12
        hegemony = 12
    }

    # The population the yearly pulse serves (count tier and above)
    is_shown = {
        OR = {
            eotg_is_aug_tier1 = yes
            eotg_is_aug_tier2 = yes
        }
        highest_held_title_tier >= tier_county
    }

    is_valid_showing_failures_only = {
        is_available_adult = yes
    }

    is_valid = {
        custom_tooltip = {
            text = eotg_decision_aug_pursue_next_stage_settling_tt
            NOT = { has_character_flag = eotg_flag_aug_settling }
        }
        custom_tooltip = {
            text = eotg_decision_aug_pursue_next_stage_suppressed_tt
            NOT = { has_character_flag = eotg_flag_suppress_progression }
        }
        trigger_if = {
            limit = { eotg_is_aug_tier1 = yes }
            capital_county ?= { development_level >= eotg_aug_dev_gate_enhanced }
            eotg_aug_can_afford_upgrade = yes
        }
        trigger_else = {
            capital_county ?= { development_level >= eotg_aug_dev_gate_overclocked }
            gold >= major_gold_value
        }
    }

    # Declining in the fired event must not be spammable
    cooldown = { years = 2 }

    effect = {
        custom_tooltip = eotg_decision_aug_pursue_next_stage_tooltip
        add_character_flag = { flag = eotg_flag_aug_sought_upgrade  days = 60 }
        if = {
            limit = { eotg_is_aug_tier1 = yes }
            trigger_event = eotg_aug_tier1.002
        }
        else = {
            trigger_event = eotg_aug_tier2.003
        }
    }

    # Cheapest tests first (balance spec §5.2): a plain has_trait rejects
    # every unaugmented AI before any XP, war, stress or gold check runs.
    ai_potential = {
        has_trait = eotg_cybernetics
        NOT = { has_trait = zealous }
        NOT = { has_trait = content }
        is_at_war = no
        stress_level < 2
        trigger_if = {
            limit = { eotg_is_aug_tier1 = yes }
            short_term_gold >= medium_gold_value
        }
        trigger_else = {
            short_term_gold >= major_gold_value
        }
    }
    ai_will_do = {
        base = 15
        modifier = { add = 20  has_trait = ambitious }
        modifier = { add = 10  has_trait = brave }
        modifier = { add = 10  has_trait = arrogant }
        modifier = { add = 10  has_trait = cynical }
        modifier = { add = -10  has_trait = craven }
        modifier = { add = -10  has_trait = paranoid }
        # AI prudence at Enhanced: the AI may read the band; the player never
        # sees it (hidden-risk rule).
        modifier = {
            factor = 0.5
            eotg_is_aug_tier2 = yes
            eotg_aug_pressure_flicker = no
        }
    }
}
