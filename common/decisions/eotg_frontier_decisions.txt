# -- eotg_frontier_decisions.txt ----------------------------------------------
# Decisions for Frontier Systems v1 (docs/specs/frontier_v1.md §3.5, §8).
#
# Shapes copied from common/decisions/eotg_augmentation_decisions.txt
# (verified against 1.20): ai_check_interval_by_tier with every tier key
# (0 = never), ai_potential + ai_will_do, custom_description around anything
# that would otherwise render as a raw variable (eotg_lint L006).
# The player chooses the Region (owner request 2026-10-08; supersedes spec
# §8.1 "decisions cannot take a Region as a target"). Every decision that
# picks a Region carries vanilla's county-picker widget, shape copied from
# pam_decisions.txt:1395-1437 (pam_send_legatine_mission_decision) and
# :4092-4133 (raise khachkar), ai_select_item from
# 50_holy_site_decisions.txt:147: gui = "decision_view_widget_select_realm_county"
# (base game; the mod ships no GUI), controller = select_scope_object,
# setup_items fills item_list with exactly the Regions that pass the
# decision's eligibility trigger, is_item_valid re-checks it, default_item
# and ai_select_item are the old automatic pick (so a player who just
# confirms, and the AI, behave as before), is_valid_showing_failures_only
# guards scope:selected_item (pam pattern), and the effect re-saves
# scope:selected_item as scope:eotg_frontier_county so every downstream
# event is unchanged. scope:selected_item can be unset while the effect
# tooltip renders (pam_decisions.txt:5912), hence the `?=` in effects.
#
# Confirmed against vanilla 1.20 (verification §3): the held-title iterators
# with tier = tier_county (V13), the picture decision_realm.dds, and
# debug_only = yes in is_shown (V10; test_decision.txt:9): the debug
# decisions show only when the game runs with -debug_mode (E4).
# -----------------------------------------------------------------------------

# Establish a Frontier (spec §8.1). The event .001 picks the project type and
# charges the cost (as Seek Augmentation: no `cost`, the event pays).
eotg_decision_frontier_establish = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    ai_check_interval_by_tier = {
        barony = 0
        county = 60
        duchy = 48
        kingdom = 48
        empire = 48
        hegemony = 48
    }

    is_shown = {
        eotg_frontier_holds_establishable = yes
    }

    is_valid = {
        custom_description = {
            text = eotg_decision_frontier_establish_gold_tt
            gold >= eotg_frontier_establish_cost_value
        }
        # Open question Q7: not while at war
        custom_description = {
            text = eotg_decision_frontier_establish_peace_tt
            is_at_war = no
        }
    }

    is_valid_showing_failures_only = {
        custom_description = {
            text = eotg_decision_frontier_establish_pick_tt
            scope:selected_item ?= {
                tier = tier_county
                holder = root
                eotg_frontier_can_establish = yes
            }
        }
    }

    # The Region picker (header). Default and AI: the most developed one.
    widget = {
        gui = "decision_view_widget_select_realm_county"
        controller = select_scope_object

        setup_items = {
            scope:actor = {
                every_held_title = {
                    limit = {
                        tier = tier_county
                        eotg_frontier_can_establish = yes
                    }
                    add_to_list = item_list
                }
            }
        }

        default_item = {
            scope:actor = {
                ordered_held_title = {
                    limit = {
                        tier = tier_county
                        eotg_frontier_can_establish = yes
                    }
                    order_by = development_level
                    save_scope_as = selected_item
                }
            }
        }

        is_item_valid = {
            custom_tooltip = {
                text = eotg_decision_frontier_establish_pick_tt
                tier = tier_county
                holder = scope:actor
                eotg_frontier_can_establish = yes
            }
        }

        ai_select_item = {
            scope:actor = {
                ordered_held_title = {
                    limit = {
                        tier = tier_county
                        eotg_frontier_can_establish = yes
                    }
                    order_by = development_level
                    save_scope_as = selected_item
                }
            }
        }
    }

    effect = {
        scope:selected_item ?= { save_scope_as = eotg_frontier_county }
        # The founder candidate: the best steward at court, or the taker
        ordered_courtier = {
            limit = {
                is_adult = yes
                is_imprisoned = no
            }
            order_by = stewardship
            save_scope_as = eotg_frontier_candidate
        }
        if = {
            limit = {
                OR = {
                    NOT = { exists = scope:eotg_frontier_candidate }
                    scope:eotg_frontier_candidate = { stewardship <= root.stewardship }
                }
            }
            save_scope_as = eotg_frontier_candidate
        }
        custom_tooltip = eotg_decision_frontier_establish_effect_tt
        trigger_event = eotg_frontier.001
    }

    ai_potential = {
        eotg_frontier_holds_establishable = yes
        eotg_frontier_holds_frontier = no
        eotg_frontier_ai_room = yes
        is_at_war = no
    }
    ai_will_do = {
        base = 10
        modifier = { add = 10  stewardship >= 12 }
        modifier = { add = 10  has_trait = ambitious }
        modifier = { add = 5  has_trait = diligent }
        modifier = { add = -10  has_trait = lazy }
        modifier = { add = -10  has_trait = content }
        modifier = {
            factor = 0
            gold < {
                value = eotg_frontier_establish_cost_value
                multiply = 2
            }
        }
    }
}

# Withdraw sponsorship (spec §8.3; open Q9): ends this character's backing.
eotg_decision_frontier_withdraw = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    ai_check_interval_by_tier = {
        barony = 0
        county = 24
        duchy = 24
        kingdom = 24
        empire = 24
        hegemony = 24
    }

    is_shown = {
        eotg_frontier_is_sponsor = yes
    }

    effect = {
        custom_tooltip = eotg_decision_frontier_withdraw_effect_tt
        hidden_effect = {
            var:eotg_frontier_sponsoring = {
                save_scope_as = eotg_frontier_county
                if = {
                    limit = { var:eotg_frontier_sponsor ?= root }
                    eotg_frontier_clear_sponsor_effect = yes
                    eotg_frontier_add_strain_effect = { CAUSE = sponsor_lapsed }
                    holder = {
                        send_interface_toast = {
                            title = eotg_frontier_toast_sponsor_withdrew
                            left_icon = root
                        }
                    }
                }
            }
            # A stale back-reference never blocks a new sponsorship
            if = {
                limit = { has_variable = eotg_frontier_sponsoring }
                remove_variable = eotg_frontier_sponsoring
            }
        }
    }

    ai_potential = {
        eotg_frontier_is_sponsor = yes
    }
    ai_will_do = {
        base = 0
        modifier = {
            add = 50
            gold < {
                value = eotg_frontier_sponsor_payment_value
                multiply = 2
            }
        }
        modifier = { add = 20  is_at_war = yes }
    }
}

                    order_by = var:eotg_frontier_progress
                    save_scope_as = selected_item
                }
            }
        }
    }

    effect = {
        scope:selected_item ?= { save_scope_as = eotg_frontier_county }
        custom_tooltip = eotg_decision_frontier_build_effect_tt
        trigger_event = eotg_frontier.021
    }

    ai_potential = {
        eotg_frontier_holds_buildable = yes
        gold >= {
            value = eotg_frontier_infra_cost_value
            multiply = 2
        }
    }
    ai_will_do = {
        base = 15
        modifier = {
            add = 10
            any_held_title = {
                tier = tier_county
                eotg_frontier_is_frontier = yes
                var:eotg_frontier_strain >= 4
            }
        }
        modifier = { add = -10  has_trait = greedy }
    }
}

# Send an Expedition (spec §4): one step of exploration in an Unknown (first)
# or Partial Region the taker holds. A new decision rather than a Survey
# extension: Survey needs trait room and runs once in five years; exploration
# needs neither, and this is the seam a future Exploration system extends.
# One expedition a year per Region (the decision's own gate).
eotg_decision_frontier_explore = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    ai_check_interval_by_tier = {
        barony = 0
        county = 60
        duchy = 48
        kingdom = 48
        empire = 48
        hegemony = 48
    }

    is_shown = {
        # Unclaimed Regions (frontier_v3.md §17): never a placeholder
        eotg_is_unclaimed_holder = no
        eotg_frontier_holds_explorable = yes
    }

    is_valid_showing_failures_only = {
        custom_description = {
            text = eotg_decision_frontier_explore_pick_tt
            scope:selected_item ?= {
                tier = tier_county
                eotg_frontier_can_target_expedition = { ACTOR = root }
            }
        }
    }

    cost = {
        gold = {
            value = eotg_frontier_explore_cost_value
        }
    }

    # The Region picker (header). The items are the taker's own explorable
    # Regions plus the explorable Unclaimed Regions within reach, so not only
    # the realm: the widget's GUI only reads each item's Title (name and
    # coat of arms), and the realm limit in its header comment is the vanilla
    # callers' choice, not the widget's. All targeting stays in the one
    # trigger and the one effect (spec §4.1): items and validity are
    # eotg_frontier_can_target_expedition; default and AI are
    # eotg_frontier_pick_expedition_target_effect, the old automatic pick.
    widget = {
        gui = "decision_view_widget_select_realm_county"
        controller = select_scope_object

        setup_items = {
            scope:actor = {
                every_held_title = {
                    limit = {
                        tier = tier_county
                        eotg_frontier_can_target_expedition = { ACTOR = scope:actor }
                    }
                    add_to_list = item_list
                }
            }
            every_in_global_list = {
                variable = eotg_unclaimed_counties
                limit = {
                    eotg_frontier_can_target_expedition = { ACTOR = scope:actor }
                }
                add_to_list = item_list
            }
        }

        default_item = {
            eotg_frontier_pick_expedition_target_effect = { ACTOR = scope:actor }
            scope:eotg_frontier_county ?= { save_scope_as = selected_item }
        }

        is_item_valid = {
            custom_tooltip = {
                text = eotg_decision_frontier_explore_pick_tt
                tier = tier_county
                eotg_frontier_can_target_expedition = { ACTOR = scope:actor }
            }
        }

        ai_select_item = {
            eotg_frontier_pick_expedition_target_effect = { ACTOR = scope:actor }
            scope:eotg_frontier_county ?= { save_scope_as = selected_item }
        }
    }

    effect = {
        # The Region chosen in the widget; eotg_frontier_can_target_expedition
        # (spec §4.1) is the only targeting rule, via setup_items and
        # is_valid_showing_failures_only
        scope:selected_item ?= { save_scope_as = eotg_frontier_county }
        custom_tooltip = eotg_decision_frontier_explore_effect_tt
        hidden_effect = {
            scope:eotg_frontier_county ?= {
                set_variable = { name = eotg_frontier_expedition_sent  value = yes  years = 1 }
            }
        }
        trigger_event = eotg_frontier.042
    }

    ai_potential = {
        eotg_frontier_holds_explorable = yes
        gold >= {
            value = eotg_frontier_explore_cost_value
            multiply = 3
        }
    }
    ai_will_do = {
        base = 10
        modifier = { add = 10  learning >= 12 }
    }
}
