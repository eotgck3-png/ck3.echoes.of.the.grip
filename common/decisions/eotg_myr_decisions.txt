# ── eotg_myr_decisions.txt ───────────────────────────────────────────────────
# Decisions for the Myr Cluster Wars struggle.
#
# eotg_decision_myr_settlement — ends the struggle from the settlement phase
#   (the Accord of the Clusters). Available to powerful involved rulers only.
# ─────────────────────────────────────────────────────────────────────────────

eotg_decision_myr_settlement = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
    }
    decision_group_type = major
    ai_check_interval = 60

    is_shown = {
        any_character_struggle = {
            is_struggle_type = eotg_myr_struggle
            involvement = involved
            is_struggle_phase = struggle_eotg_myr_phase_settlement
        }
        highest_held_title_tier >= tier_kingdom
    }

    is_valid = {
        prestige >= 1000
        is_at_war = no
    }

    is_valid_showing_failures_only = {
        is_at_war = no
    }

    cost = { prestige = 1000 }

    effect = {
        random_character_struggle = {
            limit = { is_struggle_type = eotg_myr_struggle }
            change_struggle_phase = struggle_eotg_myr_phase_conclusion
        }
        add_prestige = 500
        custom_tooltip = eotg_decision_myr_settlement_tooltip
    }

    ai_potential = {
        is_independent_ruler = yes
    }
    ai_will_do = {
        base = 20
        modifier = { add = 30  prestige >= 2000 }
        modifier = { add = -20  has_trait = ambitious }
        modifier = { add = 15  has_trait = content }
    }
}
