# Grand Underlord demands loyalty tax from all Bosses.
eotg_cartel_demand_loyalty_tax = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}
	decision_group_type = major

	desc = eotg_cartel_demand_loyalty_tax_desc
	selection_tooltip = eotg_cartel_demand_loyalty_tax_tooltip

	is_shown = {
		eotg_st_cartel_tax_demand_valid = yes
	}

	is_valid = {
		eotg_st_cartel_tax_demand_valid = yes
		is_imprisoned = no
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
	}

	effect = {
		eotg_se_cartel_demand_loyalty_tax = yes
		custom_tooltip = eotg_cartel_demand_loyalty_tax_effect_tooltip
	}

	ai_check_interval = 12
	ai_will_do = {
		base = 60
		modifier = { add = 40  dread < 30 }
		modifier = { add = -20  dread > 60 }
	}
}

# Grand Underlord issues a Purge Demand — targets the most defiant Boss for elimination.
eotg_cartel_issue_purge_demand = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}
	decision_group_type = major

	desc = eotg_cartel_issue_purge_demand_desc
	selection_tooltip = eotg_cartel_issue_purge_demand_tooltip

	is_shown = {
		eotg_st_is_grand_underlord = yes
		any_vassal = {
			eotg_st_is_boss = yes
			has_character_modifier = eotg_cartel_defiant_boss
		}
	}

	is_valid = {
		eotg_st_is_grand_underlord = yes
		is_imprisoned = no
		dread >= 30
		any_vassal = {
			eotg_st_is_boss = yes
			has_character_modifier = eotg_cartel_defiant_boss
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		dread >= 30
	}

	effect = {
		add_dread = 10
		random_vassal = {
			limit = {
				eotg_st_is_boss = yes
				has_character_modifier = eotg_cartel_defiant_boss
			}
			save_scope_as = eotg_purge_target
			trigger_event = { id = eotg_cartel_faction.0004  days = 3 }
		}
		custom_tooltip = eotg_cartel_purge_demand_effect_tooltip
	}

	cooldown = { years = 5 }

	ai_check_interval = 36
	ai_will_do = {
		base = 15
		modifier = { add = 30  has_character_modifier = eotg_cartel_shakedown_standoff }
		modifier = { add = 20  dread > 50 }
	}
}
