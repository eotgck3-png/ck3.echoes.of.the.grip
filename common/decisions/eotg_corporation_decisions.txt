# Governments v2, Phase G2, Corporation (docs/specs/governments_v2_g2_corporation.md
# §4, §6).
#   Batch CO3: eotg_decision_corporation_restructure (top, a sink) and
#              eotg_decision_corporation_court_board (vassal tiers, a source
#              that ERODES the CEO's hold: the board is contested).
#   Batch CO4: eotg_decision_corporation_adopt (entry) and
#              eotg_decision_corporation_dissolve_board (exit).
# UNVALIDATED cloud build (no Tiger, PX or game run).
#
# Signature resource: var:eotg_corp_board_influence. Every decision reads it
# (through eotg_corporation_board_influence_value) and moves it ONLY through
# eotg_corporation_change_board_influence_effect (spec §2.2). Not a `cost`
# key: gated in is_valid, paid in `effect`, each move with a custom tooltip.
# AMOUNT is always a NAMED value (eotg_corporation_values.txt).
# Cooldowns: each decision's own `cooldown` (or changed_government_recently).
# Shape: the shipped eotg_cartel_decisions.txt. Map-agnostic.

# Push Through a Restructuring (spec §4.1): the CEO spends 25 board influence
# for +10% income and -10% build cost for 5 years; every Corporation director
# takes eotg_restructured_opinion (-10).
eotg_decision_corporation_restructure = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    desc = eotg_decision_corporation_restructure_desc
    selection_tooltip = eotg_decision_corporation_restructure_tooltip
    confirm_text = eotg_decision_corporation_restructure_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 60
        duchy = 36
        kingdom = 36
        empire = 36
        hegemony = 36
    }

    # The only cadence authority (spec §4.1)
    cooldown = { years = 5 }

    is_shown = {
        eotg_corporation_role_top = yes
    }

    is_valid = {
        custom_tooltip = {
            text = eotg_decision_corporation_restructure_influence_tt
            eotg_corporation_board_influence_value >= eotg_corporation_restructure_cost_value
        }
    }

    effect = {
        custom_tooltip = eotg_decision_corporation_restructure_spend_tt
        hidden_effect = {
            eotg_corporation_change_board_influence_effect = { AMOUNT = eotg_corporation_restructure_spend_value }
        }
        add_character_modifier = {
            modifier = eotg_corporation_restructured_modifier
            years = 5
        }
        custom_tooltip = eotg_decision_corporation_restructure_effect_tt
        save_scope_as = eotg_corporation_chief
        hidden_effect = {
            every_vassal = {
                limit = { government_has_flag = eotg_government_is_corporation }
                add_opinion = {
                    modifier = eotg_restructured_opinion
                    target = scope:eotg_corporation_chief
                }
            }
        }
    }

    ai_potential = {
        eotg_corporation_role_top = yes
    }

    ai_will_do = {
        base = 10
        modifier = {
            add = 20
            eotg_corporation_board_influence_value >= 60
        }
        modifier = {
            add = 10
            has_trait = greedy
        }
        modifier = {
            add = 10
            has_trait = ambitious
        }
        modifier = { # keep 20 after the spend, so the election hold survives
            factor = 0
            eotg_corporation_board_influence_value < eotg_corporation_restructure_ai_floor_value
        }
    }
}

# Court the Board (spec §4.2): a Corporation director at any tier spends
# medium gold for +15 board influence; a Corporation CEO liege loses 5 ("every
# director's gain loosens the chief's hold"). High influence counts in the
# board election (batch CO5).
# OQ3 (default, orchestrator, 2026-11-03, pending owner): the CEO loses
# influence at most once a year (eotg_corp_board_eroded, a 1-year variable on
# the CEO), and only while the director now holds more influence than the CEO.
# The director's post-gain value is saved once as eotg_corp_director_influence;
# no other name is saved in this effect.
eotg_decision_corporation_court_board = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    desc = eotg_decision_corporation_court_board_desc
    selection_tooltip = eotg_decision_corporation_court_board_tooltip
    confirm_text = eotg_decision_corporation_court_board_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 60
        duchy = 48
        kingdom = 48
        empire = 0
        hegemony = 0
    }

    # The only cadence authority (spec §4.2)
    cooldown = { years = 5 }

    is_shown = {
        eotg_corporation_role_vassal = yes # county, duchy and kingdom vassals (v1 lesson 4)
    }

    cost = {
        gold = {
            value = medium_gold_value
        }
    }

    effect = {
        custom_tooltip = eotg_decision_corporation_court_board_gain_tt
        hidden_effect = {
            eotg_corporation_change_board_influence_effect = { AMOUNT = eotg_corporation_court_gain_value }
        }
        save_scope_value_as = {
            name = eotg_corp_director_influence
            value = eotg_corporation_board_influence_value
        }
        if = {
            limit = {
                liege ?= {
                    eotg_corporation_role_top = yes
                    NOT = { has_variable = eotg_corp_board_eroded }
                    eotg_corporation_board_influence_value < scope:eotg_corp_director_influence
                }
            }
            custom_tooltip = eotg_decision_corporation_court_board_erosion_tt
            hidden_effect = {
                liege = {
                    eotg_corporation_change_board_influence_effect = { AMOUNT = eotg_corporation_court_erosion_value }
                    set_variable = {
                        name = eotg_corp_board_eroded
                        years = 1
                    }
                }
            }
        }
    }

    ai_potential = {
        eotg_corporation_role_vassal = yes
    }

    ai_will_do = {
        base = 5
        modifier = {
            add = 15
            has_trait = ambitious
        }
        modifier = { # the liege dislikes root (spec §4.2); form: G/common/decisions/30_court_decisions.txt:371-377
            add = 10
            liege ?= {
                opinion = {
                    target = root
                    value < 0
                }
            }
        }
        modifier = {
            add = 10
            eotg_corporation_board_influence_value < 30
        }
        modifier = {
            factor = 0
            gold < medium_gold_value
        }
        modifier = {
            factor = 0
            eotg_corporation_board_influence_value >= 80
        }
    }
}

# Incorporate the Realm (spec §6.1). Entry: an independent feudal or clan
# ruler with stewardship >= 12 and capital to put on the books. ONLY THE RULER
# CONVERTS; init sets board influence 50. The capital holding is NOT changed
# (OQ10: record V-CO10 first).
# AI: OFF (OQ7, as the Cartel).
eotg_decision_corporation_adopt = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    decision_group_type = major
    desc = eotg_decision_corporation_adopt_desc
    selection_tooltip = eotg_decision_corporation_adopt_tooltip
    confirm_text = eotg_decision_corporation_adopt_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 72
        duchy = 72
        kingdom = 72
        empire = 72
        hegemony = 72
    }

    is_shown = {
        is_landed = yes
        highest_held_title_tier > tier_barony
        is_independent_ruler = yes
        OR = {
            government_has_flag = government_is_feudal
            government_has_flag = government_is_clan
        }
        NOR = {
            government_has_flag = eotg_government_is_cartel
            government_has_flag = eotg_government_is_pmc
            government_has_flag = eotg_government_is_corporation
            government_has_flag = eotg_government_is_gobcorp
            government_has_flag = eotg_government_is_unclaimed
        }
    }

    is_valid = {
        custom_tooltip = {
            text = eotg_decision_corporation_adopt_stewardship_tt
            stewardship >= 12
        }
        custom_tooltip = {
            text = eotg_decision_corporation_adopt_gold_tt
            gold >= medium_gold_value
        }
    }

    is_valid_showing_failures_only = {
        custom_tooltip = {
            text = not_changed_government_recently_tt
            NOT = { has_variable = changed_government_recently }
        }
        is_at_war = no
        is_tributary = no
    }

    cost = {
        prestige = 200
    }

    effect = {
        set_variable = {
            name = changed_government_recently
            years = 5
        }
        change_government = eotg_corporation_government
    }

    ai_potential = {
        always = no # OQ7: player-only until V-G11 (leakage) has run
    }

    ai_will_do = {
        base = 0
        modifier = {
            add = 10
            stewardship >= 16
        }
        modifier = {
            add = 10
            has_trait = greedy
        }
        modifier = {
            factor = 0
            primary_title.tier > tier_kingdom
        }
    }
}

# Dissolve the Board (spec §6.2). Exit: the hold on the board becomes personal
# legitimacy (x2, max 200); influence goes to 0 through the change effect;
# Corporation vassals go feudal with the ruler.
eotg_decision_corporation_dissolve_board = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    decision_group_type = major
    desc = eotg_decision_corporation_dissolve_board_desc
    selection_tooltip = eotg_decision_corporation_dissolve_board_tooltip
    confirm_text = eotg_decision_corporation_dissolve_board_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 120
        duchy = 120
        kingdom = 120
        empire = 120
        hegemony = 120
    }

    is_shown = {
        eotg_corporation_role_top = yes
    }

    is_valid_showing_failures_only = {
        custom_tooltip = {
            text = not_changed_government_recently_tt
            NOT = { has_variable = changed_government_recently }
        }
        is_at_war = no
    }

    cost = {
        prestige = medium_prestige_value
    }

    effect = {
        # (1) Legitimacy first: it reads influence before it is zeroed. The
        # Corporation has legitimacy = yes (eotg_corporation_government.txt).
        add_legitimacy = eotg_corporation_exit_legitimacy_value
        # (2)
        custom_tooltip = eotg_decision_corporation_dissolve_board_influence_tt
        hidden_effect = {
            eotg_corporation_change_board_influence_effect = { AMOUNT = eotg_corporation_exit_loss_value }
        }
        # (3)
        custom_tooltip = eotg_decision_corporation_dissolve_board_effect_tt
        hidden_effect = {
            every_vassal_or_below = {
                limit = { government_has_flag = eotg_government_is_corporation }
                change_government = feudal_government
            }
        }
        # (4)
        set_variable = {
            name = changed_government_recently
            years = 5
        }
        # (5)
        change_government = feudal_government
    }

    ai_potential = {
        eotg_corporation_role_top = yes
        eotg_corporation_board_influence_value < 10
    }

    ai_will_do = {
        base = 0
        modifier = {
            add = 10
            gold < 0
        }
    }
}
