# -- eotg_gobcorp_decisions.txt -----------------------------------------------
# Governments v2, Phase G2, Trade Oligarchy (docs/specs/governments_v2_g2_gobcorp.md
# §4, §6). Player text says Trade Oligarchy / Prince / First Prince / charter,
# never the internal key.
#   Batch TO3: eotg_decision_gobcorp_corner_market (top: gains, and squeezes
#              every partner Prince) and eotg_decision_gobcorp_undercut_rival
#              (vassal tiers: gains, taken from one rival Prince).
#   Batch TO4: eotg_decision_gobcorp_adopt (entry) and
#              eotg_decision_gobcorp_dissolve_charter (exit).
# UNVALIDATED cloud build (no Tiger, PX or game run).
#
# Signature resource: var:eotg_gobcorp_market_share. Every decision reads it
# (through eotg_gobcorp_market_share_value) and moves it ONLY through
# eotg_gobcorp_change_market_share_effect (spec §2.2). Not a `cost` key:
# gated in is_valid, moved in `effect`, each move with a custom tooltip.
# AMOUNT is always a NAMED value (eotg_gobcorp_values.txt).
# Cooldowns: each decision's own `cooldown` (or changed_government_recently).
# Shape: the shipped eotg_cartel_decisions.txt. Map-agnostic.
# -----------------------------------------------------------------------------

# Corner the Market (spec §4.1): the First Prince pays major gold for +15
# market share, and every Trade Oligarchy vassal Prince loses 5 and takes
# eotg_market_cornered_opinion (-10): first among equals, not above them.
# OQ10 (default, orchestrator, 2026-11-03, pending owner): each Prince is
# drained at most once a year. eotg_gobcorp_cornered (1-year variable on the
# drained Prince) is that guard; the decision's cooldown stays the cadence
# authority for the First Prince.
eotg_decision_gobcorp_corner_market = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    desc = eotg_decision_gobcorp_corner_market_desc
    selection_tooltip = eotg_decision_gobcorp_corner_market_tooltip
    confirm_text = eotg_decision_gobcorp_corner_market_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 60
        duchy = 36
        kingdom = 36
        empire = 36
        hegemony = 36
    }

    # The only cadence authority (spec §4.1)
    cooldown = { years = 5 }

    is_shown = {
        eotg_gobcorp_role_top = yes
    }

    is_valid = {
        is_at_war = no
        custom_tooltip = {
            text = eotg_decision_gobcorp_corner_market_room_tt
            eotg_gobcorp_market_share_value < 100
        }
    }

    cost = {
        gold = {
            value = major_gold_value
        }
    }

    effect = {
        custom_tooltip = eotg_decision_gobcorp_corner_market_share_tt
        hidden_effect = {
            eotg_gobcorp_change_market_share_effect = { AMOUNT = eotg_gobcorp_corner_gain_value }
        }
        custom_tooltip = eotg_decision_gobcorp_corner_market_squeeze_tt
        hidden_effect = {
            every_vassal = {
                limit = {
                    government_has_flag = eotg_government_is_gobcorp
                    NOT = { has_variable = eotg_gobcorp_cornered }
                }
                eotg_gobcorp_change_market_share_effect = { AMOUNT = eotg_gobcorp_corner_squeeze_value }
                set_variable = {
                    name = eotg_gobcorp_cornered
                    years = 1
                }
                add_opinion = {
                    modifier = eotg_market_cornered_opinion
                    target = root
                }
            }
        }
    }

    ai_potential = {
        eotg_gobcorp_role_top = yes
    }

    ai_will_do = {
        base = 10
        modifier = { # behind
            add = 20
            eotg_gobcorp_market_share_value < 40
        }
        modifier = {
            add = 10
            has_trait = greedy
        }
        modifier = { # many partners to anger (count form: eotg_augmentation_laws.txt:90-93)
            add = -10
            any_vassal = {
                count >= 3
                government_has_flag = eotg_government_is_gobcorp
            }
        }
        modifier = {
            factor = 0
            gold < major_gold_value
        }
    }
}

# Undercut a Rival Prince (spec §4.2): a Trade Oligarchy vassal at any tier
# pays minor gold for +10 market share, taken from ONE rival Prince under the
# same liege (-10 and eotg_undercut_opinion, -15). The rival is a weighted
# random pick, bigger rivals likelier: no picker widget.
# A `weight` block on random_vassal is vanilla:
# G/events/activities/hold_court_activity/hold_court_events_general.txt:1439-1449.
eotg_decision_gobcorp_undercut_rival = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    desc = eotg_decision_gobcorp_undercut_rival_desc
    selection_tooltip = eotg_decision_gobcorp_undercut_rival_tooltip
    confirm_text = eotg_decision_gobcorp_undercut_rival_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 60
        duchy = 48
        kingdom = 48
        empire = 0
        hegemony = 0
    }

    # The only cadence authority (spec §4.2)
    cooldown = { years = 5 }

    is_shown = {
        eotg_gobcorp_role_vassal = yes # county, duchy and kingdom vassals (v1 lesson 4)
    }

    is_valid = {
        custom_tooltip = {
            text = eotg_decision_gobcorp_undercut_rival_rival_exists_tt
            liege ?= {
                any_vassal = {
                    eotg_gobcorp_is_rival_prince_of = { PRINCE = root }
                }
            }
        }
    }

    cost = {
        gold = {
            value = minor_gold_value
        }
    }

    effect = {
        liege = {
            random_vassal = {
                limit = {
                    eotg_gobcorp_is_rival_prince_of = { PRINCE = root }
                }
                weight = {
                    base = 0
                    modifier = {
                        add = eotg_gobcorp_rival_weight_value
                    }
                }
                save_scope_as = eotg_gobcorp_rival
            }
        }
        custom_tooltip = eotg_decision_gobcorp_undercut_rival_gain_tt
        hidden_effect = {
            eotg_gobcorp_change_market_share_effect = { AMOUNT = eotg_gobcorp_undercut_gain_value }
        }
        custom_tooltip = eotg_decision_gobcorp_undercut_rival_rival_tt
        hidden_effect = {
            if = {
                limit = { exists = scope:eotg_gobcorp_rival }
                scope:eotg_gobcorp_rival = {
                    eotg_gobcorp_change_market_share_effect = { AMOUNT = eotg_gobcorp_undercut_loss_value }
                    add_opinion = {
                        modifier = eotg_undercut_opinion
                        target = root
                    }
                }
                send_interface_toast = { # shape: eotg_cartel_walk_off_effect
                    type = event_toast_effect_good
                    title = eotg_decision_gobcorp_undercut_rival_toast
                    left_icon = root
                    right_icon = scope:eotg_gobcorp_rival
                    custom_tooltip = eotg_decision_gobcorp_undercut_rival_toast_desc
                }
            }
        }
    }

    ai_potential = {
        eotg_gobcorp_role_vassal = yes
        NOT = { has_trait = compassionate }
    }

    ai_will_do = {
        base = 10
        modifier = {
            add = 15
            has_trait = greedy
        }
        modifier = {
            add = 10
            has_trait = ambitious
        }
        modifier = {
            factor = 0
            gold < minor_gold_value
        }
    }
}

# Sign the Charter (spec §6.1). Entry: an independent feudal or clan ruler
# with medium gold and stewardship >= 10. ONLY THE RULER CONVERTS; init sets
# market share 40. The capital holding is not changed (Corporation spec OQ10).
# AI: OFF (OQ7, as the Cartel).
eotg_decision_gobcorp_adopt = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    decision_group_type = major
    desc = eotg_decision_gobcorp_adopt_desc
    selection_tooltip = eotg_decision_gobcorp_adopt_tooltip
    confirm_text = eotg_decision_gobcorp_adopt_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 72
        duchy = 72
        kingdom = 72
        empire = 72
        hegemony = 72
    }

    is_shown = {
        is_landed = yes
        highest_held_title_tier > tier_barony
        is_independent_ruler = yes
        OR = {
            government_has_flag = government_is_feudal
            government_has_flag = government_is_clan
        }
        NOR = {
            government_has_flag = eotg_government_is_cartel
            government_has_flag = eotg_government_is_pmc
            government_has_flag = eotg_government_is_corporation
            government_has_flag = eotg_government_is_gobcorp
            government_has_flag = eotg_government_is_unclaimed
        }
    }

    is_valid = {
        custom_tooltip = {
            text = eotg_decision_gobcorp_adopt_gold_tt
            gold >= medium_gold_value
        }
        custom_tooltip = {
            text = eotg_decision_gobcorp_adopt_stewardship_tt
            stewardship >= 10
        }
    }

    is_valid_showing_failures_only = {
        custom_tooltip = {
            text = not_changed_government_recently_tt
            NOT = { has_variable = changed_government_recently }
        }
        is_at_war = no
        is_tributary = no
    }

    cost = {
        prestige = 200
    }

    effect = {
        set_variable = {
            name = changed_government_recently
            years = 5
        }
        change_government = eotg_gobcorp_government
    }

    ai_potential = {
        always = no # OQ7: player-only until V-G11 (leakage) has run
    }

    ai_will_do = {
        base = 0
        modifier = {
            add = 10
            has_trait = greedy
        }
        modifier = {
            add = 10
            stewardship >= 14
        }
        modifier = {
            factor = 0
            primary_title.tier > tier_kingdom
        }
    }
}

# Dissolve the Charter (spec §6.2). Exit: the book is sold, gold = share x 5
# (max 500); share goes to 0 through the change effect; Trade Oligarchy
# vassals go feudal with the ruler.
# Spec §6.2's "+10 if at war for long" AI term is replaced by "+10 in debt":
# the decision needs peace anyway, and no war-length trigger is used in v2.
eotg_decision_gobcorp_dissolve_charter = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    decision_group_type = major
    desc = eotg_decision_gobcorp_dissolve_charter_desc
    selection_tooltip = eotg_decision_gobcorp_dissolve_charter_tooltip
    confirm_text = eotg_decision_gobcorp_dissolve_charter_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 120
        duchy = 120
        kingdom = 120
        empire = 120
        hegemony = 120
    }

    is_shown = {
        eotg_gobcorp_role_top = yes
    }

    is_valid_showing_failures_only = {
        custom_tooltip = {
            text = not_changed_government_recently_tt
            NOT = { has_variable = changed_government_recently }
        }
        is_at_war = no
    }

    cost = {
        prestige = medium_prestige_value
    }

    effect = {
        # (1) Gold first: it reads share before it is zeroed
        add_gold = eotg_gobcorp_exit_gold_value
        # (2)
        custom_tooltip = eotg_decision_gobcorp_dissolve_charter_share_tt
        hidden_effect = {
            eotg_gobcorp_change_market_share_effect = { AMOUNT = eotg_gobcorp_exit_loss_value }
        }
        # (3)
        custom_tooltip = eotg_decision_gobcorp_dissolve_charter_effect_tt
        hidden_effect = {
            every_vassal_or_below = {
                limit = { government_has_flag = eotg_government_is_gobcorp }
                change_government = feudal_government
            }
        }
        # (4)
        set_variable = {
            name = changed_government_recently
            years = 5
        }
        # (5)
        change_government = feudal_government
    }

    ai_potential = {
        eotg_gobcorp_role_top = yes
        eotg_gobcorp_market_share_value < 10
    }

    ai_will_do = {
        base = 0
        modifier = {
            add = 10
            gold < 0
        }
    }
}
