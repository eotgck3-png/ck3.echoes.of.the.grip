# -- eotg_pmc_decisions.txt ---------------------------------------------------
# Governments v2, Phase G2, PMC (docs/specs/governments_v2_g2_pmc.md §4, §6).
#   Batch P3: eotg_decision_pmc_take_contract (top, a source),
#             eotg_decision_pmc_deliver_results (vassal tiers, a source),
#             eotg_decision_pmc_recruit_on_reputation (top, a sink; OQ3).
#   Batch P4: eotg_decision_pmc_adopt (entry) and eotg_decision_pmc_stand_down
#             (exit).
# UNVALIDATED cloud build (no Tiger, PX or game run).
#
# Signature resource: var:eotg_pmc_standing. Every decision here reads it
# (is_valid or ai_will_do, through eotg_pmc_standing_value) and moves it ONLY
# through eotg_pmc_change_standing_effect (spec §2.2). The adopt decision
# leaves it to init (start value 40).
# Standing is not a decision `cost` key, so spends are gated in is_valid and
# paid in `effect`. The change effect shows no tooltip of its own, so every
# move has a custom tooltip naming it (spec §2.3).
# AMOUNT is always a NAMED value (eotg_pmc_values.txt), never an inline { }
# block (Tiger error(macro); Cartel QA C2, 2026-10-20).
# Cooldowns: each decision's own `cooldown` (or vanilla's
# changed_government_recently for the two transitions) is the only authority.
# Shape: the shipped eotg_cartel_decisions.txt (C3 for P3, C4 for P4), whose
# vanilla citations are in that file. Loc keys: <key>, _desc, _tooltip,
# _confirm. Map-agnostic: no title, culture, faith or character key.
# -----------------------------------------------------------------------------

# Take a Contract (spec §4.1): the core verb. The company sells its force to an
# unnamed client: gold that READS standing (50 + 3 x standing), standing +10,
# and the force is away for 2 years (levy -25%).
eotg_decision_pmc_take_contract = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    desc = eotg_decision_pmc_take_contract_desc
    selection_tooltip = eotg_decision_pmc_take_contract_tooltip
    confirm_text = eotg_decision_pmc_take_contract_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 36
        duchy = 36
        kingdom = 36
        empire = 36
        hegemony = 36
    }

    # The only cadence authority for Take a Contract (spec §4.1)
    cooldown = { years = 3 }

    is_shown = {
        eotg_pmc_role_top = yes
    }

    is_valid = {
        is_at_war = no
        custom_tooltip = { # a state, not a cooldown: the force is still away
            text = eotg_decision_pmc_take_contract_free_tt
            NOT = { has_character_modifier = eotg_pmc_under_contract_modifier }
        }
    }

    effect = {
        # Pay first: it reads standing before the gain
        add_gold = eotg_pmc_contract_pay_value
        custom_tooltip = eotg_decision_pmc_take_contract_standing_tt
        hidden_effect = {
            eotg_pmc_change_standing_effect = { AMOUNT = eotg_pmc_contract_gain_value }
        }
        add_character_modifier = {
            modifier = eotg_pmc_under_contract_modifier
            years = 2
        }
    }

    ai_potential = {
        eotg_pmc_role_top = yes
    }

    ai_will_do = {
        base = 20
        modifier = { # needs the money
            add = 20
            gold < medium_gold_value
        }
        modifier = { # a good name pays well
            add = 10
            eotg_pmc_standing_value >= 50
        }
        modifier = {
            add = 10
            has_trait = greedy
        }
        modifier = { # never send the force away while at war
            factor = 0
            is_at_war = yes
        }
    }
}

# Deliver Results (spec §4.2): a PMC vassal at any tier pays its crews a
# bonus (minor gold) and earns +15 standing; a PMC liege gets +5 ("results
# travel up the chain").
eotg_decision_pmc_deliver_results = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    desc = eotg_decision_pmc_deliver_results_desc
    selection_tooltip = eotg_decision_pmc_deliver_results_tooltip
    confirm_text = eotg_decision_pmc_deliver_results_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 60
        duchy = 48
        kingdom = 48
        empire = 0
        hegemony = 0
    }

    # The only cadence authority for Deliver Results (spec §4.2)
    cooldown = { years = 5 }

    is_shown = {
        eotg_pmc_role_vassal = yes # county, duchy and kingdom vassals (v1 lesson 4)
    }

    is_valid = {
        is_at_war = no
    }

    cost = {
        gold = {
            value = minor_gold_value
        }
    }

    effect = {
        custom_tooltip = eotg_decision_pmc_deliver_results_standing_tt
        hidden_effect = {
            eotg_pmc_change_standing_effect = { AMOUNT = eotg_pmc_deliver_gain_value }
        }
        if = {
            limit = {
                liege ?= { government_has_flag = eotg_government_is_pmc }
            }
            custom_tooltip = eotg_decision_pmc_deliver_results_liege_tt
            hidden_effect = {
                liege = {
                    eotg_pmc_change_standing_effect = { AMOUNT = eotg_pmc_deliver_liege_value }
                }
            }
        }
    }

    ai_potential = {
        eotg_pmc_role_vassal = yes
    }

    ai_will_do = {
        base = 10
        modifier = { # the name needs work
            add = 20
            eotg_pmc_standing_value < 30
        }
        modifier = {
            add = 10
            has_trait = ambitious
        }
        modifier = { # good enough already
            factor = 0
            eotg_pmc_standing_value >= 80
        }
        modifier = {
            factor = 0
            gold < minor_gold_value
        }
    }
}

# Recruit on Reputation (spec §4.3, OQ3): the player-chosen sink. 20 standing
# buys a skilled Operator for the court (a generated character: map-agnostic).
# create_character + add_courtier shape: eotg_augmentation_effects.txt:1477-1484.
eotg_decision_pmc_recruit_on_reputation = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    desc = eotg_decision_pmc_recruit_on_reputation_desc
    selection_tooltip = eotg_decision_pmc_recruit_on_reputation_tooltip
    confirm_text = eotg_decision_pmc_recruit_on_reputation_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 36
        duchy = 36
        kingdom = 36
        empire = 36
        hegemony = 36
    }

    # The only cadence authority for Recruit (spec §4.3)
    cooldown = { years = 5 }

    is_shown = {
        eotg_pmc_role_top = yes
    }

    is_valid = {
        custom_tooltip = {
            text = eotg_decision_pmc_recruit_on_reputation_cost_tt
            eotg_pmc_standing_value >= eotg_pmc_recruit_cost_value
        }
    }

    effect = {
        custom_tooltip = eotg_decision_pmc_recruit_on_reputation_standing_tt
        hidden_effect = {
            eotg_pmc_change_standing_effect = { AMOUNT = eotg_pmc_recruit_spend_value }
        }
        custom_tooltip = eotg_decision_pmc_recruit_on_reputation_effect_tt
        hidden_effect = {
            create_character = {
                template = eotg_pmc_operator_template
                location = root.capital_province
                culture = root.culture
                faith = root.faith
                save_scope_as = eotg_pmc_recruit
            }
            add_courtier = scope:eotg_pmc_recruit
            send_interface_toast = { # shape: eotg_cartel_interactions.txt on_accept
                type = event_toast_effect_good
                title = eotg_decision_pmc_recruit_on_reputation_toast
                left_icon = scope:eotg_pmc_recruit
                custom_tooltip = eotg_decision_pmc_recruit_on_reputation_toast_desc
            }
        }
    }

    ai_potential = {
        eotg_pmc_role_top = yes
    }

    ai_will_do = {
        base = 5
        modifier = { # a name worth spending
            add = 10
            eotg_pmc_standing_value >= 60
        }
        modifier = { # room at court (spec §4.3); form: G/common/character_interactions/00_tributary_interactions.txt:3314
            add = 10
            number_of_knights < max_number_of_knights
        }
        modifier = { # keep a reserve (spec §7)
            factor = 0
            eotg_pmc_standing_value < 40
        }
    }
}

# Run the Realm as a Company (spec §6.1). Entry: an independent feudal or clan
# ruler who already commands (martial >= 12) turns their own government PMC.
# ONLY THE RULER CONVERTS. on_government_change runs the G1 law net and
# eotg_pmc_init_effect (standing 40).
# AI: OFF (spec OQ7, as the Cartel's OQ4, owner 2026-10-20).
# Spec §6.1's knight-count condition: at least 3 (number_of_knights:
# G/common/character_interactions/00_tributary_interactions.txt:3314).
eotg_decision_pmc_adopt = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    decision_group_type = major
    desc = eotg_decision_pmc_adopt_desc
    selection_tooltip = eotg_decision_pmc_adopt_tooltip
    confirm_text = eotg_decision_pmc_adopt_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 72
        duchy = 72
        kingdom = 72
        empire = 72
        hegemony = 72
    }

    is_shown = {
        is_landed = yes
        highest_held_title_tier > tier_barony
        is_independent_ruler = yes
        OR = {
            government_has_flag = government_is_feudal
            government_has_flag = government_is_clan
        }
        NOR = {
            government_has_flag = eotg_government_is_cartel
            government_has_flag = eotg_government_is_pmc
            government_has_flag = eotg_government_is_corporation
            government_has_flag = eotg_government_is_gobcorp
            government_has_flag = eotg_government_is_unclaimed
        }
    }

    is_valid = {
        custom_tooltip = {
            text = eotg_decision_pmc_adopt_martial_tt
            martial >= 12
        }
        custom_tooltip = {
            text = eotg_decision_pmc_adopt_knights_tt
            number_of_knights >= 3
        }
    }

    is_valid_showing_failures_only = {
        custom_tooltip = { # vanilla key, as eotg_decision_cartel_adopt
            text = not_changed_government_recently_tt
            NOT = { has_variable = changed_government_recently }
        }
        is_at_war = no
        is_tributary = no
    }

    cost = {
        prestige = 200 # as eotg_decision_cartel_adopt
    }

    effect = {
        set_variable = { # vanilla guard variable, as eotg_decision_cartel_adopt
            name = changed_government_recently
            years = 5
        }
        change_government = eotg_pmc_government
    }

    ai_potential = {
        always = no # OQ7: player-only until V-G11 (leakage) has run
    }

    # Kept for when OQ7 allows AI adoption (spec §6.1)
    ai_will_do = {
        base = 0
        modifier = {
            add = 10
            martial >= 16
        }
        modifier = {
            factor = 0
            primary_title.tier > tier_duchy
        }
    }
}

# Stand Down the Company (spec §6.2). Exit: standing converts to prestige (x4,
# max 400), standing goes to 0 through the change effect, PMC vassals go
# feudal with the ruler. Teardown then runs through on_government_change; the
# (now zero) variable stays dormant.
eotg_decision_pmc_stand_down = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
    }
    decision_group_type = major
    desc = eotg_decision_pmc_stand_down_desc
    selection_tooltip = eotg_decision_pmc_stand_down_tooltip
    confirm_text = eotg_decision_pmc_stand_down_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 120
        duchy = 120
        kingdom = 120
        empire = 120
        hegemony = 120
    }

    is_shown = {
        eotg_pmc_role_top = yes
    }

    is_valid_showing_failures_only = {
        custom_tooltip = {
            text = not_changed_government_recently_tt
            NOT = { has_variable = changed_government_recently }
        }
        is_at_war = no
        custom_tooltip = { # finish the job first
            text = eotg_decision_pmc_stand_down_contract_tt
            NOT = { has_character_modifier = eotg_pmc_under_contract_modifier }
        }
    }

    cost = {
        prestige = medium_prestige_value # as eotg_decision_cartel_go_legitimate
    }

    effect = {
        # (1) Prestige first: the value reads standing before it is zeroed
        add_prestige = eotg_pmc_exit_prestige_value
        # (2) The name is spent
        custom_tooltip = eotg_decision_pmc_stand_down_standing_tt
        hidden_effect = {
            eotg_pmc_change_standing_effect = { AMOUNT = eotg_pmc_exit_standing_loss_value }
        }
        # (3) No orphan PMC vassals under a feudal liege (as the Cartel's exit)
        custom_tooltip = eotg_decision_pmc_stand_down_effect_tt
        hidden_effect = {
            every_vassal_or_below = {
                limit = { government_has_flag = eotg_government_is_pmc }
                change_government = feudal_government
            }
        }
        # (4) Vanilla guard variable
        set_variable = {
            name = changed_government_recently
            years = 5
        }
        # (5)
        change_government = feudal_government
    }

    ai_potential = {
        eotg_pmc_role_top = yes
        eotg_pmc_standing_value < 10
    }

    ai_will_do = {
        base = 0
        modifier = {
            add = 10
            has_trait = content
        }
        modifier = {
            add = 10
            martial < 8
        }
    }
}
