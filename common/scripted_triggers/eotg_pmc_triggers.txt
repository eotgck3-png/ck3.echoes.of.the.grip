# The top: a landed, independent PMC ruler at any tier
eotg_pmc_role_top = {
    is_landed = yes
    government_has_flag = eotg_government_is_pmc
    is_independent_ruler = yes
}

# PMC vassal whose highest title is a county
eotg_pmc_role_county = {
    is_landed = yes
    government_has_flag = eotg_government_is_pmc
    is_independent_ruler = no
    highest_held_title_tier = tier_county
}

# PMC vassal whose highest title is a duchy
eotg_pmc_role_duchy = {
    is_landed = yes
    government_has_flag = eotg_government_is_pmc
    is_independent_ruler = no
    highest_held_title_tier = tier_duchy
}

# PMC vassal whose highest title is a kingdom
eotg_pmc_role_kingdom = {
    is_landed = yes
    government_has_flag = eotg_government_is_pmc
    is_independent_ruler = no
    highest_held_title_tier = tier_kingdom
}

# Any PMC vassal tier (the Gate 2 every-tier rule, v1 lesson 4)
eotg_pmc_role_vassal = {
    OR = {
        eotg_pmc_role_county = yes
        eotg_pmc_role_duchy = yes
        eotg_pmc_role_kingdom = yes
    }
}
