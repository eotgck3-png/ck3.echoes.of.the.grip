# Scripted triggers for the Cybernetic Augmentation system.

# True if the character has any augmentation trait (Phase 3b adds
# eotg_total_integration, spec phase3 §5)
eotg_is_augmented_any = {
    OR = {
        has_trait = eotg_cybernetics
        has_trait = eotg_neurofractured
        has_trait = eotg_total_integration
    }
}

# Tier tests on the eotg_cybernetics track (XP is only ever 0 / 50 / 100).
# Tier 1 = Augmented (XP 0)
eotg_is_aug_tier1 = {
    has_trait = eotg_cybernetics
    NOT = {
        has_trait_xp = {
            trait = eotg_cybernetics
            value >= 50
        }
    }
}

# Tier 2 = Enhanced (XP 50)
eotg_is_aug_tier2 = {
    has_trait = eotg_cybernetics
    has_trait_xp = {
        trait = eotg_cybernetics
        value >= 50
    }
    NOT = {
        has_trait_xp = {
            trait = eotg_cybernetics
            value >= 100
        }
    }
}

# Tier 3 = Overclocked (XP 100)
eotg_is_aug_tier3 = {
    has_trait = eotg_cybernetics
    has_trait_xp = {
        trait = eotg_cybernetics
        value >= 100
    }
}

# Eligible for initial augmentation
# eotg_flag_suppress_progression also blocks initiation: eotg_aug_init.005 (b, c)
# set it as "resist / delay", which in v1 had no effect on an unaugmented ruler.
# The flag test is wrapped so eotg_decision_seek_augmentation's is_valid
# tooltip names the reason instead of the raw flag (vanilla
# common/decisions/10_culture_conversion_decisions.txt:25-27,
# custom_description around NOT = { has_character_flag }).
eotg_can_receive_augmented = {
    # Dead scopes fail without a tooltip line. is_alive has only global trigger
    # loc and renders as BUG text in a perspective tooltip (QA
    # loc_bug_hunt_2026-10-05); a trigger_if limit is never shown. CK3 has no
    # hidden_trigger; shape: vanilla 00_bastard_triggers.txt:5-8.
    trigger_if = {
        limit = { is_alive = no }
        always = no
    }
    is_adult = yes
    NOT = { eotg_is_augmented_any = yes }
    custom_description = {
        text = eotg_can_receive_augmented_suppressed_tt
        NOT = { has_character_flag = eotg_flag_suppress_progression }
    }
    exists = capital_county
    capital_county = {
        development_level >= 10
    }
}

# Eligible to progress Augmented -> Enhanced.
# Balance spec §5.1: the development gate is a script value
# (common/script_values/eotg_augmentation_values.txt; vanilla RHS script-value
# shape, 10_religious_decisions.txt:3054), and the 5-year
# eotg_flag_aug_settling (set by eotg_aug_initiate_effect and
# eotg_aug_set_integration_effect) is a minimum time at tier. A world-state
# gate, not a cooldown: read here, by eotg_decision_aug_pursue_next_stage's
# is_valid, and through these triggers by the on_action Upgrade branches;
# never by an event trigger.
eotg_can_progress_to_enhanced = {
    eotg_is_aug_tier1 = yes
    NOT = { has_character_flag = eotg_flag_suppress_progression }
    NOT = { has_character_flag = eotg_flag_aug_settling }
    exists = capital_county
    capital_county = {
        development_level >= eotg_aug_dev_gate_enhanced
    }
}

# Eligible to progress Enhanced -> Overclocked (same shape)
eotg_can_progress_to_overclocked = {
    eotg_is_aug_tier2 = yes
    NOT = { has_character_flag = eotg_flag_suppress_progression }
    NOT = { has_character_flag = eotg_flag_aug_settling }
    exists = capital_county
    capital_county = {
        development_level >= eotg_aug_dev_gate_overclocked
    }
}

# (eotg_flag_aug_excision_free: eotg_fracture.027's excision outcome, or a
# successor who paid in eotg_aug_end.031.c), else major_gold_value, halved by
# the Heir's Arc promise (eotg_flag_aug_promised_treatment). Mirrors
# eotg_aug_excision_pay_effect. Users: eotg_decision_aug_excision (is_valid),

                multiply = 0.5

        }
    }
}

# -- Realm (docs/specs/cybernetics_v2_realm.md §3.1, G7 / G10 / G6) ---------

# The character lives under their top liege's Augmentation law ($LAW$ = ban /
# license / tolerate / favor). Realm-local (LAW AT 866): the ruler at the head
# of the realm sets it, nothing sits above. An independent ruler governs
# themself; a courtier, knight or spouse follows their employer's top liege.
# Outside any realm: no law (reads as Tolerate; nothing tests tolerate
# positively). The only reader of the law group outside the laws' own
# ai_will_do (spec §9 item 6).
eotg_aug_under_policy = {
    top_liege ?= { has_realm_law = eotg_aug_policy_$LAW$ }
}

# Sanctioned clinics operate: not under a Ban.
# Users: the clinic options at every provider-choice point and the clinic
# send options (spec §3.2); eotg_aug_borrow_technician_interaction.
eotg_aug_clinic_open = {
    NOT = { eotg_aug_under_policy = { LAW = ban } }
}

# The procedure just rolled is a crime under the patient's realm's law: any
# work under a Ban, back-street work under a License. Patient scope; reads
# scope:eotg_proc_provider, which the roll saved (spec deviation: the spec's
# PROVIDER parameter would turn that scope value into a literal and back).
# User: eotg_aug_contraband_effect.
eotg_aug_is_contraband = {
    OR = {
        eotg_aug_under_policy = { LAW = ban }
        AND = {
            eotg_aug_under_policy = { LAW = license }
            scope:eotg_proc_provider ?= flag:backstreet
        }
    }
}

# Refusing $ACTOR$'s order to remove implants is a crime (recipient scope):
# the actor is the recipient's liege or above, and the recipient's realm bans
# augmentation, or licenses it and the recipient's hardware is unlicensed.
# Vanilla shape: refusing_conversion_is_crime_trigger with
# target_is_liege_or_above (00_religious_interactions.txt:160-180, 640-672).
# User: eotg_aug_demand_removal_interaction.
eotg_aug_refusal_is_crime = {
    target_is_liege_or_above = $ACTOR$
    OR = {
        eotg_aug_under_policy = { LAW = ban }
        AND = {
            eotg_aug_under_policy = { LAW = license }
            has_character_modifier = eotg_mod_illegal_implants
        }
    }
}

        eotg_aug_borrowed_technician_valid = yes
    }
}

# Surgeon access, AND the surgeon eotg_aug_save_surgeon_effect would pick
# (same order: own technician, borrowed technician, court physician, the
# character) is not $PATIENT$. Run in the scope of whoever provides the
# surgeon. This keeps the procedures QA fix (the retired
# eotg_aug_has_court_physician: a physician ruler operating on themself at the
# physician's price without the self penalty) and the interactions guard
# (the recipient is not the actor's own physician) now that a technician can
# be the surgeon (realm spec §3.2 conflict, reported).
# `this != $PATIENT$`, not NOT = { this = }: the negated form has no loc in a
# perspective tooltip (QA loc_bug_hunt_2026-10-05; vanilla
# 00_activity_triggers.txt:47 `this != root`).
# Users: the "My own physician" options with root as patient (proc.001.b,
# proc.003.b, tier1.002.f, tier2.003.f, init.018.c, tamper.004.b, int.002.b,
# and the desc_physician lines), the interactions' physician send options,
# eotg_aug_realm.001 (b).
eotg_aug_has_surgeon_for = {
    eotg_aug_has_surgeon_access = yes
    trigger_if = {
        limit = { eotg_aug_own_technician_valid = yes }
        court_position:eotg_aug_implant_technician_court_position = {
            this != $PATIENT$
        }
    }
    trigger_else_if = {
        limit = { eotg_aug_borrowed_technician_valid = yes }
        var:eotg_aug_borrowed_technician = {
            this != $PATIENT$
        }
    }
    trigger_else_if = {
        limit = { employs_court_position = court_physician_court_position }
        court_position:court_physician_court_position = {
            this != $PATIENT$
        }
    }
    trigger_else = {
        this != $PATIENT$
    }
}

# An augmented tournament entrant (character scope): in the contestant subset,

eotg_aug_tournament_entrant = {

}
