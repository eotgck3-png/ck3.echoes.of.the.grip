# -- eotg_augmentation_triggers.txt -------------------------------------------
# Scripted triggers for the Cybernetic Augmentation system.
# -----------------------------------------------------------------------------

# True if the character has any augmentation trait (Phase 3b adds
# eotg_total_integration, spec phase3 §5)
eotg_is_augmented_any = {
    OR = {
        has_trait = eotg_cybernetics
        has_trait = eotg_neurofractured
        has_trait = eotg_total_integration
    }
}

# Tier tests on the eotg_cybernetics track (XP is only ever 0 / 50 / 100).
# Tier 1 = Augmented (XP 0)
eotg_is_aug_tier1 = {
    has_trait = eotg_cybernetics
    NOT = {
        has_trait_xp = {
            trait = eotg_cybernetics
            value >= 50
        }
    }
}

# Tier 2 = Enhanced (XP 50)
eotg_is_aug_tier2 = {
    has_trait = eotg_cybernetics
    has_trait_xp = {
        trait = eotg_cybernetics
        value >= 50
    }
    NOT = {
        has_trait_xp = {
            trait = eotg_cybernetics
            value >= 100
        }
    }
}

# Tier 3 = Overclocked (XP 100)
eotg_is_aug_tier3 = {
    has_trait = eotg_cybernetics
    has_trait_xp = {
        trait = eotg_cybernetics
        value >= 100
    }
}

# Eligible for initial augmentation
# eotg_flag_suppress_progression also blocks initiation: eotg_aug_init.005 (b, c)
# set it as "resist / delay", which in v1 had no effect on an unaugmented ruler.
# The flag test is wrapped so eotg_decision_seek_augmentation's is_valid
# tooltip names the reason instead of the raw flag (vanilla
# common/decisions/10_culture_conversion_decisions.txt:25-27,
# custom_description around NOT = { has_character_flag }).
eotg_can_receive_augmented = {
    is_adult = yes
    is_alive = yes
    NOT = { eotg_is_augmented_any = yes }
    custom_description = {
        text = eotg_can_receive_augmented_suppressed_tt
        NOT = { has_character_flag = eotg_flag_suppress_progression }
    }
    exists = capital_county
    capital_county = {
        development_level >= 10
    }
}

# Eligible to progress Augmented -> Enhanced.
# Balance spec §5.1: the development gate is a script value
# (common/script_values/eotg_augmentation_values.txt; vanilla RHS script-value
# shape, 10_religious_decisions.txt:3054), and the 5-year
# eotg_flag_aug_settling (set by eotg_aug_initiate_effect and
# eotg_aug_set_integration_effect) is a minimum time at tier. A world-state
# gate, not a cooldown: read here, by eotg_decision_aug_pursue_next_stage's
# is_valid, and through these triggers by the on_action Upgrade branches;
# never by an event trigger.
eotg_can_progress_to_enhanced = {
    eotg_is_aug_tier1 = yes
    NOT = { has_character_flag = eotg_flag_suppress_progression }
    NOT = { has_character_flag = eotg_flag_aug_settling }
    exists = capital_county
    capital_county = {
        development_level >= eotg_aug_dev_gate_enhanced
    }
}

# Eligible to progress Enhanced -> Overclocked (same shape)
eotg_can_progress_to_overclocked = {
    eotg_is_aug_tier2 = yes
    NOT = { has_character_flag = eotg_flag_suppress_progression }
    NOT = { has_character_flag = eotg_flag_aug_settling }
    exists = capital_county
    capital_county = {
        development_level >= eotg_aug_dev_gate_overclocked
    }
}

# Neurofracture risk threshold -- checked yearly for Overclocked characters.
# Fixed at 80 (cybernetics_v2 B2): stress now acts through graded accrual in
# eotg_on_yearly_aug_overclocked_check, not through a lower threshold.
eotg_neurofracture_threshold_met = {
    eotg_is_aug_tier3 = yes
    has_variable = eotg_fracture_risk
    var:eotg_fracture_risk >= 80
}

# -- Pressure bands (cybernetics_v2 phase 0 §2.1) -----------------------------
# Read eotg_fracture_risk as a band, never as a number (hidden-risk rule).
# Named for the Neurofractured bands; also used at Overclocked. A missing
# variable counts as 0.
eotg_aug_pressure_flicker = {
    OR = {
        NOT = { has_variable = eotg_fracture_risk }
        var:eotg_fracture_risk < 30
    }
}

eotg_aug_pressure_fracture = {
    has_variable = eotg_fracture_risk
    var:eotg_fracture_risk >= 30
    var:eotg_fracture_risk < 60
}

eotg_aug_pressure_storm = {
    has_variable = eotg_fracture_risk
    var:eotg_fracture_risk >= 60
}

# -- Shared state tests (phase 0 §2.1) ----------------------------------------
# Any vanilla physical-loss trait the implants can replace.
# Counterpart: eotg_aug_restore_loss_effect.
eotg_has_physical_loss = {
    OR = {
        has_trait = maimed
        has_trait = one_legged
        has_trait = one_eyed
        has_trait = blind
    }
}

# A physician to hand: the character is one, or employs one.
# employs_court_position form: vanilla events/activities/hold_court_activity/
# hold_court_events_general.txt (court_physician_court_position).
eotg_has_physician_access = {
    OR = {
        has_trait = lifestyle_physician
        employs_court_position = court_physician_court_position
    }
}

# A court physician who is someone other than the event's root (procedures
# QA, self-physician fix). The "My own physician." options price and roll a
# separate surgeon; a lifestyle_physician with no court physician takes the
# self option (e) at the self price and penalty instead. Filled-position test:
# employs_court_position (as eotg_has_physician_access); holder link as vanilla
# events/court_events/01_ep3_court_events_3.txt:379-412. Evaluate in root's
# scope only.
# Callers: eotg_aug_proc.001 (b, desc_physician), eotg_aug_proc.003 (b,
# desc_physician), eotg_aug_tier1.002 (f), eotg_aug_tier2.003 (f).
eotg_aug_has_court_physician = {
    employs_court_position = court_physician_court_position
    court_position:court_physician_court_position ?= {
        NOT = { this = root }
    }
}

# Who an episode can hit. Evaluated in the courtier's scope inside
# eotg_aug_pick_victim_effect; root is the event's root (the ruler).
# `NOT = { this = root }` excludes the event root on purpose: the episode's
# owner is never their own victim, even when the picker runs on a saved scope
# (where root is still the ruler whose event this is).
eotg_aug_victim_candidate = {
    is_alive = yes
    is_imprisoned = no
    NOT = { this = root }
    NOT = { has_trait = eotg_neurofractured }
}

# AI augmentation interest
eotg_ai_wants_augmentation = {
    is_ai = yes
    OR = {
        has_trait = ambitious
        has_trait = brave
        has_trait = greedy
    }
    NOT = {
        OR = {
            has_trait = zealous
            has_trait = content
        }
    }
}

# -- Phase 3a (docs/specs/cybernetics_v2_phase3.md) ---------------------------
# Someone who can hold a Containment Regency for $RULER$: alive, adult, free,
# and in the ruler's court or a direct vassal (vanilla diarchs are courtiers or
# vassals). Used by the NF on_action Storm branch (through
# eotg_aug_containment_possible), eotg_fracture.025's keeper pick,
# eotg_fracture.026 (a) and the Heir's Arc regency options, so the branch and
# the events never drift apart.
eotg_aug_keeper_candidate = {
    is_alive = yes
    is_adult = yes
    is_imprisoned = no
    OR = {
        is_courtier_of = $RULER$
        is_vassal_of = $RULER$
    }
}

# The Containment Regency can be offered to $RULER$ (run in the ruler's
# scope): vanilla eligibility, no diarchy yet, and a keeper (the primary
# spouse, else the adult primary heir). Mirrored by the Storm branch and
# eotg_fracture.025's trigger.
eotg_aug_containment_possible = {
    basic_eligible_for_diarchy_trigger = yes
    has_active_diarchy = no
    OR = {
        primary_spouse ?= { eotg_aug_keeper_candidate = { RULER = $RULER$ } }
        primary_heir ?= { eotg_aug_keeper_candidate = { RULER = $RULER$ } }
    }
}

# The Heir's Arc story exists for this character (thread T3).

}

# -- Phase 3b (docs/specs/cybernetics_v2_phase3.md §6-8) ----------------------
# Excision can be paid for: free when the surgeons are already in the room
# (eotg_flag_aug_excision_free: eotg_fracture.027's excision outcome, or a
# successor who paid in eotg_aug_end.031.c), else major_gold_value, halved by
# the Heir's Arc promise (eotg_flag_aug_promised_treatment). Mirrors
# eotg_aug_excision_pay_effect. Users: eotg_decision_aug_excision (is_valid),
# eotg_aug_end.001 (a, c, d).
# The free-excision flag is wrapped for the decision tooltip (as
# eotg_can_receive_augmented).
eotg_aug_can_afford_excision = {
    OR = {
        custom_description = {
            text = eotg_aug_can_afford_excision_free_tt
            has_character_flag = eotg_flag_aug_excision_free
        }
        gold >= {
            value = major_gold_value
            if = {
                limit = { has_character_flag = eotg_flag_aug_promised_treatment }
                multiply = 0.5
            }
        }
    }
}

# A Warden can be appointed for $RULER$ (run in the ruler's scope): the
# primary spouse, the adult primary heir or the chancellor, each a valid
# keeper. Users: eotg_decision_aug_appoint_warden (is_valid),
# eotg_aug_end.020 (trigger).
eotg_aug_has_warden_candidate = {
    OR = {
        primary_spouse ?= { eotg_aug_keeper_candidate = { RULER = $RULER$ } }
        primary_heir ?= { eotg_aug_keeper_candidate = { RULER = $RULER$ } }
        cp:councillor_chancellor ?= { eotg_aug_keeper_candidate = { RULER = $RULER$ } }
    }
}

# -- Phase 4 (docs/specs/cybernetics_v2_phase4.md) ----------------------------
# The First Upgrade can be paid for (mirror: eotg_aug_pay_upgrade_effect).
# Procedures spec §4.6: MULT is the provider's price multiplier (1 at the
# clinic; eotg_aug_price_mult_* otherwise).
# Users: eotg_on_yearly_aug_tier1_check (The Upgrade branch, MULT = 1),
# eotg_aug_tier1.002 (a, d, e: 1; f: physician; g: back streets),
# eotg_decision_aug_pursue_next_stage (is_valid: back streets, the cheapest).
eotg_aug_can_afford_upgrade = {
    gold >= {
        value = medium_gold_value
        if = {
            limit = { has_character_flag = eotg_flag_aug_replace_more }
            multiply = 0.85
        }
        multiply = $MULT$
    }
}

# Someone the syndicate's Errand (eotg_aug_patron.004) can name: the victim
# candidate terms (eotg_aug_victim_candidate) against $RULER$, never the
# ruler's close family or spouse. Evaluated in the courtier's scope. A
# parameter, not root, because the Patron story's tick (story scope) also
# asks. Users: eotg_story_aug_patron (tick), eotg_aug_patron.004.
eotg_aug_patron_critic_candidate = {
    is_alive = yes
    is_adult = yes    # QA round 2 M10: the syndicate never names a child
    is_imprisoned = no
    NOT = { this = $RULER$ }
    NOT = { has_trait = eotg_neurofractured }
    NOT = { is_close_family_of = $RULER$ }
    NOT = { is_spouse_of = $RULER$ }
    # CB-27 M8: never the Patron story's own envoy (story var eotg_envoy),
    # who is a courtier like any other. Compared against a temporary scope so
    # every caller (the tick, patron.004's trigger and its pick) gets it with
    # no new parameter. Vanilla shape: events/court_events/
    # 01_ep3_court_events.txt:557-570 (save_temporary_scope_as in a limit,
    # then `var:hostage_target ?= scope:hostage_temp` in any_owned_story).
    save_temporary_scope_as = eotg_critic_check
    NOT = {
        $RULER$ = {
            any_owned_story = {
                story_type = eotg_story_aug_patron
                var:eotg_envoy ?= scope:eotg_critic_check
            }
        }
    }
}

# The Patron envoy is alive and still at the owner's court, so the Final
# Demand (eotg_aug_patron.006) can show them, say they came, and let the
# owner kill them in person. Evaluated in the event's root (the story owner)
# after patron.006's immediate has saved scope:eotg_patron_envoy.
# CB-27 M7. Vanilla precedent: events/story_cycles/peasant_affair/
# story_cycle_peasant_affair_events.txt:1173-1178 (an option that acts on a
# story character requires them alive and at root's court).
# Users: eotg_aug_patron.006 (right portrait, desc, options c and e),
# eotg_aug_patron_betray_effect (the kill).
eotg_aug_patron_envoy_present = {
    scope:eotg_patron_envoy ?= {
        is_alive = yes
        is_courtier_of = root
    }
}

# The syndicate's bill can be paid (mirror: eotg_aug_patron_pay_effect). Run in
# the ruler's scope; scope:eotg_patron_story must be saved.
eotg_aug_patron_can_pay = {
    gold >= {
        value = $BASE$
        if = {
            limit = { scope:eotg_patron_story.var:eotg_terms = flag:greedy }
            multiply = 1.5
        }
        else_if = {
            limit = { scope:eotg_patron_story.var:eotg_terms = flag:read }
            multiply = 0.75
        }
        multiply = $MULT$
    }
}

# The injury named by var:eotg_aug_repair_injury (set for 365 days by
# eotg_on_trait_gained_aug_repair) is still held. A world-state guard, not a

    OR = {
        AND = {
            var:eotg_aug_repair_injury ?= flag:wounded
            has_trait = wounded_3
        }
        AND = {
            var:eotg_aug_repair_injury ?= flag:maimed
            has_trait = maimed
        }
        AND = {
            var:eotg_aug_repair_injury ?= flag:one_legged
            has_trait = one_legged
        }
        AND = {
            var:eotg_aug_repair_injury ?= flag:one_eyed
            has_trait = one_eyed
        }
        AND = {
            var:eotg_aug_repair_injury ?= flag:blind
            has_trait = blind
        }
