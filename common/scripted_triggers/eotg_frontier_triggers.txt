# -- eotg_frontier_triggers.txt -----------------------------------------------
# Scripted triggers for Frontier Systems v1 (docs/specs/frontier_v1.md §3.3).
#
# Frontier state lives on the COUNTY TITLE (a Region), never on a character
# (spec §R). Title variables are confirmed in vanilla 1.20 (verification V1:
# province_on_actions.txt:904). TEST-IN-GAME (V1): they persist through
# save/reload and a holder change. Fallback if not: a story cycle holds the
# state (never the founder).
#
# Comparisons against flag values use `?=`, as the mod's v2 script does
# (events/eotg_augmentation_procedures.txt, eotg_aug_proc.001).
# -----------------------------------------------------------------------------

# -- State (county scope) ------------------------------------------------------
eotg_frontier_is_unsettled = {
    var:eotg_frontier_state ?= flag:unsettled
}

eotg_frontier_is_frontier = {
    var:eotg_frontier_state ?= flag:frontier
}

eotg_frontier_is_abandoned = {
    var:eotg_frontier_state ?= flag:abandoned
}

# A project can start here: Unsettled, or Abandoned and resettlable
# (spec §6.6), and the Region's capital is not occupied (title_province,
# is_occupied: confirmed, verification V7, V11).
eotg_frontier_can_establish = {
    OR = {
        eotg_frontier_is_unsettled = yes
        eotg_frontier_is_abandoned = yes
    }
    title_province = { is_occupied = no }
}

# Type eligibility (spec §5.1). Six types need only eotg_frontier_can_establish;
# Administrative also needs this: the holder is a duke or above, or the Region
# is not their capital Region. A new type with its own test adds one trigger
# here and one option trigger in eotg_frontier.001 (no identifier-building
# parameters: spec §16 V14).
eotg_frontier_admin_eligible = {
    eotg_frontier_can_establish = yes
    OR = {
        holder = { highest_held_title_tier >= tier_duchy }
        NOT = { this = holder.capital_county }
    }
}

# The founder can still lead the project (spec §4.1, §R; ruling Q2: an invalid
# founder is the strain cause No Founder). Orchestrator ruling (verification
# Q3): alive, adult, free, and any of:
#   - the founder is the holder;
#   - the founder is in the holder's realm (employer, or vassal or below);
#   - the holder is vassal or below of the founder (the founder granted the
#     Region away: the founder stays valid).
# Each `prev` is one step from the county (into the founder, or the holder).
eotg_frontier_has_valid_founder = {
    exists = var:eotg_frontier_founder
    var:eotg_frontier_founder = {
        is_alive = yes
        is_adult = yes
        is_imprisoned = no
    }
    OR = {
        holder = var:eotg_frontier_founder
        var:eotg_frontier_founder = { employer ?= prev.holder }
        var:eotg_frontier_founder = { is_vassal_or_below_of = prev.holder }
        holder = { is_vassal_or_below_of = prev.var:eotg_frontier_founder }
    }
}

# A sponsor that can still back the project (spec §4.2, §4.3)
eotg_frontier_has_valid_sponsor = {
    exists = var:eotg_frontier_sponsor
    var:eotg_frontier_sponsor = { is_alive = yes }
}

# Establishment has reached Settled, and the type's floors are met (spec §5.2;
# the development floor is relative, Q4)
eotg_frontier_completion_met = {
    eotg_frontier_is_frontier = yes
    var:eotg_frontier_progress >= 100
    development_level >= eotg_frontier_min_dev_value
    county_control >= eotg_frontier_min_control_value
}

# -- Character scope -----------------------------------------------------------
# any_held_title with tier = tier_county: confirmed (verification V13).
eotg_frontier_holds_frontier = {
    any_held_title = {
        tier = tier_county
        eotg_frontier_is_frontier = yes
    }
}

eotg_frontier_holds_establishable = {
    any_held_title = {
        tier = tier_county
        eotg_frontier_can_establish = yes
    }
}

# Sponsors a Frontier. A sponsor backs one Frontier at a time
# (eotg_frontier_can_sponsor), and carries the back-reference
# var:eotg_frontier_sponsoring = that Region, written only by
# eotg_frontier_set_sponsor_effect / eotg_frontier_clear_sponsor_effect.
# A character variable, not a scope-dependent search: correct in any scope.
eotg_frontier_is_sponsor = {
    has_variable = eotg_frontier_sponsoring
}

# Can back a Frontier (spec §3.3): a landed ruler with gold to spare who
# sponsors none yet. Read in the would-be sponsor's scope.
eotg_frontier_can_sponsor = {
    is_landed = yes
    highest_held_title_tier >= tier_county
    gold >= {
        value = eotg_frontier_sponsor_payment_value
        multiply = 2
    }
    eotg_frontier_is_sponsor = no
}

# -- Global ---------------------------------------------------------------------
# Room for another AI Frontier (spec §8.4: at most eotg_frontier_ai_cap_value
# active on the map). Read from the active list itself, the one record of
# active Frontiers (verification Q-B1; `count` form: yearly_on_actions.txt:846).
eotg_frontier_ai_room = {
    NOT = {
        any_in_global_list = {
            variable = eotg_frontier_active
            count >= eotg_frontier_ai_cap_value
        }
    }
}

# An AI ruler who would back a Frontier: can sponsor, and has gold for four
# payments (the Sponsor decision's own ai_potential; verification notes:
# AI-to-AI offers apply the same gate). Character scope.
eotg_frontier_ai_would_sponsor = {
    is_ai = yes
    eotg_frontier_can_sponsor = yes
    gold >= {
        value = eotg_frontier_sponsor_payment_value
        multiply = 4
    }
}

# == Phase 2 (docs/specs/frontier_v2.md §3.3) ================================
# UNVERIFIED-VANILLA: a script value on the LEFT of a comparison
# (eotg_frontier_trait_count_value < eotg_frontier_trait_max_value). Phase 1
# only put script values on the right (spec frontier_v2.md §15 item 5).

# -- Traits and infrastructure (county scope) ----------------------------------
# Room for one more known trait (spec §5.1: at most two)
eotg_frontier_trait_room = {
    eotg_frontier_trait_count_value < eotg_frontier_trait_max_value
}

# A Frontier that can build: a free slot, and nothing built here this year
# (spec §4.1)
eotg_frontier_infra_room = {
    eotg_frontier_is_frontier = yes
    eotg_frontier_infra_count_value < eotg_frontier_infra_slots_value
    NOT = { has_variable = eotg_frontier_built }
}

# A Region a survey can learn more about (spec §8): Unsettled, a Frontier or
# Abandoned, with trait room, not surveyed in the last five years
eotg_frontier_can_survey = {
    OR = {
        eotg_frontier_is_unsettled = yes
        eotg_frontier_is_frontier = yes
        eotg_frontier_is_abandoned = yes
    }
    eotg_frontier_trait_room = yes
    NOT = { has_variable = eotg_frontier_surveyed }
}

# -- Strain conditions (county scope; read by the yearly tick) ----------------
# Dangerous: raiders and worse take their chance while the hold is weak.
# Countered by a Garrison Post, or by a Military project (spec §5.2).
eotg_frontier_danger_strains = {
    has_county_modifier = eotg_frontier_trait_dangerous
    county_control < 60
    NOT = { has_county_modifier = eotg_frontier_infra_garrison_post }
    NOT = { var:eotg_frontier_type ?= flag:military }
}

# A quiet year may ease strain, unless the place itself is hostile and nothing
# shelters the settlers from it (spec §5.2)
eotg_frontier_quiet_eases = {
    OR = {
        NOT = { has_county_modifier = eotg_frontier_trait_hostile }
        has_county_modifier = eotg_frontier_infra_orbital_station
    }
}

# War adds strain, except to a Military project, built for it (spec §6.1)
eotg_frontier_war_strains = {
    NOT = { var:eotg_frontier_type ?= flag:military }
}

# -- Character scope -----------------------------------------------------------
eotg_frontier_holds_surveyable = {
    any_held_title = {
        tier = tier_county
        eotg_frontier_can_survey = yes
    }
}

eotg_frontier_holds_buildable = {
    any_held_title = {
        tier = tier_county
        eotg_frontier_infra_room = yes
    }
}
