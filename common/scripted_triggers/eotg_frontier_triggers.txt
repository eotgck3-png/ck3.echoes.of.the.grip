# -- eotg_frontier_triggers.txt -----------------------------------------------
# Scripted triggers for Frontier Systems v1 (docs/specs/frontier_v1.md §3.3).
#
# Frontier state lives on the COUNTY TITLE (a Region), never on a character
# (spec §R). Title variables are confirmed in vanilla 1.20 (verification V1:
# province_on_actions.txt:904). TEST-IN-GAME (V1): they persist through
# save/reload and a holder change. Fallback if not: a story cycle holds the
# state (never the founder).
#
# Comparisons against flag values use `?=`, as the mod's v2 script does
# (events/eotg_augmentation_procedures.txt, eotg_aug_proc.001).
# -----------------------------------------------------------------------------

# -- State (county scope) ------------------------------------------------------
eotg_frontier_is_unsettled = {
    var:eotg_frontier_state ?= flag:unsettled
}

eotg_frontier_is_frontier = {
    var:eotg_frontier_state ?= flag:frontier
}

eotg_frontier_is_abandoned = {
    var:eotg_frontier_state ?= flag:abandoned
}

# A project can start here: Unsettled, or Abandoned and resettlable
# (spec §6.6), and the Region's capital is not occupied (title_province,
# is_occupied: confirmed, verification V7, V11).
eotg_frontier_can_establish = {
    OR = {
        eotg_frontier_is_unsettled = yes
        eotg_frontier_is_abandoned = yes
    }
    title_province = { is_occupied = no }
    # Phase 3a (frontier_v3.md §4): not until explored past Unknown
    eotg_frontier_is_unknown = no
    # Unclaimed Regions (frontier_unclaimed_regions.md §3.6): never while
    # unclaimed. Defence in depth: the decision only offers counties the taker
    # holds, so a claim always comes first.
    eotg_unclaimed_is_unclaimed = no
}

# Type eligibility (spec §5.1). Six types need only eotg_frontier_can_establish;
# Administrative also needs this: the holder is a duke or above, or the Region
# is not their capital Region. A new type with its own test adds one trigger
# here and one option trigger in eotg_frontier.001 (no identifier-building
# parameters: spec §16 V14).
eotg_frontier_admin_eligible = {
    eotg_frontier_can_establish = yes
    OR = {
        holder = { highest_held_title_tier >= tier_duchy }
        NOT = { this = holder.capital_county }
    }
}

# The founder can still lead the project (spec §4.1, §R; ruling Q2: an invalid
# founder is the strain cause No Founder). Orchestrator ruling (verification
# Q3): alive, adult, free, and any of:
#   - the founder is the holder;
#   - the founder is in the holder's realm (employer, or vassal or below);
#   - the holder is vassal or below of the founder (the founder granted the
#     Region away: the founder stays valid).
# Each `prev` is one step from the county (into the founder, or the holder).
eotg_frontier_has_valid_founder = {
    exists = var:eotg_frontier_founder
    var:eotg_frontier_founder = {
        is_alive = yes
        is_adult = yes
        is_imprisoned = no
    }
    OR = {
        holder = var:eotg_frontier_founder
        var:eotg_frontier_founder = { employer ?= prev.holder }
        var:eotg_frontier_founder = { is_vassal_or_below_of = prev.holder }
        holder = { is_vassal_or_below_of = prev.var:eotg_frontier_founder }
    }
}

# frontier_v4.md A1 "Founder first" (owner H1): the Frontier's founder may be
# the Local Ruler of the new $HOLDING$ System under $HOLDER$. COUNTY scope.
# $HOLDING$: city_holding or church_holding. $HOLDER$: the county holder
# (scope:eotg_frontier_holder in the grant; root in .004's tooltip). The
# founder is read from var:eotg_frontier_founder, which survives until the
# settle clears the project. Every case that fails falls back to the type's
# figure (§A1 edge-case table):
#   - the holder, dead, a child or a prisoner: no (validity, as for strain);
#   - a courtier elsewhere (a vassal's court, another realm, a guest): no;
#     vanilla grants only to the granter's own people (hold_court 6060's
#     random_knight, 8300's random_vassal_or_below);
#   - a direct vassal seated in a different holding type, a sub-vassal, or
#     the holder's liege: no (vanilla recasts a landed vassal's government
#     only for the AI, hold_court_events_joe.txt:295-305);
#   - an Unsworn placeholder or folk: no (M1 guards).
# Yes: a landless courtier of the holder (vanilla 6060: a landless knight is
# granted the city, hold_court_events_general.txt:6723-6729), or a direct
# vassal already seated in $HOLDING$ (keeps their government).
eotg_frontier_founder_can_rule = {
    exists = var:eotg_frontier_founder
    eotg_frontier_has_valid_founder = yes            # alive, adult, free, tied to the holder
    var:eotg_frontier_founder = {
        NOT = { this = $HOLDER$ }
        # Orchestrator ruling, 2026-10-11 (batch 2 QA finding 3): never the
        # holder's spouse or primary heir, who would otherwise become a city or
        # church vassal (the heir pulled off to govern a System). They fall back
        # to the type's figure, as every other excluded case. Forms per spec
        # §A1: is_primary_heir_of (vanilla coronation.txt:3297); primary_spouse
        # ?= (vanilla 00_marriage_interaction_effects.txt:48), false with no
        # spouse. `prev` is the founder: the scope this block is in, one step
        # before $HOLDER$
        NOT = { is_primary_heir_of = $HOLDER$ }
        NOT = { $HOLDER$ = { primary_spouse ?= prev } }
        eotg_is_unclaimed_folk = no
        eotg_is_unclaimed_holder = no
        # Fix I1b F3: the vanilla grant gender checks. A church goes through
        # can_be_granted_theocratic_titles_by (the granter's clergy gender;
        # 00_scripted_triggers.txt:368-384), anything else through
        # can_be_granted_titles_by (the faith's ruling gender, or the realm
        # gender law under an administrative government; :305-366). A founder
        # who fails falls back to the type's figure, as every excluded case.
        # Parameter comparison by flag: vanilla 00_animal_effects.txt:1678.
        trigger_if = {
            limit = { flag:$HOLDING$ = flag:church_holding }
            can_be_granted_theocratic_titles_by = { RULER = $HOLDER$ }
        }
        trigger_else = {
            can_be_granted_titles_by = { RULER = $HOLDER$ }
        }
        OR = {
            AND = {                                  # a landless courtier of the holder
                is_landed = no
                employer ?= $HOLDER$
            }
            AND = {                                  # a direct vassal already seated in this holding type
                is_landed = yes
                liege ?= $HOLDER$
                capital_province ?= { has_holding_type = $HOLDING$ }
            }
        }
    }
}

# A sponsor that can still back the project (spec §4.2, §4.3). COUNTY scope.
# Investigation rulings 2026-10-18 §6 (I2c, F2): also free, still a landed
# count or above, and not at war with the holder. Every reader (the payment,
# the completion reward, the event triggers) agrees through this one trigger.
# Precedent: war voids a backing (vanilla 01_pam_projects.txt:5551-5561);
# imprisonment voids action (befriend_scheme.txt:50-57). `prev` is the
# county, one step up from the holder (the eotg_frontier_faiths_differ form).
# A holder change is not tested here: eotg_frontier_check_sponsor_effect
# reads var:eotg_frontier_sponsor_holder at the yearly tick.
eotg_frontier_has_valid_sponsor = {
    exists = var:eotg_frontier_sponsor
    var:eotg_frontier_sponsor = {
        is_alive = yes
        is_imprisoned = no
        # Orchestrator ruling 2026-10-20 (I2c QA S1): a backer who may hold no
        # county (eotg_frontier_can_back: a company captain, a holy order's
        # leader, a head of faith without land) is exempt from the landed test.
        # The faith and order ties are the variables
        # eotg_frontier_set_faith_backer_effect sets for those found by
        # eotg_frontier_find_faith_backer_effect
        OR = {
            eotg_frontier_is_company_captain = yes
            has_variable = eotg_frontier_backs_for_order
            has_variable = eotg_frontier_backs_for_faith
            AND = {
                is_landed = yes
                highest_held_title_tier >= tier_county
            }
        }
    }
    holder ?= { NOT = { is_at_war_with = prev.var:eotg_frontier_sponsor } }
}

# frontier_v5.md §3.4.2 (round 1): the settlers do not share one faith.
# COUNTY scope. True when the Region's faith differs from its holder's, or a
# valid founder's faith differs from the Region's. Map-agnostic: compares
# scopes and never names a faith. `prev` is the county, one step up from the
# founder (the eotg_frontier_has_valid_founder pattern). Read by the flavour
# roll's .055 entry and repeated in .055's trigger (world state, so a
# conversion during the delay voids the event).
eotg_frontier_faiths_differ = {
    OR = {
        NOT = { faith = holder.faith }
        AND = {
            eotg_frontier_has_valid_founder = yes
            var:eotg_frontier_founder = { NOT = { faith = prev.faith } }
        }
    }
}

# frontier_v5.md §3.4.2 (round 1): the same, for culture (.056). COUNTY scope.
eotg_frontier_cultures_differ = {
    OR = {
        NOT = { culture = holder.culture }
        AND = {
            eotg_frontier_has_valid_founder = yes
            var:eotg_frontier_founder = { NOT = { culture = prev.culture } }
        }
    }
}

# Establishment has reached Settled, and the type's floors are met (spec §5.2;
# the development floor is relative, Q4)
eotg_frontier_completion_met = {
    eotg_frontier_is_frontier = yes
    var:eotg_frontier_progress >= 100
    development_level >= eotg_frontier_min_dev_value
    county_control >= eotg_frontier_min_control_value
}

# Top-bar nudge for Establish a Frontier (integration_v1.md §4.2, E1a): the

# Sponsors a Frontier. A sponsor backs one Frontier at a time
# (eotg_frontier_can_sponsor), and carries the back-reference
# var:eotg_frontier_sponsoring = that Region, written only by
# eotg_frontier_set_sponsor_effect / eotg_frontier_clear_sponsor_effect.
# A character variable, not a scope-dependent search: correct in any scope.
eotg_frontier_is_sponsor = {
    has_variable = eotg_frontier_sponsoring
}

# Can back a Frontier (spec §3.3): a landed ruler with gold to spare who
# sponsors none yet. Read in the would-be sponsor's scope.
eotg_frontier_can_sponsor = {
    is_landed = yes
    is_imprisoned = no                               # I2c (investigation rulings §6.3)
    highest_held_title_tier >= tier_county
    gold >= {
        value = eotg_frontier_sponsor_payment_value
        multiply = 2
    }
    eotg_frontier_is_sponsor = no
}

# -- Global ---------------------------------------------------------------------

# active on the map). Read from the active list itself, the one record of

eotg_frontier_ai_room = {

        }
    }

# -- Strain conditions (county scope; read by the yearly tick) ----------------
# Dangerous: raiders and worse take their chance while the hold is weak,
# meaning below the project's own completion control floor (verification 4: a
# project that meets its floor is not on a clock). Countered by a Garrison
# Post, or by a Military project (spec §5.2).
eotg_frontier_danger_strains = {
    has_county_modifier = eotg_frontier_trait_dangerous
    county_control < eotg_frontier_min_control_value
    NOT = { has_county_modifier = eotg_frontier_infra_garrison_post }
    NOT = { var:eotg_frontier_type ?= flag:military }
    # Phase 3a (frontier_v3.md §5): a hired escort counters it too
    NOT = { has_county_modifier = eotg_frontier_mod_merc_escort }
}

# A quiet year may ease strain, unless the place itself is hostile and nothing
# shelters the settlers from it (spec §5.2)
eotg_frontier_quiet_eases = {
    OR = {
        NOT = { has_county_modifier = eotg_frontier_trait_hostile }
        has_county_modifier = eotg_frontier_infra_orbital_station
    }
}

# War adds strain, except to a Military project, built for it (spec §6.1)
eotg_frontier_war_strains = {
    NOT = { var:eotg_frontier_type ?= flag:military }
}

# frontier_v4.md A4.1: a war reaches the Region only if the holder leads it,
# or hostile armies stand in the Region; a distant ally's call does not.
# Occupation stays its own cause. COUNTY scope; reads scope:eotg_frontier_holder
# (saved by the tick at entry). Precedents: any_character_war /
# is_war_leader (00_war.txt:718-724), any_army_in_location
# (yearly_on_actions.txt:1896-1898), army_owner (army_on_actions.txt:651-653).
# Fix I1b F1: a vassal's Region is also reached by its top liege's war. The
# war state and the hostile army's owner are tested against the top liege as
# well as the holder, as vanilla tells a liege's war from one's own
# (governance_on_actions.txt:1038-1039: `liege = { is_at_war = yes }` beside
# `is_at_war = no`). Leading the war stays the holder's own test (A4.1: a
# distant call does not reach the Region; only armies in it do). For an
# independent holder, top_liege is the holder, so nothing changes.
eotg_frontier_war_reaches = {
    eotg_frontier_war_strains = yes
    OR = {
        scope:eotg_frontier_holder = { is_at_war = yes }
        scope:eotg_frontier_holder.top_liege = { is_at_war = yes }
    }
    OR = {
        scope:eotg_frontier_holder = {
            any_character_war = { is_war_leader = prev }
        }
        any_county_province = {
            any_army_in_location = {
                army_owner ?= {
                    OR = {
                        is_at_war_with = scope:eotg_frontier_holder
                        is_at_war_with = scope:eotg_frontier_holder.top_liege
                    }
                }
            }
        }
    }
}

# An expedition can go: Unknown or Partial, none this year

    OR = {
        eotg_frontier_is_unknown = yes
        eotg_frontier_is_partial = yes

# THE expedition targeting rule (spec §4.1): may $ACTOR$ send an expedition to
# this Region? County scope. An explorable Region $ACTOR$ holds, or (Unclaimed
# Regions build) an explorable Unclaimed Region within $ACTOR$'s reach, by the
# claim interaction's own reach test (frontier_unclaimed_regions.md §8.2).
# Exploring gives no claim and no say over it (LAW AT 866); claiming stays the
# interaction's job. Every caller goes through this one trigger.
eotg_frontier_can_target_expedition = {
    eotg_frontier_can_explore = yes
    OR = {
        holder = $ACTOR$
        AND = {
            eotg_unclaimed_is_unclaimed = yes
            eotg_unclaimed_in_reach = { ACTOR = $ACTOR$ }
        }
    }
}

# Character scope, called with the decision taker as root (is_shown,
# ai_potential): there is a Region this character may send an expedition to
eotg_frontier_holds_explorable = {
    OR = {
        any_held_title = {
            tier = tier_county
            eotg_frontier_can_target_expedition = { ACTOR = root }
        }
        # Unclaimed Regions within reach (frontier_v3.md §4.1); the list is
        # the system's own record (eotg_unclaimed_counties)
        any_in_global_list = {
            variable = eotg_unclaimed_counties
            eotg_frontier_can_target_expedition = { ACTOR = root }
        }
    }
}

# -- Backers who may hold no county (frontier_v3.md §5, §6) --------------------
# A mercenary captain or a faith leader who could back a Frontier: alive,
# backing nothing yet, with gold for two payments. Narrower than
# eotg_frontier_can_sponsor (which also needs a landed count or above, for
# .003); .040 and .041 call eotg_frontier_set_sponsor_effect with it, so
# payments, lapse, withdrawal and the hooks work unchanged. Character scope.
eotg_frontier_can_back = {
    is_alive = yes
    eotg_frontier_is_sponsor = no
    gold >= {
        value = eotg_frontier_sponsor_payment_value
        multiply = 2
    }
}

# Leads a mercenary company. Character scope. government_is_mercenary:
# confirmed against vanilla 1.20 (spec frontier_v3.md §15 item 1). Fix I1b X5:
# a PMC sells force too; its identity flag is eotg_government_is_pmc
# (common/governments/eotg_pmc_government.txt, flags block).
eotg_frontier_is_company_captain = {
    OR = {
        government_has_flag = government_is_mercenary
        government_has_flag = eotg_government_is_pmc
    }
}
