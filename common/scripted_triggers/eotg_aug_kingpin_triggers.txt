# -- eotg_aug_kingpin_triggers.txt --------------------------------------------
# Scripted triggers for the Neurofractured Kingpin (docs/specs/
# cybernetics_v2_kingpin.md §3, §5.1). Batches A and B. Map-agnostic: no
# title, province, culture or faith keys; every title comes from a scope.
#
# Batch B added the claim-path triggers (eotg_aug_kp_root_is_independent,
# eotg_aug_kp_liege_claim_possible, eotg_aug_kp_neighbour_target,
# eotg_aug_kp_vassal_seat_target) at the end of the file. War immunity: no
# claim target is an Unclaimed placeholder (eotg_is_unclaimed_holder /
# eotg_is_unclaimed_folk; docs/specs/frontier_unclaimed_regions.md §4.3).
# -----------------------------------------------------------------------------

# Entry eligibility (spec §5.1.1), plus one kingpin at a time and once per
# life (§5.1.3). World state only: the 10-year cooldown flag is read by
# eotg_on_yearly_aug_kingpin_check alone (cooldown authority, invariant 4).
# eotg_flag_aug_had_kingpin is the permanent once-per-life marker set in the
# story's on_setup, not a cooldown (precedent eotg_flag_aug_had_patron).
# Users: eotg_on_yearly_aug_kingpin_check, eotg_aug_kingpin.001 (trigger).
eotg_aug_kp_can_start = {
    is_landed = yes
    highest_held_title_tier >= tier_county
    is_adult = yes
    is_imprisoned = no
    eotg_is_unclaimed_folk = no
    # Seamless exposure (event_quality_v1 §12.5.3 B13 item 2): the kingpin
    # beats are first person; a Seamless ruler does not start the story
    NOT = { has_trait = eotg_total_integration }
    NOT = { government_has_flag = government_is_landless_adventurer }
    # The same development floor as eotg_can_receive_augmented: implants are
    # traded where the capital is developed enough
    exists = capital_county
    capital_county = { development_level >= 10 }
    NOT = { any_owned_story = { story_type = eotg_story_aug_kingpin } }
    NOT = { has_character_flag = eotg_flag_aug_had_kingpin }
}

# The character owns a running kingpin story whose kingpin is alive. World
# state, read by every kingpin event after .001 so a delayed event never
# fires into an ended story or onto a dead kingpin (the story's maintenance
# ends those). Not a cooldown.
eotg_aug_kp_has_story = {
    any_owned_story = {
        story_type = eotg_story_aug_kingpin
        var:eotg_kp ?= { is_alive = yes }
    }
}

# The kingpin is the character's own prisoner (custody leaves that execute or
# keep them in the cells: .060, .061, .076 a). Evaluated in the ruler's scope.
eotg_aug_kp_in_my_cells = {
    any_owned_story = {
        story_type = eotg_story_aug_kingpin
        var:eotg_kp ?= {
            is_alive = yes
            is_imprisoned_by = root
        }
    }
}

# The root is a vassal: the liege-claim branches are open (spec §5.1.1).
# Batch A reader: eotg_aug_kingpin.003 (e_front, "Your files on [liege]").
eotg_aug_kp_root_is_vassal = {
    is_independent_ruler = no
    exists = liege
}

# A county the kingpin can be given (L10) or can seize (L12): held by the
# ruler, not the capital, tier county (spec §3). Title scope; $RULER$ is the
# ruler whose county it is (root in events).
# Users: .012 c, .013 d, .014 c, .069, .071, eotg_aug_kp_grant_county_effect,
# eotg_aug_kp_seize_county_effect.
eotg_aug_kp_seizable_county = {
    tier = tier_county
    holder = $RULER$
    NOT = { this = $RULER$.capital_county }
}

# Someone the violence errand (.005) can name: an adult courtier of $RULER$,
# not the ruler, not their close family or spouse, free, and not the
# kingpin's planted informant (index §1 rule 7: a chosen target).
# Users: .005 (immediate), the story tick's middle pool.
eotg_aug_kp_errand_target = {
    is_alive = yes
    is_adult = yes
    is_imprisoned = no
    NOT = { this = $RULER$ }
    NOT = { is_close_family_of = $RULER$ }
    NOT = { is_spouse_of = $RULER$ }
    NOT = { has_character_flag = eotg_flag_aug_kingpin }
    NOT = { has_trait = eotg_neurofractured }
}

# The purge mind's mark (.006 purge): one of the ruler's courtiers the
# kingpin's implant has them going for. Same terms as the errand, and never
# a child. Users: .006 (immediate), the story tick's middle pool.
eotg_aug_kp_purge_target = {
    eotg_aug_kp_errand_target = { RULER = $RULER$ }
    age >= 16
}

# A courtier who can be the kingpin's informant when none was planted
# (.007): adult, free, not family, with some intrigue.
# Users: .007 (immediate), the story tick's middle pool.
eotg_aug_kp_informant_candidate = {
    eotg_aug_kp_errand_target = { RULER = $RULER$ }
    intrigue >= 6
}

# Profile read (spec §3): the kingpin's axis $AXIS$ (kind / method / reach /
# mind) holds flag:$VALUE$. Needs scope:eotg_kp (saved by
# eotg_aug_kp_save_scopes_effect in events, and at the start of the tick).
eotg_aug_kp_is = {
    scope:eotg_kp ?= { var:eotg_kp_$AXIS$ ?= flag:$VALUE$ }
}

# -- The claim path (batch B; spec §5.1.1, §5.8 .003 / .030-.032) -------------

# The root is independent: the fallback claim branches (.031 The Neighbour's
# Seat, .032 The Vassal's Seat) replace the liege claim (spec §5.1.1).
# Users: .003 (desc, b_indep, c_indep), .065, .067.
eotg_aug_kp_root_is_independent = {
    is_independent_ruler = yes
}

# A vassal root can raise a claimant faction against the liege for the
# liege's primary title (.003 b / c -> .030). Mirrors what vanilla checks
# when the faction is created, so the option is not a dead end:
# common_can_character_create_trigger (common/scripted_triggers/
# 00_faction_triggers.txt: adult, no truce with the target, the target holds
# no strong hook on root and is not allied) and common_character_validity_
# trigger (the target is the liege, tier above barony). Not already in a
# faction (.030 creates one). Empire-tier claims stay open (owner ruling Q4).
# The faction itself is checked at .030 a/b with can_create_faction (it needs
# var:claiming_title, set in .030's immediate).
# Users: .003 (desc_vassal, b, c, e_syndicate), .030 (trigger).
eotg_aug_kp_liege_claim_possible = {
    eotg_aug_kp_root_is_vassal = yes
    is_adult = yes
    is_imprisoned = no
    highest_held_title_tier > tier_barony
    NOT = { exists = joined_faction }
    NOT = { has_truce = liege }
    NOT = { is_at_war_with = liege }
    liege = {
        is_landed = yes
        exists = primary_title
        eotg_is_unclaimed_holder = no
        NOT = { has_strong_hook = root }
        NOT = { is_allied_to = root }
    }
}

# Character scope: a neighbouring independent ruler whose primary title the
# kingpin can claim with vanilla claim_cb (.031). $RULER$ = the independent
# root. claim_cb needs a non-titular title (common/casus_belli_types/
# 00_claim.txt is_allowed_claim_title: is_titular = no). War immunity: never
# an Unclaimed placeholder. Truces are one-way: the root must not hold one
# against them (PX triggers.log has_truce).
# Users: .003 (b_indep), .031 (trigger, immediate).
eotg_aug_kp_neighbour_target = {
    is_alive = yes
    is_independent_ruler = yes
    NOT = { this = $RULER$ }
    is_landed = yes
    highest_held_title_tier >= tier_county
    exists = primary_title
    primary_title = { is_titular = no }
    eotg_is_unclaimed_holder = no
    eotg_is_unclaimed_folk = no
    NOT = { is_at_war_with = $RULER$ }
    NOT = { is_allied_to = $RULER$ }
    $RULER$ = { NOT = { has_truce = prev } }
}

# Character scope: a landed direct vassal of $RULER$ whose seat the root can
# hand to the kingpin by decree (.032): not the ruler's close family or
# spouse, primary title at least a county and below the ruler's tier, free,
# not at war with the ruler, never an Unclaimed placeholder.
# Users: .003 (c_indep), .032 (trigger, immediate).
eotg_aug_kp_vassal_seat_target = {
    is_alive = yes
    is_vassal_of = $RULER$
    is_landed = yes
    is_imprisoned = no
    exists = primary_title
    primary_title = {
        tier >= tier_county
        is_titular = no
    }
    highest_held_title_tier < $RULER$.highest_held_title_tier
    NOT = { is_close_family_of = $RULER$ }
    NOT = { is_spouse_of = $RULER$ }
    NOT = { has_character_flag = eotg_flag_aug_kingpin }
    eotg_is_unclaimed_holder = no
    eotg_is_unclaimed_folk = no
    NOT = { is_at_war_with = $RULER$ }
}

# A claim option of .003 is open for the root: the liege claim for a vassal,
# or a neighbour's or a vassal's seat for an independent ruler. Root scope.
# Users: .003 (desc: .desc_independent; e_syndicate yields to the claim
# options so at most five options show, spec §5.8 .003 "Max 5 visible").
eotg_aug_kp_claim_open = {
    OR = {
        eotg_aug_kp_liege_claim_possible = yes
        AND = {
            eotg_aug_kp_root_is_independent = yes
            OR = {
                any_neighboring_top_liege_realm_owner = { eotg_aug_kp_neighbour_target = { RULER = root } }
                any_vassal = { eotg_aug_kp_vassal_seat_target = { RULER = root } }
            }
        }
    }
}
