# ── eotg_elven_monarchy_triggers.txt ─────────────────────────────────────────
# Scripted triggers for the Elven Monarchy government.
# ─────────────────────────────────────────────────────────────────────────────

eotg_st_is_elven_monarch = {
	has_government = eotg_elven_monarchy_government
}

# Star Warden: county-tier Elven Monarchy vassal
eotg_st_is_star_warden = {
	has_government = eotg_elven_monarchy_government
	is_landed = yes
	highest_held_title_tier = tier_county
}

# Celestial Lord: duchy-tier Elven Monarchy vassal
eotg_st_is_celestial_lord = {
	has_government = eotg_elven_monarchy_government
	highest_held_title_tier = tier_duchy
}

# Eternal Regent: kingdom-tier Elven Monarchy vassal
eotg_st_is_eternal_regent = {
	has_government = eotg_elven_monarchy_government
	highest_held_title_tier = tier_kingdom
}

# Starborn Sovereign: empire-tier Elven Monarchy ruler
eotg_st_is_starborn_sovereign = {
	has_government = eotg_elven_monarchy_government
	highest_held_title_tier = tier_empire
}

# Character is of elven heritage (expand list as cultures are defined)
eotg_st_is_elven_character = {
	culture = culture:eotg_culture_elrossi
}

# Character is NOT of elven heritage (used for council exclusion checks)
eotg_st_is_non_elven = {
	NOT = { eotg_st_is_elven_character = yes }
}

# Starborn Sovereign can issue a Cultural Appeal (not on cooldown)
eotg_st_elven_appeal_valid = {
	eotg_st_is_starborn_sovereign = yes
	NOT = { has_character_flag = eotg_elven_appeal_cooldown }
}

# Eternal Court confirmation is pending on this character
eotg_st_elven_court_pending = {
	has_character_flag = eotg_elven_court_awaiting_confirmation
}

# Eternal Court has strong vassal support (>= 3 powerful vassals approve)
eotg_st_elven_court_strong_support = {
	eotg_sv_elven_court_support_count >= 3
}

# Eternal Court has some support (>= 1 but < 3)
eotg_st_elven_court_moderate_support = {
	eotg_sv_elven_court_support_count >= 1
	NOT = { eotg_sv_elven_court_support_count >= 3 }
}
